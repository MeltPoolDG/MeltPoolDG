120 void
121 print_relative_norm(const GenericDataOut<dim, number> &generic_data_out,
122 dealii::Function<dim> &reference_function,
123 const std::string &norm_name = "norm") const
124 {
125 const dealii::ConditionalOStream pcout(std::cout,
126 dealii::Utilities::MPI::this_mpi_process(
127 this->mpi_communicator) == 0 and
128 parameters.base.verbosity_level >= 1);
129
130 const auto &dof_vector = generic_data_out.get_vector("density");
131 const auto &dof_handler = generic_data_out.get_dof_handler("density");
132
133 reference_function.set_time(generic_data_out.get_time());
134
135 number errors_squared[3] = {};
136
137 std::map<dealii::types::global_dof_index, dealii::Point<dim, number>> support_points =
138 dealii::DoFTools::map_dofs_to_support_points(generic_data_out.get_mapping(), dof_handler);
139 for (const auto &cell : dof_handler.active_cell_iterators())
140 {
141 if (cell->is_artificial() or cell->is_ghost())
142 continue;
143 const unsigned int nodes_per_cell = cell->get_fe().dofs_per_cell / (dim + 2);
144 std::vector<dealii::types::global_dof_index> local_dof_indices(
145 cell->get_fe().dofs_per_cell);
146 cell->get_dof_indices(local_dof_indices);
147
148 for (unsigned int i = 0; i < local_dof_indices.size(); ++i)
149 {
150 switch (unsigned int component = i / nodes_per_cell)
151 {
152 case 0: {
153 const number error =
154 dof_vector[local_dof_indices[i]] -
155 reference_function.value(support_points[local_dof_indices[i]], 0);
156 errors_squared[0] += error * error;
157 break;
158 }
159 case dim + 1: {
160 const number error =
161 dof_vector[local_dof_indices[i]] -
162 reference_function.value(support_points[local_dof_indices[i]], dim + 1);
163 errors_squared[2] += error * error;
164 break;
165 }
166 default: {
167 Assert(component > 0 and component < dim + 1, dealii::ExcInternalError());
168 const number error =
169 dof_vector[local_dof_indices[i]] -
170 reference_function.value(support_points[local_dof_indices[i]], component);
171 errors_squared[1] += error * error;
172 }
173 }
174 }
175 }
176
177 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
178 std::array<number, 3> errors;
179 for (unsigned int d = 0; d < 3; ++d)
180 errors[d] = std::sqrt(errors_squared[d]);
181
182 std::ostringstream output;
183 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
184 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
185 Journal::print_line(pcout, output.str(), "compressible_flow");
186 }
187 };
