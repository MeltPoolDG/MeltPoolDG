188 inline DEAL_II_ALWAYS_INLINE //
189 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
190 CompressibleFlowViscousKernels<dim, number>::calculate_viscous_stress_tensor(
191 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> &grad_u) const
192 {
193 const dealii::VectorizedArray<number> div_u = 2. / 3. * dealii::trace(grad_u);
194
195 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> out;
196 for (unsigned int d = 0; d < dim; ++d)
197 {
198 for (unsigned int e = 0; e < dim; ++e)
199 out[d][e] = material.data.dynamic_viscosity * (grad_u[d][e] + grad_u[e][d]);
200 out[d][d] -= material.data.dynamic_viscosity * div_u;
201 }
202
203 return out;
204 }

206 template <int dim, typename number>
207 inline DEAL_II_ALWAYS_INLINE //
208 auto
209 CompressibleFlowViscousKernels<dim, number>::calculate_viscous_flux(
210 const ConservedVariablesType &conserved_variables,
211 const ConservedVariablesGradType &grad_conserved_variables) const
212 -> ConservedVariablesGradType
213 {
214 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
215 calculate_velocity<dim, number>(conserved_variables);
216
217 const auto grad_u = calculate_grad_velocity(conserved_variables, grad_conserved_variables);
218
219 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> viscous_stress =
220 calculate_viscous_stress_tensor(grad_u);
221
222 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> neg_heat_flux =
223 material.data.thermal_conductivity *
224 material.eos_utils->calculate_grad_T(conserved_variables, grad_conserved_variables);
225
226 ConservedVariablesGradType flux;
227 for (unsigned int d = 0; d < dim; ++d)
228 {
229 // density
230 flux[0][d] = 0.0;
231
232 // momentum
233 for (unsigned int e = 0; e < dim; ++e)
234 flux[e + 1][d] = viscous_stress[e][d];
235
236 // energy
237 flux[dim + 1][d] = neg_heat_flux[d];
238
239 for (unsigned int e = 0; e < dim; ++e)
240 flux[dim + 1][d] += velocity[e] * viscous_stress[d][e];
241 }
242
243 return flux;
244 }

246 template <int dim, typename number>
247 inline DEAL_II_ALWAYS_INLINE //
248 auto
249 CompressibleFlowViscousKernels<dim, number>::calculate_viscous_numerical_flux(
250 const ConservedVariablesType &u_m,
251 const ConservedVariablesType &u_p,
252 const ConservedVariablesGradType &grad_u_m,
253 const ConservedVariablesGradType &grad_u_p,
254 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
255 const dealii::VectorizedArray<number> penalty_parameter) const -> ConservedVariablesType
256 {
257 const auto flux_m = calculate_viscous_flux(u_m, grad_u_m);
258
259 const auto flux_p = calculate_viscous_flux(u_p, grad_u_p);
260
261 return contract_average_tensor_with_vector<dim + 2, dim, number>(flux_m, flux_p, normal) -
262 penalty_parameter * material.data.dynamic_viscosity / material.data.reference_density *
263 (u_m - u_p);
264 }

266 template <int dim, typename number>
267 inline DEAL_II_ALWAYS_INLINE //
268 auto
269 CompressibleFlowViscousKernels<dim, number>::calculate_viscous_numerical_flux_gradient(
270 const ConservedVariablesType &u_m,
271 const ConservedVariablesType &u_p,
272 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
273 -> std::pair<ConservedVariablesGradType, ConservedVariablesGradType>
274 {
275 ConservedVariablesGradType jump_u;
276 for (unsigned int e = 0; e < dim + 2; ++e)
277 for (unsigned int d = 0; d < dim; ++d)
278 jump_u[e][d] = (u_m[e] - u_p[e]) * normal[d];
279
280 // use jumps instead of gradients for evaluating the viscous flux
281 const ConservedVariablesGradType flux_m = 0.5 * calculate_viscous_flux(u_m, jump_u);
282 const ConservedVariablesGradType flux_p = 0.5 * calculate_viscous_flux(u_p, jump_u);
283
284 return std::make_pair(flux_m, flux_p);
285 }

287 template <int dim, typename number>
288 inline DEAL_II_ALWAYS_INLINE //
289 auto
290 CompressibleFlowViscousKernels<dim, number>::calculate_jacobian_viscous_numerical_flux(
291 const std::pair<ConservedVariablesType, ConservedVariablesType> &w_q,
292 const std::pair<ConservedVariablesGradType, ConservedVariablesGradType> &grad_w_q,
293 const std::pair<ConservedVariablesType, ConservedVariablesType> &delta_w_q,
294 const std::pair<ConservedVariablesGradType, ConservedVariablesGradType> &grad_delta_w_q,
295 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
296 dealii::VectorizedArray<number> penalty_parameter) const -> ConservedVariablesGradType
297 {
298 ConservedVariablesGradType flux_p =
299 0.5 * calculate_jacobian_viscous_flux(w_q.second,
300 grad_w_q.second,
301 delta_w_q.second,
302 grad_delta_w_q.second);
303 ConservedVariablesGradType flux_m = 0.5 * calculate_jacobian_viscous_flux(w_q.first,
304 grad_w_q.first,
305 delta_w_q.first,
306 grad_delta_w_q.first);
307
308 ConservedVariablesGradType flux;
309 for (unsigned int i = 0; i < dim + 2; ++i)
310 flux[i] = (flux_p[i] + flux_m[i]);
311
312 flux -=
313 calculate_jacobian_viscous_numerical_flux_jump_term(delta_w_q, normal, penalty_parameter);
314 return flux;
315 }

317 template <int dim, typename number>
318 inline DEAL_II_ALWAYS_INLINE //
319 auto
320 CompressibleFlowViscousKernels<dim, number>::calculate_jacobian_viscous_flux(
321 const ConservedVariablesType &w_q,
322 const ConservedVariablesGradType &grad_w_q,
323 const ConservedVariablesType &delta_w_q,
324 const ConservedVariablesGradType &grad_delta_w_q) const -> ConservedVariablesGradType
325 {
326 ConservedVariablesGradType viscous_differential_change;
327
328 // precompute values
329 dealii::VectorizedArray<number> rho_inv = 1.0 / w_q[0];
330 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> m_q;
331 for (unsigned int i = 0; i < dim; ++i)
332 m_q[i] = w_q[i + 1];
333 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> delta_m_q;
334 for (unsigned int i = 0; i < dim; ++i)
335 delta_m_q[i] = delta_w_q[i + 1];
336
337 /* change in density flux */
338 viscous_differential_change[0] = dealii::Tensor<1, dim, dealii::VectorizedArray<number>>();
339
340 /* change in momentum flux */
341 // density part
342 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> grad_m_q;
343 for (unsigned int i = 0; i < dim; ++i)
344 grad_m_q[i] = grad_w_q[i + 1];
345 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> grad_delta_m_q;
346 for (unsigned int i = 0; i < dim; ++i)
347 grad_delta_m_q[i] = grad_delta_w_q[i + 1];
348
349 auto v_q = calculate_velocity<dim, number>(w_q);
350 const auto grad_v_q_temp = calculate_grad_velocity(w_q, grad_w_q);
351 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> grad_v_q;
352 for (unsigned int i = 0; i < dim; ++i)
353 for (unsigned int j = 0; j < dim; ++j)
354 grad_v_q[i][j] = grad_v_q_temp[i][j];
355
356 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> param_a =
357 rho_inv * (-delta_w_q[0] * grad_v_q);
358 param_a += rho_inv * rho_inv * delta_w_q[0] * dyadic_product(v_q, grad_w_q[0]);
359 param_a -= rho_inv * dyadic_product(v_q, grad_delta_w_q[0]);
360
361 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> param_c =
362 param_a;
363 param_c += transpose(param_a);
364 param_c -= 2. / 3. * trace(param_a) * identity<dim, dealii::VectorizedArray<number>>();
365 param_c *= material.data.dynamic_viscosity;
366
367 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> param_b =
368 rho_inv * grad_delta_m_q;
369 param_b -= rho_inv * rho_inv * dyadic_product(delta_m_q, grad_w_q[0]);
370
371 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> param_d =
372 param_b;
373 param_d += transpose(param_b);
374 param_d -= 2. / 3. * trace(param_b) * identity<dim, dealii::VectorizedArray<number>>();
375 param_d *= material.data.dynamic_viscosity;
376
377
378 for (unsigned int i = 0; i < dim; ++i)
379 {
380 viscous_differential_change[i + 1] = param_c[i] + param_d[i];
381 }
382
383 /* change in energy density flux */
384 const auto tau_temp = calculate_viscous_stress_tensor(grad_v_q_temp);
385
386 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> tau;
387 for (unsigned int i = 0; i < dim; ++i)
388 for (unsigned int j = 0; j < dim; ++j)
389 tau[i][j] = tau_temp[i][j];
390 // density part
391 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> rho_energy_density;
392 rho_energy_density += matrix_vector_product(param_c, m_q) * rho_inv;
393 rho_energy_density -= rho_inv * rho_inv * matrix_vector_product(tau, m_q) * delta_w_q[0];
394
395
396 rho_energy_density +=
397 lambda_div_c * (-rho_inv * rho_inv * grad_w_q[dim + 1] * delta_w_q[0] +
398 rho_inv * rho_inv * rho_inv * w_q[dim + 1] * delta_w_q[0] * grad_w_q[0] +
399 grad_w_q[0] * w_q[dim + 1] * rho_inv * rho_inv * rho_inv * delta_w_q[0] -
400 w_q[dim + 1] * rho_inv * rho_inv * grad_delta_w_q[0] +
401 matrix_vector_product(grad_v_q, m_q) * rho_inv * rho_inv * delta_w_q[0]);
402
403 rho_energy_density -=
404 lambda_div_c * rho_inv * rho_inv *
405 (rho_inv * delta_w_q[0] * matrix_vector_product(dyadic_product(v_q, grad_w_q[0]), m_q) -
406 matrix_vector_product(dyadic_product(v_q, grad_delta_w_q[0]), m_q));
407
408
409 rho_energy_density +=
410 lambda_div_c * rho_inv * rho_inv * delta_w_q[0] * matrix_vector_product(grad_v_q, m_q);
411
412
413 // momentum part
414 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> momentum_energy_density;
415 momentum_energy_density += rho_inv * matrix_vector_product(param_d, m_q);
416 momentum_energy_density += rho_inv * matrix_vector_product(tau, delta_m_q);
417 momentum_energy_density -= lambda_div_c * rho_inv * matrix_vector_product(grad_v_q, delta_m_q);
418 momentum_energy_density -=
419 lambda_div_c * rho_inv * rho_inv * matrix_vector_product(grad_delta_m_q, m_q);
420 momentum_energy_density += lambda_div_c * rho_inv * rho_inv * rho_inv *
421 matrix_vector_product(dyadic_product(delta_m_q, grad_w_q[0]), m_q);
422
423
424 // energy density part
425 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> energy_energy_density;
426 energy_energy_density += rho_inv * grad_delta_w_q[dim + 1];
427 energy_energy_density -= grad_w_q[0] * rho_inv * rho_inv * delta_w_q[dim + 1];
428 energy_energy_density *= lambda_div_c;
429
430 // build the sum
431 viscous_differential_change[dim + 1] =
432 rho_energy_density + momentum_energy_density + energy_energy_density;
433 return viscous_differential_change;
434 }

437 inline DEAL_II_ALWAYS_INLINE //
438 auto
439 CompressibleFlowViscousKernels<dim, number>::
440 calculate_jacobian_viscous_numerical_flux_jump_term(
