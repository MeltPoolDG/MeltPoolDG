15 // In order to smoothly activate the recoil pressure, a scaling coefficient is computed and
16 // multiplied with the recoil pressure coefficient
17 //
18 // -
19 // | 0 if T <= T_ac
20 // |
21 // | T - T_ac
22 // scaling_coeff = | -------- if T_ac < T < T_v
23 // | T_v - T_ac
24 // |
25 // | 1 if T >= T_v
26 // -
27 //
28 // with @p T_ac the activation temperature of the recoil pressure and @p T_v the boiling temperature.
29
30 template <typename number>
31 inline dealii::VectorizedArray<number>
32 compute_scaling_coeff(const dealii::VectorizedArray<number> &T,
33 const number T_ac,
34 const number T_v)
35 {
36 return compare_and_apply_mask<dealii::SIMDComparison::less_than>(
37 T,
38 T_ac,
39 0.0,
40 compare_and_apply_mask<dealii::SIMDComparison::greater_than_or_equal>(
41 T, T_v, 1., (T - T_ac) / (T_v - T_ac)));
42 }
43
44 template <typename number>
45 inline number
46 compute_scaling_coeff(const number T, const number T_ac, const number T_v)
47 {
48 return (T < T_ac) ? 0.0 : (T >= T_v) ? 1. : (T - T_ac) / (T_v - T_ac);
49 }
