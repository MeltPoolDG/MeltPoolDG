195 template <int dim, typename number>
196 inline DEAL_II_ALWAYS_INLINE //
197 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
198 calculate_velocity(
199 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
200 {
201 const dealii::VectorizedArray<number> inverse_density =
202 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
203
204 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
205 for (unsigned int d = 0; d < dim; ++d)
206 velocity[d] = conserved_variables[1 + d] * inverse_density;
207
208 return velocity;
209 }

211 template <int dim, typename number>
212 inline DEAL_II_ALWAYS_INLINE //
213 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
214 calculate_grad_velocity(
215 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
216 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
217 &grad_conserved_variables)
218 {
219 const dealii::VectorizedArray<number> inverse_density =
220 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
221 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
222 calculate_velocity<dim, number>(conserved_variables);
223
224 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
225 for (unsigned int d = 0; d < dim; ++d)
226 grad_rho[d] = grad_conserved_variables[0][d];
227
228 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
229 for (unsigned int d = 0; d < dim; ++d)
230 for (unsigned int e = 0; e < dim; ++e)
231 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
232
233 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
234 for (unsigned int d = 0; d < dim; ++d)
235 for (unsigned int e = 0; e < dim; ++e)
236 grad_velocity[d][e] =
237 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
238
239 return grad_velocity;
240 }

242 template <int dim, typename Number>
243 void
244 calculate_penalty_parameter(
245 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
246 const dealii::MatrixFree<dim, Number> &matrix_free,
247 const std::string &domain_representation_type,
248 const unsigned int dof_index,
249 const Number scaling_factor)
250 {
251 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
252 array_penalty_parameter.resize(n_cells);
253
254 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
255 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
256 unsigned int const degree = fe.degree;
257
258 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
259 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
260 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
261 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
262
263 auto const reference_cells =
264 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
265 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
266
267 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
268 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
269
270 auto const face_quadrature =
271 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
272 1);
273 dealii::FEFaceValues<dim> fe_face_values(mapping,
274 fe,
275 face_quadrature,
276 dealii::update_JxW_values);
277
278 if (domain_representation_type == "fitted")
279 {
280 for (unsigned int i = 0; i < n_cells; ++i)
281 {
282 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
283 {
284 typename dealii::DoFHandler<dim>::cell_iterator cell =
285 matrix_free.get_cell_iterator(i, v, dof_index);
286 fe_values.reinit(cell);
287
288 // calculate cell volume
289 Number volume = 0;
290 for (unsigned int q = 0; q < quadrature.size(); ++q)
291 {
292 volume += fe_values.JxW(q);
293 }
294
295 // calculate surface area
296 Number surface_area = 0;
297 for (unsigned int const f : cell->face_indices())
298 {
299 fe_face_values.reinit(cell, f);
300 Number const factor =
301 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
302 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
303 {
304 surface_area += fe_face_values.JxW(q) * factor;
305 }
306 }
307
308 array_penalty_parameter[i][v] = surface_area / volume * fac;
309 }
310 }
311 }
312 else if (domain_representation_type == "cut")
313 {
314 for (unsigned int i = 0; i < n_cells; ++i)
315 {
316 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
317 {
318 typename dealii::DoFHandler<dim>::cell_iterator cell =
319 matrix_free.get_cell_iterator(i, v, dof_index);
320
321 // simplified computation for hypercube elements
322 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
323 }
324 }
325 }
326 else
327 AssertThrow(false,
328 dealii::ExcMessage("The domain representation type '" +
329 domain_representation_type + "' is not supported."));
330 }

332 template <int dim, typename number>
333 void
334 update_primitive_variables_solution(
335 dealii::LinearAlgebra::distributed::Vector<number> &solution_primitive_variables,
336 const dealii::LinearAlgebra::distributed::Vector<number> &solution,
337 const ScratchData<dim, dim, number> &scratch_data,
338 const unsigned int dof_idx,
339 const unsigned int quad_idx,
340 const CompressibleFlowMaterial<dim, number> *material_liquid,
341 const CompressibleFlowMaterial<dim, number> *material_gas)
342 {
343 const dealii::MatrixFree<dim, number, dealii::VectorizedArray<number>> &matrix_free =
344 scratch_data.get_matrix_free();
345 unsigned int n_support_points_per_cell = scratch_data.get_n_dofs_per_cell(dof_idx) / (dim + 2);
346
347 auto process_cell = [&](CutUtil::CellCategory category,
348 unsigned int first_component,
349 const CompressibleFlowMaterial<dim, number> *material,
350 const unsigned int cell_batch) {
351 if (!material)
352 return;
353 auto eval = FECellIntegrator<dim, dim + 2, number>(
354 matrix_free, dof_idx, quad_idx, first_component, category);
355 eval.reinit(cell_batch);
356 eval.read_dof_values(solution);
357
358 for (unsigned int i = 0; i < n_support_points_per_cell; ++i)
359 {
360 const auto &u_cons = eval.get_dof_value(i);
361 auto u_prim = material->eos_utils->convert_conservative_into_primitive_variables(u_cons);
362 eval.submit_dof_value(u_prim, i);
363 }
364
365 eval.set_dof_values(solution_primitive_variables);
366 };
367
368 if (!matrix_free.get_dof_info(dof_idx).cell_active_fe_index.empty())
369 {
370 // using hp::FECollection (cutFEM/DG)
371 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
372 {
373 const auto cell_category = matrix_free.get_cell_category(cell_batch);
374
375 if (cell_category == CutUtil::CellCategory::liquid)
376 {
377 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
378 }
379 else if (cell_category == CutUtil::CellCategory::gas)
380 {
381 process_cell(CutUtil::CellCategory::gas, dim + 2, material_gas, cell_batch);
382 }
383 else if (cell_category == CutUtil::CellCategory::intersected)
384 {
385 process_cell(CutUtil::CellCategory::intersected, 0, material_liquid, cell_batch);
386 process_cell(CutUtil::CellCategory::intersected, dim + 2, material_gas, cell_batch);
387 }
388 }
389 }
390 else
391 {
392 // not using hp::FECollection (fitted mesh)
393 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
394 {
395 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
396 }
397 }
