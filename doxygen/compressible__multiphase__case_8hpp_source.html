111 dealii::Function<dim> &reference_function,
112 const std::string &norm_name = "norm") const
113 {
114 const dealii::ConditionalOStream pcout(std::cout,
115 dealii::Utilities::MPI::this_mpi_process(
116 this->mpi_communicator) == 0 and
117 parameters.base.verbosity_level >= 1);
118
119 const auto &dof_vector = generic_data_out.get_vector("density_liquid");
120 const auto &dof_handler = generic_data_out.get_dof_handler("density_liquid");
121
122 const unsigned int n_dofs_per_element_per_phase =
123 0.5 * dof_handler.get_fe_collection().max_dofs_per_cell();
124 reference_function.set_time(generic_data_out.get_time());
125 number errors_squared[3] = {};
126
127 // reading DoF-vector and DoFHandler of level-set
128 // (an assert is thrown in the getter functions, if 'level_set' can not be found in
129 // the entries of 'generic_data_out')
130 const auto &level_set = generic_data_out.get_vector("level_set");
131 const auto &level_set_dof_handler = generic_data_out.get_dof_handler("level_set");
132
133 // generate FESystem
134 const unsigned int fe_degree = dof_handler.get_fe_collection().max_degree();
135 dealii::FE_DGQ<dim> fe_q(fe_degree);
136 dealii::FESystem<dim> fe_point_temp(fe_q, dim + 2);
137
138 // classify cells
139 dealii::NonMatching::MeshClassifier<dim> mesh_classifier(level_set_dof_handler, level_set);
140 mesh_classifier.reclassify();
141
142 // compute quadrature
143 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>>
144 mapping_info_cell_liquid(generic_data_out.get_mapping(),
145 dealii::update_values | dealii::update_JxW_values);
146 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>>
147 mapping_info_cell_gas(generic_data_out.get_mapping(),
148 dealii::update_values | dealii::update_JxW_values);
149
150 dealii::hp::QCollection<1> q_collection((dealii::QGauss<1>(fe_degree + 1)));
151 dealii::NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(
152 q_collection, level_set_dof_handler, level_set);
153
154 auto physical_liquid = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
155 return (mesh_classifier.location_to_level_set(i) ==
156 dealii::NonMatching::LocationToLevelSet::intersected) or
157 (mesh_classifier.location_to_level_set(i) ==
158 dealii::NonMatching::LocationToLevelSet::outside);
159 };
160 auto physical_gas = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
161 return (mesh_classifier.location_to_level_set(i) ==
162 dealii::NonMatching::LocationToLevelSet::intersected) or
163 (mesh_classifier.location_to_level_set(i) ==
164 dealii::NonMatching::LocationToLevelSet::inside);
165 };
166
167 const auto physical_active_cell_iterators_liquid =
168 dof_handler.active_cell_iterators() | physical_liquid;
169 const auto physical_active_cell_iterators_gas =
170 dof_handler.active_cell_iterators() | physical_gas;
171
172 std::vector<dealii::Quadrature<dim>> quad_vec_cell_liquid;
173 std::vector<dealii::Quadrature<dim>> quad_vec_cell_gas;
174
175 const unsigned int n_active_cells = std::distance(dof_handler.active_cell_iterators().begin(),
176 dof_handler.active_cell_iterators().end());
177
178 for (const auto &cell : physical_active_cell_iterators_liquid)
179 {
180 quadrature_generator.generate(cell);
181 // liquid phase has positive level-set
182 quad_vec_cell_liquid.push_back(quadrature_generator.get_outside_quadrature());
183 }
184
185 for (const auto &cell : physical_active_cell_iterators_gas)
186 {
187 quadrature_generator.generate(cell);
188 // gas phase has negative level-set
189 quad_vec_cell_gas.push_back(quadrature_generator.get_inside_quadrature());
190 }
191
192 mapping_info_cell_liquid.reinit_cells(physical_active_cell_iterators_liquid,
193 quad_vec_cell_liquid,
194 n_active_cells);
195 mapping_info_cell_gas.reinit_cells(physical_active_cell_iterators_gas,
196 quad_vec_cell_gas,
197 n_active_cells);
198
199 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>>
200 fe_point_eval_liquid(mapping_info_cell_liquid, fe_point_temp);
201 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>>
202 fe_point_eval_gas(mapping_info_cell_gas, fe_point_temp);
203
204 auto compute_error_norm = [&](auto &fe_point_eval,
205 auto &physical_domain,
206 const bool is_gas_phase) {
207 const unsigned int offset = is_gas_phase ? 0 : dim + 2;
208
209 for (const auto &cell : dof_handler.active_cell_iterators())
210 {
211 if (cell->is_artificial() or cell->is_ghost() or !physical_domain(cell))
212 continue;
213
214 const unsigned int n_dofs_per_cell = cell->get_fe().dofs_per_cell;
215 std::vector<number> solution_values_in(n_dofs_per_element_per_phase);
216
217 if (n_dofs_per_cell != n_dofs_per_element_per_phase)
218 {
219 // For an intersected element, we have to extract the Dofs, which are relevant for
220 // the current phase
221 std::vector<number> solution_values_in_tmp(n_dofs_per_cell);
222 cell->get_dof_values(dof_vector,
223 solution_values_in_tmp.begin(),
224 solution_values_in_tmp.end());
225
226 if (is_gas_phase)
227 solution_values_in.assign(solution_values_in_tmp.begin() +
228 n_dofs_per_element_per_phase,
229 solution_values_in_tmp.end());
230 else
231 solution_values_in.assign(solution_values_in_tmp.begin(),
232 solution_values_in_tmp.begin() +
233 n_dofs_per_element_per_phase);
234 }
235 else
236 cell->get_dof_values(dof_vector,
237 solution_values_in.begin(),
238 solution_values_in.end());
239
240 fe_point_eval.reinit(cell->active_cell_index());
241 fe_point_eval.evaluate(solution_values_in, dealii::EvaluationFlags::values);
242
243 dealii::VectorizedArray<number> local_errors_squared[3] = {};
244
245 for (const unsigned int q : fe_point_eval.quadrature_point_indices())
246 {
247 const auto solution = fe_point_eval.get_value(q);
248 dealii::VectorizedArray<number> error[dim + 2] = {};
249 for (unsigned int i = 0; i < dim + 2; ++i)
250 {
251 error[i] =
252 VectorTools::evaluate_function_at_vectorized_points<dim, number>(
253 reference_function, fe_point_eval.quadrature_point(q), i + offset) -
254 solution[i];
255 }
256
257 const auto JxW = fe_point_eval.JxW(q);
258 local_errors_squared[0] = error[0] * error[0] * JxW;
259
260 for (unsigned int d = 0; d < dim; ++d)
261 local_errors_squared[1] = (error[d + 1] * error[d + 1]) * JxW;
262
263 local_errors_squared[2] = (error[dim + 1] * error[dim + 1]) * JxW;
264
265 for (unsigned int v = 0; v < fe_point_eval.n_active_entries_per_quadrature_batch(q);
266 ++v)
267 for (unsigned int d = 0; d < 3; ++d)
268 errors_squared[d] += local_errors_squared[d][v];
269 }
270 }
271 };
272
273 compute_error_norm(fe_point_eval_liquid, physical_liquid, false);
274 compute_error_norm(fe_point_eval_gas, physical_gas, true);
275
276 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
277 std::array<number, 3> errors{};
278 for (unsigned int d = 0; d < 3; ++d)
279 errors[d] = std::sqrt(errors_squared[d]);
280
281 std::ostringstream output;
282 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
283 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
284 Journal::print_line(pcout, output.str(), "compressible_multiphase");
285 }
