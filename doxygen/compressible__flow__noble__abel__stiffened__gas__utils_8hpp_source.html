34 inline DEAL_II_ALWAYS_INLINE //
35 dealii::VectorizedArray<number>
36 calculate_thermodynamic_pressure(

40 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
41 calculate_velocity<dim, number>(conserved_variables);

43 return ((material_data.gamma - 1.) *

61 inline DEAL_II_ALWAYS_INLINE //
62 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
63 calculate_grad_T(
64 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
65 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
66 &grad_conserved_variables) override
67 {
68 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> u =
69 calculate_velocity<dim, number>(conserved_variables);
70 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_u =
71 calculate_grad_velocity<dim, number>(conserved_variables, grad_conserved_variables);
72 const dealii::VectorizedArray<number> rho = conserved_variables[0];
73 const dealii::VectorizedArray<number> inv_rho = dealii::VectorizedArray<number>(1.) / rho;
74 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho =
75 grad_conserved_variables[0];
76 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_E =
77 inv_rho *
78 (grad_conserved_variables[dim + 1] - inv_rho * conserved_variables[dim + 1] * grad_rho);
79
80 return material_data.gamma / material_data.specific_isobaric_heat *
81 (grad_E - grad_u * u + material_data.eos_data.p_inf * inv_rho * inv_rho * grad_rho);
82 }

131 inline DEAL_II_ALWAYS_INLINE //
132 CompressibleFlowTypes::ConservedVariablesType<dim, number>
133 convert_primitive_into_conservative_variables(
134 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &u_prim) override
135 {
136 CompressibleFlowTypes::ConservedVariablesType<dim, number> u_cons;
137
138 // density
139 u_cons[0] = (u_prim[0] + material_data.eos_data.p_inf) /
140 (material_data.specific_isobaric_heat * u_prim[dim + 1] *
141 (material_data.gamma - 1.) / material_data.gamma +
142 material_data.eos_data.b * (u_prim[0] + material_data.eos_data.p_inf));
143 // momentum
144 for (unsigned int i = 1; i < dim + 1; i++)
145 u_cons[i] = u_prim[i] * u_cons[0];
146 // total energy
147 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
148 calculate_velocity<dim, number>(u_cons);
149 u_cons[dim + 1] =
150 u_cons[0] * (material_data.specific_isobaric_heat / material_data.gamma * u_prim[dim + 1] +
151 material_data.eos_data.p_inf * (1. / u_cons[0] - material_data.eos_data.b) +
152 material_data.eos_data.q + 0.5 * scalar_product(velocity, velocity));
153
154 return u_cons;
155 }
