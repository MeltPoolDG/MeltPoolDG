95 void
96 print_relative_norm(const GenericDataOut<dim, number> &generic_data_out,
97 dealii::Function<dim> &reference_function,
98 const std::string &norm_name = "norm") const
99 {
100 const dealii::ConditionalOStream pcout(std::cout,
101 dealii::Utilities::MPI::this_mpi_process(
102 this->mpi_communicator) == 0 and
103 parameters.base.verbosity_level >= 1);
104
105 const auto &dof_vector = generic_data_out.get_vector("density");
106 const auto &dof_handler = generic_data_out.get_dof_handler("density");
107
108 reference_function.set_time(generic_data_out.get_time());
109
110 number errors_squared[3] = {};
111
112 std::map<dealii::types::global_dof_index, dealii::Point<dim, number>> support_points =
113 dealii::DoFTools::map_dofs_to_support_points(generic_data_out.get_mapping(), dof_handler);
114 for (const auto &cell : dof_handler.active_cell_iterators())
115 {
116 if (cell->is_artificial() or cell->is_ghost())
117 continue;
118 const unsigned int nodes_per_cell = cell->get_fe().dofs_per_cell / (dim + 2);
119 std::vector<dealii::types::global_dof_index> local_dof_indices(
120 cell->get_fe().dofs_per_cell);
121 cell->get_dof_indices(local_dof_indices);
122
123 for (unsigned int i = 0; i < local_dof_indices.size(); ++i)
124 {
125 switch (unsigned int component = i / nodes_per_cell)
126 {
127 case 0: {
128 const number error =
129 dof_vector[local_dof_indices[i]] -
130 reference_function.value(support_points[local_dof_indices[i]], 0);
131 errors_squared[0] += error * error;
132 break;
133 }
134 case dim + 1: {
135 const number error =
136 dof_vector[local_dof_indices[i]] -
137 reference_function.value(support_points[local_dof_indices[i]], dim + 1);
138 errors_squared[2] += error * error;
139 break;
140 }
141 default: {
142 Assert(component > 0 and component < dim + 1, dealii::ExcInternalError());
143 const number error =
144 dof_vector[local_dof_indices[i]] -
145 reference_function.value(support_points[local_dof_indices[i]], component);
146 errors_squared[1] += error * error;
147 }
148 }
149 }
150 }
151
152 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
153 std::array<number, 3> errors;
154 for (unsigned int d = 0; d < 3; ++d)
155 errors[d] = std::sqrt(errors_squared[d]);
156
157 std::ostringstream output;
158 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
159 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
160 Journal::print_line(pcout, output.str(), "compressible_flow");
161 }
