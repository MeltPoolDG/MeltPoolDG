151 template <typename number1, typename number2>
152 void
153 remove_duplicates(std::vector<number1> &a, std::vector<number2> &b)
154 {
155 std::unordered_set<int> seen;
156 auto it_a = a.begin();
157 auto it_b = b.begin();
158
159 while (it_a != a.end())
160 {
161 if (seen.find(*it_a) != seen.end())
162 {
163 // If element is duplicate, remove it from both vectors
164 it_a = a.erase(it_a);
165 it_b = b.erase(it_b);
166 }
167 else
168 {
169 // If element is not duplicate, mark it as seen and move iterators forward
170 seen.insert(*it_a);
171 ++it_a;
172 ++it_b;
173 }
174 }
175 }

311 void
312 generate_points_along_vector(std::vector<Point<dim>> &points_normal_to_interface,
313 const Point<dim> &starting_point,
314 const vector &unit_vec,
315 const double max_distance_per_side,
316 const unsigned int n_inc_per_side,
317 const bool bidirectional = true)
318 {

358 /*
359 * Compute a linearly extrapolated initial guess value (predictor) for the
360 * Newton-Raphson solver at time "t_n+1" from the old solution @param old_vec
361 * computed at time "t_n" and the old old solution @param old_old_vec computed
362 * at time "t_n-1". In addition the @param current_time_increment
363 * dt_n+1 = t_n+1 - t_n and the @param old_time_incrementdt_n = t_n-t_n-1 have
364 * to be provided.
365 *
366 * The predictor is computed as follows
367 *
368 * dt_n+1
369 * q_n+1^(0) = q_n + ------ * (q_n-q_n-1)
370 * dt_n
371 *
372 */
373 template <typename VectorType>
374 void
375 compute_linear_predictor(const VectorType &old_vec,
376 const VectorType &old_old_vec,
377 VectorType &predictor,
378 const double current_time_increment,
379 const double old_time_increment)
380 {
381 if (std::abs(old_time_increment) < 1e-12)
382 {
383 predictor = old_vec;
384 return;
385 }
386
387 const double fraction = current_time_increment / old_time_increment;
388
389 for (unsigned int c = 0; c < internal::n_blocks(predictor); ++c)
390 DEAL_II_OPENMP_SIMD_PRAGMA
391 for (unsigned int i = 0; i < internal::block(predictor, c).locally_owned_size(); ++i)
392 internal::block(predictor, c).local_element(i) =
393 (fraction + 1.0) * internal::block(old_vec, c).local_element(i) -
394 fraction * internal::block(old_old_vec, c).local_element(i);
395 }

424 template <int dim>
425 FullMatrix<double>
426 create_dof_interpolation_matrix(const DoFHandler<dim> &dof_handler_1, // e.g. pressure
427 const DoFHandler<dim> &dof_handler_2, // e.g. level set
428 const bool do_sort_lexicographically)
429 {
430 FullMatrix<double> dof_interpolation_matrix(dof_handler_1.get_fe().n_dofs_per_cell(),
431 dof_handler_2.get_fe().n_dofs_per_cell());
432
433 const FE_Q_iso_Q1<dim> *fe_2 =
434 dynamic_cast<const FE_Q_iso_Q1<dim> *>(&dof_handler_2.get_fe());
435
436 AssertThrow(fe_2,
437 ExcMessage("dof_handler_2 must contain finite elements of type FE_Q_iso_Q1."));
438
439 const std::vector<unsigned int> lexicographic_ls = fe_2->get_poly_space_numbering_inverse();
440
441
442 //@todo: get rid of base_element
443 if (const FE_Q<dim> *fe_1 =
444 dynamic_cast<const FE_Q<dim> *>(&dof_handler_1.get_fe().base_element(0)))
445 {
446 const std::vector<unsigned int> lexicographic_p =
447 fe_1->get_poly_space_numbering_inverse();
448 for (unsigned int j = 0; j < fe_1->dofs_per_cell; ++j)
449 {
450 const Point<dim> p =
451 fe_1->get_unit_support_points()[do_sort_lexicographically ? lexicographic_p[j] : j];
452 for (unsigned int i = 0; i < fe_2->dofs_per_cell; ++i)
453 dof_interpolation_matrix(j, i) = dof_handler_2.get_fe().shape_value(
454 do_sort_lexicographically ? lexicographic_ls[i] : i, p);
455 }
456 }
457 else if (const FE_Q_DG0<dim> *fe_1 =
458 dynamic_cast<const FE_Q_DG0<dim> *>(&dof_handler_1.get_fe().base_element(0)))
459 {
460 const std::vector<unsigned int> lexicographic_p =
461 fe_1->get_poly_space_numbering_inverse();
462
463 // Loop over all support points except the one for the discontinuous
464 // shape function in the middle of the cell (dofs_per_cell - 1).
465 for (unsigned int j = 0; j < fe_1->dofs_per_cell - 1; ++j)
466 {
467 const Point<dim> p =
468 fe_1->get_unit_support_points()[do_sort_lexicographically ? lexicographic_p[j] : j];
469 for (unsigned int i = 0; i < fe_2->dofs_per_cell; ++i)
470 dof_interpolation_matrix(j, i) = dof_handler_2.get_fe().shape_value(
471 do_sort_lexicographically ? lexicographic_ls[i] : i, p);
472 }
473 }
474 else
475 AssertThrow(
476 false,
477 ExcMessage(
478 "The operation for the requested pair of DoFHandler "
479 "is not supported. Types must be: dof_handler_1 = FE_Q_iso_Q1; dof_handler_2 = FE_Q || FE_Q_DG0."));
480
481 return dof_interpolation_matrix;
482 }

494 template <int dim>
495 void
496 compute_gradient_at_interpolated_dof_values(
497 FECellIntegrator<dim, 1, double> &values,
498 FECellIntegrator<dim, 1, double> &interpolated_values,
499 const FullMatrix<double> &interpolation_matrix)
500 {
501 // Evaluate the field Φ at the support points of its space j
502 values.evaluate(EvaluationFlags::values);
503
504 // Loop over the support points of the to be interpolated space i
505 for (unsigned int i = 0; i < interpolated_values.dofs_per_cell; ++i)
506 {
507 VectorizedArray<double> interpolated_value = 0;
508
509 // Interpolate the field Φ from the support points of the space j
510 // of the original field to the one of the interpolated field i,
511 // using the interpolation matrix P
512 // _
513 // Φ = P · Φ
514 // i ij j
515 for (unsigned int j = 0; j < values.dofs_per_cell; ++j)
516 interpolated_value += interpolation_matrix(i, j) * values.get_dof_value(j);
517
518 // Store the interpolated values at the support points of the pressure space
519 interpolated_values.submit_dof_value(interpolated_value, i);
520 }
521
522 // Evaluate the gradient from the interpolated field
523 // _
524 // ∇ Φ
525 //
526 interpolated_values.evaluate(EvaluationFlags::gradients);
527 }
