149 template <typename number1, typename number2>
150 void
151 remove_duplicates(std::vector<number1> &a, std::vector<number2> &b)
152 {
153 std::unordered_set<int> seen;
154 auto it_a = a.begin();
155 auto it_b = b.begin();
156
157 while (it_a != a.end())
158 {
159 if (seen.find(*it_a) != seen.end())
160 {
161 // If element is duplicate, remove it from both vectors
162 it_a = a.erase(it_a);
163 it_b = b.erase(it_b);
164 }
165 else
166 {
167 // If element is not duplicate, mark it as seen and move iterators forward
168 seen.insert(*it_a);
169 ++it_a;
170 ++it_b;
171 }
172 }
173 }

317 void
318 generate_points_along_vector(std::vector<dealii::Point<dim>> &points_normal_to_interface,
319 const dealii::Point<dim> &starting_point,
320 const vector &unit_vec,
321 const number max_distance_per_side,
322 const unsigned int n_inc_per_side,
323 const bool bidirectional = true)
324 {

365 /*
366 * Compute a linearly extrapolated initial guess value (predictor) for the
367 * Newton-Raphson solver at time "t_n+1" from the old solution @param old_vec
368 * computed at time "t_n" and the old old solution @param old_old_vec computed
369 * at time "t_n-1". In addition the @param current_time_increment
370 * dt_n+1 = t_n+1 - t_n and the @param old_time_incrementdt_n = t_n-t_n-1 have
371 * to be provided.
372 *
373 * The predictor is computed as follows
374 *
375 * dt_n+1
376 * q_n+1^(0) = q_n + ------ * (q_n-q_n-1)
377 * dt_n
378 *
379 */
380 template <typename VectorType, typename number>
381 void
382 compute_linear_predictor(const VectorType &old_vec,
383 const VectorType &old_old_vec,
384 VectorType &predictor,
385 const number current_time_increment,
386 const number old_time_increment)
387 {
388 if (std::abs(old_time_increment) < 1e-12)
389 {
390 predictor = old_vec;
391 return;
392 }
393
394 const number fraction = current_time_increment / old_time_increment;
395
396 for (unsigned int c = 0; c < dealii::internal::n_blocks(predictor); ++c)
397 DEAL_II_OPENMP_SIMD_PRAGMA
398 for (unsigned int i = 0; i < dealii::internal::block(predictor, c).locally_owned_size(); ++i)
399 dealii::internal::block(predictor, c).local_element(i) =
400 (fraction + 1.0) * dealii::internal::block(old_vec, c).local_element(i) -
401 fraction * dealii::internal::block(old_old_vec, c).local_element(i);
402 }

431 template <int dim, typename number>
432 dealii::FullMatrix<number>
433 create_dof_interpolation_matrix(const dealii::DoFHandler<dim> &dof_handler_1, // e.g. pressure
434 const dealii::DoFHandler<dim> &dof_handler_2, // e.g. level set
435 const bool do_sort_lexicographically)
436 {
437 dealii::FullMatrix<number> dof_interpolation_matrix(dof_handler_1.get_fe().n_dofs_per_cell(),
438 dof_handler_2.get_fe().n_dofs_per_cell());
439
440 const dealii::FE_Q_iso_Q1<dim> *fe_2 =
441 dynamic_cast<const dealii::FE_Q_iso_Q1<dim> *>(&dof_handler_2.get_fe());
442
443 AssertThrow(fe_2,
444 dealii::ExcMessage(
445 "dof_handler_2 must contain finite elements of type FE_Q_iso_Q1."));
446
447 const std::vector<unsigned int> lexicographic_ls = fe_2->get_poly_space_numbering_inverse();
448
449
450 //@todo: get rid of base_element
451 if (const dealii::FE_Q<dim> *fe_1 =
452 dynamic_cast<const dealii::FE_Q<dim> *>(&dof_handler_1.get_fe().base_element(0)))
453 {
454 const std::vector<unsigned int> lexicographic_p =
455 fe_1->get_poly_space_numbering_inverse();
456 for (unsigned int j = 0; j < fe_1->dofs_per_cell; ++j)
457 {
458 const dealii::Point<dim> p =
459 fe_1->get_unit_support_points()[do_sort_lexicographically ? lexicographic_p[j] : j];
460 for (unsigned int i = 0; i < fe_2->dofs_per_cell; ++i)
461 dof_interpolation_matrix(j, i) = dof_handler_2.get_fe().shape_value(
462 do_sort_lexicographically ? lexicographic_ls[i] : i, p);
463 }
464 }
465 else if (const dealii::FE_Q_DG0<dim> *fe_1 = dynamic_cast<const dealii::FE_Q_DG0<dim> *>(
466 &dof_handler_1.get_fe().base_element(0)))
467 {
468 const std::vector<unsigned int> lexicographic_p =
469 fe_1->get_poly_space_numbering_inverse();
470
471 // Loop over all support points except the one for the discontinuous
472 // shape function in the middle of the cell (dofs_per_cell - 1).
473 for (unsigned int j = 0; j < fe_1->dofs_per_cell - 1; ++j)
474 {
475 const dealii::Point<dim> p =
476 fe_1->get_unit_support_points()[do_sort_lexicographically ? lexicographic_p[j] : j];
477 for (unsigned int i = 0; i < fe_2->dofs_per_cell; ++i)
478 dof_interpolation_matrix(j, i) = dof_handler_2.get_fe().shape_value(
479 do_sort_lexicographically ? lexicographic_ls[i] : i, p);
480 }
481 }
482 else
483 AssertThrow(false,
484 dealii::ExcMessage(
485 "The operation for the requested pair of DoFHandler "
486 "is not supported. Types must be: dof_handler_1 = FE_Q_iso_Q1;"
487 " dof_handler_2 = FE_Q || FE_Q_DG0."));
488
489 return dof_interpolation_matrix;
490 }

502 template <int dim, typename number>
503 void
504 compute_gradient_at_interpolated_dof_values(
505 dealii::FECellIntegrator<dim, 1, number> &values,
506 dealii::FECellIntegrator<dim, 1, number> &interpolated_values,
507 const dealii::FullMatrix<number> &interpolation_matrix)
508 {
509 // Evaluate the field Φ at the support points of its space j
510 values.evaluate(dealii::EvaluationFlags::values);
511
512 // Loop over the support points of the to be interpolated space i
513 for (unsigned int i = 0; i < interpolated_values.dofs_per_cell; ++i)
514 {
515 dealii::VectorizedArray<number> interpolated_value = 0;
516
517 // Interpolate the field Φ from the support points of the space j
518 // of the original field to the one of the interpolated field i,
519 // using the interpolation matrix P
520 // _
521 // Φ = P · Φ
522 // i ij j
523 for (unsigned int j = 0; j < values.dofs_per_cell; ++j)
524 interpolated_value += interpolation_matrix(i, j) * values.get_dof_value(j);
525
526 // Store the interpolated values at the support points of the pressure space
527 interpolated_values.submit_dof_value(interpolated_value, i);
528 }
529
530 // Evaluate the gradient from the interpolated field
531 // _
532 // ∇ Φ
533 //
534 interpolated_values.evaluate(dealii::EvaluationFlags::gradients);
535 }
