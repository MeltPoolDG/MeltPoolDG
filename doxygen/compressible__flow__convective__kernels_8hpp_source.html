113 inline DEAL_II_ALWAYS_INLINE //
114 auto
115 CompressibleFlowConvectiveKernels<dim, number, is_gas_phase>::calculate_convective_flux(
116 const ConservedVariablesType &conserved_variables) const -> ConservedVariablesGradType
117 {
118 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
119 calculate_velocity<dim, number>(conserved_variables);
120 const dealii::VectorizedArray<number> pressure =
121 EOS::calculate_thermodynamic_pressure<dim, number, is_gas_phase>(conserved_variables,
122 flow_data);
123
124 ConservedVariablesGradType flux;
125 for (unsigned int d = 0; d < dim; ++d)
126 {
127 flux[0][d] = conserved_variables[1 + d];
128 for (unsigned int e = 0; e < dim; ++e)
129 flux[e + 1][d] = conserved_variables[e + 1] * velocity[d];
130 flux[d + 1][d] += pressure;
131 flux[dim + 1][d] = velocity[d] * (conserved_variables[dim + 1] + pressure);
132 }
133 return flux;
134 }

137 inline DEAL_II_ALWAYS_INLINE //
138 auto
139 CompressibleFlowConvectiveKernels<dim, number, is_gas_phase>::
140 calculate_convective_numerical_flux(
141 const ConservedVariablesType &u_m,
142 const ConservedVariablesType &u_p,
143 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
144 -> ConservedVariablesType
145 {
146 const auto velocity_m = calculate_velocity<dim, number>(u_m);
147 const auto velocity_p = calculate_velocity<dim, number>(u_p);
148
149 const auto flux_m = calculate_convective_flux(u_m);
150 const auto flux_p = calculate_convective_flux(u_p);
151
152 const auto sound_speed_p =
153 EOS::calculate_speed_of_sound<dim, number, is_gas_phase>(u_p, flow_data);
154 const auto sound_speed_m =
155 EOS::calculate_speed_of_sound<dim, number, is_gas_phase>(u_m, flow_data);
156
157 switch (flow_data.numerical_flux_type)
158 {
159 case NumericalFluxType::lax_friedrichs_modified: {
160 const auto sound_speed_p2 = sound_speed_p * sound_speed_p;
161 const auto sound_speed_m2 = sound_speed_m * sound_speed_m;
162
163 const auto lambda =
164 0.5 * std::sqrt(std::max(velocity_p.norm_square() + sound_speed_p2,
165 velocity_m.norm_square() + sound_speed_m2));
166
167 return UtilityFunctions::contract_average_tensor_with_vector<dim + 2, dim, number>(
168 flux_m, flux_p, normal) +
169 0.5 * lambda * (u_m - u_p);
170 }
171 case NumericalFluxType::lax_friedrichs_exact: {
172 const auto lambda = std::max(std::abs(velocity_p * normal) + sound_speed_p,
173 std::abs(velocity_m * normal) + sound_speed_m);
174
175 return UtilityFunctions::contract_average_tensor_with_vector<dim + 2, dim, number>(
176 flux_m, flux_p, normal) +
177 0.5 * lambda * (u_m - u_p);
178 }
179 case NumericalFluxType::harten_lax_vanleer: {
180 const auto avg_velocity_normal = 0.5 * ((velocity_m + velocity_p) * normal);
181 const auto avg_c = std::sqrt(std::abs(0.5 * (sound_speed_m + sound_speed_p)));
182 const dealii::VectorizedArray<number> s_pos =
183 std::max(dealii::VectorizedArray<number>(), avg_velocity_normal + avg_c);
184 const dealii::VectorizedArray<number> s_neg =
185 std::min(dealii::VectorizedArray<number>(), avg_velocity_normal - avg_c);
186 const dealii::VectorizedArray<number> inverse_s =
187 dealii::VectorizedArray<number>(1.) / (s_pos - s_neg);
188
189 return inverse_s *
190 ((s_pos *
191 UtilityFunctions::contract_tensor_with_vector<dim + 2, dim, number>(flux_m,
192 normal) -
193 s_neg * UtilityFunctions::contract_tensor_with_vector<dim + 2, dim, number>(
194 flux_p, normal)) -
195 s_pos * s_neg * (u_m - u_p));
196 }
197 default: {
198 Assert(false, dealii::ExcNotImplemented());
199 return {};
200 }
201 }
202 }

205 inline DEAL_II_ALWAYS_INLINE //
206 auto
207 CompressibleFlowConvectiveKernels<dim, number, is_gas_phase>::
208 calculate_jacobian_convective_numerical_flux(
209 const std::pair<ConservedVariablesType, ConservedVariablesType> &w_q,
210 const std::pair<ConservedVariablesType, ConservedVariablesType> &delta_w_q,
211 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
212 -> ConservedVariablesGradType
213 {
214 // For now only the exact and modified Lax-Friedrichs flux are supported
215 AssertThrow(flow_data.numerical_flux_type == NumericalFluxType::lax_friedrichs_exact or
216 flow_data.numerical_flux_type == NumericalFluxType::lax_friedrichs_modified,
217 dealii::ExcMessage(
218 "The chosen convective numerical flux type is not supported within the "
219 "analytic Jacobian computation."));
220
221 // average
222 ConservedVariablesGradType flux_p =
223 calculate_jacobian_convective_flux(w_q.second, delta_w_q.second);
224 ConservedVariablesGradType flux_m =
225 calculate_jacobian_convective_flux(w_q.first, delta_w_q.first);
226
227 ConservedVariablesGradType flux;
228 for (unsigned int i = 0; i < dim + 2; ++i)
229 flux[i] = 0.5 * (flux_p[i] + flux_m[i]);
230
231 flux += calculate_jacobian_convective_numerical_flux_jump_term(w_q, delta_w_q, normal);
232
233 return flux;
234 }

236 template <int dim, typename number, bool is_gas_phase>
237 inline DEAL_II_ALWAYS_INLINE //
238 auto
239 CompressibleFlowConvectiveKernels<dim, number, is_gas_phase>::
240 calculate_jacobian_convective_flux(const ConservedVariablesType &w_q,
241 const ConservedVariablesType &delta_w_q) const
242 -> ConservedVariablesGradType
243 {
244 ConservedVariablesGradType convective_differential_change;
245
246 // precompute values
247 dealii::VectorizedArray<number> rho_inv = 1.0 / w_q[0];
248 dealii::VectorizedArray<number> momentum_norm_squared = 0.0;
249 dealii::VectorizedArray<number> momentum_times_delta_momentum_squared = 0.0;
250 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> momentum;
251 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> delta_momentum;
252 for (unsigned int i = 1; i < dim + 1; ++i)
253 {
254 momentum_norm_squared += w_q[i] * w_q[i];
255 momentum_times_delta_momentum_squared += delta_w_q[i] * w_q[i];
256 momentum[i - 1] = w_q[i];
257 delta_momentum[i - 1] = delta_w_q[i];
258 }
259
260 //** change in density flux **//
261 for (unsigned int dimension = 0; dimension < dim; ++dimension)
262 convective_differential_change[0][dimension] = delta_w_q[dimension + 1];
263
264 //** change in momentum flux **//
265 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> helper;
266 // density direction
267 helper -=
268 rho_inv * rho_inv * delta_w_q[0] *
269 VectorTools::dyadic_product<dim, dim, dealii::VectorizedArray<number>>(w_q.begin_raw() + 1,
270 w_q.begin_raw() + 1);
271 helper += rho_inv * rho_inv * rs_div_c * momentum_norm_squared * delta_w_q[0] * 0.5 *
272 VectorTools::identity<dim, dealii::VectorizedArray<number>>();
273 // momentum direction
274 helper += rho_inv * VectorTools::dyadic_product<dim, dim, dealii::VectorizedArray<number>>(
275 delta_w_q.begin_raw() + 1, w_q.begin_raw() + 1);
276 helper += rho_inv * VectorTools::dyadic_product<dim, dim, dealii::VectorizedArray<number>>(
277 w_q.begin_raw() + 1, delta_w_q.begin_raw() + 1);
278 helper -= rs_div_c * rho_inv * momentum_times_delta_momentum_squared *
279 VectorTools::identity<dim, dealii::VectorizedArray<number>>();
280 // energy direction
281 helper +=
282 rs_div_c * delta_w_q[dim + 1] * VectorTools::identity<dim, dealii::VectorizedArray<number>>();
283
284 for (unsigned int i = 0; i < dim; ++i)
285 {
286 convective_differential_change[i + 1] += helper[i];
287 }
288
289 //** change in energy density flux **//
290 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> helper_energy;
291 // density part
292 helper_energy -= (w_q[dim + 1] + rs_div_c * (w_q[dim + 1] - rho_inv * momentum_norm_squared)) *
293 rho_inv * rho_inv * delta_w_q[0] * momentum;
294 // momentum part
295 helper_energy +=
296 (w_q[dim + 1] + rs_div_c * (w_q[dim + 1] - 0.5 * rho_inv * momentum_norm_squared)) * rho_inv *
297 delta_momentum;
298 helper_energy -=
299 rs_div_c * momentum_times_delta_momentum_squared * rho_inv * rho_inv * momentum;
300 // energy part
301 helper_energy += (1 + rs_div_c) * delta_w_q[dim + 1] * rho_inv * momentum;
302 convective_differential_change[dim + 1] = helper_energy;
303
304 return convective_differential_change;
305 }

308 inline DEAL_II_ALWAYS_INLINE //
309 auto
310 CompressibleFlowConvectiveKernels<dim, number, is_gas_phase>::
311 calculate_jacobian_convective_numerical_flux_jump_term(
312 const std::pair<ConservedVariablesType, ConservedVariablesType> &w_q,
313 const std::pair<ConservedVariablesType, ConservedVariablesType> &delta_w_q,
314 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
315 -> ConservedVariablesGradType
316 {
317 // define aliases
318 const ConservedVariablesType &w_m = w_q.first;
319 const ConservedVariablesType &w_p = w_q.second;
320 const ConservedVariablesType &delta_w_m = delta_w_q.first;
321 const ConservedVariablesType &delta_w_p = delta_w_q.second;
322 // jump
323 std::function<
324 ConservedVariablesType(const ConservedVariablesType &,
325 const ConservedVariablesType &,
326 const ConservedVariablesType &,
327 const ConservedVariablesType &,
328 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &)>
329 compute_lambda_times_jump;
330 ConservedVariablesGradType flux;
331
332 switch (flow_data.numerical_flux_type)
333 {
334 case NumericalFluxType::lax_friedrichs_exact: {
335 compute_lambda_times_jump =
336 [&flow_data =
337 flow_data](const ConservedVariablesType &w_p,
338 const ConservedVariablesType &w_m,
339 const ConservedVariablesType &delta_w_p,
340 const ConservedVariablesType &delta_w_m,
341 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal)
342 -> ConservedVariablesType {
343 const auto velocity_m = calculate_velocity<dim, number>(w_m);
344 const auto velocity_p = calculate_velocity<dim, number>(w_p);
345
346 const auto pressure_m =
347 EOS::calculate_thermodynamic_pressure<dim, number>(w_m, flow_data);
348 const auto pressure_p =
349 EOS::calculate_thermodynamic_pressure<dim, number>(w_p, flow_data);
350
351 const auto lambda =
352 std::max(std::abs(velocity_p * normal) +
353 std::sqrt(flow_data.material.gas.gamma * pressure_p / w_p[0]),
354 std::abs(velocity_m * normal) +
355 std::sqrt(flow_data.material.gas.gamma * pressure_m / w_m[0]));
356 return lambda * (delta_w_m - delta_w_p);
357 };
358 break;
359 }
360 case NumericalFluxType::lax_friedrichs_modified: {
361 compute_lambda_times_jump =
362 [this](const ConservedVariablesType &w_p,
363 const ConservedVariablesType &w_m,
364 const ConservedVariablesType &delta_w_p,
365 const ConservedVariablesType &delta_w_m,
366 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &)
367 -> ConservedVariablesType {
368 const auto velocity_m = calculate_velocity<dim, number>(w_m);
369 const auto velocity_p = calculate_velocity<dim, number>(w_p);
370
371 const auto pressure_m =
372 EOS::calculate_thermodynamic_pressure<dim, number>(w_m, flow_data);
373 const auto pressure_p =
374 EOS::calculate_thermodynamic_pressure<dim, number>(w_p, flow_data);
375
376 const auto lambda =
377 0.5 *
378 std::sqrt(std::max(velocity_p.norm_square() +
379 std::abs(flow_data.material.gas.gamma * pressure_p / w_p[0]),
380 velocity_m.norm_square() +
381 std::abs(flow_data.material.gas.gamma * pressure_m / w_m[0])));
382 return lambda * (delta_w_m - delta_w_p);
383 };
384 break;
385 }
386 default:
387 DEAL_II_ASSERT_UNREACHABLE();
388 }
389
390 // We do not ue the square root of machine precision here as this prevents the Newton solver to
391 // convergence
392 constexpr number epsilon = 1e-3;
393 constexpr number epsilon_inv = 1e3;
394 switch (flow_data.linearization_jump_convective_flux)
395 {
396 case LinearizedConvectiveFluxJumpType::complete_fd: {
397 flux -= VectorTools::dyadic_product(
398 0.5 * epsilon_inv * compute_lambda_times_jump(w_p, w_m, w_p, w_m, normal), normal);
399 flux +=
400 VectorTools::dyadic_product(0.5 * epsilon_inv *
401 compute_lambda_times_jump(w_p + epsilon * delta_w_p,
402 w_m + epsilon * delta_w_m,
403 w_p + epsilon * delta_w_p,
404 w_m + epsilon * delta_w_m,
405 normal),
406 normal);
407 break;
408 }
409 case LinearizedConvectiveFluxJumpType::lambda_fd: {
410 flux += VectorTools::dyadic_product(
411 0.5 * compute_lambda_times_jump(w_p, w_m, delta_w_p, delta_w_m, normal), normal);
412
413 auto helper = compute_lambda_times_jump(
414 w_p + epsilon * delta_w_p, w_m + epsilon * delta_w_m, w_p, w_m, normal);
415 helper -= compute_lambda_times_jump(w_p, w_m, w_p, w_m, normal);
416 flux += VectorTools::dyadic_product(0.5 / epsilon * helper, normal);
417 break;
418 }
419 case LinearizedConvectiveFluxJumpType::analytic: {
420 flux += VectorTools::dyadic_product(
421 0.5 * compute_lambda_times_jump(w_p, w_m, delta_w_p, delta_w_m, normal), normal);
422
423 const auto linearize_speed_of_sound =
424 [this](const ConservedVariablesType &w_q,
425 const ConservedVariablesType &delta_w_q,
426 const dealii::VectorizedArray<number> &c,
427 const dealii::VectorizedArray<number> &) -> dealii::VectorizedArray<number> {
428 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> m_q;
429 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> delta_m_q;
430 for (unsigned int i = 0; i < dim; ++i)
431 {
432 m_q[i] = w_q[i + 1];
433 delta_m_q[i] = delta_w_q[i + 1];
434 }
435 dealii::VectorizedArray<number> rho_inv = 1. / w_q[0];
436 dealii::VectorizedArray<number> lin_c =
437 delta_w_q[0] / (2.0 * c) * flow_data.material.gas.gamma * rho_inv * rho_inv *
438 rs_div_c * (rho_inv * m_q * m_q - w_q[dim + 1]);
439 lin_c -= flow_data.material.gas.gamma / (2.0 * c) * rho_inv * rho_inv * rs_div_c *
440 m_q * delta_m_q;
441 lin_c +=
442 flow_data.material.gas.gamma / (2.0 * c) * rho_inv * rs_div_c * delta_w_q[dim + 1];
443 return lin_c;
444 };
445
446 const auto norm_lin_velocity =
447 [this](const ConservedVariablesType &w_q,
448 const ConservedVariablesType &delta_w_q,
449 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal)
450 -> dealii::VectorizedArray<number> {
451 auto rho_inv = 1. / w_q[dim + 1];
452 auto u = calculate_velocity<dim, number>(w_q);
453 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> delta_m_q;
454 for (unsigned int i = 0; i < dim; ++i)
455 delta_m_q[i] = delta_w_q[i + 1];
456 auto lin_velocity = delta_m_q * rho_inv - u * rho_inv * delta_w_q[0];
457 return lin_velocity * normal;
458 };
459
460 const auto signum =
461 [](
462 const dealii::VectorizedArray<number> &a) -> const dealii::VectorizedArray<number> {
463 return dealii::compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
464 a, dealii::VectorizedArray<number>(0.0), 1.0, -1.0);
465 };
466
467 const auto velocity_m = calculate_velocity<dim, number>(w_m);
468 const auto velocity_p = calculate_velocity<dim, number>(w_p);
469
470 const auto pressure_m =
471 EOS::calculate_thermodynamic_pressure<dim, number>(w_m, flow_data);
472 const auto pressure_p =
473 EOS::calculate_thermodynamic_pressure<dim, number>(w_p, flow_data);
474
475 const auto speed_of_sound_m =
476 std::sqrt(flow_data.material.gas.gamma * pressure_m / w_m[0]);
477 const auto speed_of_sound_p =
478 std::sqrt(flow_data.material.gas.gamma * pressure_p / w_p[0]);
479
480 const auto lin_lambda_p =
481 signum(velocity_p * normal) * norm_lin_velocity(w_p, delta_w_p, normal) +
482 linearize_speed_of_sound(w_p, delta_w_p, speed_of_sound_p, pressure_p);
483 const auto lin_lambda_m =
484 signum(velocity_m * normal) * norm_lin_velocity(w_m, delta_w_m, normal) +
485 linearize_speed_of_sound(w_m, delta_w_m, speed_of_sound_m, pressure_m);
486
487 const auto lin_lambda =
488 dealii::compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
489 velocity_p.norm() + speed_of_sound_p,
490 velocity_m.norm() + speed_of_sound_m,
491 lin_lambda_p,
492 lin_lambda_m);
493
494 for (unsigned int i = 0; i < dim + 2; ++i)
495 flux[i] += 0.5 * lin_lambda * (w_m[i] - w_p[i]) * normal;
496
497 break;
498 }
499 default: {
500 AssertThrow(
501 false,
502 dealii::ExcMessage(
503 "The provided linearization scheme of the jump operator term in the convective"
504 " numerical flux is not supported."));
505 }
506 }
507 return flux;
508 }
