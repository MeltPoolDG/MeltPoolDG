74 number
75 value(const dealii::Point<dim, number> &p, const unsigned int component) const final
76 {
77 std::array<unsigned int, dim + 2> gas_components;
78 std::array<unsigned int, dim + 2> liquid_components;
79
80 for (unsigned int i = 0; i < dim + 2; ++i)
81 {
82 if (gas_phase_is_first)
83 {
84 gas_components[i] = i;
85 liquid_components[i] = i + dim + 2;
86 }
87 else
88 {
89 gas_components[i] = i + dim + 2;
90 liquid_components[i] = i;
91 }
92 }
93
94 // gas phase
95 if (component == gas_components[0])
96 return parsing_function_gas->value(p, 0);
97 else if (component == gas_components[1])
98 return parsing_function_gas->value(p, 1);
99 else if (component == gas_components[2])
100 return parsing_function_gas->value(p, 2);
101
102 // liquid phase
103 else if (component == liquid_components[0])
104 return parsing_function_liquid->value(p, 0);
105 else if (component == liquid_components[1])
106 return parsing_function_liquid->value(p, 1);
107 else if (component == liquid_components[2])
108 return parsing_function_liquid->value(p, 2);
109 else
110 return 0.;
111 }

150 void
151 create_spatial_discretization() override
152 {
153 // TODO: no distributed triangulation possible for dim=1
154 this->triangulation =
155 std::make_shared<dealii::parallel::shared::Triangulation<dim>>(this->mpi_communicator);
156
157 dealii::Point<dim, number> lower_left;
158 lower_left[0] = -5.e-4;
159 for (unsigned int d = 1; d < dim; ++d)
160 lower_left[d] = 0.;
161
162 dealii::Point<dim, number> upper_right;
163 upper_right[0] = 5.e-4;
164 for (unsigned int d = 1; d < dim; ++d)
165 upper_right[d] = 0.;
166
167 std::vector<unsigned int> subdivisions(dim, 1);
168 subdivisions[0] = 20;
169 for (unsigned int d = 1; d < dim; ++d)
170 subdivisions[d] = 1;
171
172 dealii::GridGenerator::subdivided_hyper_rectangle(
173 *this->triangulation, subdivisions, lower_left, upper_right, true);
174
175 this->triangulation->refine_global(this->parameters.base.global_refinements);

181 void
182 set_boundary_conditions() override
183 {
184 auto inflow_outflow_solution =
185 std::make_shared<InitialField<dim, number>>(initial_conditions.gas,
186 initial_conditions.liquid);
187
188 // face numbering according to the deal.II colorize flag
189 const auto [lower_bc, upper_bc, left_bc, right_bc, front_bc, back_bc] =
190 get_colorized_rectangle_boundary_ids<dim>();
191
192 // left boundary
193 this->attach_boundary_condition({lower_bc, inflow_outflow_solution},
194 left_boundary_condition,
195 "compressible_multiphase_flow");
196 // right boundary
197 this->attach_boundary_condition({upper_bc, inflow_outflow_solution},
198 right_boundary_condition,
199 "compressible_multiphase_flow");

205 void
206 set_field_conditions() override
207 {
208 // The solution vector is ordered, such that the liquid phase is the first phase and the gas
209 // phase is the second phase.
210 auto initial_condition =
211 std::make_shared<InitialField<dim, number>>(initial_conditions.gas,
212 initial_conditions.liquid,
213 false /*gas_phase_is_first*/);
214 this->attach_initial_condition(initial_condition, "compressible_multiphase_flow");
215
216 // set level-set function
217 dealii::Point<dim, number> p;
218 // avoid phase interface colliding with element face (bug in dealii has to be fixed)
219 p[0] = 0.0135468746e-4;
220
221 dealii::Tensor<1, dim, number> normal;
222 normal[0] = -1.;
223
224 const auto level_set =
225 std::make_shared<dealii::Functions::SignedDistance::Plane<dim>>(p, normal);
226 this->attach_field_function(level_set, "level_set", "compressible_multiphase_flow");
227 }
