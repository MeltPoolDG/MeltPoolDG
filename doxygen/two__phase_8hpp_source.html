112 explicit ConservedVariablesInitialField(std::string ic_gas_phase,
113 std::string ic_liquid_phase,
114 const bool gas_phase_is_first = true)
115 : dealii::Function<dim, number>(2 * (dim + 2))
116 , gas_phase_is_first(gas_phase_is_first)

126 (dim == 2 ? std::string("x,y") : std::string("x")),

141 number
142 value(const dealii::Point<dim, number> &p, const unsigned int component) const final
143 {
144 std::array<unsigned int, dim + 2> gas_components;
145 std::array<unsigned int, dim + 2> liquid_components;
146
147 for (unsigned int i = 0; i < dim + 2; ++i)
148 {
149 if (gas_phase_is_first)
150 {
151 gas_components[i] = i;
152 liquid_components[i] = i + dim + 2;
153 }
154 else
155 {
156 gas_components[i] = i + dim + 2;
157 liquid_components[i] = i;
158 }
159 }
160
161 // gas phase
162 if (component == gas_components[0])
163 return parsing_function_gas->value(p, 0);
164 else if (component == gas_components[1])
165 return parsing_function_gas->value(p, 1);
166 else if (component == gas_components[2])
167 return parsing_function_gas->value(p, 2);
168
169 // liquid phase
170 else if (component == liquid_components[0])
171 return parsing_function_liquid->value(p, 0);
172 else if (component == liquid_components[1])
173 return parsing_function_liquid->value(p, 1);
174 else if (component == liquid_components[2])
175 return parsing_function_liquid->value(p, 2);
176 else
177 return 0.;
178 }
179
180 private:
182 bool gas_phase_is_first;

188 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_gas =

190
192 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_liquid =
193 std::make_unique<dealii::FunctionParser<dim>>(dim + 2);

217 void
218 create_spatial_discretization() override
219 {
220 // TODO: no distributed triangulation possible for dim=1
221 this->triangulation =
222 std::make_shared<dealii::parallel::shared::Triangulation<dim>>(this->mpi_communicator);
223
224 dealii::Point<dim, number> lower_left;
225 lower_left[0] = -5.e-4;
226 for (unsigned int d = 1; d < dim; ++d)
227 lower_left[d] = 0.;
228
229 dealii::Point<dim, number> upper_right;
230 upper_right[0] = 5.e-4;
231 for (unsigned int d = 1; d < dim; ++d)
232 upper_right[d] = 0.;
233
234 std::vector<unsigned int> subdivisions(dim, 1);
235 subdivisions[0] = 20;
236 for (unsigned int d = 1; d < dim; ++d)
237 subdivisions[d] = 1;
238
239 dealii::GridGenerator::subdivided_hyper_rectangle(
240 *this->triangulation, subdivisions, lower_left, upper_right, true);
241
242 this->triangulation->refine_global(this->parameters.base.global_refinements);

248 void
249 set_boundary_conditions() override
250 {
251 // face numbering according to the deal.II colorize flag
252 const auto [lower_bc, upper_bc, left_bc, right_bc, front_bc, back_bc] =
253 get_colorized_rectangle_boundary_ids<dim>();
254
255 // left boundary
256 if (left_boundary_condition == "outflow_fixed_energy" or
257 left_boundary_condition == "outflow_fixed_pressure")
258 {
259 auto boundary_condition =
260 std::make_shared<SingleVariableTwoPhaseField<dim, number>>(initial_conditions.gas,
261 initial_conditions.liquid);
262 this->attach_boundary_condition({lower_bc, boundary_condition},
263 left_boundary_condition,
264 "compressible_multiphase_flow");
265 }
266 else
267 {
268 auto boundary_condition = std::make_shared<ConservedVariablesInitialField<dim, number>>(
269 initial_conditions.gas, initial_conditions.liquid);
270 this->attach_boundary_condition({lower_bc, boundary_condition},
271 left_boundary_condition,
272 "compressible_multiphase_flow");
273 }
274
275 // right boundary
276 if (right_boundary_condition == "outflow_fixed_energy" or
277 right_boundary_condition == "outflow_fixed_pressure")
278 {
279 auto boundary_condition =
280 std::make_shared<SingleVariableTwoPhaseField<dim, number>>(initial_conditions.gas,
281 initial_conditions.liquid);
282 this->attach_boundary_condition({upper_bc, boundary_condition},
283 right_boundary_condition,
284 "compressible_multiphase_flow");

299 void
300 set_field_conditions() override
301 {
302 // The solution vector is ordered, such that the liquid phase is the first phase and the gas
303 // phase is the second phase.
304 auto initial_condition =
305 std::make_shared<ConservedVariablesInitialField<dim, number>>(initial_conditions.gas,
306 initial_conditions.liquid,
307 false /*gas_phase_is_first*/);
308 this->attach_initial_condition(initial_condition, "compressible_multiphase_flow");
309
310 // set level-set function
311 dealii::Point<dim, number> p;
312 // avoid phase interface colliding with element face (bug in dealii has to be fixed)
313 p[0] = 0.0135468746e-4;
314
315 dealii::Tensor<1, dim, number> normal;
316 normal[0] = -1.;
317
318 const auto level_set =
319 std::make_shared<dealii::Functions::SignedDistance::Plane<dim>>(p, normal);
320 this->attach_field_function(level_set, "level_set", "compressible_multiphase_flow");
321 }
