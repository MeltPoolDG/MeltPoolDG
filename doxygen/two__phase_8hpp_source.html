28 template <int dim, typename number>
29 class InitialField : public dealii::Function<dim, number>
30 {
31 public:
32 explicit InitialField(std::string ic_gas_phase,
33 std::string ic_liquid_phase,
34 const bool gas_phase_is_first = true)
35 : dealii::Function<dim, number>(2 * (dim + 2))
36 , gas_phase_is_first(gas_phase_is_first)
37 {
38 parsing_function_gas->initialize(dim == 3 ?
39 std::string("x,y,z") :
40 (dim == 2 ? std::string("x,y") : std::string("x")),
41 ic_gas_phase,
42 constants,
43 false);
44 parsing_function_liquid->initialize(dim == 3 ?
45 std::string("x,y,z") :
46 (dim == 2 ? std::string("x,y") : std::string("x")),
47 ic_liquid_phase,
48 constants,
49 false);
50
51 // Currently, only 1D simulations are possible
52 Assert(dim == 1, dealii::ExcNotImplemented());
53 }

55 number
56 value(const dealii::Point<dim, number> &p, const unsigned int component) const final
57 {
58 std::array<unsigned int, dim + 2> gas_components;
59 std::array<unsigned int, dim + 2> liquid_components;
60
61 for (unsigned int i = 0; i < dim + 2; ++i)
62 {
63 if (gas_phase_is_first)
64 {
65 gas_components[i] = i;
66 liquid_components[i] = i + dim + 2;
67 }
68 else
69 {
70 gas_components[i] = i + dim + 2;
71 liquid_components[i] = i;
72 }
73 }
74
75 // gas phase
76 if (component == gas_components[0])
77 return parsing_function_gas->value(p, 0);
78 else if (component == gas_components[1])
79 return parsing_function_gas->value(p, 1);
80 else if (component == gas_components[2])
81 return parsing_function_gas->value(p, 2);
82
83 // liquid phase
84 else if (component == liquid_components[0])
85 return parsing_function_liquid->value(p, 0);
86 else if (component == liquid_components[1])
87 return parsing_function_liquid->value(p, 1);
88 else if (component == liquid_components[2])
89 return parsing_function_liquid->value(p, 2);
90 else
91 return 0.;
92 }
93
94 private:
95 bool gas_phase_is_first;
96 std::map<std::string, number> constants;
97 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_gas =
98 std::make_unique<dealii::FunctionParser<dim>>(dim + 2);
99 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_liquid =
100 std::make_unique<dealii::FunctionParser<dim>>(dim + 2);
101 };

111 void
112 create_spatial_discretization() override
113 {
114 // TODO: no distributed triangulation possible for dim=1
115 this->triangulation =
116 std::make_shared<dealii::parallel::shared::Triangulation<dim>>(this->mpi_communicator);
117
118 dealii::Point<dim, number> lower_left;
119 lower_left[0] = -5.e-4;
120 for (unsigned int d = 1; d < dim; ++d)
121 lower_left[d] = 0.;
122
123 dealii::Point<dim, number> upper_right;
124 upper_right[0] = 5.e-4;
125 for (unsigned int d = 1; d < dim; ++d)
126 upper_right[d] = 0.;
127
128 std::vector<unsigned int> subdivisions(dim, 1);
129 subdivisions[0] = 20;
130 for (unsigned int d = 1; d < dim; ++d)
131 subdivisions[d] = 1;
132
133 dealii::GridGenerator::subdivided_hyper_rectangle(
134 *this->triangulation, subdivisions, lower_left, upper_right, true);
135
136 this->triangulation->refine_global(this->parameters.base.global_refinements);
137 }

139 void
140 set_boundary_conditions() override
141 {
142 auto inflow_outflow_solution =
143 std::make_shared<InitialField<dim, number>>(initial_conditions.gas,
144 initial_conditions.liquid);
145
146 // face numbering according to the deal.II colorize flag
147 const auto [lower_bc, upper_bc, left_bc, right_bc, front_bc, back_bc] =
148 get_colorized_rectangle_boundary_ids<dim>();
149
150 // left boundary
151 this->attach_boundary_condition({lower_bc, inflow_outflow_solution},
152 left_boundary_condition,
153 "compressible_multiphase_flow");
154 // right boundary
155 this->attach_boundary_condition({upper_bc, inflow_outflow_solution},
156 right_boundary_condition,
157 "compressible_multiphase_flow");
158 }

160 void
161 set_field_conditions() override
162 {
163 // The solution vector is ordered, such that the liquid phase is the first phase and the gas
164 // phase is the second phase.
165 auto initial_condition =
166 std::make_shared<InitialField<dim, number>>(initial_conditions.gas,
167 initial_conditions.liquid,
168 false /*gas_phase_is_first*/);
169 this->attach_initial_condition(initial_condition, "compressible_multiphase_flow");
170
171 // set level-set function
172 dealii::Point<dim, number> p;
173 // avoid phase interface colliding with element face (bug in dealii has to be fixed)
174 p[0] = 0.0135468746e-4;
175
176 dealii::Tensor<1, dim, number> normal;
177 normal[0] = -1.;
178
179 const auto level_set =
180 std::make_shared<dealii::Functions::SignedDistance::Plane<dim>>(p, normal);
181 this->attach_field_function(level_set, "level_set", "compressible_multiphase_flow");
182 }
