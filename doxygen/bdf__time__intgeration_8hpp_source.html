180 // Perform the initial steps with a bdf scheme of lower order, i.e. 1st step: BDF1, 2nd
181 // step BDF2 and so on until the desired BDF scheme is reached
182 static unsigned int step_count = 0;
183 if (step_count != required_solution_history_size() - 1)
184 {
185 ++step_count;
186 set_up_bdf_parameters(step_count);
187 }
188
189 Assert(solution_history.size() > step_count,
190 dealii::ExcMessage(
191 "The size of the solution history object does not fit the requirements of the "
192 "chosen time integration scheme."));
193
194 DEAL_II_OPENMP_SIMD_PRAGMA
195 for (unsigned int i = 0; i < summed_old_solution.locally_owned_size(); ++i)
196 {
197 summed_old_solution.local_element(i) = 0.;
198 for (unsigned int j = step_count; j > 0; --j)
199 {
200 summed_old_solution.local_element(i) +=
201 bdf_prefactors_old_solutions[j - 1] *
202 solution_history.get_solution(j).local_element(i);
203 }
204 }
