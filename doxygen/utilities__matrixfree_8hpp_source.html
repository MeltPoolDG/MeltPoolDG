47 // The dof index that is used for the DoF Vector from the matrix-vector
48 // product must be switched to the one without Dirichlet boundary
49 // conditions, such that inhomogeneities are not zeroed out during
50 // read_dof_values().
51 operator_base.reset_dof_index(dof_no_bc_idx);
52
53 DoFVectorType temp_rhs;
54 scratch_data.initialize_dof_vector(temp_rhs, dof_no_bc_idx);
55
56 // This copy is necessary since we observed problems with incompatible
57 // communication pattern when using periodic BC.
58 SrcRhsVectorType temp_src;
59 scratch_data.initialize_dof_vector(temp_src, dof_no_bc_idx);
60 temp_src = src;
61
62 DoFVectorType bc_values;
63 scratch_data.initialize_dof_vector(bc_values, dof_no_bc_idx);
64
65 // set inhomogeneity
66 if (additional_inhomogeneous_constraints)
67 {
68 const auto &bc = *additional_inhomogeneous_constraints;
69 for (unsigned int i = 0; i < bc.first.size(); ++i)
70 bc_values.local_element(bc.first[i]) = bc.second[i];
71 }
72
73 scratch_data.get_constraint(dof_idx).distribute(bc_values);
74
75 // perform matrix-vector multiplication (with unconstrained system and constrained set in
76 // Vector)
77 operator_base.vmult(temp_rhs, bc_values);
78
79 // Modify right-hand side
80 temp_rhs *= -1.0;
81 operator_base.create_rhs(temp_rhs, temp_src);
82 /*
83 * Zero-out constrained values
84 */
85 if (zero_out)
86 rhs = temp_rhs;
87 else
88 rhs += temp_rhs;
89
90 // zero-out values of additional inhomogeneity
91 if (additional_inhomogeneous_constraints)
92 {
93 const auto &bc = *additional_inhomogeneous_constraints;
94 for (const auto &i : bc.first)
95 {
96 rhs.local_element(i) = 0;
97 }
98 }
99
100 scratch_data.get_constraint(dof_idx).set_zero(rhs);
101
102 operator_base.reset_dof_index(dof_idx);
103 }
