19 template <int dim, typename number>
20 class IsentropicVortexExactSolution : public dealii::Function<dim, number>
21 {
22 public:
23 IsentropicVortexExactSolution(const number time, const number gamma)
24 : dealii::Function<dim, number>(dim + 2, time)
25 , gamma(gamma)
26 {}
27
28 number
29 value(const dealii::Point<dim, number> &x, const unsigned int component) const override
30 {
31 AssertDimension(dim, 2);
32
33 constexpr number beta = 5;
34 const number current_time = this->get_time();
35 dealii::Point<dim, number> x0;
36 x0[0] = 5.;
37 const number radius_sqr =
38 (x - x0).norm_square() - 2. * (x[0] - x0[0]) * current_time + current_time * current_time;
39 const number factor = beta / (dealii::numbers::PI * 2) * std::exp(1. - radius_sqr);
40 const number density_log =
41 std::log2(std::abs(1. - (gamma - 1.) / gamma * 0.25 * factor * factor));
42 const number density = std::exp2(density_log * (1. / (gamma - 1.)));
43 const number u = 1. - factor * (x[1] - x0[1]);
44 const number v = factor * (x[0] - current_time - x0[0]);
45
46 if (component == 0)
47 return density;
48 else if (component == 1)
49 return density * u;
50 else if (component == 2)
51 return density * v;
52 else
53 {
54 const number pressure = std::exp2(density_log * (gamma / (gamma - 1.)));
55 return pressure / (gamma - 1.) + 0.5 * (density * u * u + density * v * v);
56 }
57 }
58
59 private:
60 const number gamma;
61 };
