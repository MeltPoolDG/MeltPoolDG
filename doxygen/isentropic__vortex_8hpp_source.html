19 template <int dim>
20 class IsentropicVortexExactSolution : public Function<dim>
21 {
22 public:
23 IsentropicVortexExactSolution(const double time, const double gamma)
24 : Function<dim>(dim + 2, time)
25 , gamma(gamma)
26 {}
27
28 double
29 value(const Point<dim> &x, const unsigned int component) const override
30 {
31 AssertDimension(dim, 2);
32
33 constexpr double beta = 5;
34 const double current_time = this->get_time();
35 Point<dim> x0;
36 x0[0] = 5.;
37 const double radius_sqr =
38 (x - x0).norm_square() - 2. * (x[0] - x0[0]) * current_time + current_time * current_time;
39 const double factor = beta / (numbers::PI * 2) * std::exp(1. - radius_sqr);
40 const double density_log =
41 std::log2(std::abs(1. - (gamma - 1.) / gamma * 0.25 * factor * factor));
42 const double density = std::exp2(density_log * (1. / (gamma - 1.)));
43 const double u = 1. - factor * (x[1] - x0[1]);
44 const double v = factor * (x[0] - current_time - x0[0]);
45
46 if (component == 0)
47 return density;
48 else if (component == 1)
49 return density * u;
50 else if (component == 2)
51 return density * v;
52 else
53 {
54 const double pressure = std::exp2(density_log * (gamma / (gamma - 1.)));
55 return pressure / (gamma - 1.) + 0.5 * (density * u * u + density * v * v);
56 }
57 }
58
59 private:
60 const double gamma;
61 };
