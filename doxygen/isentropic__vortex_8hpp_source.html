42 number
43 value(const dealii::Point<dim, number> &x, const unsigned int component) const override
44 {
45 AssertDimension(dim, 2);
46
47 constexpr number beta = 5;
48 const number current_time = this->get_time();
49 dealii::Point<dim, number> x0;
50 x0[0] = 5.;
51 const number radius_sqr =
52 (x - x0).norm_square() - 2. * (x[0] - x0[0]) * current_time + current_time * current_time;
53 const number factor = beta / (dealii::numbers::PI * 2) * std::exp(1. - radius_sqr);
54 const number density_log =
55 std::log2(std::abs(1. - (gamma - 1.) / gamma * 0.25 * factor * factor));
56 const number density = std::exp2(density_log * (1. / (gamma - 1.)));
57 const number u = 1. - factor * (x[1] - x0[1]);
58 const number v = factor * (x[0] - current_time - x0[0]);
59
60 if (component == 0)
61 return density;
62 else if (component == 1)
63 return density * u;
64 else if (component == 2)
65 return density * v;
66 else
67 {
68 const number pressure = std::exp2(density_log * (gamma / (gamma - 1.)));
69 return pressure / (gamma - 1.) + 0.5 * (density * u * u + density * v * v);
70 }
71 }
