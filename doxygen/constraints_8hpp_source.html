19 template <int dim, int spacedim, typename number>
20 void
21 check_constraints([[maybe_unused]] const DoFHandler<dim, spacedim> &dof_handler,
22 [[maybe_unused]] const AffineConstraints<number> &constraints)
23 {
24#ifndef DEBUG
25 return;
26#endif
27
28 IndexSet locally_active_dofs;
29 DoFTools::extract_locally_active_dofs(dof_handler, locally_active_dofs);
30
31 Assert(constraints.is_consistent_in_parallel(
32 dealii::Utilities::MPI::all_gather(dof_handler.get_communicator(),
33 dof_handler.locally_owned_dofs()),
34 locally_active_dofs,
35 dof_handler.get_communicator()),
36 ExcInternalError());
37 }

52 template <int dim>
53 void
54 fill_DBC(ScratchData<dim> &scratch_data,
55 const std::map<dealii::types::boundary_id, std::shared_ptr<Function<dim>>> &bc_data,
56 const unsigned int dof_idx,
57 const bool set_inhomogeneities = true,
58 const bool close = true)
59 {
60 if (!bc_data.empty())
61 {
62 for (const auto &bc : bc_data)
63 {
64 if (set_inhomogeneities)
65 dealii::VectorTools::interpolate_boundary_values<dim>(
66 scratch_data.get_mapping(),
67 scratch_data.get_dof_handler(dof_idx),
68 bc.first,
69 *bc.second,
70 scratch_data.get_constraint(dof_idx));
71 else
72 dealii::DoFTools::make_zero_boundary_constraints<dim>(
73 scratch_data.get_dof_handler(dof_idx),
74 bc.first,
75 scratch_data.get_constraint(dof_idx));
76 }
77 }
78
79 if (close)
80 {
81 scratch_data.get_constraint(dof_idx).close();
82 check_constraints(scratch_data.get_dof_handler(dof_idx),
83 scratch_data.get_constraint(dof_idx));
84 }
85 }
