187 template <int dim, typename number>
188 inline DEAL_II_ALWAYS_INLINE //
189 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
190 CompressibleFlowCalculators<dim, number>::calculate_velocity(
191 const ConservedVariablesType &conserved_variables)
192 {
193 const dealii::VectorizedArray<number> inverse_density =
194 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
195
196 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
197 for (unsigned int d = 0; d < dim; ++d)
198 velocity[d] = conserved_variables[1 + d] * inverse_density;
199
200 return velocity;
201 }

204 template <int dim, typename number>
205 inline DEAL_II_ALWAYS_INLINE //
206 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
207 CompressibleFlowCalculators<dim, number>::calculate_grad_velocity(
208 const ConservedVariablesType &conserved_variables,
209 const ConservedVariablesGradType &grad_conserved_variables)
210 {
211 const dealii::VectorizedArray<number> inverse_density =
212 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
213 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
214 calculate_velocity(conserved_variables);
215
216 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
217 for (unsigned int d = 0; d < dim; ++d)
218 grad_rho[d] = grad_conserved_variables[0][d];
219
220 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
221 for (unsigned int d = 0; d < dim; ++d)
222 for (unsigned int e = 0; e < dim; ++e)
223 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
224
225 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
226 for (unsigned int d = 0; d < dim; ++d)
227 for (unsigned int e = 0; e < dim; ++e)
228 grad_velocity[d][e] =
229 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
230
231 return grad_velocity;
232 }

236 inline DEAL_II_ALWAYS_INLINE //
237 dealii::VectorizedArray<number>
238 CompressibleFlowCalculators<dim, number>::calculate_pressure(
239 const ConservedVariablesType &conserved_variables,
240 number gamma)
241 {
242 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
243 calculate_velocity(conserved_variables);
244 return (gamma - 1.) * (conserved_variables[dim + 1] -
245 conserved_variables[0] * 0.5 * scalar_product(velocity, velocity));
246 }

248 template <int dim, typename number>
249 inline DEAL_II_ALWAYS_INLINE //
250 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
251 CompressibleFlowCalculators<dim, number>::calculate_grad_T(
252 const ConservedVariablesType &conserved_variables,
253 const ConservedVariablesGradType &grad_conserved_variables) const
254 {
255 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> u =
256 calculate_velocity(conserved_variables);
257 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_u =
258 calculate_grad_velocity(conserved_variables, grad_conserved_variables);
259 const dealii::VectorizedArray<number> rho = conserved_variables[0];
260 const dealii::VectorizedArray<number> inv_rho = dealii::VectorizedArray<number>(1.) / rho;
261 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho =
262 grad_conserved_variables[0];
263 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_E =
264 inv_rho *
265 (grad_conserved_variables[dim + 1] - inv_rho * conserved_variables[dim + 1] * grad_rho);
266 return (comp_flow_data.gamma - 1.0) / comp_flow_data.specific_gas_constant *
267 (grad_E - grad_u * u);
268 }

273 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesType
274 CompressibleFlowCalculators<dim, number>::contract_tensor_with_normal(
275 const ConservedVariablesGradType &grad,
276 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
277 {
278 ConservedVariablesType result;
279
280 result[0] = grad[0] * normal;
281
282 for (unsigned int e = 0; e < dim; ++e)

291 template <int dim, typename number>
292 DEAL_II_ALWAYS_INLINE //
293 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesType
294 CompressibleFlowCalculators<dim, number>::contract_average_tensor_with_normal(
295 const ConservedVariablesGradType &grad_m,
296 const ConservedVariablesGradType &grad_p,
297 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
298 {
299 ConservedVariablesType result;
300
301 result[0] = (grad_m[0] + grad_p[0]) * normal;
302
303 for (unsigned int e = 0; e < dim; ++e)
304 result[e + 1] = (grad_m[e + 1] + grad_p[e + 1]) * normal;
305
306 result[dim + 1] = (grad_m[dim + 1] + grad_p[dim + 1]) * normal;
307
308 return 0.5 * result;
309 }

312 template <int dim, typename number>
313 inline DEAL_II_ALWAYS_INLINE //
314 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesGradType
315 CompressibleFlowCalculators<dim, number>::calculate_convective_flux(
316 const ConservedVariablesType &conserved_variables) const
317 {
318 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
319 calculate_velocity(conserved_variables);
320 const dealii::VectorizedArray<number> pressure =
321 calculate_pressure(conserved_variables, comp_flow_data.gamma);
322
323 ConservedVariablesGradType flux;
324 for (unsigned int d = 0; d < dim; ++d)
325 {
326 flux[0][d] = conserved_variables[1 + d];
327 for (unsigned int e = 0; e < dim; ++e)
328 flux[e + 1][d] = conserved_variables[e + 1] * velocity[d];
329 flux[d + 1][d] += pressure;
330 flux[dim + 1][d] = velocity[d] * (conserved_variables[dim + 1] + pressure);
331 }
332 return flux;
333 }

335 template <int dim, typename number>
336 inline DEAL_II_ALWAYS_INLINE //
337 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesType
338 CompressibleFlowCalculators<dim, number>::calculate_convective_numerical_flux(
339 const ConservedVariablesType &u_m,
340 const ConservedVariablesType &u_p,
341 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
342 {
343 const auto velocity_m = calculate_velocity(u_m);
344 const auto velocity_p = calculate_velocity(u_p);
345
346 const auto pressure_m = calculate_pressure(u_m, comp_flow_data.gamma);
347 const auto pressure_p = calculate_pressure(u_p, comp_flow_data.gamma);
348
349 const auto flux_m = calculate_convective_flux(u_m);
350 const auto flux_p = calculate_convective_flux(u_p);
351
352 if (comp_flow_data.numerical_flux_type == "lax_friedrichs_modified")
353 {
354 const auto lambda =
355 0.5 * std::sqrt(std::max(velocity_p.norm_square() +
356 std::abs(comp_flow_data.gamma * pressure_p * (1. / u_p[0])),
357 velocity_m.norm_square() +
358 std::abs(comp_flow_data.gamma * pressure_m * (1. / u_m[0]))));
359
360 return contract_average_tensor_with_normal(flux_m, flux_p, normal) +
361 0.5 * lambda * (u_m - u_p);
362 }
363 else if (comp_flow_data.numerical_flux_type == "lax_friedrichs_exact")
364 {
365 const auto lambda =
366 std::max(std::abs(velocity_p * normal) +
367 std::sqrt(comp_flow_data.gamma * pressure_p * (1. / u_p[0])),
368 std::abs(velocity_m * normal) +
369 std::sqrt(comp_flow_data.gamma * pressure_m * (1. / u_m[0])));
370
371 return contract_average_tensor_with_normal(flux_m, flux_p, normal) +
372 0.5 * lambda * (u_m - u_p);
373 }
374 else if (comp_flow_data.numerical_flux_type == "harten_lax_vanleer")
375 {
376 const auto avg_velocity_normal = 0.5 * ((velocity_m + velocity_p) * normal);
377 const auto avg_c = std::sqrt(std::abs(
378 0.5 * comp_flow_data.gamma * (pressure_p * (1. / u_p[0]) + pressure_m * (1. / u_m[0]))));
379 const dealii::VectorizedArray<number> s_pos =
380 std::max(VectorizedArray<number>(), avg_velocity_normal + avg_c);
381 const dealii::VectorizedArray<number> s_neg =
382 std::min(VectorizedArray<number>(), avg_velocity_normal - avg_c);
383 const dealii::VectorizedArray<number> inverse_s =
384 dealii::VectorizedArray<number>(1.) / (s_pos - s_neg);
385
386 return inverse_s * ((s_pos * contract_tensor_with_normal(flux_m, normal) -
387 s_neg * contract_tensor_with_normal(flux_p, normal)) -
388 s_pos * s_neg * (u_m - u_p));
389 }
390 else
391 {
392 Assert(false, ExcNotImplemented());
393 return {};
394 }
395 }

397 template <int dim, typename number>
398 inline DEAL_II_ALWAYS_INLINE //
399 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
400 CompressibleFlowCalculators<dim, number>::calculate_viscous_stress_tensor(
401 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> &grad_u) const
402 {
403 const dealii::VectorizedArray<number> div_u = (2. / 3.) * trace(grad_u);
404
405 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> out;
406 for (unsigned int d = 0; d < dim; ++d)
407 {
408 for (unsigned int e = 0; e < dim; ++e)
409 out[d][e] = comp_flow_data.dynamic_viscosity * (grad_u[d][e] + grad_u[e][d]);
410 out[d][d] -= comp_flow_data.dynamic_viscosity * div_u;
411 }
412
413 return out;
414 }

416 template <int dim, typename number>
417 inline DEAL_II_ALWAYS_INLINE //
418 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesGradType
419 CompressibleFlowCalculators<dim, number>::calculate_viscous_flux(
420 const ConservedVariablesType &conserved_variables,
421 const ConservedVariablesGradType &grad_conserved_variables) const
422 {
423 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
424 calculate_velocity(conserved_variables);
425
426 const auto grad_u = calculate_grad_velocity(conserved_variables, grad_conserved_variables);
427
428 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> viscous_stress =
429 calculate_viscous_stress_tensor(grad_u);
430
431 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> neg_heat_flux =
432 comp_flow_data.thermal_conductivity *
433 calculate_grad_T(conserved_variables, grad_conserved_variables);
434
435 ConservedVariablesGradType flux;
436 for (unsigned int d = 0; d < dim; ++d)
437 {
438 // density
439 flux[0][d] = 0.0;
440
441 // momentum
442 for (unsigned int e = 0; e < dim; ++e)
443 flux[e + 1][d] = viscous_stress[e][d];
444
445 // energy
446 flux[dim + 1][d] = neg_heat_flux[d];
447
448 for (unsigned int e = 0; e < dim; ++e)
449 flux[dim + 1][d] += velocity[e] * viscous_stress[d][e];
450 }
451
452 return flux;
453 }

455 template <int dim, typename number>
456 inline DEAL_II_ALWAYS_INLINE //
457 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesType
458 CompressibleFlowCalculators<dim, number>::calculate_viscous_numerical_flux(
459 const ConservedVariablesType &u_m,
460 const ConservedVariablesType &u_p,
461 const ConservedVariablesGradType &grad_u_m,
462 const ConservedVariablesGradType &grad_u_p,
463 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
464 const dealii::VectorizedArray<number> penalty_parameter) const
465 {
466 const auto flux_m = calculate_viscous_flux(u_m, grad_u_m);
467
468 const auto flux_p = calculate_viscous_flux(u_p, grad_u_p);
469
470 return contract_average_tensor_with_normal(flux_m, flux_p, normal) -
471 penalty_parameter * comp_flow_data.dynamic_viscosity / comp_flow_data.reference_density *
472 (u_m - u_p);
473 }

475 template <int dim, typename number>
476 inline DEAL_II_ALWAYS_INLINE //
477 std::pair<typename CompressibleFlowCalculators<dim, number>::ConservedVariablesGradType,
478 typename CompressibleFlowCalculators<dim, number>::ConservedVariablesGradType>
479 CompressibleFlowCalculators<dim, number>::calculate_viscous_numerical_flux_gradient(
480 const ConservedVariablesType &u_m,
481 const ConservedVariablesType &u_p,
482 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal) const
483 {
484 ConservedVariablesGradType jump_u;
485 for (unsigned int e = 0; e < dim + 2; ++e)
486 for (unsigned int d = 0; d < dim; ++d)
487 jump_u[e][d] = (u_m[e] - u_p[e]) * normal[d];
488
489 // use jumps instead of gradients for evaluating the viscous flux
490 const ConservedVariablesGradType flux_m = 0.5 * calculate_viscous_flux(u_m, jump_u);
491 const ConservedVariablesGradType flux_p = 0.5 * calculate_viscous_flux(u_p, jump_u);
492
493 return std::make_pair(flux_m, flux_p);
494 }
