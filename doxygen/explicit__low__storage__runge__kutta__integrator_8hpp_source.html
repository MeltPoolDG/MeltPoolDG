198 Assert(solution_history.size() >= required_solution_history_size(),
199 dealii::ExcMessage(

202 TimerOutput::Scope timer_section(timer, "Explicit Runge-Kutta time integration");
203 rk_register_ri = 0.;
204 if (stage_pre_processing)
205 stage_pre_processing(current_time, rk_register_ri, solution_history.get_current_solution());
206 pde_operator.apply_operator(
207 current_time,
208 rk_register_ri,
209 solution_history.get_current_solution(),
210 [&](const unsigned int start_range, const unsigned int end_range) {
211 DEAL_II_OPENMP_SIMD_PRAGMA
212 for (unsigned int i = start_range; i < end_range; ++i)
213 {
214 const number k_i = rk_register_ri.local_element(i);
215 const number sol_i = solution_history.get_current_solution().local_element(i);
216 solution_history.get_current_solution().local_element(i) =
217 sol_i + bi[0] * time_step * k_i;
218 rk_register_ri.local_element(i) = sol_i + ai[0] * time_step * k_i;
219 }
220 });
221 if (stage_post_processing)
222 stage_post_processing(current_time + ci[1] * time_step, rk_register_ri, rk_register_ri);
223
