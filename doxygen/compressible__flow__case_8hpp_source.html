104 const dealii::ConditionalOStream pcout(std::cout,
105 Utilities::MPI::this_mpi_process(
106 this->mpi_communicator) == 0 and
107 parameters.base.verbosity_level >= 1);
108
109 const auto &dof_vector = generic_data_out.get_vector("density");
110 const auto &dof_handler = generic_data_out.get_dof_handler("density");
111
112 reference_function.set_time(generic_data_out.get_time());
113
114 number errors_squared[3] = {};
115
116 bool is_cut = false;
117 if (dof_handler.get_fe_collection().size() > 1)
118 is_cut = true;
119
120 if (not is_cut)
121 {
122 // in the fitted mesh case, we can use the DoF values for error computation
123
124 std::map<types::global_dof_index, Point<dim, number>> support_points =
125 DoFTools::map_dofs_to_support_points(generic_data_out.get_mapping(), dof_handler);
126 for (const auto &cell : dof_handler.active_cell_iterators())
127 {
128 if (cell->is_artificial() or cell->is_ghost())
129 continue;
130 const unsigned int nodes_per_cell = cell->get_fe().dofs_per_cell / (dim + 2);
131 std::vector<types::global_dof_index> local_dof_indices(cell->get_fe().dofs_per_cell);
132 cell->get_dof_indices(local_dof_indices);
133
134 for (unsigned int i = 0; i < local_dof_indices.size(); ++i)
135 {
136 switch (unsigned int component = i / nodes_per_cell)
137 {
138 case 0: {
139 const number error =
140 dof_vector[local_dof_indices[i]] -
141 reference_function.value(support_points[local_dof_indices[i]], 0);
142 errors_squared[0] += error * error;
143 break;
144 }
145 case dim + 1: {
146 const number error =
147 dof_vector[local_dof_indices[i]] -
148 reference_function.value(support_points[local_dof_indices[i]], dim + 1);
149 errors_squared[2] += error * error;
150 break;
151 }
152 default: {
153 Assert(component > 0 and component < dim + 1, dealii::ExcInternalError());
154 const number error =
155 dof_vector[local_dof_indices[i]] -
156 reference_function.value(support_points[local_dof_indices[i]],
157 component);
158 errors_squared[1] += error * error;
159 }
160 }
161 }
162 }
163 }
164 else
165 {
166 // if cutDG is used, we evaluate the error at the quadrature points
167
168 // reading DoF-vector and DoFHandler of level-set
169 // (an assert is thrown in the getter functions, if 'level_set' can not be found in
170 // the entries of 'generic_data_out')
171 const auto &level_set = generic_data_out.get_vector("level_set");
172 const auto &level_set_dof_handler = generic_data_out.get_dof_handler("level_set");
173
174 // generate FESystem
175 const unsigned int fe_degree = dof_handler.get_fe_collection().max_degree();
176 FE_DGQ<dim> fe_q(fe_degree);
177 FESystem<dim> fe_point_temp(fe_q, dim + 2);
178
179 // classify cells
180 NonMatching::MeshClassifier<dim> mesh_classifier(level_set_dof_handler, level_set);
181 mesh_classifier.reclassify();
182
183 // compute quadrature
184 NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>> mapping_info_cell(
185 generic_data_out.get_mapping(), update_values | update_JxW_values);
186
187 hp::QCollection<1> q_collection((dealii::QGauss<1>(fe_degree + 1)));
188 NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(q_collection,
189 level_set_dof_handler,
190 level_set);
191
192 auto physical = [&](const typename DoFHandler<dim>::active_cell_iterator &i) {
193 return (mesh_classifier.location_to_level_set(i) ==
194 NonMatching::LocationToLevelSet::intersected) or
195 (mesh_classifier.location_to_level_set(i) ==
196 NonMatching::LocationToLevelSet::
197 outside /*active phase has positive level-set*/);
198 };
199
200 const auto physical_active_cell_iterators =
201 dof_handler.active_cell_iterators() | physical;
202
203 std::vector<Quadrature<dim>> quad_vec_cell;
204
205 unsigned int n_active_cells = std::distance(dof_handler.active_cell_iterators().begin(),
206 dof_handler.active_cell_iterators().end());
207
208 for (const auto &cell : physical_active_cell_iterators)
209 {
210 quadrature_generator.generate(cell);
211 // active phase has positive level-set
212 quad_vec_cell.push_back(quadrature_generator.get_outside_quadrature());
213 }
214
215 mapping_info_cell.reinit_cells(physical_active_cell_iterators,
216 quad_vec_cell,
217 n_active_cells);
218
219 // cell-loop for error-norm computation
220 FEPointEvaluation<dim + 2, dim, dim, VectorizedArray<number>> fe_point_eval(
221 mapping_info_cell, fe_point_temp);
222
223 for (const auto &cell : dof_handler.active_cell_iterators())
224 {
225 if (cell->is_artificial() or cell->is_ghost() or !physical(cell))
226 continue;
227
228 std::vector<number> solution_values_in(cell->get_fe().dofs_per_cell);
229
230 cell->get_dof_values(dof_vector,
231 solution_values_in.begin(),
232 solution_values_in.end());
233
234 fe_point_eval.reinit(cell->active_cell_index());
235 fe_point_eval.evaluate(solution_values_in, EvaluationFlags::values);
236
237 VectorizedArray<number> local_errors_squared[3] = {};
238
239 for (const unsigned int q : fe_point_eval.quadrature_point_indices())
240 {
241 const auto error =
242 VectorTools::evaluate_function_at_vectorized_points<dim, number, dim + 2>(
243 reference_function, fe_point_eval.quadrature_point(q)) -
244 fe_point_eval.get_value(q);
245
246 const auto JxW = fe_point_eval.JxW(q);
247 local_errors_squared[0] = error[0] * error[0] * JxW;
248
249 for (unsigned int d = 0; d < dim; ++d)
250 local_errors_squared[1] = (error[d + 1] * error[d + 1]) * JxW;
251 local_errors_squared[2] = (error[dim + 1] * error[dim + 1]) * JxW;
252
253 for (unsigned int v = 0;
254 v < fe_point_eval.n_active_entries_per_quadrature_batch(q);
255 ++v)
256 for (unsigned int d = 0; d < 3; ++d)
257 errors_squared[d] += local_errors_squared[d][v];
258 }
259 }
260 }
261
262 Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
263 std::array<number, 3> errors;
264 for (unsigned int d = 0; d < 3; ++d)
265 errors[d] = std::sqrt(errors_squared[d]);
266
267 std::ostringstream output;
268 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
269 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
270 Journal::print_line(pcout, output.str(), "compressible_flow");
271 }
