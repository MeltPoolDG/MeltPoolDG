156 void
157 print_relative_norm(const GenericDataOut<dim, number> &generic_data_out,
158 dealii::Function<dim> &reference_function,
159 const std::string &norm_name = "norm") const
160 {
161 const dealii::ConditionalOStream pcout(std::cout,
162 dealii::Utilities::MPI::this_mpi_process(
163 this->mpi_communicator) == 0 and
164 parameters.base.verbosity_level >= 1);
165
166 const auto &dof_vector = generic_data_out.get_vector("density");
167 const auto &dof_handler = generic_data_out.get_dof_handler("density");
168
169 reference_function.set_time(generic_data_out.get_time());
170
171 number errors_squared[3] = {};
172
173 bool is_cut = false;
174 if (dof_handler.get_fe_collection().size() > 1)
175 is_cut = true;
176
177 if (not is_cut)
178 {
179 // in the fitted mesh case, we can use the DoF values for error computation
180
181 std::map<dealii::types::global_dof_index, dealii::Point<dim, number>> support_points =
182 dealii::DoFTools::map_dofs_to_support_points(generic_data_out.get_mapping(),
183 dof_handler);
184 for (const auto &cell : dof_handler.active_cell_iterators())
185 {
186 if (cell->is_artificial() or cell->is_ghost())
187 continue;
188 const unsigned int nodes_per_cell = cell->get_fe().dofs_per_cell / (dim + 2);
189 std::vector<dealii::types::global_dof_index> local_dof_indices(
190 cell->get_fe().dofs_per_cell);
191 cell->get_dof_indices(local_dof_indices);
192
193 for (unsigned int i = 0; i < local_dof_indices.size(); ++i)
194 {
195 switch (unsigned int component = i / nodes_per_cell)
196 {
197 case 0: {
198 const number error =
199 dof_vector[local_dof_indices[i]] -
200 reference_function.value(support_points[local_dof_indices[i]], 0);
201 errors_squared[0] += error * error;
202 break;
203 }
204 case dim + 1: {
205 const number error =
206 dof_vector[local_dof_indices[i]] -
207 reference_function.value(support_points[local_dof_indices[i]], dim + 1);
208 errors_squared[2] += error * error;
209 break;
210 }
211 default: {
212 Assert(component > 0 and component < dim + 1, dealii::ExcInternalError());
213 const number error =
214 dof_vector[local_dof_indices[i]] -
215 reference_function.value(support_points[local_dof_indices[i]],
216 component);
217 errors_squared[1] += error * error;
218 }
219 }
220 }
221 }
222 }
223 else
224 {
225 // if cutDG is used, we evaluate the error at the quadrature points
226
227 // reading DoF-vector and DoFHandler of level-set
228 // (an assert is thrown in the getter functions, if 'level_set' can not be found in
229 // the entries of 'generic_data_out')
230 const auto &level_set = generic_data_out.get_vector("level_set");
231 const auto &level_set_dof_handler = generic_data_out.get_dof_handler("level_set");
232
233 // generate FESystem
234 const unsigned int fe_degree = dof_handler.get_fe_collection().max_degree();
235 dealii::FE_DGQ<dim> fe_q(fe_degree);
236 dealii::FESystem<dim> fe_point_temp(fe_q, dim + 2);
237
238 // classify cells
239 dealii::NonMatching::MeshClassifier<dim> mesh_classifier(level_set_dof_handler,
240 level_set);
241 mesh_classifier.reclassify();
242
243 // compute quadrature
244 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>>
245 mapping_info_cell(generic_data_out.get_mapping(),
246 dealii::update_values | dealii::update_JxW_values);
247
248 dealii::hp::QCollection<1> q_collection((dealii::QGauss<1>(fe_degree + 1)));
249 dealii::NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(
250 q_collection, level_set_dof_handler, level_set);
251
252 auto physical = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
253 return (mesh_classifier.location_to_level_set(i) ==
254 dealii::NonMatching::LocationToLevelSet::intersected) or
255 (mesh_classifier.location_to_level_set(i) ==
256 dealii::NonMatching::LocationToLevelSet::
257 outside /*active phase has positive level-set*/);
258 };
259
260 const auto physical_active_cell_iterators =
261 dof_handler.active_cell_iterators() | physical;
262
263 std::vector<dealii::Quadrature<dim>> quad_vec_cell;
264
265 unsigned int n_active_cells = std::distance(dof_handler.active_cell_iterators().begin(),
266 dof_handler.active_cell_iterators().end());
267
268 for (const auto &cell : physical_active_cell_iterators)
269 {
270 quadrature_generator.generate(cell);
271 // active phase has positive level-set
272 quad_vec_cell.push_back(quadrature_generator.get_outside_quadrature());
273 }
274
275 mapping_info_cell.reinit_cells(physical_active_cell_iterators,
276 quad_vec_cell,
277 n_active_cells);
278
279 // cell-loop for error-norm computation
280 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>>
281 fe_point_eval(mapping_info_cell, fe_point_temp);
282
283 for (const auto &cell : dof_handler.active_cell_iterators())
284 {
285 if (cell->is_artificial() or cell->is_ghost() or (not physical(cell)))
286 continue;
287
288 std::vector<number> solution_values_in(cell->get_fe().dofs_per_cell);
289
290 cell->get_dof_values(dof_vector,
291 solution_values_in.begin(),
292 solution_values_in.end());
293
294 fe_point_eval.reinit(cell->active_cell_index());
295 fe_point_eval.evaluate(solution_values_in, dealii::EvaluationFlags::values);
296
297 dealii::VectorizedArray<number> local_errors_squared[3] = {};
298
299 for (const unsigned int q : fe_point_eval.quadrature_point_indices())
300 {
301 const auto error =
302 VectorTools::evaluate_function_at_vectorized_points<dim, number, dim + 2>(
303 reference_function, fe_point_eval.quadrature_point(q)) -
304 fe_point_eval.get_value(q);
305
306 const auto JxW = fe_point_eval.JxW(q);
307 local_errors_squared[0] = error[0] * error[0] * JxW;
308
309 for (unsigned int d = 0; d < dim; ++d)
310 local_errors_squared[1] = (error[d + 1] * error[d + 1]) * JxW;
311 local_errors_squared[2] = (error[dim + 1] * error[dim + 1]) * JxW;
312
313 for (unsigned int v = 0;
314 v < fe_point_eval.n_active_entries_per_quadrature_batch(q);
315 ++v)
316 for (unsigned int d = 0; d < 3; ++d)
317 errors_squared[d] += local_errors_squared[d][v];
318 }
319 }
320 }
321
322 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
323 std::array<number, 3> errors;
324 for (unsigned int d = 0; d < 3; ++d)
325 errors[d] = std::sqrt(errors_squared[d]);
326
327 std::ostringstream output;
328 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
329 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
330 Journal::print_line(pcout, output.str(), "compressible_flow");
331 }
