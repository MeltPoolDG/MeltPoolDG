27 template <int dim>
28 class LevelSetHeaviside : public dealii::Function<dim>
29 {
30 public:
31 LevelSetHeaviside(const InterfaceCase interface_case_in,
32 const std::pair<double, double> &interface_case_info_in,
33 const double epsilon_cell_in)
34 : dealii::Function<dim>(1)
35 , eps(epsilon_cell_in)
36 , interface_case(interface_case_in)
37 , interface_case_info(interface_case_info_in)
38 {}
39
40 double
41 value(const dealii::Point<dim> &p, const unsigned int) const override
42 {
43 if (interface_case == InterfaceCase::straight)
44 {
45 const auto y = p[dim - 1];
46 const auto current_time = this->get_time();
47 return UtilityFunctions::CharacteristicFunctions::heaviside(
48 level +
49 interface_case_info.first *
50 (current_time < interface_case_info.second ? current_time :
51 interface_case_info.second) /
52 interface_case_info.second -
53 y,
54 eps); // 0 side of H() stands for gas, 1 side of H() stands for liquid
55 }
56 else if (interface_case == InterfaceCase::single_powder_particle)
57 {
58 // say we inherit from a straight interface case
59 const auto y = p[dim - 1];
60
61 double straight_value = UtilityFunctions::CharacteristicFunctions::heaviside(
62 level - y,
63 eps); // 0 side of H() stands for gas, 1 side of H() stands for liquid
64
65 dealii::Point<dim> sphere_center;
66 sphere_center[dim - 1] = interface_case_info.first;
67
68 // now add a powder particle with gradient:
69 const dealii::Functions::SignedDistance::Sphere<dim> distance_sphere(
70 sphere_center, interface_case_info.second);
71 double powder_particle_value =
72 UtilityFunctions::CharacteristicFunctions::heaviside(-distance_sphere.value(p), eps);
73 return std::max(straight_value, powder_particle_value);
74 }
75 else
76 {
77 AssertThrow(false, dealii::ExcNotImplemented());
78 return 0.0;
79 }
80 return 0.0;
81 }
82
83 private:
84 const double eps = 0.1;
85 const double level = 0.0;
86 InterfaceCase interface_case = InterfaceCase::straight;
87 std::pair<double, double> interface_case_info = std::pair<double, double>(0., 0.);
88 };

149 void
150 SimulationRadTrans<dim, Problem>::create_spatial_discretization()
151 {
152 if (this->parameters.base.fe.type == FiniteElementType::FE_SimplexP || dim == 1)
153 {
154#ifdef DEAL_II_WITH_METIS
155 this->triangulation = std::make_shared<dealii::parallel::shared::Triangulation<dim>>(
156 this->mpi_communicator,
157 dealii::Triangulation<dim>::MeshSmoothing::none,
158 true,
159 dealii::parallel::shared::Triangulation<dim>::Settings::partition_metis);
160#else
161 AssertThrow(
162 false,
163 dealii::ExcMessage(
164 "Missing Metis support of the deal.II installation. "
165 "Configure deal.II with -D DEAL_II_WITH_METIS='ON' to execute this example."));
166#endif
167 }
168 else
169 {
170 this->triangulation = std::make_shared<dealii::parallel::distributed::Triangulation<dim>>(
171 this->mpi_communicator);
172 }
173
174 const dealii::Point<dim> bottom_left =
175 dim == 1 ? dealii::Point<dim>(domain_y_min) :
176 dim == 2 ? dealii::Point<dim>(domain_x_min, domain_y_min) :
177 dealii::Point<dim>(domain_x_min, domain_x_min, domain_y_min);
178 const dealii::Point<dim> top_right =
179 dim == 1 ? dealii::Point<dim>(domain_y_max) :
180 dim == 2 ? dealii::Point<dim>(domain_x_max, domain_y_max) :
181 dealii::Point<dim>(domain_x_max, domain_x_max, domain_y_max);
182 if (this->parameters.base.fe.type == FiniteElementType::FE_SimplexP)
183 {
184 std::vector<unsigned int> subdivisions(
185 dim, 5 * dealii::Utilities::pow(2, this->parameters.base.global_refinements));
186 subdivisions[dim - 1] *= 2;
187 for (int d = 0; d < dim; d++)
188 subdivisions[d] *= cell_repetitions[d];
189
190 dealii::GridGenerator::subdivided_hyper_rectangle_with_simplices(
191 *this->triangulation, subdivisions, bottom_left, top_right, true /*colorize*/);
192 }
193 else
194 {
195 dealii::GridGenerator::subdivided_hyper_rectangle(
196 *this->triangulation, cell_repetitions, bottom_left, top_right, true /*colorize*/);
197 }
198 }

224 template <int dim, typename Problem>
225 void
226 SimulationRadTrans<dim, Problem>::set_field_conditions()
227 {
228 // pass simulation-specific parameters to the simulation class.
229 // Done after json parsing, is relevant for heaviside
230 if (interface_case == InterfaceCase::straight)
231 interface_case_info = std::pair<double, double>(speed, end_time);
232 else if (interface_case == InterfaceCase::single_powder_particle)
233 interface_case_info =
234 std::pair<double, double>(powder_particle_offset, powder_particle_radius);
235
236 // determine the interface epsilon parameter from minimum mesh size
237 double thickness_scale_factor = 2.5;
238 const double epsilon_cell = dealii::GridTools::minimal_cell_diameter(*this->triangulation) /
239 std::sqrt(dim) * thickness_scale_factor;
240
241 // attach the prescribed heaviside function field
242 this->attach_initial_condition(std::make_shared<LevelSetHeaviside<dim>>(interface_case,
243 interface_case_info,
244 epsilon_cell),
245 "prescribed_heaviside");
246
247 if (this->parameters.base.problem_name == "radiative_transport")
248 this->attach_initial_condition(std::make_shared<dealii::Functions::ZeroFunction<dim>>(),
249 "intensity");
250
251 // this is only used to TODO generate the comparison solution with the gaussian laser
252 if (this->parameters.base.problem_name == "heat_transfer")
253 {
254 // attach dummy initial conditions for the heat transfer operation
255 this->attach_initial_condition(std::make_shared<dealii::Functions::ZeroFunction<dim>>(),
256 "heat_transfer");
257 }
