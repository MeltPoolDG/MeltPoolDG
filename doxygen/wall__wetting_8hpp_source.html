231 void
232 set_field_conditions() final
233 {
234 // Compute normal diffusion factor (we consider a static mesh)
235 cell_size_min =
236 dealii::GridTools::minimal_cell_diameter(*this->triangulation) / std::sqrt(dim);
237 epsilon = this->parameters.reinit.compute_interface_thickness_parameter_epsilon(
238 cell_size_min / this->parameters.reinit.fe.get_n_subdivisions());
239
240 // Compute time-step and change data value
241 /* dt = h^2 / [2 * (epsilon_n + epsilon_t)] * time_step_factor */
242 number time_step = time_step_factor * 0.5 * dealii::Utilities::fixed_power<2>(cell_size_min) /
243 (epsilon * (1 + this->parameters.reinit.tangential_diffusion_factor));
244 this->parameters.time_stepping.time_step_size = time_step;
245
246 // Compute normal vector filtering factor and change data value
247 gamma = dealii::Utilities::fixed_power<2>(
248 gamma_factor * this->parameters.reinit.interface_thickness_parameter.value);
249 this->parameters.normal_vec.filter_parameter = gamma;
250
251 // Initial level-set condition
252 this->attach_initial_condition(std::make_shared<InitialLevelSet<dim, number>>(x_interface,
253 epsilon),
254 "level_set");
255 }

326 void
327 compute_contact_angle_at_boundary(number &contact_angle,
328 const GenericDataOut<dim, double> &generic_data_out) const
329 {
330 const auto &dof_handler = generic_data_out.get_dof_handler("psi");
331 const auto &mapping = generic_data_out.get_mapping();
332 const auto &level_set_vector = generic_data_out.get_vector("psi");
333
334 number contact_angle_sum = 0.0;
335 number total_weight = 0.0;
336
337 const unsigned int n_subdivisions = 2;
338 const number tolerance = 3 * epsilon;
339 const number contour_value = 0.0;
340
341 // Bottom wall normal
342 dealii::Tensor<1, dim> wall_normal;
343 wall_normal[1] = 1.0; // unit normal in +y direction
344
345 // Tolerance identifying value at the bottom wall
346 const number tolerance_bottom_wall = 0.25 * cell_size_min;
347
348 // Evaluate the contact angle
349 LevelSet::Tools::evaluate_at_interface<dim, number>(
350 dof_handler,
351 mapping,
352 level_set_vector,
353 [&](const auto &cell,
354 const auto &interface_points,
355 const auto &reference_points,
356 const auto & /*JxW*/) {
357 const auto &fe = dof_handler.get_fe();
358 dealii::FEValues<dim> fe_values(mapping,
359 fe,
360 dealii::Quadrature<dim>(reference_points),
361 dealii::update_gradients);
362 fe_values.reinit(cell);
363 std::vector<dealii::Tensor<1, dim>> grad_phi(reference_points.size());
364 fe_values.get_function_gradients(level_set_vector, grad_phi);
365
366 for (unsigned int i = 0; i < reference_points.size(); ++i)
367 {
368 // Extract real points and level-set indicator gradient values
369 const auto &point_i = interface_points[i];
370 const auto &grad_phi_i = grad_phi[i];
371
372 // Check if point is at the bottom wall
373 const number distance_to_wall = std::abs(point_i[dim - 1] - y_min);
374 if (distance_to_wall < tolerance_bottom_wall)
375 {
376 // Compute contact angle with normal vector and grad_phi scalar product
377 const number cos_alpha = grad_phi_i * wall_normal / (grad_phi_i.norm() + 1e-16);
378 const number alpha_rad = std::acos(cos_alpha);
379
380 // Sum weighted contributions, so that if multiple points are considered, the
381 // closest to the bottom wall have more weight
382 const number weight = (tolerance - distance_to_wall) / tolerance;
383 contact_angle_sum += weight * alpha_rad * 180.0 / dealii::numbers::PI;
384 total_weight += weight;
385 }
386 }
387 },
388 contour_value,
389 n_subdivisions,
390 tolerance,
391 true /*use_mca=*/);
392
393 // Sum results from all processes
394 const number global_contact_angle_sum =
395 dealii::Utilities::MPI::sum(contact_angle_sum, MPI_COMM_WORLD);
396 const number global_total_weight = dealii::Utilities::MPI::sum(total_weight, MPI_COMM_WORLD);
397 contact_angle = (global_total_weight > 0) ? global_contact_angle_sum / global_total_weight :
398 std::numeric_limits<number>::quiet_NaN();
399 }
400
401 private:
402 /* Domain dimensions */
404 number x_min = 0.0;
406 number x_max = 2.0;
408 number y_min = x_min;
410 number y_max = x_max;
411
413 number x_interface = 1.0;
414
416 number contact_angle_deg = 45.0;
417
420 number gamma_factor = 1.0;
