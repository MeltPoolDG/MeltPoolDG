108 template <int dim>
109 inline VectorizedArray<double>
110 compute_cell_size_dependent_filter_parameter_mf(const ScratchData<dim> &scratch_data,
111 const unsigned int dof_idx,
112 const unsigned int cell_idx,
113 const double scale_factor)
114 {
115 const double n_subdivisions =
116 scratch_data.is_FE_Q_iso_Q_1(dof_idx) ? scratch_data.get_degree(dof_idx) : 1;
117 return Utilities::fixed_power<2>(
118 std::max(VectorizedArray<double>(scratch_data.get_min_cell_size()),
119 scratch_data.get_cell_sizes()[cell_idx] / (double)n_subdivisions)) *
120 scale_factor;
121 }

132 template <int dim, typename cell_type>
133 double
134 compute_cell_size_dependent_filter_parameter(const ScratchData<dim> &scratch_data,
135 const unsigned int dof_idx,
136 const cell_type cell,
137 const double scale_factor)
138 {
139 const double n_subdivisions =
140 scratch_data.is_FE_Q_iso_Q_1(dof_idx) ? scratch_data.get_degree(dof_idx) : 1;
141
142 return Utilities::fixed_power<2>(
143 std::max(scratch_data.get_min_cell_size(),
144 cell->diameter() / (std::sqrt(dim) * n_subdivisions))) *
145 scale_factor;
146 }
