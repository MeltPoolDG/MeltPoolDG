167 void
168 post(const FiniteElementData &base_fe_data, const unsigned int base_verbosity_level)
169 {
170 fe.post(base_fe_data);
171 AssertThrow(fe.type == FiniteElementType::FE_DGQ,
172 dealii::ExcMessage(
173 "The compressible flow solver only supports elements of type 'FE_DGQ'."));
174
175 if (cut.two_phase)
176 interface_jump_conditions.post();
177
178 // set default time integration scheme for cut
179 if (domain_representation_type == "cut")
180 {
181 if (time_integrator.integrator_type == TimeIntegratorSchemes::not_initialized)
182 time_integrator.integrator_type = TimeIntegratorSchemes::explicit_euler;
183 AssertThrow(
184 time_integrator.integrator_type == TimeIntegratorSchemes::explicit_euler,
185 dealii::ExcMessage(
186 "The cut compressible flow solver only supports explicit Euler time integration."));
187 }
188
189 // Synchronize verbosity with base verbosity if not set explicitly.
190 if (verbosity_level < 0)
191 verbosity_level = base_verbosity_level;
192
193 // Set thermal conductivity, if not explicitly set by the user.
194 // For physical consistency, set thermal conductivity based on the user-defined dynamic
195 // viscosity, gamma and specific gas constant. The Prandtl number is currently set
196 // constant to Pr=0.71 for the gas phase (air) and to Pr=0.01 for the liquid phase (metal).
197 if (material.gas.thermal_conductivity == std::numeric_limits<number>::max())
198 material.gas.thermal_conductivity = material.gas.dynamic_viscosity * material.gas.gamma *
199 material.gas.specific_gas_constant /
200 (material.gas.gamma - 1.) * 1. / 0.71;
201 if (cut.two_phase and
202 material.liquid.thermal_conductivity == std::numeric_limits<number>::max())
203 material.liquid.thermal_conductivity =
204 material.liquid.dynamic_viscosity * material.liquid.gamma *
205 material.liquid.specific_gas_constant / (material.liquid.gamma - 1.) * 1. / 0.01;
206
207 // Ensure that parameters are set for advanced equations of state
208 if (material.gas.eos_data.type == EquationOfState::stiffened_gas)
209 AssertThrow(material.liquid.eos_data.p_inf != std::numeric_limits<number>::max(),
210 dealii::ExcMessage(
211 "The parameter p_inf is required for the stiffened gas EOS."));
212 else if (material.gas.eos_data.type == EquationOfState::noble_abel_stiffened_gas)
213 AssertThrow(material.liquid.eos_data.p_inf != std::numeric_limits<number>::max() and
214 material.liquid.eos_data.b != std::numeric_limits<number>::max() and
215 material.liquid.eos_data.q != std::numeric_limits<number>::min(),
216 dealii::ExcMessage(
217 "The parameters p_inf, b and q are required for the Noble-Abel stiffened"
218 " gas EOS."));
219
220 if (cut.two_phase)
221 {
222 if (material.liquid.eos_data.type == EquationOfState::stiffened_gas)
223 AssertThrow(material.liquid.eos_data.p_inf != std::numeric_limits<number>::max(),
224 dealii::ExcMessage(
225 "The parameter p_inf is required for the stiffened gas EOS."));
226 else if (material.liquid.eos_data.type == EquationOfState::noble_abel_stiffened_gas)
227 AssertThrow((material.liquid.eos_data.p_inf != std::numeric_limits<number>::max()) and
228 (material.liquid.eos_data.b != std::numeric_limits<number>::max()) and
229 (material.liquid.eos_data.q != std::numeric_limits<number>::min()),
230 dealii::ExcMessage(
231 "The parameters p_inf, b and q are required for the Noble-Abel stiffened"
232 " gas EOS."));
233 }
234
235 // Advanced EOS are currently only allowed for explicit time integration.
236 if (material.gas.eos_data.type != EquationOfState::ideal_gas)
237 AssertThrow(
238 !MeltPoolDG::time_integrator_scheme_is_explicit(time_integrator.integrator_type),
239 dealii::ExcMessage("Only the ideal gas EOS is allowed for implicit time integration."));
240 }
