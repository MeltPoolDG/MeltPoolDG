32 inline DEAL_II_ALWAYS_INLINE //
33 dealii::VectorizedArray<number>
34 calculate_thermodynamic_pressure(

36 override
37 {
38 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
39 calculate_velocity<dim, number>(conserved_variables);

41 return (material_data.gamma - 1.) *

56 inline DEAL_II_ALWAYS_INLINE //
57 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
58 calculate_grad_T(
59 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
60 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
61 &grad_conserved_variables) override
62 {
63 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> u =
64 calculate_velocity<dim, number>(conserved_variables);
65 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_u =
66 calculate_grad_velocity<dim, number>(conserved_variables, grad_conserved_variables);
67 const dealii::VectorizedArray<number> rho = conserved_variables[0];
68 const dealii::VectorizedArray<number> inv_rho = dealii::VectorizedArray<number>(1.) / rho;
69 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho =
70 grad_conserved_variables[0];
71 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_E =
72 inv_rho *
73 (grad_conserved_variables[dim + 1] - inv_rho * conserved_variables[dim + 1] * grad_rho);
74
75 return material_data.gamma / material_data.specific_isobaric_heat *
76 (grad_E - grad_u * u + material_data.eos_data.p_inf * inv_rho * inv_rho * grad_rho);
77 }

124 inline DEAL_II_ALWAYS_INLINE //
125 CompressibleFlowTypes::ConservedVariablesType<dim, number>
126 convert_primitive_into_conservative_variables(
127 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &u_prim) override
128 {
129 CompressibleFlowTypes::ConservedVariablesType<dim, number> u_cons;
130
131 // density
132 u_cons[0] =
133 (u_prim[0] + material_data.eos_data.p_inf) * material_data.gamma /
134 (material_data.specific_isobaric_heat * u_prim[dim + 1] * (material_data.gamma - 1.));
135 // momentum
136 for (unsigned int i = 1; i < dim + 1; i++)
137 u_cons[i] = u_prim[i] * u_cons[0];
138 // total energy
139 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
140 calculate_velocity<dim, number>(u_cons);
141 u_cons[dim + 1] =
142 u_cons[0] *
143 (material_data.specific_isobaric_heat / material_data.gamma * u_prim[dim + 1] +
144 material_data.eos_data.p_inf / u_cons[0] + 0.5 * scalar_product(velocity, velocity));
145
146 return u_cons;
147 }
