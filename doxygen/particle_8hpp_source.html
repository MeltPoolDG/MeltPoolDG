35 public:
36 // Number of components of the angular velocity: one in 2D, three in 3D.
37 static constexpr unsigned int size_angular_velocity = dim - 3 % dim;
38
39 // Total number of scalar properties associated with each obstacle particle.
40 // Breakdown:
41 // - 2 * size_angular_velocity: angular velocity and angular acceleration
42 // - 2 * dim: velocity and acceleration
43 // - dim: (directional) force acting on the particle
44 // - 5: radius, volume, density, mass, moment of intertia, particle id
45 static constexpr unsigned int n_obstacle_properties =
46 2 * size_angular_velocity + 2 * dim + dim + 6;

51 enum Properties
52 {
53 velocity = 0,
54 acceleration = dim,
55 angular_velocity = acceleration + dim,
56 angular_acceleration = angular_velocity + size_angular_velocity,
57 force = angular_acceleration + size_angular_velocity,
58 radius = force + dim,
59 volume,
60 density,
61 mass,
62 moment_of_inertia,
63 // TODO: This is temporary and should be replaced by a proper deal.II ghost particle
64 // implementation
65 particle_id, // used to identify particles received from other processes
66 };

440template <int dim, typename number>
441std::pair<std::vector<number>, dealii::Point<dim, number>>
442MeltPoolDG::SphericalParticle<dim, number>::read_state_input(const std::string &input_data)
443{
444 dealii::Point<dim, number> particle_location;
445 std::vector<number> particle_properties(n_obstacle_properties, 0.0);
446 std::string temp;
447 std::istringstream data_string(input_data);
448 // particle position
449 for (int i = 0; i < dim; i++)
450 {
451 std::getline(data_string, temp, ',');
452 particle_location[i] = std::stod(temp);
453 }
454
455 // set particle translational velocity and angular velocity to zero by default
456 for (unsigned int i = Properties::velocity; i < Properties::angular_acceleration; ++i)
457 {
458 particle_properties[i] = 0;
459 }
460
461 // density and radius
462 std::getline(data_string, temp, ',');
463 particle_properties[Properties::density] = std::stod(temp);
464 std::getline(data_string, temp, ',');
465 particle_properties[Properties::radius] = std::stod(temp);
466
467 // Compute remaining variables
468 if constexpr (dim == 3)
469 {
470 particle_properties[Properties::volume] =
471 4.0 / 3.0 * M_PI * std::pow(particle_properties[Properties::radius], 3);
472 particle_properties[Properties::mass] =
473 particle_properties[Properties::volume] * particle_properties[Properties::density];
474 particle_properties[Properties::moment_of_inertia] =
475 0.4 * particle_properties[Properties::mass] *
476 std::pow(particle_properties[Properties::radius], 2);
477 }
478 else if constexpr (dim == 2)
479 {
480 particle_properties[Properties::volume] =
481 M_PI * std::pow(particle_properties[Properties::radius], 2);
482 particle_properties[Properties::mass] =
483 particle_properties[Properties::volume] * particle_properties[Properties::density];
484 particle_properties[Properties::moment_of_inertia] =
485 0.5 * particle_properties[Properties::mass] *
486 std::pow(particle_properties[Properties::radius], 2);
487 }
488
489 return {particle_properties, particle_location};
490}

665template <int dim, typename number>
666template <typename VectorizedArrayType>
667dealii::Tensor<1, dim, VectorizedArrayType>
668MeltPoolDG::SphericalParticle<dim, number>::get_velocity(
669 const dealii::Particles::ParticleAccessor<dim> &particle,
670 const dealii::Point<dim, VectorizedArrayType> &location)
671{
672 Assert(dim == 2 or dim == 3, dealii::ExcMessage("Invalid dimension!"));
673
674 dealii::Point<dim, dealii::VectorizedArray<number>> vectorized_particle_location;
675 for (auto i = 0; i < dim; ++i)
676 vectorized_particle_location[i] = dealii::VectorizedArray<number>(particle.get_location()[i]);
677
678 dealii::Tensor<1, dim, VectorizedArrayType> distance_to_center =
679 location - vectorized_particle_location;
680 if constexpr (dim == 2)
681 {
682 return dealii::cross_product_2d(get_angular_velocity(particle)[0] * distance_to_center) +
683 get_velocity<VectorizedArrayType>(particle);
684 }
685 if constexpr (dim == 3)
686 {
687 return get_velocity<VectorizedArrayType>(particle) +
688 dealii::cross_product_3d(get_angular_velocity(particle), distance_to_center);
689 }
690 AssertThrow(false, dealii::ExcInternalError());
691}
