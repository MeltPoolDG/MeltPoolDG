327template <int dim, typename number>
328std::pair<std::vector<number>, dealii::Point<dim, number>>
329MeltPoolDG::SphericalParticle<dim, number>::read_state_input(const std::string &input_data)
330{
331 dealii::Point<dim, number> particle_location;
332 std::vector<number> particle_properties(n_obstacle_properties, 0.0);
333 std::string temp;
334 std::istringstream data_string(input_data);
335 // particle position
336 for (unsigned int i = 0; i < dim; i++)
337 {
338 std::getline(data_string, temp, ',');
339 particle_location[i] = std::stod(temp);
340 }
341
342 // particle velocity
343 for (unsigned int i = Properties::velocity; i < Properties::acceleration; ++i)
344 {
345 std::getline(data_string, temp, ',');
346 particle_properties[i] = std::stod(temp);
347 }
348
349 // particle angular velocity
350 for (unsigned int i = Properties::angular_velocity; i < Properties::angular_acceleration; ++i)
351 {
352 std::getline(data_string, temp, ',');
353 particle_properties[i] = std::stod(temp);
354 }
355
356 // density and radius
357 std::getline(data_string, temp, ',');
358 particle_properties[Properties::density] = std::stod(temp);
359 std::getline(data_string, temp, ',');
360 particle_properties[Properties::radius] = std::stod(temp);
361
362 // Compute remaining variables
363 if constexpr (dim == 3)
364 {
365 particle_properties[Properties::volume] =
366 4.0 / 3.0 * M_PI * std::pow(particle_properties[Properties::radius], 3);
367 particle_properties[Properties::mass] =
368 particle_properties[Properties::volume] * particle_properties[Properties::density];
369 particle_properties[Properties::moment_of_inertia] =
370 0.4 * particle_properties[Properties::mass] *
371 std::pow(particle_properties[Properties::radius], 2);
372 }
373 else if constexpr (dim == 2)
374 {
375 particle_properties[Properties::volume] =
376 M_PI * std::pow(particle_properties[Properties::radius], 2);
377 particle_properties[Properties::mass] =
378 particle_properties[Properties::volume] * particle_properties[Properties::density];
379 particle_properties[Properties::moment_of_inertia] =
380 0.5 * particle_properties[Properties::mass] *
381 std::pow(particle_properties[Properties::radius], 2);
382 }
383
384 return {particle_properties, particle_location};
385}

512template <int dim, typename number>
513template <typename VectorizedArrayType>
514dealii::Tensor<1, dim, VectorizedArrayType>
515MeltPoolDG::SphericalParticle<dim, number>::get_velocity(
516 const dealii::Particles::ParticleAccessor<dim> &particle,
517 const dealii::Point<dim, VectorizedArrayType> &location)
518{
519 Assert(dim == 2 or dim == 3, dealii::ExcMessage("Invalid dimension!"));
520
521 dealii::Point<dim, dealii::VectorizedArray<number>> vectorized_particle_location;
522 for (auto i = 0; i < dim; ++i)
523 vectorized_particle_location[i] = dealii::VectorizedArray<number>(particle.get_location()[i]);
524
525 dealii::Tensor<1, dim, VectorizedArrayType> distance_to_center =
526 location - vectorized_particle_location;
527 if constexpr (dim == 2)
528 {
529 return dealii::cross_product_2d(get_angular_velocity(particle)[0] * distance_to_center) +
530 get_velocity<VectorizedArrayType>(particle);
531 }
532 if constexpr (dim == 3)
533 {
534 return get_velocity<VectorizedArrayType>(particle) +
535 dealii::cross_product_3d(get_angular_velocity(particle), distance_to_center);
536 }
537 AssertThrow(false, dealii::ExcInternalError());
538}
