39 // Total number of scalar properties associated with each obstacle particle.
40 // Breakdown:
41 // - 2 * size_angular_velocity: angular velocity and angular acceleration

43 // - dim: (directional) force acting on the particle

398template <int dim, typename number>
399std::pair<std::vector<number>, dealii::Point<dim, number>>
400MeltPoolDG::SphericalParticle<dim, number>::read_state_input(const std::string &input_data)
401{
402 dealii::Point<dim, number> particle_location;
403 std::vector<number> particle_properties(n_obstacle_properties, 0.0);
404 std::string temp;
405 std::istringstream data_string(input_data);
406 // particle position
407 for (unsigned int i = 0; i < dim; i++)
408 {
409 std::getline(data_string, temp, ',');
410 particle_location[i] = std::stod(temp);
411 }
412
413 // particle velocity
414 for (unsigned int i = Properties::velocity; i < Properties::acceleration; ++i)
415 {
416 std::getline(data_string, temp, ',');
417 particle_properties[i] = std::stod(temp);
418 }
419
420 // particle angular velocity
421 for (unsigned int i = Properties::angular_velocity; i < Properties::angular_acceleration; ++i)
422 {
423 std::getline(data_string, temp, ',');
424 particle_properties[i] = std::stod(temp);
425 }
426
427 // density and radius
428 std::getline(data_string, temp, ',');
429 particle_properties[Properties::density] = std::stod(temp);
430 std::getline(data_string, temp, ',');
431 particle_properties[Properties::radius] = std::stod(temp);
432
433 // Compute remaining variables
434 if constexpr (dim == 3)
435 {
436 particle_properties[Properties::volume] =
437 4.0 / 3.0 * M_PI * std::pow(particle_properties[Properties::radius], 3);
438 particle_properties[Properties::mass] =
439 particle_properties[Properties::volume] * particle_properties[Properties::density];
440 particle_properties[Properties::moment_of_inertia] =
441 0.4 * particle_properties[Properties::mass] *
442 std::pow(particle_properties[Properties::radius], 2);
443 }
444 else if constexpr (dim == 2)
445 {
446 particle_properties[Properties::volume] =
447 M_PI * std::pow(particle_properties[Properties::radius], 2);
448 particle_properties[Properties::mass] =
449 particle_properties[Properties::volume] * particle_properties[Properties::density];
450 particle_properties[Properties::moment_of_inertia] =
451 0.5 * particle_properties[Properties::mass] *
452 std::pow(particle_properties[Properties::radius], 2);
453 }
454
455 return {particle_properties, particle_location};
456}

583template <int dim, typename number>
584template <typename VectorizedArrayType>
585dealii::Tensor<1, dim, VectorizedArrayType>
586MeltPoolDG::SphericalParticle<dim, number>::get_velocity(
587 const dealii::Particles::ParticleAccessor<dim> &particle,
588 const dealii::Point<dim, VectorizedArrayType> &location)
589{
590 Assert(dim == 2 or dim == 3, dealii::ExcMessage("Invalid dimension!"));
591
592 dealii::Point<dim, dealii::VectorizedArray<number>> vectorized_particle_location;
593 for (auto i = 0; i < dim; ++i)
594 vectorized_particle_location[i] = dealii::VectorizedArray<number>(particle.get_location()[i]);
595
596 dealii::Tensor<1, dim, VectorizedArrayType> distance_to_center =
597 location - vectorized_particle_location;
598 if constexpr (dim == 2)
599 {
600 return dealii::cross_product_2d(get_angular_velocity(particle)[0] * distance_to_center) +
601 get_velocity<VectorizedArrayType>(particle);
602 }
603 if constexpr (dim == 3)
604 {
605 return get_velocity<VectorizedArrayType>(particle) +
606 dealii::cross_product_3d(get_angular_velocity(particle), distance_to_center);
607 }
608 AssertThrow(false, dealii::ExcInternalError());
609}
