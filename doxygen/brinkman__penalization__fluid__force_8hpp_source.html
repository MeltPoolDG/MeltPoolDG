165{
166 // In order to loop over all obstacles in the property pool we need to ensure that there are no
167 // 'dead' slots inside the property pool which could provide wrong data. In other words we do
168 // not want empty slots resulting from deregistering particles.
169 // TODO: What happens if we removed one handle and added one with a new handle, then the for
170 // loop would fail but the assert wouldn't be triggered!
171 Assert(cell_relevant_obstacles.n_slots() == cell_relevant_obstacles.n_registered_slots(),
172 dealii::ExcInternalError());
173
174 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> fluid_velocity;
175 for (int d = 0; d < dim; ++d)
176 fluid_velocity[d] = w_q[d + 1];
177
178 // TODO Penalty term for mass balance equation?
179
180 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> momentum_fluid_force;
181 for (typename dealii::Particles::PropertyPool<dim>::Handle obstacle_handle = 0;
182 obstacle_handle < cell_relevant_obstacles.n_slots();
183 ++obstacle_handle)
184 {
185 // TODO: Consider obstacle velocity
186 momentum_fluid_force = -mask_function(q_point, cell_relevant_obstacles, obstacle_handle) /
187 brinkman_penalization_data.permeability * fluid_velocity;
188 }
189
190 ConservedVariablesType fluid_force;
191 for (int d = 0; d < dim; ++d)
192 fluid_force[d + 1] = momentum_fluid_force[d];
193 return fluid_force;
