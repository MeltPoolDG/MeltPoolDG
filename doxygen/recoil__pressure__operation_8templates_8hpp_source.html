35 // In order to smoothly activate the recoil pressure, a scaling coefficient is computed and
36 // multiplied with the recoil pressure coefficient
37 //
38 // -
39 // | 0 if T <= T_ac
40 // |
41 // | T - T_ac
42 // scaling_coeff = | -------- if T_ac < T < T_v
43 // | T_v - T_ac
44 // |
45 // | 1 if T >= T_v
46 // -
47 //
48 // with @p T_ac the activation temperature of the recoil pressure and @p T_v the boiling temperature.
49
50 template <typename number>
51 inline VectorizedArray<number>
52 compute_scaling_coeff(const VectorizedArray<number> &T, const number T_ac, const number T_v)
53 {
54 return compare_and_apply_mask<SIMDComparison::less_than>(
55 T,
56 T_ac,
57 0.0,
58 compare_and_apply_mask<SIMDComparison::greater_than_or_equal>(
59 T, T_v, 1., (T - T_ac) / (T_v - T_ac)));
60 }
61
62 template <typename number>
63 inline number
64 compute_scaling_coeff(const number T, const number T_ac, const number T_v)
65 {
66 return (T < T_ac) ? 0.0 : (T >= T_v) ? 1. : (T - T_ac) / (T_v - T_ac);
67 }
