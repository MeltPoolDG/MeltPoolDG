166 template <typename number>
167 dealii::LinearAlgebra::distributed::Vector<number>
168 merge_two_indicator_fields(const dealii::LinearAlgebra::distributed::Vector<number> &indicator_1,
169 const dealii::LinearAlgebra::distributed::Vector<number> &indicator_2,
170 BooleanType type = BooleanType::Union,
171 const number indicator_value_interior = 1.0,
172 const number indicator_value_exterior = -1.0)
173 {
174 AssertThrow(indicator_1.size() == indicator_2.size(),
175 dealii::ExcMessage(
176 "The two level set vectors to be merged must be of equal length."));
177
178 AssertThrow(indicator_value_interior != indicator_value_exterior,
179 dealii::ExcMessage(
180 "The indicator value in the interior of the implicit geometry must be "
181 "different than the one in the exterior. Make sure that the function "
182 "arguments indicator_value_interior and indicator_value_exterior are "
183 "set correctly."));
184
185 AssertThrow(indicator_1.linfty_norm() ==
186 std::max(indicator_value_exterior, indicator_value_interior),
187 dealii::ExcMessage(
188 "The maximum indicator value does not correspond to the given bounds "
189 "of indicator_value_interior and indicator_value_exterior. Make sure that "
190 "the indicator_value_interior and indicator_value_exterior comply with the "
191 "indicator field."));
192
193 dealii::LinearAlgebra::distributed::Vector<number> merge_indicator(indicator_1);
194
195 bool is_interior_larger_than_exterior = indicator_value_interior > indicator_value_exterior;
196
197 for (const auto &i : indicator_1.locally_owned_elements())
198 switch (type)
199 {
200 case BooleanType::Union:
201 merge_indicator[i] = is_interior_larger_than_exterior ?
202 std::max(indicator_1[i], indicator_2[i]) :
203 std::min(indicator_1[i], indicator_2[i]);
204 break;
205 case BooleanType::Intersection:
206 merge_indicator[i] = is_interior_larger_than_exterior ?
207 std::min(indicator_1[i], indicator_2[i]) :
208 std::max(indicator_1[i], indicator_2[i]);
209 break;
210 case BooleanType::Subtraction:
211 merge_indicator[i] =
212 indicator_1[i] == indicator_2[i] && indicator_1[i] == indicator_value_interior ?
213 indicator_value_exterior :
214 indicator_1[i];
215 break;
216 default:
217 AssertThrow(false, dealii::ExcNotImplemented());
218 }
219
220 merge_indicator.compress(dealii::VectorOperation::insert);
221
222 return merge_indicator;
223 }

774 cell->get_dof_values(normal_vector.block(i), buffer.begin(), buffer.end());
775
776 for (unsigned int c = 0; c < cell->get_fe().n_dofs_per_cell(); ++c)
777 buffer_dim[fe_normal.component_to_system_index(i, c)] = buffer[c];
778 }
779
780 // normalize
781 for (unsigned int c = 0; c < cell->get_fe().n_dofs_per_cell(); ++c)
782 {
783 number norm = 0.0;
784 for (int i = 0; i < dim; ++i)
785 norm += dealii::Utilities::fixed_power<2>(
786 buffer_dim[fe_normal.component_to_system_index(i, c)]);
787
788 norm = std::sqrt(norm);
789
790 if (norm > 1e-16)
791 {
792 for (int i = 0; i < dim; ++i)
793 buffer_dim[fe_normal.component_to_system_index(i, c)] /= norm;
794 }
795 }
796
797 phi_normal.evaluate(dealii::make_array_view(buffer_dim), dealii::EvaluationFlags::values);
798 }
799
800 /*
801 * loop over generated points from MCA
802 */
803 for (unsigned int p_index = 0; p_index < real_points.size(); ++p_index)
804 {
805 /*
806 * generate points (x) along normal
807 *
808 * x
809 * |
810 * x
811 * |
812 * --------------- ls = contour_value
813 * |
814 * x
815 * |
816 * x
817 */
818 std::vector<dealii::Point<dim>> points_normal_to_interface;
819
820 const auto unit_normal = phi_normal.get_value(p_index);
821
822 UtilityFunctions::generate_points_along_vector<dim>(points_normal_to_interface,
823 real_points[p_index],
824 unit_normal,
825 max_distance_per_side,
826 n_inc_per_side,
827 bidirectional);
828
829 // check if point is outside domain and if so then project it back to
830 // the domain
831 //
832 // @todo: compute intersection between normal and bounding box and use
833 // this point
834 for (unsigned int d = 0; d < dim; ++d)
835 {
836 for (unsigned int counter = 0; counter < points_normal_to_interface.size(); counter++)
837 {
838 if (points_normal_to_interface[counter][d] < boundary_points.first[d])
839 {
840 points_normal_to_interface.erase(points_normal_to_interface.begin() +
841 counter);
842 continue;
843 }
844 else if (points_normal_to_interface[counter][d] > boundary_points.second[d])
845 points_normal_to_interface.erase(points_normal_to_interface.begin() + counter);
846 }
847 }
848
849 global_points_normal_to_interface.insert(global_points_normal_to_interface.end(),
850 points_normal_to_interface.begin(),
851 points_normal_to_interface.end());
852 global_points_normal_to_interface_pointer.emplace_back(
853 global_points_normal_to_interface.size());
854 }
855 };

858 mapping,
859 level_set_vector,
860 collect_points_along_normal,
861 contour_value, /*contour value*/
862 n_subdivisions_MCA /*n_subdivisions*/);
863
864 if (not level_set_is_ghosted)
865 level_set_vector.zero_out_ghost_values();
866 if (not normal_vector_is_ghosted)
867 normal_vector.zero_out_ghost_values();

888 const bool has_ghost_elements = level_set_heaviside.has_ghost_elements();
889

899 const number min_ls = *std::min_element(hs_local.begin(), hs_local.end());

901 }
902 }
903
904 if (not has_ghost_elements)
905 level_set_heaviside.zero_out_ghost_values();
906
907 // communicate local data to ghost cells

909 dealii::TriangulationType::serial)
910 {
911 using active_cell_iterator = typename dealii::Triangulation<dim>::active_cell_iterator;
912 auto pack = [](const active_cell_iterator &cell) -> unsigned int {
913 return cell->material_id();
914 };
915
916 auto unpack = [](const active_cell_iterator &cell, const unsigned int material_id) -> void {
917 cell->set_material_id(material_id);
918 };
919 if (const auto tria_shared = dynamic_cast<dealii::parallel::shared::Triangulation<dim> *>(
920 const_cast<dealii::Triangulation<dim> *>(&scratch_data.get_triangulation())))
921 {
922 AssertThrow(
923 tria_shared->with_artificial_cells(),
924 dealii::ExcMessage(
925 "For a shared triangulation, make sure that you allow to create artificial cells."
926 "Otherwise every cell is a ghost cell, which prohibits the use of this functionality."));
927 }
928
929 dealii::GridTools::exchange_cell_data_to_ghosts<unsigned int, dealii::Triangulation<dim>>(
930 scratch_data.get_triangulation(), pack, unpack);
931 }
