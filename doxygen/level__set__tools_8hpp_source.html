164 template <typename number>
165 dealii::LinearAlgebra::distributed::Vector<number>
166 merge_two_indicator_fields(const dealii::LinearAlgebra::distributed::Vector<number> &indicator_1,
167 const dealii::LinearAlgebra::distributed::Vector<number> &indicator_2,
168 BooleanType type = BooleanType::Union,
169 const number indicator_value_interior = 1.0,
170 const number indicator_value_exterior = -1.0)
171 {
172 AssertThrow(indicator_1.size() == indicator_2.size(),
173 dealii::ExcMessage(
174 "The two level set vectors to be merged must be of equal length."));
175
176 AssertThrow(indicator_value_interior != indicator_value_exterior,
177 dealii::ExcMessage(
178 "The indicator value in the interior of the implicit geometry must be "
179 "different than the one in the exterior. Make sure that the function "
180 "arguments indicator_value_interior and indicator_value_exterior are "
181 "set correctly."));
182
183 AssertThrow(indicator_1.linfty_norm() ==
184 std::max(indicator_value_exterior, indicator_value_interior),
185 dealii::ExcMessage(
186 "The maximum indicator value does not correspond to the given bounds "
187 "of indicator_value_interior and indicator_value_exterior. Make sure that "
188 "the indicator_value_interior and indicator_value_exterior comply with the "
189 "indicator field."));
190
191 dealii::LinearAlgebra::distributed::Vector<number> merge_indicator(indicator_1);
192
193 bool is_interior_larger_than_exterior = indicator_value_interior > indicator_value_exterior;
194
195 for (const auto &i : indicator_1.locally_owned_elements())
196 switch (type)
197 {
198 case BooleanType::Union:
199 merge_indicator[i] = is_interior_larger_than_exterior ?
200 std::max(indicator_1[i], indicator_2[i]) :
201 std::min(indicator_1[i], indicator_2[i]);
202 break;
203 case BooleanType::Intersection:
204 merge_indicator[i] = is_interior_larger_than_exterior ?
205 std::min(indicator_1[i], indicator_2[i]) :
206 std::max(indicator_1[i], indicator_2[i]);
207 break;
208 case BooleanType::Subtraction:
209 merge_indicator[i] =
210 indicator_1[i] == indicator_2[i] && indicator_1[i] == indicator_value_interior ?
211 indicator_value_exterior :
212 indicator_1[i];
213 break;
214 default:
215 AssertThrow(false, dealii::ExcNotImplemented());
216 }
217
218 merge_indicator.compress(dealii::VectorOperation::insert);
219
220 return merge_indicator;
221 }

312 // point location (at the reference cell) and weight
313 const auto fu_mca = [&]() -> std::tuple<std::vector<dealii::Point<dim>>,
314 std::vector<dealii::Point<dim>>,

390 }
391 }
392 }
393 return {points_real, points, weights};
394 };

413 points,

422 evaluate_at_interface_points(cell, points_real, points, weights);

424 }
425
426 // check if total nodal weights are equal to 1
427 if (dim == 1)
428 for (const auto &[k, w] : total_nodal_weights)
429 AssertThrow(std::abs(w - 1) < 1e-10,

431 "The MCA delivered only one cell corresponding to the corner node " +
432 std::to_string(k) + ". The total nodal weight is " + std::to_string(w) +
433 ". Abort..."));
434
435 if (not is_ghosted)
436 level_set_vector.zero_out_ghost_values();

484 template <int dim, typename number>
485 void
486 generate_points_along_normal(
487 std::vector<dealii::Point<dim>> &global_points_normal_to_interface,
488 std::vector<unsigned int> &global_points_normal_to_interface_pointer,
489 const dealii::DoFHandler<dim> &dof_handler_ls,
490 const dealii::FESystem<dim> &fe_normal,
491 const dealii::Mapping<dim> &mapping,
492 const dealii::LinearAlgebra::distributed::Vector<number> &level_set_vector,
493 const dealii::LinearAlgebra::distributed::BlockVector<number> &normal_vector,
494 const number max_distance_per_side,
495 const unsigned int n_inc_per_side,
496 const bool bidirectional = true,
497 const number contour_value = 0.0,
498 const unsigned int n_subdivisions_MCA = 1)
499 {
500 const bool level_set_is_ghosted = level_set_vector.has_ghost_elements();
501 if (not level_set_is_ghosted)
502 level_set_vector.update_ghost_values();
503
504 const bool normal_vector_is_ghosted = normal_vector.has_ghost_elements();
505 if (not normal_vector_is_ghosted)
506 normal_vector.update_ghost_values();
507
508 dealii::FEPointEvaluation<dim, dim> phi_normal(mapping, fe_normal, dealii::update_values);
509
510 std::vector<number> buffer;
511 std::vector<number> buffer_dim;
512 std::vector<dealii::types::global_dof_index> local_dof_indices;
513
514 global_points_normal_to_interface.clear();
515 global_points_normal_to_interface_pointer.clear();
516
517 global_points_normal_to_interface_pointer = {0};
518
519 const auto bounding_box =
520 dealii::GridTools::compute_bounding_box(dof_handler_ls.get_triangulation());
521 const auto boundary_points = bounding_box.get_boundary_points();
522
523
524 const auto collect_points_along_normal = [&](const auto &cell,
525 const auto &real_points,
526 const auto &unit_points,
527 [[maybe_unused]] const auto &weights) {
528 local_dof_indices.resize(cell->get_fe().n_dofs_per_cell());
529 buffer.resize(cell->get_fe().n_dofs_per_cell());
530 cell->get_dof_indices(local_dof_indices);
531
532 phi_normal.reinit(cell, unit_points);
533 /*
534 * gather_evaluate normal for the points at the interface
535 */
536 {
537 buffer_dim.resize(fe_normal.n_dofs_per_cell());
538 for (int i = 0; i < dim; ++i)
539 {
540 cell->get_dof_values(normal_vector.block(i), buffer.begin(), buffer.end());
541
542 for (unsigned int c = 0; c < cell->get_fe().n_dofs_per_cell(); ++c)
543 buffer_dim[fe_normal.component_to_system_index(i, c)] = buffer[c];
544 }
545
546 // normalize
547 for (unsigned int c = 0; c < cell->get_fe().n_dofs_per_cell(); ++c)
548 {
549 number norm = 0.0;
550 for (int i = 0; i < dim; ++i)
551 norm += dealii::Utilities::fixed_power<2>(
552 buffer_dim[fe_normal.component_to_system_index(i, c)]);
553
554 norm = std::sqrt(norm);
555
556 if (norm > 1e-16)
557 {
558 for (int i = 0; i < dim; ++i)
559 buffer_dim[fe_normal.component_to_system_index(i, c)] /= norm;
560 }
561 }
562
563 phi_normal.evaluate(dealii::make_array_view(buffer_dim), dealii::EvaluationFlags::values);
564 }
565
566 /*
567 * loop over generated points from MCA
568 */
569 for (unsigned int p_index = 0; p_index < real_points.size(); ++p_index)
570 {
571 /*
572 * generate points (x) along normal
573 *
574 * x
575 * |
576 * x
577 * |
578 * --------------- ls = contour_value
579 * |
580 * x
581 * |
582 * x
583 */
584 std::vector<dealii::Point<dim>> points_normal_to_interface;
585
586 const auto unit_normal = phi_normal.get_value(p_index);
587
588 UtilityFunctions::generate_points_along_vector<dim>(points_normal_to_interface,
589 real_points[p_index],
590 unit_normal,
591 max_distance_per_side,
592 n_inc_per_side,
593 bidirectional);
594
595 // check if point is outside domain and if so then project it back to
596 // the domain
597 //
598 // @todo: compute intersection between normal and bounding box and use
599 // this point
600 for (unsigned int d = 0; d < dim; ++d)
601 {
602 for (unsigned int counter = 0; counter < points_normal_to_interface.size(); counter++)
603 {
604 if (points_normal_to_interface[counter][d] < boundary_points.first[d])
605 {
606 points_normal_to_interface.erase(points_normal_to_interface.begin() +
607 counter);
608 continue;
609 }
610 else if (points_normal_to_interface[counter][d] > boundary_points.second[d])
611 points_normal_to_interface.erase(points_normal_to_interface.begin() + counter);
612 }
613 }
614
615 global_points_normal_to_interface.insert(global_points_normal_to_interface.end(),
616 points_normal_to_interface.begin(),
617 points_normal_to_interface.end());
618 global_points_normal_to_interface_pointer.emplace_back(
619 global_points_normal_to_interface.size());
620 }
621 };
622
623 evaluate_at_interface<dim, number>(dof_handler_ls,
624 mapping,
625 level_set_vector,
626 collect_points_along_normal,
627 contour_value, /*contour value*/
628 n_subdivisions_MCA /*n_subdivisions*/);
629
630 if (not level_set_is_ghosted)
631 level_set_vector.zero_out_ghost_values();
632 if (not normal_vector_is_ghosted)
633 normal_vector.zero_out_ghost_values();
634 }
