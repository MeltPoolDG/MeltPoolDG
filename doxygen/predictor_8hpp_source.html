31 template <typename MatrixType>
32 void
33 vmult(const MatrixType &matrix, VectorType &solution, const VectorType &rhs) const
34 {
35 const unsigned int n_old_solutions = old_solution.size();
36 internal_vectors.resize(n_old_solutions);
37 FullMatrix<double> projection_matrix(n_old_solutions, n_old_solutions);
38 unsigned int step = 0;
39 for (; step < n_old_solutions; ++step)
40 {
41 internal_vectors[step].reinit(rhs, true);
42 matrix.vmult(internal_vectors[step], *old_solution[step]);
43 // modified Gram-Schmidt
44 projection_matrix(0, step) = internal_vectors[step] * internal_vectors[0];
45 for (unsigned int j = 0; j < step; ++j)
46 {
47 projection_matrix(j + 1, step) = internal_vectors[step].add_and_dot(
48 -projection_matrix(j, step) /
49 projection_matrix(j, j), // TODO: check for division by zero?
50 internal_vectors[j],
51 internal_vectors[j + 1]);
52 }
53
54 if (projection_matrix(step, step) < 1e-16 * projection_matrix(0, 0))
55 break;
56 }
57
58 // Solve least-squares system
59 std::vector<double> project_sol(step);
60 for (unsigned int s = 0; s < step; ++s)
61 project_sol[s] = internal_vectors[s] * rhs;
62
63 for (int s = step - 1; s >= 0; --s)
64 {
65 double sum = project_sol[s];
66 for (unsigned int j = s + 1; j < step; ++j)
67 sum -= project_sol[j] * projection_matrix(s, j);
68 project_sol[s] = sum / projection_matrix(s, s); // TODO check for division by zero?
69 }
70
71 // extrapolate solution from old values
72 solution.reinit(rhs, true);
73 solution.equ(project_sol[0], *old_solution[0]);
74 for (unsigned int s = 1; s < step; ++s)
75 solution.add(project_sol[s], *old_solution[s]);
76 }
77
78 private:
79 std::vector<const VectorType *> old_solution;
80
81 mutable std::vector<VectorType> internal_vectors;
82 };

117 {
