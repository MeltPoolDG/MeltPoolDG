29 template <int dim>
30 class InitialField : public dealii::Function<dim>
31 {
32 public:
33 explicit InitialField(std::string ic_gas_phase,
34 std::string ic_liquid_phase,
35 const bool gas_phase_is_first = true)
36 : dealii::Function<dim>(2 * (dim + 2))
37 , gas_phase_is_first(gas_phase_is_first)
38 {
39 parsing_function_gas->initialize(dim == 3 ?
40 std::string("x,y,z") :
41 (dim == 2 ? std::string("x,y") : std::string("x")),
42 ic_gas_phase,
43 constants,
44 false);
45 parsing_function_liquid->initialize(dim == 3 ?
46 std::string("x,y,z") :
47 (dim == 2 ? std::string("x,y") : std::string("x")),
48 ic_liquid_phase,
49 constants,
50 false);
51
52 // Currently, only 1D simulations are possible
53 Assert(dim == 1, dealii::ExcNotImplemented());
54 }
55
56 double
57 value(const dealii::Point<dim> &p, const unsigned int component) const final
58 {
59 unsigned int gas_components[dim + 2];
60 unsigned int liquid_components[dim + 2];
61
62 for (unsigned int i = 0; i < dim + 2; ++i)
63 {
64 if (gas_phase_is_first)
65 {
66 gas_components[i] = i;
67 liquid_components[i] = i + dim + 2;
68 }
69 else
70 {
71 gas_components[i] = i + dim + 2;
72 liquid_components[i] = i;
73 }
74 }
75
76 // gas phase
77 if (component == gas_components[0])
78 return parsing_function_gas->value(p, 0);
79 else if (component == gas_components[1])
80 return parsing_function_gas->value(p, 1);
81 else if (component == gas_components[2])
82 return parsing_function_gas->value(p, 2);
83
84 // liquid phase
85 else if (component == liquid_components[0])
86 return parsing_function_liquid->value(p, 0);
87 else if (component == liquid_components[1])
88 return parsing_function_liquid->value(p, 1);
89 else if (component == liquid_components[2])
90 return parsing_function_liquid->value(p, 2);
91 else
92 return 0.;
93 }
94
95 private:
96 bool gas_phase_is_first;
97 std::map<std::string, double> constants;
98 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_gas =
99 std::make_unique<dealii::FunctionParser<dim>>(dim + 2);
100 std::unique_ptr<dealii::FunctionParser<dim>> parsing_function_liquid =
101 std::make_unique<dealii::FunctionParser<dim>>(dim + 2);
102 };

112 void
113 create_spatial_discretization() override
114 {
115 // TODO: no distributed triangulation possible for dim=1
116 this->triangulation =
117 std::make_shared<parallel::shared::Triangulation<dim>>(this->mpi_communicator);
118
119 dealii::Point<dim> lower_left;
120 lower_left[0] = -5.;
121 for (unsigned int d = 1; d < dim; ++d)
122 lower_left[d] = 0.;
123
124 dealii::Point<dim> upper_right;
125 upper_right[0] = 5.;
126 for (unsigned int d = 1; d < dim; ++d)
127 upper_right[d] = 0.;
128
129 std::vector<unsigned int> subdivisions(dim, 1);
130 subdivisions[0] = 20;
131 for (unsigned int d = 1; d < dim; ++d)
132 subdivisions[d] = 1;
133
134 GridGenerator::subdivided_hyper_rectangle(
135 *this->triangulation, subdivisions, lower_left, upper_right, true);
136
137 this->triangulation->refine_global(this->parameters.base.global_refinements);
