20 template <int dim, typename number = double, bool is_gas_phase = true>
21 inline DEAL_II_ALWAYS_INLINE //
22 dealii::VectorizedArray<number>
23 calculate_thermodynamic_pressure(
24 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
25 const CompressibleFlowData<number> &flow_data)
26 {
27 const CompressibleFluidMaterialPhaseData<number> &material_data =
28 is_gas_phase ? flow_data.material.gas : flow_data.material.liquid;
29
30 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
31 MeltPoolDG::Flow::calculate_velocity<dim, number>(conserved_variables);
32
33 switch (material_data.eos_data.type)
34 {
35 case EquationOfState::ideal_gas: {
36 return (material_data.gamma - 1.) *
37 (conserved_variables[dim + 1] -
38 conserved_variables[0] * 0.5 * scalar_product(velocity, velocity));
39 }
40 case EquationOfState::stiffened_gas: {
41 return (material_data.gamma - 1.) *
42 (conserved_variables[dim + 1] -
43 conserved_variables[0] * 0.5 * scalar_product(velocity, velocity)) -
44 material_data.gamma * material_data.eos_data.p_inf;
45 }
46 case EquationOfState::noble_abel_stiffened_gas: {
47 return ((material_data.gamma - 1.) *
48 (conserved_variables[dim + 1] -
49 0.5 * conserved_variables[0] * scalar_product(velocity, velocity) -
50 conserved_variables[0] * material_data.eos_data.q) -
51 material_data.gamma * material_data.eos_data.p_inf *
52 (1. - conserved_variables[0] * material_data.eos_data.b)) /
53 (1. - conserved_variables[0] * material_data.eos_data.b);
54 }
55 default:
56 AssertThrow(false, ExcNotImplemented("The given EOS is not supported."));
57 }
58 }

70 template <int dim, typename number = double, bool is_gas_phase = true>
71 inline DEAL_II_ALWAYS_INLINE //
72 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
73 calculate_grad_T(
74 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
75 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
76 &grad_conserved_variables,
77 const CompressibleFlowData<number> &flow_data)
78 {
79 const CompressibleFluidMaterialPhaseData<number> &material_data =
80 is_gas_phase ? flow_data.material.gas : flow_data.material.liquid;
81
82 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> u =
83 calculate_velocity<dim, number>(conserved_variables);
84 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_u =
85 calculate_grad_velocity<dim, number>(conserved_variables, grad_conserved_variables);
86 const dealii::VectorizedArray<number> rho = conserved_variables[0];
87 const dealii::VectorizedArray<number> inv_rho = dealii::VectorizedArray<number>(1.) / rho;
88 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho =
89 grad_conserved_variables[0];
90 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_E =
91 inv_rho *
92 (grad_conserved_variables[dim + 1] - inv_rho * conserved_variables[dim + 1] * grad_rho);
93
94 switch (material_data.eos_data.type)
95 {
96 case EquationOfState::ideal_gas: {
97 return (material_data.gamma - 1.0) / material_data.specific_gas_constant *
98 (grad_E - grad_u * u);
99 }
100 case EquationOfState::stiffened_gas:
101 case EquationOfState::noble_abel_stiffened_gas: {
102 return material_data.gamma / material_data.specific_isobaric_heat *
103 (grad_E - grad_u * u +
104 material_data.eos_data.p_inf * inv_rho * inv_rho * grad_rho);
105 }
106 default:
107 AssertThrow(false, ExcNotImplemented("The given EOS is not supported."));
108 }
109 }

119 template <int dim, typename number = double, bool is_gas_phase = true>
120 inline DEAL_II_ALWAYS_INLINE //
121 dealii::VectorizedArray<number>
122 calculate_speed_of_sound(
123 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
124 const CompressibleFlowData<number> &flow_data)
125 {
126 const CompressibleFluidMaterialPhaseData<number> &material_data =
127 is_gas_phase ? flow_data.material.gas : flow_data.material.liquid;
128
129 const auto pressure =
130 calculate_thermodynamic_pressure<dim, number, is_gas_phase>(conserved_variables, flow_data);
131 const auto density = conserved_variables[0];
132
133 switch (material_data.eos_data.type)
134 {
135 case EquationOfState::ideal_gas: {
136 return std::sqrt(material_data.gamma * pressure / density);
137 }
138 case EquationOfState::stiffened_gas: {
139 return std::sqrt(material_data.gamma * (pressure + material_data.eos_data.p_inf) /
140 density);
141 }
142 case EquationOfState::noble_abel_stiffened_gas: {
143 return std::sqrt(material_data.gamma * (pressure + material_data.eos_data.p_inf) /
144 (density * (1. - density * material_data.eos_data.b)));
145 }
146 default:
147 AssertThrow(false, ExcNotImplemented("The given EOS is not supported."));
148 }
149 }

159 template <int dim, typename number = double, bool is_gas_phase = true>
160 inline DEAL_II_ALWAYS_INLINE //
161 dealii::VectorizedArray<number>
162 calculate_temperature(
163 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
164 const CompressibleFlowData<number> &flow_data)
165 {
166 const CompressibleFluidMaterialPhaseData<number> &material_data =
167 is_gas_phase ? flow_data.material.gas : flow_data.material.liquid;
168
169 const auto pressure =
170 calculate_thermodynamic_pressure<dim, number, is_gas_phase>(conserved_variables, flow_data);
171 const auto density = conserved_variables[0];
172
173 switch (material_data.eos_data.type)
174 {
175 case EquationOfState::ideal_gas: {
176 return pressure / (material_data.specific_gas_constant * density);
177 }
178 case EquationOfState::stiffened_gas: {
179 return (pressure + material_data.eos_data.p_inf) * material_data.gamma /
180 (material_data.specific_isobaric_heat * density * (material_data.gamma - 1.));
181 }
182 case EquationOfState::noble_abel_stiffened_gas: {
183 return (pressure + material_data.eos_data.p_inf) *
184 (1. / density - material_data.eos_data.b) * material_data.gamma /
185 ((material_data.gamma - 1.) * material_data.specific_isobaric_heat);
186 }
187 default:
188 AssertThrow(false, ExcNotImplemented("The given EOS is not supported."));
189 }

204 template <int dim, typename number = double, bool is_gas_phase = true>
205 inline DEAL_II_ALWAYS_INLINE //
206 dealii::Tensor<2, dim, VectorizedArray<number>>
207 calculate_stress_tensor(
208 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
209 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
210 &grad_conserved_variables,
211 const CompressibleFlowData<number> &flow_data,
212 const auto &viscous_terms)
213 {
214 const auto grad_vel =
215 Flow::calculate_grad_velocity<dim, number>(conserved_variables, grad_conserved_variables);
216
217 const auto viscous_stress_tensor = viscous_terms.calculate_viscous_stress_tensor(grad_vel);
218
219 const auto pressure_tensor =
220 calculate_thermodynamic_pressure<dim, number, is_gas_phase>(conserved_variables, flow_data) *
221 dealii::unit_symmetric_tensor<dim, VectorizedArray<number>>();
222
223 return viscous_stress_tensor - pressure_tensor;
224 }

235 template <int dim, typename number, bool is_gas_phase>
236 inline DEAL_II_ALWAYS_INLINE //
237 CompressibleFlowTypes::ConservedVariablesType<dim, number>
238 convert_conservative_into_primitive_variables(
239 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &u_cons,
240 const Flow::CompressibleFlowData<number> &flow_data)
241 {
242 CompressibleFlowTypes::ConservedVariablesType<dim, number> u_prim;
243
244 // pressure
245 u_prim[0] = calculate_thermodynamic_pressure<dim, number, is_gas_phase>(u_cons, flow_data);
246
247 // velocity
248 for (unsigned int i = 1; i < dim + 1; i++)
249 u_prim[i] = u_cons[i] / u_cons[0];
250
251 // temperature
252 u_prim[dim + 1] = calculate_temperature<dim, number, is_gas_phase>(u_cons, flow_data);
253
254 return u_prim;
255 }

266 template <int dim, typename number, bool is_gas_phase>
267 inline DEAL_II_ALWAYS_INLINE //
268 CompressibleFlowTypes::ConservedVariablesType<dim, number>
269 convert_primitive_into_conservative_variables(
270 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &u_prim,
271 const Flow::CompressibleFlowData<number> &flow_data)
272 {
273 CompressibleFlowTypes::ConservedVariablesType<dim, number> u_cons;
274
275 const CompressibleFluidMaterialPhaseData<number> &material_data =
276 is_gas_phase ? flow_data.material.gas : flow_data.material.liquid;
277
278 switch (material_data.eos_data.type)
279 {
280 case EquationOfState::ideal_gas: {
281 // density
282 u_cons[0] = u_prim[0] / (material_data.specific_gas_constant * u_prim[dim + 1]);
283 // momentum
284 for (unsigned int i = 1; i < dim + 1; i++)
285 u_cons[i] = u_prim[i] * u_cons[0];
286 // total energy
287 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
288 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_cons);
289 u_cons[dim + 1] = u_cons[0] * (material_data.specific_gas_constant /
290 (material_data.gamma - 1.) * u_prim[dim + 1] +
291 0.5 * scalar_product(velocity, velocity));
292 break;
293 }
294 case EquationOfState::stiffened_gas: {
295 // density
296 u_cons[0] =
297 (u_prim[0] + material_data.eos_data.p_inf) * material_data.gamma /
298 (material_data.specific_isobaric_heat * u_prim[dim + 1] * (material_data.gamma - 1.));
299 // momentum
300 for (unsigned int i = 1; i < dim + 1; i++)
301 u_cons[i] = u_prim[i] * u_cons[0];
302 // total energy
303 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
304 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_cons);
305 u_cons[dim + 1] =
306 u_cons[0] *
307 (material_data.specific_isobaric_heat / material_data.gamma * u_prim[dim + 1] +
308 material_data.eos_data.p_inf / u_cons[0] + 0.5 * scalar_product(velocity, velocity));
309 break;
310 }
311 case EquationOfState::noble_abel_stiffened_gas: {
312 // density
313 u_cons[0] = (u_prim[0] + material_data.eos_data.p_inf) /
314 (material_data.specific_isobaric_heat * u_prim[dim + 1] *
315 (material_data.gamma - 1.) / material_data.gamma +
316 material_data.eos_data.b * (u_prim[0] + material_data.eos_data.p_inf));
317 // momentum
318 for (unsigned int i = 1; i < dim + 1; i++)
319 u_cons[i] = u_prim[i] * u_cons[0];
320 // total energy
321 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
322 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_cons);
323 u_cons[dim + 1] =
324 u_cons[0] *
325 (material_data.specific_isobaric_heat / material_data.gamma * u_prim[dim + 1] +
326 material_data.eos_data.p_inf * (1. / u_cons[0] - material_data.eos_data.b) +
327 material_data.eos_data.q + 0.5 * scalar_product(velocity, velocity));
328 break;
329 }
330 default:
331 AssertThrow(false, ExcNotImplemented("The given EOS is not supported."));
332 }
333
334 return u_cons;
335 }

346 template <int dim, typename number>
347 inline DEAL_II_ALWAYS_INLINE //
348 VectorizedArray<number>
349 compute_inner_energy_from_pressure(
350 const VectorizedArray<number> &pressure,
351 const VectorizedArray<number> &density,
352 const Flow::CompressibleFluidMaterialPhaseData<number> &material_data)
353 {
354 switch (material_data.eos_data.type)
355 {
356 case EquationOfState::ideal_gas: {
357 return pressure / (material_data.gamma - 1.);
358 }
359 case EquationOfState::stiffened_gas: {
360 return (pressure + material_data.gamma * material_data.eos_data.p_inf) /
361 (material_data.gamma - 1.);

364 return (pressure + material_data.gamma * material_data.eos_data.p_inf) /
365 (material_data.gamma - 1.) * (1. - density * material_data.eos_data.b) +
366 density * material_data.eos_data.q;
367 }
