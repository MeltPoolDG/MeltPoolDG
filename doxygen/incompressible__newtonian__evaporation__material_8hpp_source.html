106 Tensor<2, dim, VectorizedArray<number>>
107 get_tau() final
108 {
109 const auto mask = compare_and_apply_mask<SIMDComparison::less_than>(
110 hs, 1.0, compare_and_apply_mask<SIMDComparison::greater_than>(hs, 0, 1.0, 0.0), 0);
111
112 return 2. * viscosity *
113 (0.5 * (grad_u + transpose(grad_u)) - mask * div_u * outer_product(normal, normal));
114 }
