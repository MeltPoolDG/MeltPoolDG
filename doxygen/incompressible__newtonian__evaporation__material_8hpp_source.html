99 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
100 get_tau() final

102 const auto mask = compare_and_apply_mask<SIMDComparison::less_than>(
103 hs, 1.0, compare_and_apply_mask<SIMDComparison::greater_than>(hs, 0, 1.0, 0.0), 0);
