71 template <int dim, typename number, int n_components = dim>
72 dealii::Vector<number>
73 convert_matrix_free_cell_aligned_vector_to_vector(
74 const dealii::MatrixFree<dim, number> &mf,
75 const dealii::AlignedVector<dealii::VectorizedArray<number>> &cell_aligned_vector,
76 const dealii::Triangulation<dim> &tria,
77 const unsigned dof_idx = 0,
78 const unsigned quad_idx = 0)
79 {
80 dealii::Vector<number> vec_out(tria.n_active_cells());
81 for (unsigned cell_batch = 0; cell_batch < mf.n_cell_batches(); ++cell_batch)
82 {
83 FECellIntegrator<dim, n_components, number> phi(mf, dof_idx, quad_idx);
84 phi.reinit(cell_batch);
85 const dealii::VectorizedArray<number> cell_batch_indicator =
86 phi.read_cell_data(cell_aligned_vector);
87
88 for (unsigned lane = 0; lane < mf.n_active_entries_per_cell_batch(cell_batch); ++lane)
89 {
90 const auto cell = mf.get_cell_iterator(cell_batch, lane);
91 vec_out(cell->active_cell_index()) = cell_batch_indicator[lane];
92 }
93 }
94 return vec_out;
95 }
