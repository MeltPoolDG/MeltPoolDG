86 number
87 value(const dealii::Point<dim, number> &loc, const unsigned int component) const override
88 {
89 const auto inflow_ramp_up_scaling = [&]() -> number {
90 if (this->get_time() > inflow_ramp_up_duration or inflow_ramp_up_duration == 0.)
91 return 1.;
92
93 switch (ramp_up_type)
94 {
95 case RampUpType::linear:
96 return this->get_time() / inflow_ramp_up_duration;
97 break;
98 case RampUpType::exponential:
99 return (std::exp(this->get_time() / inflow_ramp_up_duration) - 1) /
100 (std::numbers::e - 1);
101 break;
102 case RampUpType::cosine:
103 return 0.5 *
104 (1 - std::cos(std::numbers::pi * this->get_time() / inflow_ramp_up_duration));
105 break;
106 default:
107 AssertThrow(false, dealii::ExcInternalError());
108 }
109 };
110
111 const auto compute_conserved_energy = [&](const dealii::Point<dim, number> &loc) -> number {
112 dealii::Vector<number> velocity_vec(dim);
113 velocity->vector_value(loc, velocity_vec);
114 velocity_vec *= inflow_ramp_up_scaling() * density->value(loc, 0);
115 return density->value(loc, 0) * (energy->value(loc, 0) + 0.5 * velocity_vec.norm_sqr());
116 };
117
118 switch (component)
119 {
120 case 0:
121 return density->value(loc, 0);
122 break;
123 case 1:
124 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 0);
125 break;
126 case 2:
127 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 1);
128 break;
129 case 3:
130 if constexpr (dim == 3)
131 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 2);
132 else
133 return compute_conserved_energy(loc);
134 break;
135 case 4:
136 if constexpr (dim == 3)
137 return compute_conserved_energy(loc);
138 else
139 AssertThrow(false, dealii::ExcInternalError());
140 break;
141 default:
142 AssertThrow(false, dealii::ExcInternalError());
143 }
144 }

205 std::shared_ptr<dealii::Function<dim>>
206 create_boundary_function(const number start_time)
207 {
208 switch (type)
209 {
210 case (Flow::CompressibleBoundaryConditionType::inflow): {
211 auto density_boundary_function =
212 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
213 density_boundary_function->initialize(
214 dealii::FunctionParser<dim>::default_variable_names(),
215 density,
216 typename dealii::FunctionParser<dim>::ConstMap());
217
218 auto velocity_boundary_function = std::make_shared<dealii::FunctionParser<dim>>(dim);
219 velocity_boundary_function->initialize(
220 dealii::FunctionParser<dim>::default_variable_names(),
221 velocity,
222 typename dealii::FunctionParser<dim>::ConstMap());
223
224 auto energy_boundary_function =
225 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
226 energy_boundary_function->initialize(
227 dealii::FunctionParser<dim>::default_variable_names(),
228 energy,
229 typename dealii::FunctionParser<dim>::ConstMap());
230
231 return std::make_shared<ConservativeFunction<dim, number>>(start_time,
232 density_boundary_function,
233 velocity_boundary_function,
234 energy_boundary_function,
235 inflow_ramp_up_duration,
236 inflow_ramp_up_type);
237 }
238 case (Flow::CompressibleBoundaryConditionType::subsonic_outflow_fixed_energy): {
239 auto energy_boundary_function =
240 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
241 energy_boundary_function->initialize(
242 dealii::FunctionParser<dim>::default_variable_names(),
243 energy,
244 typename dealii::FunctionParser<dim>::ConstMap());
245 return energy_boundary_function;
246 }
247 case (Flow::CompressibleBoundaryConditionType::subsonic_outflow_fixed_pressure): {
248 auto pressure_boundary_function =
249 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
250 pressure_boundary_function->initialize(
251 dealii::FunctionParser<dim>::default_variable_names(),
252 pressure,
253 typename dealii::FunctionParser<dim>::ConstMap());
254 return pressure_boundary_function;
255 }
256 default:
257 return std::make_shared<dealii::Functions::ConstantFunction<dim>>(0.);
258 }
