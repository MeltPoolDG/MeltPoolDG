386 template <CellEvaluatorType<dim,
387 dim + 2,
388 number,
389 dealii::VectorizedArray<number>> Integrator>
390 inline DEAL_II_ALWAYS_INLINE //
391 std::tuple<typename CompressibleFlowOperatorBase<dim, number>::ConservedVariablesType,
392 typename CompressibleFlowOperatorBase<dim, number>::ConservedVariablesGradType>
393 CompressibleFlowOperatorBase<dim, number>::rhs_cell_integral_kernel(
394 const Integrator &evaluator,
395 const unsigned int q,
396 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> *constant_body_force) const
397 {
398 const auto w_q = evaluator.get_value(q);
399
400 auto flux = calculator_functions.calculate_convective_flux(w_q);
401
402 if (this->comp_flow_data.dynamic_viscosity > 0)
403 {
404 const auto grad_w_q = evaluator.get_gradient(q);
405 flux -= calculator_functions.calculate_viscous_flux(w_q, grad_w_q);
406 }
407
408 dealii::Tensor<1, dim + 2, dealii::VectorizedArray<number>> forcing;
409
410 if (this->body_force.get() != nullptr)
411 {
412 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> force =
413 constant_body_force ?
414 *constant_body_force :
415 VectorTools::evaluate_function_at_vectorized_points(*this->body_force,
416 evaluator.quadrature_point(q));
417 for (unsigned int d = 0; d < dim; ++d)
418 forcing[d + 1] = w_q[0] * force[d];
419 for (unsigned int d = 0; d < dim; ++d)
420 forcing[dim + 1] += force[d] * w_q[d + 1];
421 }
422
423 return {forcing, flux};
424 }

435 typename CompressibleFlowOperatorBase<dim, number>::ConservedVariablesGradType,
436 typename CompressibleFlowOperatorBase<dim, number>::ConservedVariablesGradType>
437 CompressibleFlowOperatorBase<dim, number>::rhs_face_integral_kernel(
438 const Integrator &evaluator_m,
439 const Integrator &evaluator_p,
440 const unsigned int q,
441 const dealii::VectorizedArray<number> penalty_parameter) const
442 {
443 auto numerical_flux =
444 calculator_functions.calculate_convective_numerical_flux(evaluator_m.get_value(q),
445 evaluator_p.get_value(q),
446 evaluator_m.normal_vector(q));
447
448 if (this->comp_flow_data.dynamic_viscosity > 0)
449 numerical_flux -=
450 calculator_functions.calculate_viscous_numerical_flux(evaluator_m.get_value(q),
451 evaluator_p.get_value(q),
452 evaluator_m.get_gradient(q),
453 evaluator_p.get_gradient(q),
454 evaluator_m.normal_vector(q),
455 penalty_parameter);
456
457 std::pair<ConservedVariablesGradType, ConservedVariablesGradType> viscous_numerical_flux;
458
459 // interior penalty
460 if (this->comp_flow_data.dynamic_viscosity > 0)
461 {
462 viscous_numerical_flux = calculator_functions.calculate_viscous_numerical_flux_gradient(
463 evaluator_m.get_value(q), evaluator_p.get_value(q), evaluator_m.normal_vector(q));
464 }
465
466 return {-numerical_flux,
467 numerical_flux,
468 viscous_numerical_flux.first,
469 viscous_numerical_flux.second};
470 }

498
499 auto flux = calculator_functions.calculate_convective_numerical_flux(w_m, w_p, normal);
500
501 if (this->comp_flow_data.dynamic_viscosity > 0)
502 flux -= calculator_functions.calculate_viscous_numerical_flux(
503 w_m, w_p, grad_w_m, grad_w_p, normal, penalty_parameter);
