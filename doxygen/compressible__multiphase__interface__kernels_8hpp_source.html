77 dim == 1,
78 Flow::Idx1D,
79 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;

81 ConservedVariablesType total_flux_liquid;
82 ConservedVariablesType total_flux_gas;
83
84 // TODO: investigate weighting factors!
85 const dealii::VectorizedArray<number> omega_mass_1 = 0.5;
86 const dealii::VectorizedArray<number> omega_mass_2 = 1. - omega_mass_1;
87
89 // mass conservation //
91
92 const dealii::VectorizedArray<number> interface_mass_flux_conservation_term =
93 (u_liquid[Idx::density] / u_gas[Idx::density] * u_gas[Idx::momentum_x] -
94 u_gas[Idx::density] / u_liquid[Idx::density] * u_liquid[Idx::momentum_x] +
95 (u_gas[Idx::density] * u_gas[Idx::density] -
96 u_liquid[Idx::density] * u_liquid[Idx::density]) /
97 (u_liquid[Idx::density] * u_gas[Idx::density]) *
98 multiphase_scratch_data.phase_coupling.m_dot_evap);

100 total_flux_liquid[Idx::density] = omega_mass_1 * interface_mass_flux_conservation_term;
101 total_flux_gas[Idx::density] = omega_mass_2 * interface_mass_flux_conservation_term;

103 const dealii::VectorizedArray<number> weighted_average_momentum =
104 UtilityFunctions::calculate_arithmetic_phase_weighted_average(omega_mass_1,
105 u_gas[Idx::momentum_x],
106 omega_mass_2,
107 u_liquid[Idx::momentum_x]);
108
109 total_flux_liquid[Idx::density] += weighted_average_momentum;
110 total_flux_gas[Idx::density] -= weighted_average_momentum;

112 // penalty approach for density constraint in gas phase
113 // TODO: compute target density in gas phase from Hertz-Knudsen theory
114 const dealii::VectorizedArray<number> penalty_gas_density =
115 multiphase_scratch_data.phase_coupling.penalty.coefficients.density *
116 (u_gas[Idx::density] -
117 multiphase_scratch_data.phase_coupling.penalty.target_values.density_gas_phase);
118
119 total_flux_gas[Idx::density] += penalty_gas_density;

122 // momentum conservation //
124
125 // TODO: investigate weighting factors!
126 const dealii::VectorizedArray<number> omega_mom_1_conv = 0.5;
127 const dealii::VectorizedArray<number> omega_mom_2_conv = 1. - omega_mom_1_conv;

129 const dealii::VectorizedArray<number> omega_mom_1_visc =
130 multiphase_scratch_data.material_liquid.data.dynamic_viscosity /
131 (multiphase_scratch_data.material_liquid.data.dynamic_viscosity +
132 multiphase_scratch_data.material_gas.data.dynamic_viscosity);
133 const dealii::VectorizedArray<number> omega_mom_2_visc = 1. - omega_mom_1_visc;
134
135 // compute stress tensor (pressure and viscous contributions) and convert to type
136 // dealii::VectorizedArray<number>
137 const auto grad_vel_liquid =
138 Flow::calculate_grad_velocity<dim, number>(u_liquid, grad_u_liquid);
139 const auto grad_vel_gas = Flow::calculate_grad_velocity<dim, number>(u_gas, grad_u_gas);
140
141 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> viscous_stress_tensor_liquid =

147 multiphase_scratch_data.material_liquid.eos_utils->calculate_stress_tensor(

151 u_gas, viscous_stress_tensor_gas)[0][0];
152
153 const dealii::VectorizedArray<number> jump_momentum_term_liquid =
154 u_liquid[Idx::momentum_x] * u_liquid[Idx::momentum_x] / u_liquid[Idx::density];
155 const dealii::VectorizedArray<number> jump_momentum_term_gas =
156 u_gas[Idx::momentum_x] * u_gas[Idx::momentum_x] / u_gas[Idx::density];

158 total_flux_liquid[Idx::momentum_x] +=

160 total_flux_gas[Idx::momentum_x] +=
161 omega_mom_2_conv * (jump_momentum_term_liquid - jump_momentum_term_gas);
162
163 const dealii::VectorizedArray<number> average_momentum_term_1 =
164 UtilityFunctions::calculate_arithmetic_phase_weighted_average(omega_mom_2_conv,
165 jump_momentum_term_liquid,
166 omega_mom_1_conv,
167 jump_momentum_term_gas);
168
169 total_flux_liquid[Idx::momentum_x] += average_momentum_term_1;
170 total_flux_gas[Idx::momentum_x] -= average_momentum_term_1;
171
172 const dealii::VectorizedArray<number> jump_momentum_term_2 =
173 -(u_liquid[Idx::momentum_x] / u_liquid[Idx::density] -
174 u_gas[Idx::momentum_x] / u_gas[Idx::density]) *

177 const dealii::VectorizedArray<number> average_momentum_term_2 =
178 UtilityFunctions::calculate_arithmetic_phase_weighted_average(omega_mom_2_visc,

180 omega_mom_1_visc,
181 -stress_tensor_gas);

184 jump_momentum_term_2 * omega_mom_1_visc + average_momentum_term_2;
185 total_flux_gas[Idx::momentum_x] +=
186 jump_momentum_term_2 * omega_mom_2_visc - average_momentum_term_2;
187
189 // energy conservation //
191
192 // TODO: investigate weighting factors!
193 const dealii::VectorizedArray<number> omega_energy_1_conv = 0.5;
194 const dealii::VectorizedArray<number> omega_energy_2_conv = 1. - omega_energy_1_conv;
195
196 const dealii::VectorizedArray<number> omega_energy_1_visc =
197 multiphase_scratch_data.material_liquid.data.dynamic_viscosity /
198 (multiphase_scratch_data.material_liquid.data.dynamic_viscosity +
199 multiphase_scratch_data.material_gas.data.dynamic_viscosity);

201
202 // compute velocities and convert to VectorizedArray<number>
203 const dealii::VectorizedArray<number> vel_liquid =
204 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_liquid)[0];

206 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_gas)[0];

223 const dealii::VectorizedArray<number> jump_energy_term_2 =
224 -multiphase_scratch_data.phase_coupling.m_dot_evap *

292 template <int dim,
293 typename number,
294 typename ConservedVariablesType,
295 typename ConservedVariablesGradType>
296 inline DEAL_II_ALWAYS_INLINE //
297 std::
298 tuple<ConservedVariablesGradType, ConservedVariablesGradType, dealii::VectorizedArray<number>>
299 calculate_convective_interface_flux_HLLP0(
300 const ConservedVariablesType &u_liquid,
301 const ConservedVariablesType &u_gas,
302 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
303 const auto &convective_terms_liquid,
304 const auto &convective_terms_gas,
305 const Flow::CompressibleMultiphaseScratchData<dim, number> &multiphase_scratch_data)
306 {
307 // Note: Variables, that are relevant for both the liquid and the gas phase, are considered as
308 // arrays of length 2 in the following. The first element refers to the liquid phase and the
309 // second element to the gas phase.
310 constexpr int liquid = 0;
311 constexpr int gas = 1;
312
313 // enumeration for conserved variables component indices
314 using Idx = std::conditional_t<
315 dim == 1,
316 Flow::Idx1D,
317 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
318
319 // 0) preliminaries
320
321 std::array<ConservedVariablesType, 2> u = {{u_liquid, u_gas}};
322 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel;
323 std::array<dealii::VectorizedArray<number>, 2> pressure;
324 std::array<dealii::VectorizedArray<number>, 2> rho;
325 std::array<dealii::VectorizedArray<number>, 2> rho_E;
326 std::array<dealii::VectorizedArray<number>, 2> speed_of_sound;
327 for (unsigned int i : {0, 1})
328 {
329 vel[i] = Flow::calculate_velocity<dim>(u[i]);
330 rho[i] = u[i][Idx::density];
331 rho_E[i] = u[i][Idx::energy];
332 }
333 pressure[liquid] =
334 multiphase_scratch_data.material_liquid.eos_utils->calculate_thermodynamic_pressure(
335 u[liquid]);
336 pressure[gas] =
337 multiphase_scratch_data.material_gas.eos_utils->calculate_thermodynamic_pressure(u[gas]);
338 speed_of_sound[liquid] =
339 multiphase_scratch_data.material_liquid.eos_utils->calculate_speed_of_sound(u[liquid]);
340 speed_of_sound[gas] =
341 multiphase_scratch_data.material_gas.eos_utils->calculate_speed_of_sound(u[gas]);
342
343 // 1) project velocity and kinetic energy into normal direction of the interface
344
345 std::array<dealii::VectorizedArray<number>, 2> vel_n;
346 std::array<dealii::VectorizedArray<number>, 2> E_n;
347 for (unsigned int i : {0, 1})
348 {
349 vel_n[i] = vel[i] * normal;
350 E_n[i] = rho_E[i] / rho[i] - 0.5 * (vel[i] * vel[i] - vel_n[i] * vel_n[i]);
351 }
352
353 // 2) shock speed estimation
354
355 std::array<dealii::VectorizedArray<number>, 2> shock_speed;
356 shock_speed[liquid] = vel_n[liquid] - speed_of_sound[liquid];
357 shock_speed[gas] = vel_n[gas] + speed_of_sound[gas];
358
359 // 3) calculate helpers using Rankine-Hugoniot conditions
360
361 std::array<dealii::VectorizedArray<number>, 2> m_hat;
362 std::array<dealii::VectorizedArray<number>, 2> I_hat;
363 std::array<dealii::VectorizedArray<number>, 2> E_hat;
364 for (unsigned int i : {0, 1})
365 {
366 m_hat[i] = rho[i] * (vel_n[i] - shock_speed[i]);
367 I_hat[i] = m_hat[i] * vel_n[i] + pressure[i];
368 E_hat[i] = m_hat[i] * E_n[i] + pressure[i] * vel_n[i];
369 }
370
371 // 4) calculate intermediate velocity states
372
373 // TODO: consider surface tension for dim>1 here
374 const dealii::VectorizedArray<number> delta_p = 0.;
375
376 // TODO: consider Hertz-Knudsen theory for evaporation mass flux here
377 std::array<dealii::VectorizedArray<number>, 2> tmp_1;
378 std::array<dealii::VectorizedArray<number>, 2> tmp_2;
379 for (unsigned int i : {0, 1})
380 {
381 tmp_1[i] = multiphase_scratch_data.phase_coupling.m_dot_evap / m_hat[i];
382 tmp_2[i] = multiphase_scratch_data.phase_coupling.m_dot_evap - m_hat[i];
383 }
384
385 std::array<dealii::VectorizedArray<number>, 2> numerator;
386 std::array<dealii::VectorizedArray<number>, 2> denominator;
387
388 numerator[liquid] = tmp_2[gas] *
389 (tmp_1[liquid] * shock_speed[liquid] - tmp_1[gas] * shock_speed[gas]) /
390 (1. - tmp_1[gas]) -
391 I_hat[liquid] - delta_p + I_hat[gas];
392 numerator[gas] = tmp_2[liquid] *
393 (tmp_1[gas] * shock_speed[gas] - tmp_1[liquid] * shock_speed[liquid]) /
394 (1. - tmp_1[liquid]) -
395 I_hat[gas] + delta_p + I_hat[liquid];
396 denominator[liquid] = tmp_2[liquid] - (1. - tmp_1[liquid]) / (1. - tmp_1[gas]) * tmp_2[gas];
397 denominator[gas] = tmp_2[gas] - (1. - tmp_1[gas]) / (1. - tmp_1[liquid]) * tmp_2[liquid];
398
399 std::array<dealii::VectorizedArray<number>, 2> vel_n_star;
400 for (unsigned int i : {0, 1})
401 vel_n_star[i] = numerator[i] / denominator[i];
402
403 // 5) calculate intermediate pressure
404
405 std::array<dealii::VectorizedArray<number>, 2> pressure_star;
406 for (unsigned int i : {0, 1})
407 pressure_star[i] = I_hat[i] - m_hat[i] * vel_n_star[i];
408
409 // 6) re-project normal velocity to Cartesian coordinates
410
411 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel_star_cartesian;
412
413 std::vector<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> tangent;
414 tangent.resize(dim - 1);
415
416 // compute tangential vector for dim=2 and dim=3
417 if constexpr (dim == 2)
418 {
419 tangent[0][0] = normal[1];
420 tangent[0][1] = -normal[0];
421 }
422 else if constexpr (dim == 3)
423 {
424 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec;
425 temp_vec[0] = 1.;
426 // if normal vector is identical with unit vector choose different unit vector to
427 // compute the tangent
428 dealii::VectorizedArray<number> tolerance = 1.e-10;
429 dealii::VectorizedArray<number> norm_diff = (temp_vec - normal).norm();
430 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec_y;
431 temp_vec_y[1] = 1.;
432 for (int i = 0; i < 3; ++i)
433 {
434 temp_vec[i] = compare_and_apply_mask<dealii::SIMDComparison::less_than>(norm_diff,
435 tolerance,
436 temp_vec_y[i],
437 temp_vec[i]);
438 }
439 tangent[0] = temp_vec - (temp_vec * normal) * normal;
440 tangent[1] = dealii::cross_product_3d(normal, tangent[0]);
441 }
442
443 for (unsigned int i : {0, 1})
444 {
445 vel_star_cartesian[i] = vel_n_star[i] * normal;
446 for (unsigned int j = 0; j < dim - 1; ++j)
447 vel_star_cartesian[i] += (vel[i] * tangent[j]) * tangent[j];
448 }
449
450 // 7) calculate conservative variable state vectors of inner states
451
452 std::array<ConservedVariablesType, 2> u_star;
453 for (unsigned int i : {0, 1})
454 {
455 u_star[i][Idx::density] = m_hat[i] / (vel_n_star[i] - shock_speed[i]);
456 for (unsigned int j = 1; j < dim + 1; j++)
457 u_star[i][j] = u_star[i][Idx::density] * vel_star_cartesian[i][0][j - 1];
458 u_star[i][Idx::energy] =
459 (E_hat[i] - pressure_star[i] * vel_n_star[i]) / (vel_n_star[i] - shock_speed[i]) -
460 0.5 * u_star[i][Idx::density] * vel_n_star[i] * vel_n_star[i] +
461 0.5 * u_star[i][Idx::density] * vel_star_cartesian[i] * vel_star_cartesian[i];
462 }
463
464 // 8) calculate phase interface velocity
465
466 dealii::VectorizedArray<number> numerator_normal_vel =
467 vel_n_star[liquid] * u_star[liquid][Idx::density] -
468 vel_n_star[gas] * u_star[gas][Idx::density];
469 dealii::VectorizedArray<number> denominator_normal_vel =
470 u_star[liquid][Idx::density] - u_star[gas][Idx::density];
471 // avoid division by zero
472 dealii::VectorizedArray<number> normal_velocity_interface =
473 compare_and_apply_mask<dealii::SIMDComparison::greater_than>(std::abs(denominator_normal_vel),
474 1.e-12,
475 numerator_normal_vel /
476 denominator_normal_vel,
477 vel_n_star[liquid]);
478
479 // 9) calculate fluxes for the two phases
480
481 std::array<ConservedVariablesGradType, 2> flux;
482 std::array<ConservedVariablesGradType, 2> conv_flux;
483 std::array<ConservedVariablesGradType, 2> shock_flux;
484
485 conv_flux[liquid] = convective_terms_liquid.calculate_convective_flux(u[liquid]);
486 conv_flux[gas] = convective_terms_gas.calculate_convective_flux(u[gas]);
487
488 for (unsigned int i : {0, 1})
489 {
490 shock_flux[i] = dyadic_product(shock_speed[i] * (u_star[i] - u[i]), normal);
491 flux[i] = conv_flux[i];
492 }
493
494 const auto zero_vec = dealii::make_vectorized_array(0.);
495 const auto one_vec = dealii::make_vectorized_array(1.);
496
497 flux[liquid] +=
498 shock_flux[liquid] * compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
499 shock_speed[liquid], zero_vec, zero_vec, one_vec);
500 flux[gas] +=
501 shock_flux[gas] * compare_and_apply_mask<dealii::SIMDComparison::less_than_or_equal>(
502 shock_speed[gas], zero_vec, zero_vec, one_vec);
503
504 return {flux[liquid], flux[gas], normal_velocity_interface};
505 }

637 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
638
639 // TODO: add contributions for surface tension, interface heat source (laser energy) and
640 // Marangoni forces
641
642 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_liquid =
643 Flow::calculate_velocity<dim>(u_liquid);
644 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_gas =
645 Flow::calculate_velocity<dim>(u_gas);
646
647 const dealii::VectorizedArray<number> vel_n_liquid = vel_liquid * normal;
648 const dealii::VectorizedArray<number> vel_n_gas = vel_gas * normal;
649
650 const auto pressure_liquid =
651 multiphase_scratch_data.material_liquid.eos_utils->calculate_thermodynamic_pressure(u_liquid);
652 const auto pressure_gas =
653 multiphase_scratch_data.material_gas.eos_utils->calculate_thermodynamic_pressure(u_gas);
654
655 // compute Robin-type viscous interface jump conditions
656
657 ConservedVariablesType J_Rob;
658
659 // TODO: Add entries for case dim>1
660 J_Rob[Idx::energy] = (pressure_liquid * vel_n_liquid - pressure_gas * vel_n_gas) +
661 multiphase_scratch_data.phase_coupling.m_dot_evap *
662 (u_liquid[Idx::energy] / u_liquid[Idx::density] -
663 u_gas[Idx::energy] / u_gas[Idx::density]) -
664 multiphase_scratch_data.phase_coupling.delta_q;
665
666 const ConservedVariablesGradType viscous_flux_liquid =
667 viscous_terms_liquid.calculate_viscous_flux(u_liquid, grad_u_liquid);
668
669 const ConservedVariablesGradType viscous_flux_gas =
670 viscous_terms_gas.calculate_viscous_flux(u_gas, grad_u_gas);
671
672 ConservedVariablesGradType total_flux_liquid = dyadic_product(J_Rob, normal);
673 total_flux_liquid += viscous_flux_gas;
674 total_flux_liquid *= visc_ave_weight_phase_liquid;
675 total_flux_liquid += visc_ave_weight_phase_gas * viscous_flux_liquid;
676
677 ConservedVariablesGradType total_flux_gas =
678 dyadic_product(J_Rob, -normal); // opposite normal direction for phase 2
679 total_flux_gas += viscous_flux_liquid;
680 total_flux_gas *= visc_ave_weight_phase_gas;
681 total_flux_gas += visc_ave_weight_phase_liquid * viscous_flux_gas;
682
683 // penalty term
684
685 const auto J_Dir_cons =
686 calculate_Dirichlet_jump_in_conservative_variables<dim, number, ConservedVariablesType>(
687 u_liquid, u_gas, multiphase_scratch_data);
688
689 const number penalty_parameter =
690 std::min(multiphase_scratch_data.material_liquid.data.dynamic_viscosity /
691 multiphase_scratch_data.material_liquid.data.reference_density,
692 multiphase_scratch_data.material_gas.data.dynamic_viscosity /
693 multiphase_scratch_data.material_gas.data.reference_density) *
694 (multiphase_scratch_data.flow_data.fe.degree + 1.) *
695 (multiphase_scratch_data.flow_data.fe.degree + 1.) / cell_size * tau;
696
697 ConservedVariablesGradType penalty_flux_liquid;
698 const auto tmp_m = u_liquid - (u_gas + J_Dir_cons);
699 penalty_flux_liquid = dyadic_product(tmp_m, normal);
700 penalty_flux_liquid *= penalty_parameter;
701
702 ConservedVariablesGradType penalty_flux_gas;
703 const auto tmp_p = u_gas - (u_liquid - J_Dir_cons);
704 penalty_flux_gas = dyadic_product(tmp_p, -normal);
705 penalty_flux_gas *= penalty_parameter;
706
707 total_flux_liquid -= penalty_flux_liquid;
708 total_flux_gas -= penalty_flux_gas;
709
710 return {contract_tensor_with_vector<dim + 2, dim, number>(total_flux_liquid, normal),
711 contract_tensor_with_vector<dim + 2, dim, number>(total_flux_gas, normal)};
712 }

842 multiphase_scratch_data.material_liquid.eos_utils->calculate_thermodynamic_pressure(u_liquid);
843 const dealii::VectorizedArray<number> pressure_gas =
844 multiphase_scratch_data.material_gas.eos_utils->calculate_thermodynamic_pressure(u_gas);

846 // compute Robin-type viscous interface jump conditions

866 (multiphase_scratch_data.material_liquid.data.thermal_conductivity +
867 multiphase_scratch_data.material_gas.data.thermal_conductivity) /
868 (2. * cell_size) *
869 ((multiphase_scratch_data.material_liquid.eos_utils->calculate_temperature(u_liquid) -
870 multiphase_scratch_data.material_gas.eos_utils->calculate_temperature(u_gas)) -

877 visc_ave_weight_phase_liquid,
878 contract_tensor_with_vector<dim + 2, dim, number>(viscous_flux_gas, normal));
