73 using Idx = std::conditional_t<
74 dim == 1,
75 Flow::Idx1D,

81 // TODO: investigate weighting factors!

83 const VectorizedArray<number> omega_2 = 0.5;

86 // mass conservation //
88
89 const VectorizedArray<number> interface_mass_flux_conservation_term =
90 (u_liquid[Idx::density] / u_gas[Idx::density] * u_gas[Idx::momentum_x] -
91 u_gas[Idx::density] / u_liquid[Idx::density] * u_liquid[Idx::momentum_x] +
92 (u_gas[Idx::density] * u_gas[Idx::density] -
93 u_liquid[Idx::density] * u_liquid[Idx::density]) /
94 (u_liquid[Idx::density] * u_gas[Idx::density]) *

97 total_flux_liquid[Idx::density] = omega_1 * interface_mass_flux_conservation_term;
98 total_flux_gas[Idx::density] = omega_2 * interface_mass_flux_conservation_term;

101 UtilityFunctions::calculate_arithmetic_phase_weighted_average(omega_1,
102 u_gas[Idx::momentum_x],
103 omega_2,
104 u_liquid[Idx::momentum_x]);
105
106 total_flux_liquid[Idx::density] += weighted_average_momentum;
107 total_flux_gas[Idx::density] -= weighted_average_momentum;
108
109 // penalty approach for density constraint in gas phase
110 // TODO: compute target density in gas phase from Hertz-Knudsen theory
111 const VectorizedArray<number> penalty_gas_density =
112 flow_data.interface_jump_conditions.penalty.coefficients.density *
113 (u_gas[Idx::density] -
114 flow_data.interface_jump_conditions.penalty.target_values.density_gas_phase);

116 total_flux_gas[Idx::density] += penalty_gas_density;

121
122 const VectorizedArray<number> jump_momentum_term =
123 (u_liquid[Idx::momentum_x] * u_liquid[Idx::momentum_x] / u_liquid[Idx::density] -
124 u_gas[Idx::momentum_x] * u_gas[Idx::momentum_x] / u_gas[Idx::density]) -
125 (1. / u_liquid[Idx::density] - 1. / u_gas[Idx::density]) *
126 flow_data.interface_jump_conditions.m_dot_evap *
127 flow_data.interface_jump_conditions.m_dot_evap;

129 total_flux_liquid[Idx::momentum_x] = jump_momentum_term * omega_1;
130 total_flux_gas[Idx::momentum_x] = jump_momentum_term * omega_2;

132 // compute stress tensor (pressure and viscous contributions) and convert to type
133 // VectorizedArray<number>
134 const VectorizedArray<number> stress_tensor_liquid =
135 Flow::EOS::calculate_stress_tensor<dim, number, false /*is_gas_phase*/>(

148 omega_2,
149 weighted_average_momentum_term_liquid,
150 omega_1,
151 weighted_average_momentum_term_gas);
152

154 total_flux_gas[Idx::momentum_x] -= weighted_average_momentum_term;

161 const VectorizedArray<number> vel_liquid =
162 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_liquid)[0];
163 const VectorizedArray<number> vel_gas =
164 MeltPoolDG::Flow::calculate_velocity<dim, number>(u_gas)[0];

166 const VectorizedArray<number> delta_q = 0.;
167 const VectorizedArray<number> jump_energy_term =

169 flow_data.interface_jump_conditions.m_dot_evap *

171 u_gas[Idx::energy] / u_gas[Idx::density]) +

236 template <int dim,
237 typename number,
238 typename ConservedVariablesType,
239 typename ConservedVariablesGradType>
240 inline DEAL_II_ALWAYS_INLINE //
241 std::tuple<ConservedVariablesType, ConservedVariablesType, VectorizedArray<number>>
242 calculate_convective_interface_flux_HLLP0(const ConservedVariablesType &u_liquid,
243 const ConservedVariablesType &u_gas,
244 const Tensor<1, dim, VectorizedArray<number>> &normal,
245 const auto &convective_terms_liquid,
246 const auto &convective_terms_gas,
247 const Flow::CompressibleFlowData<number> &flow_data)
248 {
249 // Note: Variables, that are relevant for both the liquid and the gas phase ,are considered as
250 // arrays of length 2 in the following. The first element refers to the liquid phase and the
251 // second element to the gas phase.
252 constexpr int liquid = 0;
253 constexpr int gas = 1;
254
255 // enumeration for conserved variables component indices
256 using Idx = std::conditional_t<
257 dim == 1,
258 Flow::Idx1D,
259 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
260
261 // 0) preliminaries
262
263 std::array<ConservedVariablesType, 2> u = {{u_liquid, u_gas}};
264 std::array<Tensor<1, dim, VectorizedArray<number>>, 2> vel;
265 std::array<VectorizedArray<number>, 2> pressure;
266 std::array<VectorizedArray<number>, 2> rho;
267 std::array<VectorizedArray<number>, 2> rho_E;
268 std::array<VectorizedArray<number>, 2> speed_of_sound;
269 for (unsigned int i : {0, 1})
270 {
271 vel[i] = Flow::calculate_velocity<dim>(u[i]);
272 rho[i] = u[i][0];
273 rho_E[i] = u[i][dim + 1];
274 }
275 pressure[liquid] =
276 Flow::EOS::calculate_thermodynamic_pressure<dim, number, false /*is_gas_phase*/>(u[liquid],
277 flow_data);
278 pressure[gas] =
279 Flow::EOS::calculate_thermodynamic_pressure<dim, number, true /*is_gas_phase*/>(u[gas],
280 flow_data);
281 speed_of_sound[liquid] =
282 Flow::EOS::calculate_speed_of_sound<dim, number, false /*is_gas_phase*/>(u[liquid],
283 flow_data);
284 speed_of_sound[gas] =
285 Flow::EOS::calculate_speed_of_sound<dim, number, true /*is_gas_phase*/>(u[gas], flow_data);
286
287 // 1) project velocity and kinetic energy into normal direction of the interface
288
289 std::array<VectorizedArray<number>, 2> vel_n;
290 std::array<VectorizedArray<number>, 2> E_n;
291 for (unsigned int i : {0, 1})
292 {
293 vel_n[i] = vel[i] * normal;
294 E_n[i] = rho_E[i] / rho[i] - 0.5 * (vel[i] * vel[i] - vel_n[i] * vel_n[i]);
295 }
296
297 // 2) shock speed estimation
298
299 std::array<VectorizedArray<number>, 2> shock_speed;
300 shock_speed[liquid] = vel_n[liquid] - speed_of_sound[liquid];
301 shock_speed[gas] = vel_n[gas] + speed_of_sound[gas];
302
303 // 3) calculate helpers using Rankine-Hugoniot conditions
304
305 std::array<VectorizedArray<number>, 2> m_hat;
306 std::array<VectorizedArray<number>, 2> I_hat;
307 std::array<VectorizedArray<number>, 2> E_hat;
308 for (unsigned int i : {0, 1})
309 {
310 m_hat[i] = rho[i] * (vel_n[i] - shock_speed[i]);
311 I_hat[i] = m_hat[i] * vel_n[i] + pressure[i];
312 E_hat[i] = m_hat[i] * E_n[i] + pressure[i] * vel_n[i];
313 }
314
315 // 4) calculate intermediate velocity states
316
317 // TODO: consider surface tension for dim>1 here
318 const VectorizedArray<number> delta_p = 0.;
319
320 // TODO: consider Hertz-Knudsen theory for evaporation mass flux here
321 std::array<VectorizedArray<number>, 2> tmp_1;
322 std::array<VectorizedArray<number>, 2> tmp_2;
323 for (unsigned int i : {0, 1})
324 {
325 tmp_1[i] = flow_data.interface_jump_conditions.m_dot_evap / m_hat[i];
326 tmp_2[i] = flow_data.interface_jump_conditions.m_dot_evap - m_hat[i];
327 }
328
329 std::array<VectorizedArray<number>, 2> numerator;
330 std::array<VectorizedArray<number>, 2> denominator;
331
332 numerator[liquid] = tmp_2[gas] *
333 (tmp_1[liquid] * shock_speed[liquid] - tmp_1[gas] * shock_speed[gas]) /
334 (1. - tmp_1[gas]) -
335 I_hat[liquid] - delta_p + I_hat[gas];
336 numerator[gas] = tmp_2[liquid] *
337 (tmp_1[gas] * shock_speed[gas] - tmp_1[liquid] * shock_speed[liquid]) /
338 (1. - tmp_1[liquid]) -
339 I_hat[gas] + delta_p + I_hat[liquid];
340 denominator[liquid] = tmp_2[liquid] - (1. - tmp_1[liquid]) / (1. - tmp_1[gas]) * tmp_2[gas];
341 denominator[gas] = tmp_2[gas] - (1. - tmp_1[gas]) / (1. - tmp_1[liquid]) * tmp_2[liquid];
342
343 std::array<VectorizedArray<number>, 2> vel_n_star;
344 for (unsigned int i : {0, 1})
345 vel_n_star[i] = numerator[i] / denominator[i];
346
347 // 5) calculate intermediate pressure
348
349 std::array<VectorizedArray<number>, 2> pressure_star;
350 for (unsigned int i : {0, 1})
351 pressure_star[i] = I_hat[i] - m_hat[i] * vel_n_star[i];
352
353 // 6) re-project normal velocity to Cartesian coordinates
354
355 std::array<Tensor<1, dim, VectorizedArray<number>>, 2> vel_star_cartesian;
356
357 std::vector<Tensor<1, dim, VectorizedArray<number>>> tangent;
358 tangent.resize(dim - 1);
359
360 // compute tangential vector for dim=2 and dim=3
361 if constexpr (dim == 2)
362 {
363 tangent[0][0] = normal[1];
364 tangent[0][1] = -normal[0];
365 }
366 else if constexpr (dim == 3)
367 {
368 Tensor<1, dim, VectorizedArray<number>> temp_vec;
369 temp_vec[0] = 1.;
370 // if normal vector is identical with unit vector choose different unit vector to
371 // compute the tangent
372 VectorizedArray<number> tolerance = 1.e-10;
373 VectorizedArray<number> norm_diff = (temp_vec - normal).norm();
374 Tensor<1, dim, VectorizedArray<number>> temp_vec_y;
375 temp_vec_y[1] = 1.;
376 for (int i = 0; i < 3; ++i)
377 {
378 temp_vec[i] = compare_and_apply_mask<SIMDComparison::less_than>(norm_diff,
379 tolerance,
380 temp_vec_y[i],
381 temp_vec[i]);
382 }
383 tangent[0] = temp_vec - (temp_vec * normal) * normal;
384 tangent[1] = dealii::cross_product_3d(normal, tangent[0]);
385 }
386
387 for (unsigned int i : {0, 1})
388 {
389 vel_star_cartesian[i] = vel_n_star[i] * normal;
390 for (unsigned int j = 0; j < dim - 1; ++j)
391 vel_star_cartesian[i] += (vel[i] * tangent[j]) * tangent[j];
392 }
393
394 // 7) calculate conservative variable state vectors of inner states
395
396 std::array<ConservedVariablesType, 2> u_star;
397 for (unsigned int i : {0, 1})
398 {
399 u_star[i][Idx::density] = m_hat[i] / (vel_n_star[i] - shock_speed[i]);
400 for (unsigned int j = 1; j < dim + 1; j++)
401 u_star[i][j] = u_star[i][Idx::density] * vel_star_cartesian[i][j - 1];
402 u_star[i][Idx::energy] =
403 (E_hat[i] - pressure_star[i] * vel_n_star[i]) / (vel_n_star[i] - shock_speed[i]) -
404 0.5 * u_star[i][Idx::density] * vel_n_star[i] * vel_n_star[i] +
405 0.5 * u_star[i][Idx::density] * vel_star_cartesian[i] * vel_star_cartesian[i];
406 }
407
408 // 8) calculate phase interface velocity
409
410 VectorizedArray<number> numerator_normal_vel =
411 vel_n_star[liquid] * u_star[liquid][Idx::density] -
412 vel_n_star[gas] * u_star[gas][Idx::density];
413 VectorizedArray<number> denominator_normal_vel =
414 u_star[liquid][Idx::density] - u_star[gas][Idx::density];
415 // avoid division by zero
416 VectorizedArray<number> normal_velocity_interface =
417 compare_and_apply_mask<SIMDComparison::greater_than>(std::abs(denominator_normal_vel),
418 1.e-12,
419 numerator_normal_vel /
420 denominator_normal_vel,
421 0.);
422
423 // 9) calculate fluxes for the two phases
424
425 std::array<ConservedVariablesType, 2> flux;
426 std::array<ConservedVariablesGradType, 2> conv_flux;
427 std::array<ConservedVariablesType, 2> shock_flux;
428
429 conv_flux[liquid] = convective_terms_liquid.calculate_convective_flux(u[liquid]);
430 conv_flux[gas] = convective_terms_gas.calculate_convective_flux(u[gas]);
431
432 for (unsigned int i : {0, 1})
433 {
434 shock_flux[i] = shock_speed[i] * (u_star[i] - u[i]);
435 flux[i] =
436 UtilityFunctions::contract_tensor_with_vector<dim + 2, dim, number>(conv_flux[i], normal);
437 }
438
439 const auto zero_vec = dealii::make_vectorized_array(0.);
440 const auto one_vec = dealii::make_vectorized_array(1.);
441
442 flux[liquid] += shock_flux[liquid] * compare_and_apply_mask<SIMDComparison::greater_than>(
443 shock_speed[liquid], zero_vec, one_vec, zero_vec);
444 flux[gas] += shock_flux[gas] * compare_and_apply_mask<SIMDComparison::less_than_or_equal>(
445 shock_speed[gas], zero_vec, one_vec, zero_vec);
446
447 return {flux[liquid], flux[gas], normal_velocity_interface};
448 }

550 inline DEAL_II_ALWAYS_INLINE //
551 std::pair<ConservedVariablesType, ConservedVariablesType>
552 calculate_viscous_interface_flux(const ConservedVariablesType &u_liquid,
553 const ConservedVariablesType &u_gas,
554 const ConservedVariablesGradType &grad_u_liquid,
555 const ConservedVariablesGradType &grad_u_gas,
556 const Tensor<1, dim, VectorizedArray<number>> &normal,
557 const number &alpha_1,
558 const number &alpha_2,
559 const number &tau,
560 const auto &viscous_terms_liquid,
561 const auto &viscous_terms_gas,
562 const Flow::CompressibleFlowData<number> &flow_data)
563 {
564 // enumeration for conserved variables component indices
565 using Idx = std::conditional_t<
566 dim == 1,
567 Flow::Idx1D,
568 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
569
570 // TODO: add contributions for surface tension, interface heat source (laser energy) and
571 // Marangoni forces
572
573 const Tensor<1, dim, VectorizedArray<number>> vel_liquid =
574 Flow::calculate_velocity<dim>(u_liquid);
575 const Tensor<1, dim, VectorizedArray<number>> vel_gas = Flow::calculate_velocity<dim>(u_gas);
576
577 const VectorizedArray<number> vel_n_liquid = vel_liquid * normal;
578 const VectorizedArray<number> vel_n_gas = vel_gas * normal;
579
580 const auto pressure_liquid =
581 Flow::EOS::calculate_thermodynamic_pressure<dim, number, true>(u_liquid, flow_data);
582 const auto pressure_p =
583 Flow::EOS::calculate_thermodynamic_pressure<dim, number, false>(u_gas, flow_data);
584
585 const auto grad_vel_liquid = Flow::calculate_grad_velocity(u_liquid, grad_u_liquid);
586 const auto grad_vel_p = Flow::calculate_grad_velocity(u_gas, grad_u_gas);
587
588 const Tensor<2, dim, VectorizedArray<number>> stress_tensor_liquid =
589 viscous_terms_liquid.calculate_viscous_stress_tensor(grad_vel_liquid);
590 const Tensor<2, dim, VectorizedArray<number>> stress_tensor_p =
591 viscous_terms_gas.calculate_viscous_stress_tensor(grad_vel_p);
592
593 // compute Robin-type viscous interface jump conditions
594
595 ConservedVariablesType J_Rob;
596
597 // TODO: Add entries for case dim>1
598 const VectorizedArray<number> delta_q = 0.;
599 J_Rob[Idx::energy] = (stress_tensor_liquid * vel_liquid - stress_tensor_p * vel_gas) * normal -
600 (pressure_liquid * vel_n_liquid - pressure_p * vel_n_gas) -
601 flow_data.interface_jump_conditions.m_dot_evap *
602 (u_liquid[Idx::energy] / u_liquid[Idx::density] -
603 u_gas[Idx::energy] / u_gas[Idx::density]) +
604 delta_q;
605
606 const ConservedVariablesGradType viscous_flux_liquid =
607 viscous_terms_liquid.calculate_viscous_flux(u_liquid, grad_u_liquid);
608
609 const ConservedVariablesGradType viscous_flux_gas =
610 viscous_terms_gas.calculate_viscous_flux(u_gas, grad_u_gas);
611
612 ConservedVariablesGradType total_flux_liquid = UtilityFunctions::dyadic_product(J_Rob, normal);
613 total_flux_liquid += viscous_flux_gas;
614 total_flux_liquid *= alpha_1;
615 total_flux_liquid += alpha_2 * viscous_flux_liquid;
616
617 ConservedVariablesGradType total_flux_gas =
618 UtilityFunctions::dyadic_product(J_Rob, -normal); // opposite normal direction for phase 2
619 total_flux_gas += viscous_flux_liquid;
620 total_flux_gas *= alpha_2;
621 total_flux_gas += alpha_1 * viscous_flux_gas;
622
623 // penalty term
624
625 const auto J_Dir_cons =
626 calculate_Dirichlet_jump_in_conservative_variables<dim, number, ConservedVariablesType>(
627 u_liquid, u_gas, flow_data);
628
629 const auto u_liquid_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
630 alpha_1, u_liquid, alpha_2, u_gas + J_Dir_cons);
631 const auto u_gas_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
632 alpha_2, u_gas, alpha_1, u_liquid - J_Dir_cons);
633
634 ConservedVariablesGradType penalty_flux_liquid;
635 const auto tmp_m = u_liquid - (u_gas_star + J_Dir_cons);
636 penalty_flux_liquid = UtilityFunctions::dyadic_product(tmp_m, normal);
637 penalty_flux_liquid *= tau;
638
639 ConservedVariablesGradType penalty_flux_gas;
640 const auto tmp_p = u_gas - (u_liquid_star - J_Dir_cons);
641 penalty_flux_gas = UtilityFunctions::dyadic_product(tmp_p, -normal);
642 penalty_flux_gas *= tau;
643
644 total_flux_liquid -= penalty_flux_liquid;
645 total_flux_gas -= penalty_flux_gas;
646

677 calculate_viscous_interface_flux_gradient(const ConservedVariablesType &u_liquid,
678 const ConservedVariablesType &u_gas,
679 const Tensor<1, dim, VectorizedArray<number>> &normal,
680 const number &alpha_1,
681 const number &alpha_2,
682 const auto &viscous_terms_liquid,
683 const auto &viscous_terms_gas,
684 const Flow::CompressibleFlowData<number> &flow_data)
685 {

687 calculate_Dirichlet_jump_in_conservative_variables<dim, number, ConservedVariablesType>(
688 u_liquid, u_gas, flow_data);
689
690 const auto u_liquid_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
691 alpha_1, u_liquid, alpha_2, u_gas + J_Dir_cons);
692 const auto u_gas_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
693 alpha_2, u_gas, alpha_1, u_liquid - J_Dir_cons);
694
695 auto tmp_liquid = u_liquid_star - u_liquid;
696 auto tmp_gas = u_gas_star - u_gas;
697
698 ConservedVariablesGradType arg_liquid = UtilityFunctions::dyadic_product(tmp_liquid, normal);
699 ConservedVariablesGradType arg_gas = UtilityFunctions::dyadic_product(tmp_gas, -normal);
700
701 const ConservedVariablesGradType flux_grad_liquid =
702 viscous_terms_liquid.calculate_viscous_flux(u_liquid, arg_liquid);
703 const ConservedVariablesGradType flux_grad_gas =
704 viscous_terms_gas.calculate_viscous_flux(u_gas, arg_gas);
705
706 return {flux_grad_liquid, flux_grad_gas};
