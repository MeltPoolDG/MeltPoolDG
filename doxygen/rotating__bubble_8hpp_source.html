118 void
119 set_boundary_conditions() override
120 {
121 /*
122 * create a pair of (boundary_id, dirichlet_function)
123 */
124 constexpr types::boundary_id inflow_bc = 42;
125 constexpr types::boundary_id do_nothing = 0;
126
127 this->attach_boundary_condition({inflow_bc,
128 std::make_shared<Functions::ConstantFunction<dim>>(-1.0)},
129 "dirichlet",
130 "level_set");
131 /*
132 * mark inflow edges with boundary label (no boundary on outflow edges must be prescribed
133 * due to the hyperbolic nature of the analyzed problem
134 *
135 out in
136 (-1,1) +---------------+ (1,1)
137 | : |
138 in | : | out
139 |_______________|
140 | : |
141 out | : | in
142 | : |
143 +---------------+
144 * (-1,-1) in out (1,-1)
145 */
146 if constexpr (dim > 1)
147 {
148 for (const auto &cell : this->triangulation->cell_iterators())
149 for (const auto &face : cell->face_iterators())
150 if ((face->at_boundary()))
151 {
152 const double half_line = (right_domain + left_domain) / 2;
153
154 if (face->center()[0] == left_domain && face->center()[1] > half_line)
155 face->set_boundary_id(inflow_bc);
156 else if (face->center()[0] == right_domain && face->center()[1] < half_line)
157 face->set_boundary_id(inflow_bc);
158 else if (face->center()[1] == right_domain && face->center()[0] > half_line)
159 face->set_boundary_id(inflow_bc);
160 else if (face->center()[1] == left_domain && face->center()[0] < half_line)
161 face->set_boundary_id(inflow_bc);
162 else
163 face->set_boundary_id(do_nothing);
164 }
165 }
