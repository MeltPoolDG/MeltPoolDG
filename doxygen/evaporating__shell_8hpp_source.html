69 template <int dim>
70 class RadialBoundaryVelocity : public dealii::Function<dim>
71 {
72 public:
73 RadialBoundaryVelocity(const dealii::Point<dim> &center)
74 : dealii::Function<dim>(dim)
75 , center(center)
76 {}
77
78 double
79 value(const dealii::Point<dim> &p, const unsigned int comp) const override
80 {
81 const double v = velocity;
82
83 const double radius = p.distance(center);
84
85 // if point is not at boundary face
86 if (std::abs(radius - inner_radius) > 1e-6)
87 return 0.0;
88
89 if constexpr (dim == 1)
90 return v;
91
92 const auto n = p - center;
93
94 if (dim == 2)
95 {
96 if (comp == 0)
97 return v * n[0] / radius;
98 else //(comp == 1)
99 return v * n[1] / radius;
100 }
101 else if (dim == 3)
102 {
103 const double phi = std::acos(n[2] / radius);
104 const double theta = std::atan2(n[1], n[0]);
105
106 if (comp == 0)
107 return v * std::cos(theta) * std::sin(phi);
108 else if (comp == 1)
109 return v * std::sin(theta) * std::sin(phi);
110 else // if (comp == 2)
111 return v * std::cos(phi);
112 }
113 else
114 {
115 AssertThrow(false, dealii::ExcNotImplemented());
116 return 0;
117 }
118 }
119
120 private:
121 const dealii::Point<dim> center;
122 };

227 void
228 do_postprocessing(const GenericDataOut<dim, double> &generic_data_out) const final
229 {
230 // Compare numerical vs. analytical solution based on the last computed time-step.
231 // This is sufficient since the test case is stationary.
232 if ((n_time_step == this->parameters.time_stepping.max_n_steps) ||
233 generic_data_out.get_time() == this->parameters.time_stepping.end_time)
234 {
235 std::cout.precision(3);
236 generic_data_out.get_vector("velocity").update_ghost_values();
237 generic_data_out.get_vector("pressure").update_ghost_values();
238
239 const auto analytical_velocity = [&](const double &r) -> double {
240 return velocity * inner_radius / r;
241 };
242
243 const auto analytical_pressure = [&](const double &r) -> double {
244 const double rho = this->parameters.material.gas.density;
245 const double mu = this->parameters.material.gas.dynamic_viscosity;
246 const double u0 = velocity;
247 return rho * u0 * u0 * inner_radius * inner_radius * 0.5 *
248 (1. / std::pow(outer_radius, 2) - 1 / (r * r)) -
249 2. * mu * u0 * inner_radius / std::pow(outer_radius, 2);
250 };
251
252 // generate number of request points and calculate analytical solution
253 std::vector<dealii::Point<dim>> req_points;
254 dealii::TableHandler table;
255 if (dealii::Utilities::MPI::this_mpi_process(this->mpi_communicator) == 0)
256 {
257 const std::vector<double> req_radii = {inner_radius,
258 (inner_radius + outer_radius) * 0.5,
259 outer_radius};
260
261 for (const auto r : req_radii)
262 {
263 auto p = dealii::Point<dim>();
264 p[0] = r;
265 req_points.emplace_back(p);
266 table.add_value("analytical pressure", analytical_pressure(r));
267 table.add_value("analytical radial velocity", analytical_velocity(r));
268 }
269 }
270
271 // read numerical results at request points
272 dealii::Utilities::MPI::RemotePointEvaluation<dim, dim> remote_point_evaluation;
273 remote_point_evaluation.reinit(req_points,
274 *this->triangulation,
275 generic_data_out.get_mapping());
276
277 const auto pressure_vals =
278 dealii::VectorTools::point_values<1>(remote_point_evaluation,
279 generic_data_out.get_dof_handler("pressure"),
280 generic_data_out.get_vector("pressure"));
281
282 const auto vel_vals =
283 dealii::VectorTools::point_values<dim>(remote_point_evaluation,
284 generic_data_out.get_dof_handler("velocity"),
285 generic_data_out.get_vector("velocity"));
286 if (dealii::Utilities::MPI::this_mpi_process(this->mpi_communicator) == 0)
287 {
288 for (unsigned int i = 0; i < vel_vals.size(); ++i)
289 {
290 table.add_value("pressure", pressure_vals[i]);
291 if constexpr (dim == 1)
292 table.add_value("radial velocity", vel_vals[i]);
293 else
294 table.add_value("radial velocity", vel_vals[i][0]);
295 }
296
297 table.set_scientific("analytical radial velocity", true);
298 table.set_scientific("analytical pressure", true);
299 table.set_scientific("radial velocity", true);
300 table.set_scientific("pressure", true);
301 std::cout << std::endl;
302 table.write_text(std::cout);
303 std::cout << std::endl;
304 }
305
306 generic_data_out.get_vector("velocity").zero_out_ghost_values();
307 generic_data_out.get_vector("pressure").zero_out_ghost_values();
308 }
309 n_time_step += 1;
310 }
