69 rhs_cell_integral_kernel(

71 const unsigned int q,
72 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> *constant_body_force,
73 const auto &convective_terms,
74 const auto &viscous_terms,
75 const CompressibleFlowScratchData<dim, number> &flow_scratch_data)
76 {
77 const auto w_q = evaluator.get_value(q);
78
79 auto flux = convective_terms.calculate_convective_flux(w_q);
80
81 if (is_viscous)
82 {
83 const auto grad_w_q = evaluator.get_gradient(q);
84 flux -= viscous_terms.calculate_viscous_flux(w_q, grad_w_q);
85 }
86
87 dealii::Tensor<1, dim + 2, dealii::VectorizedArray<number>> forcing;
88
89 if (flow_scratch_data.body_force.get() != nullptr)
90 {
91 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> force =
92 constant_body_force ?
93 *constant_body_force :
94 VectorTools::evaluate_function_at_vectorized_points(*flow_scratch_data.body_force,
95 evaluator.quadrature_point(q));
96 for (unsigned int d = 0; d < dim; ++d)
97 forcing[d + 1] = w_q[0] * force[d];
98 for (unsigned int d = 0; d < dim; ++d)
99 forcing[dim + 1] += force[d] * w_q[d + 1];
100 }
101
102 return {forcing, flux};
103 }
