65 inline DEAL_II_ALWAYS_INLINE //
66 std::tuple<CompressibleFlowTypes::ConservedVariablesType<dim, number>,

68 rhs_cell_integral_kernel(
69 const Integrator &evaluator,
70 const unsigned int q,
71 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> *constant_body_force,
72 const CompressibleFlowConvectiveKernels<dim, number> &convective_terms,
73 const CompressibleFlowViscousKernels<dim, number> &viscous_terms,
74 const CompressibleFlowScratchData<dim, number> &flow_scratch_data)
75 {
76 const auto w_q = evaluator.get_value(q);
77
78 auto flux = convective_terms.calculate_convective_flux(w_q);
79
80 if (flow_scratch_data.flow_data.dynamic_viscosity > 0)
81 {
82 const auto grad_w_q = evaluator.get_gradient(q);
83 flux -= viscous_terms.calculate_viscous_flux(w_q, grad_w_q);
84 }
85
86 dealii::Tensor<1, dim + 2, dealii::VectorizedArray<number>> forcing;
87
88 if (flow_scratch_data.body_force.get() != nullptr)
89 {
90 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> force =
91 constant_body_force ?
92 *constant_body_force :
93 VectorTools::evaluate_function_at_vectorized_points(*flow_scratch_data.body_force,
94 evaluator.quadrature_point(q));
95 for (unsigned int d = 0; d < dim; ++d)
96 forcing[d + 1] = w_q[0] * force[d];
97 for (unsigned int d = 0; d < dim; ++d)
98 forcing[dim + 1] += force[d] * w_q[d + 1];
99 }
100
101 return {forcing, flux};
102 }
