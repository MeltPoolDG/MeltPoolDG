157 template <int dim, typename number>
158 inline DEAL_II_ALWAYS_INLINE //
159 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
160 calculate_velocity(
161 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
162 {
163 const dealii::VectorizedArray<number> inverse_density =
164 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
165
166 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
167 for (unsigned int d = 0; d < dim; ++d)
168 velocity[d] = conserved_variables[1 + d] * inverse_density;
169
170 return velocity;
171 }

173 template <int dim, typename number>
174 inline DEAL_II_ALWAYS_INLINE //
175 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
176 calculate_grad_velocity(
177 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
178 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
179 &grad_conserved_variables)
180 {
181 const dealii::VectorizedArray<number> inverse_density =
182 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
183 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
184 calculate_velocity<dim, number>(conserved_variables);
185
186 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
187 for (unsigned int d = 0; d < dim; ++d)
188 grad_rho[d] = grad_conserved_variables[0][d];
189
190 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
191 for (unsigned int d = 0; d < dim; ++d)
192 for (unsigned int e = 0; e < dim; ++e)
193 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
194
195 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
196 for (unsigned int d = 0; d < dim; ++d)
197 for (unsigned int e = 0; e < dim; ++e)
198 grad_velocity[d][e] =
199 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
200
201 return grad_velocity;
202 }

204 template <int dim, typename Number>
205 void
206 calculate_penalty_parameter(
207 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
208 const dealii::MatrixFree<dim, Number> &matrix_free,
209 const std::string &domain_representation_type,
210 const unsigned int dof_index,
211 const Number scaling_factor)
212 {
213 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
214 array_penalty_parameter.resize(n_cells);
215
216 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
217 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
218 unsigned int const degree = fe.degree;
219
220 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
221 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
222 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
223 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
224
225 auto const reference_cells =
226 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
227 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
228
229 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
230 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
231
232 auto const face_quadrature =
233 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
234 1);
235 dealii::FEFaceValues<dim> fe_face_values(mapping,
236 fe,
237 face_quadrature,
238 dealii::update_JxW_values);
239
240 if (domain_representation_type == "fitted")
241 {
242 for (unsigned int i = 0; i < n_cells; ++i)
243 {
244 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
245 {
246 typename dealii::DoFHandler<dim>::cell_iterator cell =
247 matrix_free.get_cell_iterator(i, v, dof_index);
248 fe_values.reinit(cell);
249
250 // calculate cell volume
251 Number volume = 0;
252 for (unsigned int q = 0; q < quadrature.size(); ++q)
253 {
254 volume += fe_values.JxW(q);
255 }
256
257 // calculate surface area
258 Number surface_area = 0;
259 for (unsigned int const f : cell->face_indices())
260 {
261 fe_face_values.reinit(cell, f);
262 Number const factor =
263 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
264 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
265 {
266 surface_area += fe_face_values.JxW(q) * factor;
267 }
268 }
269
270 array_penalty_parameter[i][v] = surface_area / volume * fac;
271 }
272 }
273 }
274 else if (domain_representation_type == "cut")
275 {
276 for (unsigned int i = 0; i < n_cells; ++i)
277 {
278 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
279 {
280 typename dealii::DoFHandler<dim>::cell_iterator cell =
281 matrix_free.get_cell_iterator(i, v, dof_index);
282
283 // simplified computation for hypercube elements
284 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
285 }
286 }
287 }
288 else
289 AssertThrow(false,
290 dealii::ExcMessage("The domain representation type '" +
291 domain_representation_type + "' is not supported."));
292 }

294 template <int dim, typename number>
295 void
296 update_primitive_variables_solution(
297 dealii::LinearAlgebra::distributed::Vector<number> &solution_primitive_variables,
298 const dealii::LinearAlgebra::distributed::Vector<number> &solution,
299 const ScratchData<dim, dim, number> &scratch_data,
300 const unsigned int dof_idx,
301 const unsigned int quad_idx,
302 const CompressibleFlowMaterial<dim, number> *material_liquid,
303 const CompressibleFlowMaterial<dim, number> *material_gas)
304 {
305 const dealii::MatrixFree<dim, number, dealii::VectorizedArray<number>> &matrix_free =
306 scratch_data.get_matrix_free();
307 unsigned int n_support_points_per_cell = scratch_data.get_n_dofs_per_cell(dof_idx) / (dim + 2);
308
309 auto process_cell = [&](CutUtil::CellCategory category,
310 unsigned int first_component,
311 const CompressibleFlowMaterial<dim, number> *material,
312 const unsigned int cell_batch) {
313 if (!material)
314 return;
315 auto eval = FECellIntegrator<dim, dim + 2, number>(
316 matrix_free, dof_idx, quad_idx, first_component, category);
317 eval.reinit(cell_batch);
318 eval.read_dof_values(solution);
319
320 for (unsigned int i = 0; i < n_support_points_per_cell; ++i)
321 {
322 const auto &u_cons = eval.get_dof_value(i);
323 auto u_prim = material->eos_utils->convert_conservative_into_primitive_variables(u_cons);
324 eval.submit_dof_value(u_prim, i);
325 }
326
327 eval.set_dof_values(solution_primitive_variables);
328 };
329
330 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
331 {
332 const auto cell_category = matrix_free.get_cell_category(cell_batch);
333
334 if (cell_category == CutUtil::CellCategory::liquid)
335 {
336 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
337 }
338 else if (cell_category == CutUtil::CellCategory::gas)
339 {
340 process_cell(CutUtil::CellCategory::gas, dim + 2, material_gas, cell_batch);
341 }
342 else if (cell_category == CutUtil::CellCategory::intersected)
343 {
344 process_cell(CutUtil::CellCategory::intersected, 0, material_liquid, cell_batch);
345 process_cell(CutUtil::CellCategory::intersected, dim + 2, material_gas, cell_batch);
346 }
347 }
