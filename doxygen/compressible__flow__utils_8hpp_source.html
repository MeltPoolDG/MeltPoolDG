304 template <int dim, typename number>
305 inline DEAL_II_ALWAYS_INLINE //
306 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
307 calculate_velocity(
308 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
309 {
310 const dealii::VectorizedArray<number> inverse_density =
311 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
312
313 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
314 for (unsigned int d = 0; d < dim; ++d)
315 velocity[d] = conserved_variables[1 + d] * inverse_density;
316
317 return velocity;
318 }

320 template <int dim, typename number>
321 inline DEAL_II_ALWAYS_INLINE //
322 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
323 calculate_grad_velocity(
324 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
325 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
326 &grad_conserved_variables)
327 {
328 const dealii::VectorizedArray<number> inverse_density =
329 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
330 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
331 calculate_velocity<dim, number>(conserved_variables);
332
333 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
334 for (unsigned int d = 0; d < dim; ++d)
335 grad_rho[d] = grad_conserved_variables[0][d];
336
337 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
338 for (unsigned int d = 0; d < dim; ++d)
339 for (unsigned int e = 0; e < dim; ++e)
340 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
341
342 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
343 for (unsigned int d = 0; d < dim; ++d)
344 for (unsigned int e = 0; e < dim; ++e)
345 grad_velocity[d][e] =
346 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
347
348 return grad_velocity;
349 }

351 template <int dim, typename Number>
352 void
353 calculate_penalty_parameter(
354 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
355 const dealii::MatrixFree<dim, Number> &matrix_free,
356 const std::string &domain_representation_type,
357 const unsigned int dof_index,
358 const Number scaling_factor)
359 {
360 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
361 array_penalty_parameter.resize(n_cells);
362
363 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
364 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
365 unsigned int const degree = fe.degree;
366
367 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
368 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
369 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
370 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
371
372 auto const reference_cells =
373 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
374 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
375
376 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
377 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
378
379 auto const face_quadrature =
380 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
381 1);
382 dealii::FEFaceValues<dim> fe_face_values(mapping,
383 fe,
384 face_quadrature,
385 dealii::update_JxW_values);
386
387 if (domain_representation_type == "fitted")
388 {
389 for (unsigned int i = 0; i < n_cells; ++i)
390 {
391 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
392 {
393 typename dealii::DoFHandler<dim>::cell_iterator cell =
394 matrix_free.get_cell_iterator(i, v, dof_index);
395 fe_values.reinit(cell);
396
397 // calculate cell volume
398 Number volume = 0;
399 for (unsigned int q = 0; q < quadrature.size(); ++q)
400 {
401 volume += fe_values.JxW(q);
402 }
403
404 // calculate surface area
405 Number surface_area = 0;
406 for (unsigned int const f : cell->face_indices())
407 {
408 fe_face_values.reinit(cell, f);
409 Number const factor =
410 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
411 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
412 {
413 surface_area += fe_face_values.JxW(q) * factor;
414 }
415 }
416
417 array_penalty_parameter[i][v] = surface_area / volume * fac;
418 }
419 }
420 }
421 else if (domain_representation_type == "cut")
422 {
423 for (unsigned int i = 0; i < n_cells; ++i)
424 {
425 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
426 {
427 typename dealii::DoFHandler<dim>::cell_iterator cell =
428 matrix_free.get_cell_iterator(i, v, dof_index);
429
430 // simplified computation for hypercube elements
431 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
432 }
433 }
434 }
435 else
436 AssertThrow(false,
437 dealii::ExcMessage("The domain representation type '" +
438 domain_representation_type + "' is not supported."));
439 }
