133 template <int dim, typename number>
134 inline DEAL_II_ALWAYS_INLINE //
135 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
136 calculate_velocity(
137 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
138 {
139 const dealii::VectorizedArray<number> inverse_density =
140 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
141
142 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
143 for (unsigned int d = 0; d < dim; ++d)
144 velocity[d] = conserved_variables[1 + d] * inverse_density;
145
146 return velocity;
147 }

149 template <int dim, typename number>
150 inline DEAL_II_ALWAYS_INLINE //
151 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
152 calculate_grad_velocity(
153 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
154 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
155 &grad_conserved_variables)
156 {
157 const dealii::VectorizedArray<number> inverse_density =
158 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
159 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
160 calculate_velocity<dim, number>(conserved_variables);
161
162 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
163 for (unsigned int d = 0; d < dim; ++d)
164 grad_rho[d] = grad_conserved_variables[0][d];
165
166 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
167 for (unsigned int d = 0; d < dim; ++d)
168 for (unsigned int e = 0; e < dim; ++e)
169 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
170
171 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
172 for (unsigned int d = 0; d < dim; ++d)
173 for (unsigned int e = 0; e < dim; ++e)
174 grad_velocity[d][e] =
175 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
176
177 return grad_velocity;
178 }

180 template <int dim, typename Number>
181 void
182 calculate_penalty_parameter(
183 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
184 const dealii::MatrixFree<dim, Number> &matrix_free,
185 const std::string &domain_representation_type,
186 const unsigned int dof_index,
187 const Number scaling_factor)
188 {
189 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
190 array_penalty_parameter.resize(n_cells);
191
192 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
193 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
194 unsigned int const degree = fe.degree;
195
196 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
197 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
198 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
199 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
200
201 auto const reference_cells =
202 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
203 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
204
205 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
206 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
207
208 auto const face_quadrature =
209 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
210 1);
211 dealii::FEFaceValues<dim> fe_face_values(mapping,
212 fe,
213 face_quadrature,
214 dealii::update_JxW_values);
215
216 if (domain_representation_type == "fitted")
217 {
218 for (unsigned int i = 0; i < n_cells; ++i)
219 {
220 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
221 {
222 typename dealii::DoFHandler<dim>::cell_iterator cell =
223 matrix_free.get_cell_iterator(i, v, dof_index);
224 fe_values.reinit(cell);
225
226 // calculate cell volume
227 Number volume = 0;
228 for (unsigned int q = 0; q < quadrature.size(); ++q)
229 {
230 volume += fe_values.JxW(q);
231 }
232
233 // calculate surface area
234 Number surface_area = 0;
235 for (unsigned int const f : cell->face_indices())
236 {
237 fe_face_values.reinit(cell, f);
238 Number const factor =
239 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
240 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
241 {
242 surface_area += fe_face_values.JxW(q) * factor;
243 }
244 }
245
246 array_penalty_parameter[i][v] = surface_area / volume * fac;
247 }
248 }
249 }
250 else if (domain_representation_type == "cut")
251 {
252 for (unsigned int i = 0; i < n_cells; ++i)
253 {
254 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
255 {
256 typename dealii::DoFHandler<dim>::cell_iterator cell =
257 matrix_free.get_cell_iterator(i, v, dof_index);
258
259 // simplified computation for hypercube elements
260 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
261 }
262 }
263 }
264 else
265 AssertThrow(false,
266 dealii::ExcMessage("The domain representation type '" +
267 domain_representation_type + "' is not supported."));
268 }
