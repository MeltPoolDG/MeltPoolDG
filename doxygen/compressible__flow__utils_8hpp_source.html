165 template <int dim, typename number>
166 inline DEAL_II_ALWAYS_INLINE //
167 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
168 calculate_velocity(
169 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
170 {
171 const dealii::VectorizedArray<number> inverse_density =
172 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
173
174 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
175 for (unsigned int d = 0; d < dim; ++d)
176 velocity[d] = conserved_variables[1 + d] * inverse_density;
177
178 return velocity;
179 }

181 template <int dim, typename number>
182 inline DEAL_II_ALWAYS_INLINE //
183 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
184 calculate_grad_velocity(
185 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
186 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
187 &grad_conserved_variables)
188 {
189 const dealii::VectorizedArray<number> inverse_density =
190 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
191 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
192 calculate_velocity<dim, number>(conserved_variables);
193
194 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
195 for (unsigned int d = 0; d < dim; ++d)
196 grad_rho[d] = grad_conserved_variables[0][d];
197
198 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
199 for (unsigned int d = 0; d < dim; ++d)
200 for (unsigned int e = 0; e < dim; ++e)
201 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
202
203 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
204 for (unsigned int d = 0; d < dim; ++d)
205 for (unsigned int e = 0; e < dim; ++e)
206 grad_velocity[d][e] =
207 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
208
209 return grad_velocity;
210 }

212 template <int dim, typename Number>
213 void
214 calculate_penalty_parameter(
215 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
216 const dealii::MatrixFree<dim, Number> &matrix_free,
217 const std::string &domain_representation_type,
218 const unsigned int dof_index,
219 const Number scaling_factor)
220 {
221 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
222 array_penalty_parameter.resize(n_cells);
223
224 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
225 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
226 unsigned int const degree = fe.degree;
227
228 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
229 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
230 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
231 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
232
233 auto const reference_cells =
234 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
235 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
236
237 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
238 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
239
240 auto const face_quadrature =
241 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
242 1);
243 dealii::FEFaceValues<dim> fe_face_values(mapping,
244 fe,
245 face_quadrature,
246 dealii::update_JxW_values);
247
248 if (domain_representation_type == "fitted")
249 {
250 for (unsigned int i = 0; i < n_cells; ++i)
251 {
252 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
253 {
254 typename dealii::DoFHandler<dim>::cell_iterator cell =
255 matrix_free.get_cell_iterator(i, v, dof_index);
256 fe_values.reinit(cell);
257
258 // calculate cell volume
259 Number volume = 0;
260 for (unsigned int q = 0; q < quadrature.size(); ++q)
261 {
262 volume += fe_values.JxW(q);
263 }
264
265 // calculate surface area
266 Number surface_area = 0;
267 for (unsigned int const f : cell->face_indices())
268 {
269 fe_face_values.reinit(cell, f);
270 Number const factor =
271 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
272 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
273 {
274 surface_area += fe_face_values.JxW(q) * factor;
275 }
276 }
277
278 array_penalty_parameter[i][v] = surface_area / volume * fac;
279 }
280 }
281 }
282 else if (domain_representation_type == "cut")
283 {
284 for (unsigned int i = 0; i < n_cells; ++i)
285 {
286 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
287 {
288 typename dealii::DoFHandler<dim>::cell_iterator cell =
289 matrix_free.get_cell_iterator(i, v, dof_index);
290
291 // simplified computation for hypercube elements
292 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
293 }
294 }
295 }
296 else
297 AssertThrow(false,
298 dealii::ExcMessage("The domain representation type '" +
299 domain_representation_type + "' is not supported."));
300 }
