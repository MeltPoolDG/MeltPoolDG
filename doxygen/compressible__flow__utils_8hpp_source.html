226 template <int dim, typename number>
227 inline DEAL_II_ALWAYS_INLINE //
228 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
229 calculate_velocity(
230 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
231 {
232 const dealii::VectorizedArray<number> inverse_density =
233 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
234
235 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
236 for (unsigned int d = 0; d < dim; ++d)
237 velocity[d] = conserved_variables[1 + d] * inverse_density;
238
239 return velocity;
240 }

243 template <int dim, typename number>
244 inline DEAL_II_ALWAYS_INLINE //
245 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
246 calculate_grad_velocity(
247 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
248 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
249 &grad_conserved_variables)
250 {
251 const dealii::VectorizedArray<number> inverse_density =
252 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
253 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
254 calculate_velocity<dim, number>(conserved_variables);
255
256 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
257 for (unsigned int d = 0; d < dim; ++d)
258 grad_rho[d] = grad_conserved_variables[0][d];
259
260 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
261 for (unsigned int d = 0; d < dim; ++d)
262 for (unsigned int e = 0; e < dim; ++e)
263 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
264
265 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
266 for (unsigned int d = 0; d < dim; ++d)
267 for (unsigned int e = 0; e < dim; ++e)
268 grad_velocity[d][e] =
269 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
270
271 return grad_velocity;
272 }

275 template <int dim, typename number>
276 inline DEAL_II_ALWAYS_INLINE //
277 dealii::VectorizedArray<number>
278 calculate_pressure(
279 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
280 number gamma)
281 {
282 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
283 calculate_velocity<dim, number>(conserved_variables);
284 return (gamma - 1.) * (conserved_variables[dim + 1] -
285 conserved_variables[0] * 0.5 * scalar_product(velocity, velocity));
286 }

288 template <int dim, typename number>
289 inline DEAL_II_ALWAYS_INLINE //
290 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
291 calculate_grad_T(
292 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
293 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
294 &grad_conserved_variables,
295 const number gamma,
296 const number specific_gas_constant)
297 {
298 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> u =
299 calculate_velocity<dim, number>(conserved_variables);
300 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_u =
301 calculate_grad_velocity<dim, number>(conserved_variables, grad_conserved_variables);
302 const dealii::VectorizedArray<number> rho = conserved_variables[0];
303 const dealii::VectorizedArray<number> inv_rho = dealii::VectorizedArray<number>(1.) / rho;
304 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho =
305 grad_conserved_variables[0];
306 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_E =
307 inv_rho *
308 (grad_conserved_variables[dim + 1] - inv_rho * conserved_variables[dim + 1] * grad_rho);
309 return (gamma - 1.0) / specific_gas_constant * (grad_E - grad_u * u);
310 }

351 template <int dim, typename Number>
352 void
353 calculate_penalty_parameter(AlignedVector<VectorizedArray<Number>> &array_penalty_parameter,
354 const MatrixFree<dim, Number> &matrix_free,
355 const std::string &domain_representation_type,
356 const unsigned int dof_index,
357 const double scaling_factor)
358 {
359 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
360 array_penalty_parameter.resize(n_cells);
361
362 Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
363 FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
364 unsigned int const degree = fe.degree;
365
366 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
367 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
368 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
369 const double fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
370
371 auto const reference_cells =
372 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
373 AssertThrow(reference_cells.size() == 1, ExcMessage("No mixed meshes allowed."));
374
375 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
376 FEValues<dim> fe_values(mapping, fe, quadrature, update_JxW_values);
377
378 auto const face_quadrature =
379 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
380 1);
381 FEFaceValues<dim> fe_face_values(mapping, fe, face_quadrature, update_JxW_values);
382
383 if (domain_representation_type == "fitted")
384 {
385 for (unsigned int i = 0; i < n_cells; ++i)
386 {
387 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
388 {
389 typename DoFHandler<dim>::cell_iterator cell =
390 matrix_free.get_cell_iterator(i, v, dof_index);
391 fe_values.reinit(cell);
392
393 // calculate cell volume
394 Number volume = 0;
395 for (unsigned int q = 0; q < quadrature.size(); ++q)
396 {
397 volume += fe_values.JxW(q);
398 }
399
400 // calculate surface area
401 Number surface_area = 0;
402 for (unsigned int const f : cell->face_indices())
403 {
404 fe_face_values.reinit(cell, f);
405 Number const factor =
406 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
407 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
408 {
409 surface_area += fe_face_values.JxW(q) * factor;
410 }
411 }
412
413 array_penalty_parameter[i][v] = surface_area / volume * fac;
414 }
415 }
416 }
417 else if (domain_representation_type == "cut")
418 {
419 for (unsigned int i = 0; i < n_cells; ++i)
420 {
421 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
422 {
423 typename DoFHandler<dim>::cell_iterator cell =
424 matrix_free.get_cell_iterator(i, v, dof_index);
425
426 // simplified computation for hypercube elements
427 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
428 }
429 }
430 }
431 else
432 AssertThrow(false,
433 dealii::ExcMessage("The domain representation type '" +
434 domain_representation_type + "' is not supported."));
435 }
