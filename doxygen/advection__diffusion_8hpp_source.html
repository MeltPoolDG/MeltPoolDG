41 template <int dim>
42 class InitialLevelSetField : public Function<dim>
43 {
44 public:
45 InitialLevelSetField(const LevelSetType level_set_type = LevelSetType::level_set,
46 const double eps = 0.0)
47 : Function<dim>()
48 , distance_sphere(dim == 1 ? Point<dim>(0.0) :
49 (dim == 2) ? Point<dim>(0.0, 0.5) :
50 Point<dim>(0, 0, 0.5),
51 0.25)
52 , level_set_type(level_set_type)
53 , eps(eps)
54 {}
55
56 double
57 value(const Point<dim> &p, const unsigned int /*component*/) const override
58 {
59 const auto signed_distance = -distance_sphere.value(p);
60
61 switch (level_set_type)
62 {
63 case LevelSetType::level_set:
64 return UtilityFunctions::CharacteristicFunctions::tanh_characteristic_function(
65 signed_distance, eps);
66 case LevelSetType::smooth_heaviside:
67 return UtilityFunctions::CharacteristicFunctions::heaviside(signed_distance, eps);
68 case LevelSetType::heaviside:
69 return UtilityFunctions::CharacteristicFunctions::sgn(signed_distance);
70 case LevelSetType::signed_distance:
71 return signed_distance;
72 default:
73 DEAL_II_NOT_IMPLEMENTED();
74 }
75 // unreachable dummy return
76 return 0.0;
77 }
78
79 private:
80 const Functions::SignedDistance::Sphere<dim> distance_sphere;
81 const LevelSetType level_set_type;
82 const double eps;
83 };

149 void
150 set_boundary_conditions() final
151 {
152 /*
153 * create a pair of (boundary_id, dirichlet_function)
154 */
155 constexpr types::boundary_id inflow_bc = 42;
156 constexpr types::boundary_id do_nothing = 0;
157
158 auto dirichlet = std::make_shared<Functions::ConstantFunction<dim>>(-1);
159
160 if (inflow_outflow_bc)
161 {
162 this->attach_boundary_condition({inflow_bc, dirichlet},
163 "inflow_outflow",
164 "advection_diffusion");
165 this->attach_boundary_condition({do_nothing, dirichlet},
166 "inflow_outflow",
167 "advection_diffusion");
168 }
169 else
170 this->attach_boundary_condition({inflow_bc, dirichlet}, "dirichlet", "advection_diffusion");
171
172 /*
173 * mark inflow edges with boundary label (no boundary on outflow edges must be prescribed
174 * due to the hyperbolic nature of the analyzed problem
175 *
176 out in
177 (-1,1) +---------------+ (1,1)
178 | : |
179 in | : | out
180 |_______________|
181 | : |
182 out | : | in
183 | : |
184 +---------------+
185 * (-1,-1) in out (1,-1)
186 */
187 if constexpr (dim >= 2)
188 {
189 for (const auto &cell : this->triangulation->cell_iterators())
190 for (const auto &face : cell->face_iterators())
191 if ((face->at_boundary()))
192 {
193 const double half_line = (right_domain + left_domain) / 2;
194
195 if (face->center()[0] == left_domain && face->center()[dim - 1] >= half_line)
196 face->set_boundary_id(inflow_bc);
197 else if (face->center()[0] == right_domain &&
198 face->center()[dim - 1] <= half_line)
199 face->set_boundary_id(inflow_bc);
200 else if (face->center()[dim - 1] == right_domain &&
201 face->center()[0] >= half_line)
202 face->set_boundary_id(inflow_bc);
203 else if (face->center()[dim - 1] == left_domain && face->center()[0] <= half_line)
204 face->set_boundary_id(inflow_bc);
205 else
206 face->set_boundary_id(do_nothing);
207 }
208 }
209 else
210 {
211 (void)do_nothing; // suppress unused variable for 1D
212 }
213 }
