39 template <int dim, typename number>
40 class InitialLevelSetField : public Function<dim, number>
41 {
42 public:
43 InitialLevelSetField(const LevelSetType level_set_type = LevelSetType::level_set,
44 const number eps = 0.0)
45 : Function<dim>()
46 , distance_sphere(dim == 1 ? Point<dim, number>(0.0) :
47 (dim == 2) ? Point<dim, number>(0.0, 0.5) :
48 Point<dim, number>(0, 0, 0.5),

50 , level_set_type(level_set_type)

54 number
55 value(const Point<dim, number> &p, const unsigned int /*component*/) const override
56 {
57 const auto signed_distance = -distance_sphere.value(p);
58
59 switch (level_set_type)
60 {
61 case LevelSetType::level_set:
62 return UtilityFunctions::CharacteristicFunctions::tanh_characteristic_function(
63 signed_distance, eps);
64 case LevelSetType::smooth_heaviside:
65 return UtilityFunctions::CharacteristicFunctions::heaviside(signed_distance, eps);
66 case LevelSetType::heaviside:
67 return UtilityFunctions::CharacteristicFunctions::sgn(signed_distance);
68 case LevelSetType::signed_distance:
69 return signed_distance;
70 default:
71 DEAL_II_NOT_IMPLEMENTED();
72 }
73 // unreachable dummy return
74 return 0.0;
75 }
76
77 private:
78 const Functions::SignedDistance::Sphere<dim> distance_sphere;
79 const LevelSetType level_set_type;
80 const number eps;
81 };

147 void
148 set_boundary_conditions() final
149 {
150 /*
151 * create a pair of (boundary_id, dirichlet_function)
152 */
153 constexpr types::boundary_id inflow_bc = 42;
154 constexpr types::boundary_id do_nothing = 0;
155
156 auto dirichlet = std::make_shared<Functions::ConstantFunction<dim>>(-1);
157
158 if (inflow_outflow_bc)
159 {
160 this->attach_boundary_condition({inflow_bc, dirichlet},
161 "inflow_outflow",
162 "advection_diffusion");
163 this->attach_boundary_condition({do_nothing, dirichlet},
164 "inflow_outflow",
165 "advection_diffusion");
166 }
167 else
168 this->attach_boundary_condition({inflow_bc, dirichlet}, "dirichlet", "advection_diffusion");
169
170 /*
171 * mark inflow edges with boundary label (no boundary on outflow edges must be prescribed
172 * due to the hyperbolic nature of the analyzed problem
173 *
174 out in
175 (-1,1) +---------------+ (1,1)
176 | : |
177 in | : | out
178 |_______________|
179 | : |
180 out | : | in
181 | : |
182 +---------------+
183 * (-1,-1) in out (1,-1)
184 */
185 if constexpr (dim >= 2)
186 {
187 for (const auto &cell : this->triangulation->cell_iterators())
188 for (const auto &face : cell->face_iterators())
189 if ((face->at_boundary()))
190 {
191 const number half_line = (right_domain + left_domain) / 2;
192
193 if (face->center()[0] == left_domain && face->center()[dim - 1] >= half_line)
194 face->set_boundary_id(inflow_bc);
195 else if (face->center()[0] == right_domain &&
196 face->center()[dim - 1] <= half_line)
197 face->set_boundary_id(inflow_bc);
198 else if (face->center()[dim - 1] == right_domain &&
199 face->center()[0] >= half_line)
200 face->set_boundary_id(inflow_bc);
201 else if (face->center()[dim - 1] == left_domain && face->center()[0] <= half_line)
202 face->set_boundary_id(inflow_bc);
203 else
204 face->set_boundary_id(do_nothing);
205 }
206 }
207 else
208 {
209 (void)do_nothing; // suppress unused variable for 1D
210 }
211 }
