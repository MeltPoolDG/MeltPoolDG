85 if (dim > 1)

87 // create DataOut of the slice
88 dealii::MappingQ1<dim - 1, dim> mapping_slice;
89 DataOutResample<dim, dim - 1, dim> data_out(tria_slice, mapping_slice);
90
91 for (const auto &i : idx_req_vars)
92 {
93 const auto &data = generic_data_out->entries[i];

95 data_out.add_data_vector(*std::get<0>(data),
96 *std::get<1>(data),
97 std::get<2>(data),
98 std::get<3>(data));
99 }
100
101
102 data_out.update_mapping(generic_data_out->get_mapping());
103 data_out.build_patches();
104
105 // write slice data to vtu files
106 const std::string pvtu_filename =
107 data_out.write_vtu_with_pvtu_record(output_data.directory + "/",
108 output_data.paraview.filename + "_slice",
109 n_time_step,
110 tria_slice.get_communicator(),
111 output_data.paraview.n_digits_timestep,
112 output_data.paraview.n_groups);
113
114 // write a pvd file relating the pvtu-file with a simulation time
115 if (Utilities::MPI::this_mpi_process(tria_slice.get_communicator()) == 0)
116 {
117 times_and_names.emplace_back(generic_data_out->get_time(), pvtu_filename);
118 std::ofstream pvd_output(output_data.directory + "/" + output_data.paraview.filename +
119 "_slice.pvd");
120 DataOutBase::write_pvd_record(pvd_output, times_and_names);
121 }
122 }
