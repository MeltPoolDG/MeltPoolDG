149 void
150 attach_output_vectors(GenericDataOut<dim, double> &data_out) const
151 {
152 using namespace dealii;
153
154 // check, if single-phase or two-phase case is considered
155 const auto &dof_handler = operation_pimpl->get_dof_handler();
156 const bool two_phase = dof_handler.get_fe_collection().n_components() / (dim + 2) == 2;
157
158 std::vector<std::string> names;
159
160 if (not two_phase)
161 {
162 names.emplace_back("density");
163 for (unsigned int d = 0; d < dim; ++d)
164 names.emplace_back("momentum");
165 names.emplace_back("energy");
166 }
167 else
168 {
169 names.emplace_back("density_liquid");
170 for (unsigned int d = 0; d < dim; ++d)
171 names.emplace_back("momentum_liquid");
172 names.emplace_back("energy_liquid");
173
174 names.emplace_back("density_gas");
175 for (unsigned int d = 0; d < dim; ++d)
176 names.emplace_back("momentum_gas");
177 names.emplace_back("energy_gas");
178 }
179
180 std::vector<DataComponentInterpretation::DataComponentInterpretation> interpretation;
181 interpretation.push_back(DataComponentInterpretation::component_is_scalar);
182 for (unsigned int d = 0; d < dim; ++d)
183 interpretation.push_back(DataComponentInterpretation::component_is_part_of_vector);
184 interpretation.push_back(DataComponentInterpretation::component_is_scalar);
185
186 // add entries for two-phase case
187 if (two_phase)
188 {
189 interpretation.push_back(DataComponentInterpretation::component_is_scalar);
190 for (unsigned int d = 0; d < dim; ++d)
191 interpretation.push_back(DataComponentInterpretation::component_is_part_of_vector);
192 interpretation.push_back(DataComponentInterpretation::component_is_scalar);
193 }
194
195 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
196 operation_pimpl->get_solution(),
197 names,
198 interpretation);
199 }
