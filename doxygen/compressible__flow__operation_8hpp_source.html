172 // check, if single-phase or two-phase case is considered
173 const auto &dof_handler = operation_pimpl->get_dof_handler();
174 const bool two_phase = dof_handler.get_fe_collection().n_components() / (dim + 2) == 2;
175
176 std::vector<std::string> names;
177
178 if (not two_phase)

180 names.emplace_back("density");
181 for (unsigned int d = 0; d < dim; ++d)
182 names.emplace_back("momentum");
183 names.emplace_back("energy");
184 }
185 else
186 {
187 names.emplace_back("density_liquid");
188 for (unsigned int d = 0; d < dim; ++d)
189 names.emplace_back("momentum_liquid");
190 names.emplace_back("energy_liquid");
191
192 names.emplace_back("density_gas");
193 for (unsigned int d = 0; d < dim; ++d)
194 names.emplace_back("momentum_gas");
195 names.emplace_back("energy_gas");
196 }
197
198 std::vector<dealii::DataComponentInterpretation::DataComponentInterpretation> interpretation;
199 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
200 for (unsigned int d = 0; d < dim; ++d)
201 interpretation.push_back(dealii::DataComponentInterpretation::component_is_part_of_vector);
202 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
203
204 // add entries for two-phase case
205 if (two_phase)
206 {
207 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
208 for (unsigned int d = 0; d < dim; ++d)
209 interpretation.push_back(
210 dealii::DataComponentInterpretation::component_is_part_of_vector);
211 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
212 }
213
214 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
215 operation_pimpl->get_solution(),
216 names,
