184 {
185 // check, if single-phase or two-phase case is considered
186 const auto &dof_handler = operation_pimpl->get_dof_handler();
187 const bool two_phase = dof_handler.get_fe_collection().n_components() / (dim + 2) == 2;
188
189 std::vector<dealii::DataComponentInterpretation::DataComponentInterpretation> interpretation;
190 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
191 for (unsigned int d = 0; d < dim; ++d)
192 interpretation.push_back(dealii::DataComponentInterpretation::component_is_part_of_vector);
193 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
194
195 // add entries for two-phase case
196 if (two_phase)
197 {
198 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
199 for (unsigned int d = 0; d < dim; ++d)
200 interpretation.push_back(
201 dealii::DataComponentInterpretation::component_is_part_of_vector);
202 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
203 }
204
205 std::vector<std::string> names_conservative;
206 std::vector<std::string> names_primitive;
207
208 if (not two_phase)
209 {
210 names_conservative.emplace_back("density");
211 for (unsigned int d = 0; d < dim; ++d)
212 names_conservative.emplace_back("momentum");
213 names_conservative.emplace_back("energy");
214
215 names_primitive.emplace_back("pressure");
216 for (unsigned int d = 0; d < dim; ++d)
217 names_primitive.emplace_back("velocity");
218 names_primitive.emplace_back("temperature");
219 }
220 else
221 {
222 names_conservative.emplace_back("density_liquid");
223 for (unsigned int d = 0; d < dim; ++d)
224 names_conservative.emplace_back("momentum_liquid");
225 names_conservative.emplace_back("energy_liquid");
226
227 names_conservative.emplace_back("density_gas");
228 for (unsigned int d = 0; d < dim; ++d)
229 names_conservative.emplace_back("momentum_gas");
230 names_conservative.emplace_back("energy_gas");
231
232 names_primitive.emplace_back("pressure_liquid");
233 for (unsigned int d = 0; d < dim; ++d)
234 names_primitive.emplace_back("velocity_liquid");
235 names_primitive.emplace_back("temperature_liquid");
236
237 names_primitive.emplace_back("pressure_gas");
238 for (unsigned int d = 0; d < dim; ++d)
239 names_primitive.emplace_back("velocity_gas");
240 names_primitive.emplace_back("temperature_gas");
241 }
242
243 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
244 operation_pimpl->get_solution(),
245 names_conservative,
