149 void
150 attach_output_vectors(GenericDataOut<dim, number> &data_out) const
151 {
152 // check, if single-phase or two-phase case is considered
153 const auto &dof_handler = operation_pimpl->get_dof_handler();
154 const bool two_phase = dof_handler.get_fe_collection().n_components() / (dim + 2) == 2;
155
156 std::vector<std::string> names;
157
158 if (not two_phase)
159 {
160 names.emplace_back("density");
161 for (unsigned int d = 0; d < dim; ++d)
162 names.emplace_back("momentum");
163 names.emplace_back("energy");
164 }
165 else
166 {
167 names.emplace_back("density_liquid");
168 for (unsigned int d = 0; d < dim; ++d)
169 names.emplace_back("momentum_liquid");
170 names.emplace_back("energy_liquid");
171
172 names.emplace_back("density_gas");
173 for (unsigned int d = 0; d < dim; ++d)
174 names.emplace_back("momentum_gas");
175 names.emplace_back("energy_gas");
176 }
177
178 std::vector<dealii::DataComponentInterpretation::DataComponentInterpretation> interpretation;
179 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
180 for (unsigned int d = 0; d < dim; ++d)
181 interpretation.push_back(dealii::DataComponentInterpretation::component_is_part_of_vector);
182 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
183
184 // add entries for two-phase case
185 if (two_phase)
186 {
187 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
188 for (unsigned int d = 0; d < dim; ++d)
189 interpretation.push_back(
190 dealii::DataComponentInterpretation::component_is_part_of_vector);
191 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
192 }
193
194 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
195 operation_pimpl->get_solution(),
196 names,
197 interpretation);
198 }
