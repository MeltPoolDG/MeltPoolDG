511 void
512 run_simulation(const std::string &parameter_file, const MPI_Comm mpi_communicator)
513 {
514 unsigned int dim = 0;
515 std::string case_name;
516
517 // Read and process parameters
518 {
519 ParameterHandler prm;
520 ParametersType parameters;
521 parameters.process_parameters_file(prm, parameter_file);
522
523 // Print number of processes and GIT hashes if verbosity level >= 3
524 if (parameters.base.verbosity_level >= 3)
525 {
526 dealii::ConditionalOStream pcout(std::cout,
527 Utilities::MPI::this_mpi_process(mpi_communicator) == 0);
528 Journal::print_decoration_line(pcout);
529 Journal::print_line(pcout,
530 "Running simulation on " +
531 std::to_string(Utilities::MPI::n_mpi_processes(mpi_communicator)) +
532 " ranks.");
533 Journal::print_decoration_line(pcout);
534 pcout << " - deal.II:" << std::endl
535 << " * branch: " << DEAL_II_GIT_BRANCH << std::endl
536 << " * revision: " << DEAL_II_GIT_REVISION << std::endl
537 << " * short: " << DEAL_II_GIT_SHORTREV << std::endl;
538 Journal::print_decoration_line(pcout);
539 }
540
541 if (Utilities::MPI::this_mpi_process(mpi_communicator) == 0 &&
542 parameters.base.do_print_parameters)
543 parameters.print_parameters(prm, std::cout, false /*print_details*/);
544
545 dim = parameters.base.dimension;
546 case_name = parameters.base.case_name;
547 }
548
549 // Try to run the simulation based on the dimension
550 try
551 {
552 if (dim == 1)
553 {
554 auto sim =
555 get_simulation_case<CaseType<1>>(case_name, parameter_file, mpi_communicator);
556 sim->create();
557 auto problem = std::make_unique<ProblemType<1>>(std::move(sim));
558 problem->run();
559 }
560 else if (dim == 2)
561 {
562 auto sim =
563 get_simulation_case<CaseType<2>>(case_name, parameter_file, mpi_communicator);
564 sim->create();
565 auto problem = std::make_unique<ProblemType<2>>(std::move(sim));
566 problem->run();
567 }
568 else if (dim == 3)
569 {
570 auto sim =
571 get_simulation_case<CaseType<3>>(case_name, parameter_file, mpi_communicator);
572 sim->create();
573 auto problem = std::make_unique<ProblemType<3>>(std::move(sim));
574 problem->run();
575 }
576 else
577 {
578 AssertThrow(false, ExcMessage("Dimension must be 1, 2, or 3."));
579 }
580 }
581 catch (std::exception &exc)
582 {
583 std::cerr << "\n\n----------------------------------------------------" << std::endl;
584 std::cerr << "Exception on processing: " << std::endl
585 << exc.what() << std::endl
586 << "Aborting!" << std::endl
587 << "----------------------------------------------------" << std::endl;
588 }
589 catch (...)
590 {
591 std::cerr << "\n\n----------------------------------------------------" << std::endl;
592 std::cerr << "Unknown exception!" << std::endl
593 << "Aborting!" << std::endl
594 << "----------------------------------------------------" << std::endl;
595 }
596 }

598 template <typename Parameters, template <int> class Case, template <int> class Problem>
599 void
600 default_main(int argc, char *argv[], MPI_Comm mpi_comm)
601 {
602 // Ensure at least one input file is provided
603 if (argc < 2)
604 {
605 AssertThrow(false, dealii::ExcMessage("no input file specified"));
606 return;
607 }
608
609 std::string input_file = argv[argc - 1]; // Last argument as input file
610
611 // Handle help options
612 if (argc == 3 && (std::string(argv[1]) == "--help" || std::string(argv[1]) == "--help-detail"))
613 {
614 dealii::ParameterHandler prm;
615 Parameters parameters;
616 parameters.process_parameters_file(prm, input_file);
617
618 if (dealii::Utilities::MPI::this_mpi_process(mpi_comm) == 0)
619 parameters.print_parameters(prm, std::cout, std::string(argv[1]) == "--help-detail");
620
621 return;
622 }
623
624 AssertThrow(argc < 4,
625 dealii::ExcMessage(
626 "The provided number of command line parameters is not supported."));
627
628 run_simulation<Parameters, Case, Problem>(input_file, mpi_comm);
629 }
