188 {
189 if (std::abs(phi) < cutoff)
190 return eps * std::log((1. + phi) / (1. - phi));
191 else if (phi >= cutoff)
192 return eps * std::log((1. + cutoff) / (1. - cutoff));
193 else /*( phi <= -cutoff )*/
