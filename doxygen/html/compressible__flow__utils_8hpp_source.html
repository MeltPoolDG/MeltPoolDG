196 template <int dim, typename number>
197 inline DEAL_II_ALWAYS_INLINE //
198 dealii::Tensor<1, dim, dealii::VectorizedArray<number>>
199 calculate_velocity(
200 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables)
201 {
202 const dealii::VectorizedArray<number> inverse_density =
203 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
204
205 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity;
206 for (unsigned int d = 0; d < dim; ++d)
207 velocity[d] = conserved_variables[1 + d] * inverse_density;
208
209 return velocity;
210 }

212 template <int dim, typename number>
213 inline DEAL_II_ALWAYS_INLINE //
214 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
215 calculate_grad_velocity(
216 const CompressibleFlowTypes::ConservedVariablesType<dim, number> &conserved_variables,
217 const CompressibleFlowTypes::ConservedVariablesGradType<dim, number>
218 &grad_conserved_variables)
219 {
220 const dealii::VectorizedArray<number> inverse_density =
221 dealii::VectorizedArray<number>(1.) / conserved_variables[0];
222 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
223 calculate_velocity<dim, number>(conserved_variables);
224
225 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> grad_rho;
226 for (unsigned int d = 0; d < dim; ++d)
227 grad_rho[d] = grad_conserved_variables[0][d];
228
229 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_rho_velocity;
230 for (unsigned int d = 0; d < dim; ++d)
231 for (unsigned int e = 0; e < dim; ++e)
232 grad_rho_velocity[d][e] = grad_conserved_variables[1 + d][e];
233
234 dealii::Tensor<2, dim, dealii::VectorizedArray<number>> grad_velocity;
235 for (unsigned int d = 0; d < dim; ++d)
236 for (unsigned int e = 0; e < dim; ++e)
237 grad_velocity[d][e] =
238 inverse_density * (grad_rho_velocity[d][e] - velocity[d] * grad_rho[e]);
239
240 return grad_velocity;
241 }

243 template <int dim, typename Number>
244 void
245 calculate_penalty_parameter(
246 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
247 const dealii::MatrixFree<dim, Number> &matrix_free,
248 const std::string &domain_representation_type,
249 const unsigned int dof_index,
250 const Number scaling_factor)
251 {
252 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
253 array_penalty_parameter.resize(n_cells);
254
255 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
256 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
257 unsigned int const degree = fe.degree;
258
259 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
260 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
261 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
262 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
263
264 auto const reference_cells =
265 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
266 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
267
268 auto const quadrature = reference_cells[0].template get_gauss_type_quadrature<dim>(degree + 1);
269 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
270
271 auto const face_quadrature =
272 reference_cells[0].face_reference_cell(0).template get_gauss_type_quadrature<dim - 1>(degree +
273 1);
274 dealii::FEFaceValues<dim> fe_face_values(mapping,
275 fe,
276 face_quadrature,
277 dealii::update_JxW_values);
278
279 if (domain_representation_type == "fitted")
280 {
281 for (unsigned int i = 0; i < n_cells; ++i)
282 {
283 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
284 {
285 typename dealii::DoFHandler<dim>::cell_iterator cell =
286 matrix_free.get_cell_iterator(i, v, dof_index);
287 fe_values.reinit(cell);
288
289 // calculate cell volume
290 Number volume = 0;
291 for (unsigned int q = 0; q < quadrature.size(); ++q)
292 {
293 volume += fe_values.JxW(q);
294 }
295
296 // calculate surface area
297 Number surface_area = 0;
298 for (unsigned int const f : cell->face_indices())
299 {
300 fe_face_values.reinit(cell, f);
301 Number const factor =
302 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
303 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
304 {
305 surface_area += fe_face_values.JxW(q) * factor;
306 }
307 }
308
309 array_penalty_parameter[i][v] = surface_area / volume * fac;
310 }
311 }
312 }
313 else if (domain_representation_type == "cut")
314 {
315 for (unsigned int i = 0; i < n_cells; ++i)
316 {
317 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
318 {
319 typename dealii::DoFHandler<dim>::cell_iterator cell =
320 matrix_free.get_cell_iterator(i, v, dof_index);
321
322 // simplified computation for hypercube elements
323 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
324 }
325 }
326 }
327 else
328 AssertThrow(false,
329 dealii::ExcMessage("The domain representation type '" +
330 domain_representation_type + "' is not supported."));
331 }

333 template <int dim, typename number>
334 void
335 update_primitive_variables_solution(
336 dealii::LinearAlgebra::distributed::Vector<number> &solution_primitive_variables,
337 const dealii::LinearAlgebra::distributed::Vector<number> &solution,
338 const ScratchData<dim, dim, number> &scratch_data,
339 const unsigned int dof_idx,
340 const unsigned int quad_idx,
341 const CompressibleFlowMaterial<dim, number> *material_liquid,
342 const CompressibleFlowMaterial<dim, number> *material_gas)
343 {
344 const dealii::MatrixFree<dim, number, dealii::VectorizedArray<number>> &matrix_free =
345 scratch_data.get_matrix_free();
346 unsigned int n_support_points_per_cell = scratch_data.get_n_dofs_per_cell(dof_idx) / (dim + 2);
347
348 auto process_cell = [&](CutUtil::CellCategory category,
349 unsigned int first_component,
350 const CompressibleFlowMaterial<dim, number> *material,
351 const unsigned int cell_batch) {
352 if (!material)
353 return;
354 auto eval = FECellIntegrator<dim, dim + 2, number>(
355 matrix_free, dof_idx, quad_idx, first_component, category);
356 eval.reinit(cell_batch);
357 eval.read_dof_values(solution);
358
359 for (unsigned int i = 0; i < n_support_points_per_cell; ++i)
360 {
361 const auto &u_cons = eval.get_dof_value(i);
362 auto u_prim = material->eos_utils->convert_conservative_into_primitive_variables(u_cons);
363 eval.submit_dof_value(u_prim, i);
364 }
365
366 eval.set_dof_values(solution_primitive_variables);
367 };
368
369 if (!matrix_free.get_dof_info(dof_idx).cell_active_fe_index.empty())
370 {
371 // using hp::FECollection (cutFEM/DG)
372 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
373 {
374 const auto cell_category = matrix_free.get_cell_category(cell_batch);
375
376 if (cell_category == CutUtil::CellCategory::liquid)
377 {
378 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
379 }
380 else if (cell_category == CutUtil::CellCategory::gas)
381 {
382 process_cell(CutUtil::CellCategory::gas, dim + 2, material_gas, cell_batch);
383 }
384 else if (cell_category == CutUtil::CellCategory::intersected)
385 {
386 process_cell(CutUtil::CellCategory::intersected, 0, material_liquid, cell_batch);
387 process_cell(CutUtil::CellCategory::intersected, dim + 2, material_gas, cell_batch);
388 }
389 }
390 }
391 else
392 {
393 // not using hp::FECollection (fitted mesh)
394 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
395 {
396 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
397 }
398 }
