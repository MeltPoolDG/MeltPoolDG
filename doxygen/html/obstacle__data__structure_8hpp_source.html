702 template <int dim, typename number, typename ObstacleType>
703 template <typename ObstacleContainer>
704 void
705 CellListParticleHandler<dim, number, ObstacleType>::get_obstacles_in_cell(
706 const dealii::TriaIterator<dealii::CellAccessor<dim>> &cell,
707 ObstacleContainer &particles)
708 {
709 std::size_t current_size = particles.size();
710
711 // Find all relevant particles for the given cell and add them to the provided container.
712 find_relevant_particles(find_particle_cache_cell(cell), particles);
713
714 // Remove duplicates from the container while ensuring that the particles which were already
715 // present before the function call are not removed.
716 auto old_end = particles.begin() + current_size;
717 auto write_ptr = old_end;
718 for (auto read_ptr = old_end; read_ptr != particles.end(); ++read_ptr)
719 {
720 auto found_it = std::find_if(particles.begin(), old_end, [&read_ptr](const auto &particle) {
721 return particle.id() == read_ptr->id();
722 });
723 if (found_it == old_end)
724 {
725 if (write_ptr->id() != read_ptr->id())
726 *write_ptr = std::move(*read_ptr);
727 ++write_ptr;
728 }
729 }
730 particles.erase(write_ptr, particles.end());
731 }

733 template <int dim, typename number, typename ObstacleType>
734 template <typename ObstacleContainer>
735 void
736 CellListParticleHandler<dim, number, ObstacleType>::find_relevant_particles(
737 const dealii::TriaIterator<dealii::CellAccessor<dim>> &cell,
738 ObstacleContainer &particles)
739 {
740 Assert(cell->level() == cell_particle_cache.cell_level,
741 dealii::ExcMessage(
742 "The function only supports searching for particles in cells on the same level as the "
743 "level on which particles are cached."));
744
745 const auto push_back_particles_from_cell = [&](const auto &current_cell) {
746 for (dealii::Particles::ParticleIterator<dim> &particle :
747 cell_particle_cache.locally_owned_particles[current_cell->index()])
748 {
749 Assert(particle.state() == dealii::IteratorState::valid, dealii::ExcInternalError());
750 particles.emplace_back(*particle);
751 }
752
753 for (const typename dealii::Particles::PropertyPool<dim>::Handle particle_handle :
754 cell_particle_cache.ghost_particles[current_cell->index()])
755 {
756 Assert(particle_handle < ghost_particles_property_pool.n_registered_slots(),
757 dealii::ExcInternalError());
758 particles.emplace_back(ghost_particles_property_pool, particle_handle);
759 }
760 };
761
762 // Adjacent cells
763 for (const dealii::TriaIterator<dealii::CellAccessor<dim>> &current_cell :
764 triangulation_level_cache.adjacent_cells.get_adjacent_cells(cell))
765 {
766 push_back_particles_from_cell(current_cell);
767 }
768
769 // Current cell
770 push_back_particles_from_cell(cell);
771 }
