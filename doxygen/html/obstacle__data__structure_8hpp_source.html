649 template <int dim, typename number, typename ObstacleType>
650 template <typename ObstacleContainer>
651 void
652 CellListParticleHandler<dim, number, ObstacleType>::get_obstacles_in_cell(
653 const dealii::TriaIterator<dealii::CellAccessor<dim>> &cell,
654 ObstacleContainer &particles)
655 {
656 std::size_t current_size = particles.size();
657
658 // Find all relevant particles for the given cell and add them to the provided container.
659 find_relevant_particles(find_particle_cache_cell(cell), particles);
660
661 // Remove duplicates from the container while ensuring that the particles which were already
662 // present before the function call are not removed.
663 auto old_end = particles.begin() + current_size;
664 auto write_ptr = old_end;
665 for (auto read_ptr = old_end; read_ptr != particles.end(); ++read_ptr)
666 {
667 auto found_it = std::find_if(particles.begin(), old_end, [&read_ptr](const auto &particle) {
668 return particle.id() == read_ptr->id();
669 });
670 if (found_it == old_end)
671 {
672 if (write_ptr->id() != read_ptr->id())
673 *write_ptr = std::move(*read_ptr);
674 ++write_ptr;
675 }
676 }
677 particles.erase(write_ptr, particles.end());
678 }

680 template <int dim, typename number, typename ObstacleType>
681 template <typename ObstacleContainer>
682 void
683 CellListParticleHandler<dim, number, ObstacleType>::find_relevant_particles(
684 const dealii::TriaIterator<dealii::CellAccessor<dim>> &cell,
685 ObstacleContainer &particles)
686 {
687 Assert(cell->level() == cell_particle_cache.cell_level,
688 dealii::ExcMessage(
689 "The function only supports searching for particles in cells on the same level as the "
690 "level on which particles are cached."));
691
692 const auto push_back_particles_from_cell = [&](const auto &current_cell) {
693 for (dealii::Particles::ParticleIterator<dim> &particle :
694 cell_particle_cache.locally_owned_particles[current_cell->index()])
695 {
696 Assert(particle.state() == dealii::IteratorState::valid, dealii::ExcInternalError());
697 particles.emplace_back(*particle);
698 }
699
700 for (const typename dealii::Particles::PropertyPool<dim>::Handle particle_handle :
701 cell_particle_cache.ghost_particles[current_cell->index()])
702 {
703 Assert(particle_handle < ghost_particles_property_pool.n_registered_slots(),
704 dealii::ExcInternalError());
705 particles.emplace_back(ghost_particles_property_pool, particle_handle);
706 }
707 };
708
709 // Adjacent cells
710 for (const dealii::TriaIterator<dealii::CellAccessor<dim>> &current_cell :
711 triangulation_level_cache.adjacent_cells.get_adjacent_cells(cell))
712 {
713 push_back_particles_from_cell(current_cell);
714 }
715
716 // Current cell
717 push_back_particles_from_cell(cell);
718 }
