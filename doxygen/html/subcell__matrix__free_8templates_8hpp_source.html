23 template <int dim>
24 void
25 SubcellIndexInfo<dim>::reinit(const unsigned int n_subcells_1d_in)
26 {
27 n_subcells_1d = n_subcells_1d_in;
28 n_subcells = dealii::Utilities::fixed_power<dim>(n_subcells_1d);
29 // A zero-dimensional cell, i.e. the face of a one-dimensional cell, has exactly one subcell on
30 // the face.
31 if constexpr (dim > 0)
32 n_subcells_face = dealii::Utilities::fixed_power<dim - 1>(n_subcells_1d);
33 else
34 n_subcells_face = 1;
35 n_intra_cell_faces = dim * (n_subcells_1d - 1) * n_subcells_face;
36 n_haloed_subcells = dealii::Utilities::fixed_power<dim>(n_subcells_1d + 2);
37
38 // stride of the lexicographic subcell index in each direction
39 std::array<unsigned int, dim> strides;
40 unsigned int stride = 1;
41 for (unsigned int d = 0; d < dim; ++d)
42 {
43 strides[d] = stride;
44 stride *= n_subcells_1d;
45 }
46
47 // On a face normal to the direction d, deal.II orders the quadrature points along the
48 // tangential directions (d + 1) % dim and (d + 2) % dim, with the first one varying fastest.
49 // In 3D, this is the (y, z) system on the faces 0 and 1, the (z, x) system on the faces 2 and 3
50 // and the (x, y) system on the faces 4 and 5.
51 if constexpr (dim > 0)
52 for (unsigned int face_no = 0; face_no < 2 * dim; ++face_no)
53 {
54 const unsigned int direction = face_no / 2;
55 const unsigned int layer = (face_no % 2) * (n_subcells_1d - 1);
56
57 face_to_cell_subcell_indices[face_no].resize(n_subcells_face);
58 for (unsigned int q_face = 0; q_face < n_subcells_face; ++q_face)
59 {
60 unsigned int subcell = layer * strides[direction];
61 unsigned int remaining_index = q_face;
62 for (unsigned int k = 0; k + 1 < dim; ++k)
63 {
64 subcell += (remaining_index % n_subcells_1d) * strides[(direction + 1 + k) % dim];
65 remaining_index /= n_subcells_1d;
66 }
67 face_to_cell_subcell_indices[face_no][q_face] = subcell;
68 }
69 }
70
71 // Looping over the subcells in lexicographic order and skipping the last subcell in the normal
72 // direction gives the faces in the order described in FESubcellEvaluation.
73 intra_cell_face_to_subcells.clear();
74 intra_cell_face_directions.clear();
75 intra_cell_face_to_subcells.reserve(n_intra_cell_faces);
76 intra_cell_face_directions.reserve(n_intra_cell_faces);
77 for (unsigned int d = 0; d < dim; ++d)
78 for (unsigned int subcell = 0; subcell < n_subcells; ++subcell)
79 if ((subcell / strides[d]) % n_subcells_1d < n_subcells_1d - 1)
80 {
81 intra_cell_face_to_subcells.push_back({{subcell, subcell + strides[d]}});
82 intra_cell_face_directions.push_back(d);
83 }
84 AssertDimension(intra_cell_face_to_subcells.size(), n_intra_cell_faces);
85
86 subcell_cartesian_indices.resize(n_subcells);
87 for (unsigned int subcell = 0; subcell < n_subcells; ++subcell)
88 {
89 for (unsigned int d = 0; d < dim; ++d)
90 subcell_cartesian_indices[subcell][d] = (subcell / strides[d]) % n_subcells_1d;
91 }
92
93 // The haloed layout adds one halo subcell in front of and one behind the subcells of the cell
94 // in each direction, which shifts the index in each direction by one. The neighbor in the
95 // direction of the face face_no lies one stride below the subcell for even and one stride above
96 // for odd face numbers.
97 std::array<unsigned int, dim> haloed_strides;
98 unsigned int haloed_stride = 1;
99 for (unsigned int d = 0; d < dim; ++d)
100 {
101 haloed_strides[d] = haloed_stride;
102 haloed_stride *= n_subcells_1d + 2;
103 }
104
105 subcell_index_to_haloed_vector_index.resize(n_subcells);
106 subcell_index_to_haloed_neighbor_vector_index.reinit(n_subcells, 2 * dim);
107 for (unsigned int subcell = 0; subcell < n_subcells; ++subcell)
108 {
109 unsigned int haloed_index = 0;
110 for (unsigned int d = 0; d < dim; ++d)
111 haloed_index += ((subcell / strides[d]) % n_subcells_1d + 1) * haloed_strides[d];
112 subcell_index_to_haloed_vector_index[subcell] = haloed_index;
113
114 for (unsigned int face_no = 0; face_no < 2 * dim; ++face_no)
115 subcell_index_to_haloed_neighbor_vector_index[subcell][face_no] =
116 (face_no % 2 == 0) ? haloed_index - haloed_strides[face_no / 2] :
117 haloed_index + haloed_strides[face_no / 2];
118 }
119 }

121 template <int dim, typename number>
122 void
123 SubcellShapeInfo<dim, number>::reinit(const dealii::Quadrature<1> &quadrature_1d)
124 {
125 const unsigned int n_subcells_1d = quadrature_1d.size();
126 cell_index_info.reinit(n_subcells_1d);
127 face_index_info.reinit(n_subcells_1d);
128
129 // The subcell faces lie at the cumulative sums of the quadrature weights, and the geometric
130 // center of a subcell is the midpoint between its two faces.
131 subcell_centers_1d.resize(n_subcells_1d);
132 subcell_widths_1d.resize(n_subcells_1d);
133 number lower_face = 0.;
134 for (unsigned int i = 0; i < n_subcells_1d; ++i)
135 {
136 subcell_widths_1d[i] = quadrature_1d.weight(i);
137 subcell_centers_1d[i] = lower_face + 0.5 * subcell_widths_1d[i];
138 lower_face += subcell_widths_1d[i];
139 }

179 const dealii::LinearAlgebra::distributed::Vector<number> &src) const
180 {
181 Assert(matrix_free != nullptr, dealii::ExcNotInitialized());
182 Assert(src.get_partitioner()->locally_owned_range() ==

184 dealii::ExcMessage(

187
188 // The exchange is completed before returning, so it cannot interfere with the ghost exchange

192 0,
193 dealii::ArrayView<const number>(src.begin(), src.locally_owned_size()),
194 dealii::make_array_view(extra_cell_import_values),
195 dealii::make_array_view(extra_cell_ghost_values),
196 requests);
197 extra_cell_partitioner->export_to_ghosted_array_finish(
198 dealii::make_array_view(extra_cell_ghost_values), requests);
199
200 ghost_values_source = &src;

326 template <int dim, typename number>
327 void
328 SubcellMatrixFree<dim, number>::setup_face_neighbors()
329 {
330 const unsigned int n_cell_batches =
331 matrix_free->n_cell_batches() + matrix_free->n_ghost_cell_batches();
332 face_neighbors.reinit(n_cell_batches, 2 * dim);
333
334 // map from the global active cell index of an extra cell to its index in extra_cells
335 std::map<unsigned int, unsigned int> extra_cell_indices;
336 extra_cells.clear();
337
338 for (unsigned int cell_batch = 0; cell_batch < n_cell_batches; ++cell_batch)
339 for (unsigned int lane = 0; lane < matrix_free->n_active_entries_per_cell_batch(cell_batch);
340 ++lane)
341 {
342 const typename dealii::DoFHandler<dim>::cell_iterator cell =
343 matrix_free->get_cell_iterator(cell_batch, lane, dof_no);
344 const bool is_owned_cell = cell_batch < matrix_free->n_cell_batches();
345
346 for (unsigned int face_no = 0; face_no < 2 * dim; ++face_no)
347 {
348 FaceNeighbors &neighbors = face_neighbors(cell_batch, face_no);
349
350 const bool is_periodic_face = cell->has_periodic_neighbor(face_no);
351 if (cell->at_boundary(face_no) and not is_periodic_face)
352 {
353 neighbors.boundary_ids[lane] = cell->face(face_no)->boundary_id();
354 neighbors.all_held = false;
355 continue;
356 }
357
358 const typename dealii::DoFHandler<dim>::cell_iterator neighbor =
359 cell->neighbor_or_periodic_neighbor(face_no);
360
361 // The face neighbors of ghost cells outside of the ghost layer are not available, see
362 // the class description. Note that artificial cells may be coarser than the actual
363 // mesh, so this check has to come before the one for hanging nodes.
364 if (neighbor->is_active() and neighbor->is_artificial())
365 {
366 Assert(not is_owned_cell, dealii::ExcInternalError());
367 neighbors.all_held = false;
368 continue;
369 }
370
371 AssertThrow(neighbor->is_active() and neighbor->level() == cell->level(),
372 dealii::ExcNotImplemented(
373 "SubcellMatrixFree does not support hanging nodes."));
374
375 AssertThrow(
376 (is_periodic_face ? cell->periodic_neighbor_face_no(face_no) :
377 cell->neighbor_face_no(face_no)) == (face_no ^ 1),
378 dealii::ExcNotImplemented(
379 "SubcellMatrixFree requires neighboring cells with aligned reference axes, i.e., the "
380 "face shared by two neighboring cells must have the local face numbers 0<->1, 2<->3, "
381 "or 4<->5. This is violated, e.g., by neighboring cells rotated by 180 degrees "
382 "against each other."));
383
384 if constexpr (dim == 3)
385 AssertThrow(cell->combined_face_orientation(face_no) ==
386 dealii::numbers::default_geometric_orientation and
387 neighbor->combined_face_orientation(face_no ^ 1) ==
388 dealii::numbers::default_geometric_orientation,
389 dealii::ExcNotImplemented(
390 "SubcellMatrixFree requires faces in standard orientation."));
391
392 neighbors.extents[lane] = neighbor->extent_in_direction(face_no / 2);
393
394 const unsigned int neighbor_cell_id =
395 matrix_free->get_matrix_free_cell_index(neighbor);
396 if (neighbor_cell_id != dealii::numbers::invalid_unsigned_int)
397 {
398 neighbors.cell_ids[lane] = neighbor_cell_id;
399 continue;
400 }
401
402 AssertThrow(
403 not is_owned_cell,
404 dealii::ExcMessage(
405 "The face neighbor of a locally owned cell is not held by the MatrixFree object. "
406 "Set MatrixFree::AdditionalData::hold_all_faces_to_owned_cells = true and enable "
407 "face integrals."));
408
409 const unsigned int next_extra_cell = extra_cells.size();
410 const std::pair<std::map<unsigned int, unsigned int>::iterator, bool> inserted =
411 extra_cell_indices.insert({neighbor->active_cell_index(), next_extra_cell});
412 if (inserted.second)
413 extra_cells.push_back(neighbor);
414
415 neighbors.extra_cell_ids[lane] = inserted.first->second;
416 neighbors.all_held = false;
417 }
418 }
419 }

443 template <int dim, typename number>
444 void
445 SubcellMatrixFree<dim, number>::setup_extra_cells()
446 {
447 const std::vector<unsigned int> &lexicographic_numbering =
448 matrix_free->get_shape_info(dof_no, quad_no).lexicographic_numbering;
449
450 // DoF indices of the extra cells, with the DoFs of a cell in the order of FEEvaluation
451 std::vector<dealii::types::global_dof_index> extra_cell_dof_indices(extra_cells.size() *
452 n_dofs_per_cell);
453 std::vector<dealii::types::global_dof_index> cell_dof_indices(n_dofs_per_cell);
454 for (unsigned int extra_cell = 0; extra_cell < extra_cells.size(); ++extra_cell)
455 {
456 extra_cells[extra_cell]->get_dof_indices(cell_dof_indices);
457 for (unsigned int i = 0; i < n_dofs_per_cell; ++i)
458 extra_cell_dof_indices[extra_cell * n_dofs_per_cell + i] =
459 cell_dof_indices[lexicographic_numbering[i]];
460 }
461
462 std::vector<dealii::types::global_dof_index> sorted_dof_indices = extra_cell_dof_indices;
463 std::ranges::sort(sorted_dof_indices);
464 sorted_dof_indices.erase(std::unique(sorted_dof_indices.begin(), sorted_dof_indices.end()),
465 sorted_dof_indices.end());
466
467 const std::shared_ptr<const dealii::Utilities::MPI::Partitioner> &vector_partitioner =
468 matrix_free->get_vector_partitioner(dof_no);
469 dealii::IndexSet ghost_dofs(vector_partitioner->size());
470 ghost_dofs.add_indices(sorted_dof_indices.begin(), sorted_dof_indices.end());
471
472 extra_cell_partitioner = std::make_shared<const dealii::Utilities::MPI::Partitioner>(
473 vector_partitioner->locally_owned_range(),
474 ghost_dofs,
475 vector_partitioner->get_mpi_communicator());
476
477 extra_cell_dof_positions.resize(extra_cell_dof_indices.size());
478 for (unsigned int i = 0; i < extra_cell_dof_indices.size(); ++i)
479 extra_cell_dof_positions[i] =
480 extra_cell_partitioner->global_to_local(extra_cell_dof_indices[i]) -
481 extra_cell_partitioner->locally_owned_size();
482
483 extra_cell_ghost_values.resize(extra_cell_partitioner->n_ghost_indices());
484 extra_cell_import_values.resize(extra_cell_partitioner->n_import_indices());
485 }
