48 template <int dim, typename number>
49 dealii::Tensor<1, axial_dim<dim>, number>
50 compute_torque(const dealii::Tensor<1, dim, number> &force,
51 const dealii::Tensor<1, dim, number> &lever_arm)
52 {
53 dealii::Tensor<1, axial_dim<dim>, number> torque;
54 if constexpr (dim == 2)
55 torque[0] = lever_arm[0] * force[1] - lever_arm[1] * force[0];
56 if constexpr (dim == 3)
57 torque = dealii::cross_product_3d(lever_arm, force);
58 return torque;
59 }
