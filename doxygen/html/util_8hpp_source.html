292 template <int dim>
293 bool
294 face_has_ghost_penalty(const dealii::NonMatching::MeshClassifier<dim> &mesh_classifier,
295 const typename dealii::Triangulation<dim>::active_cell_iterator &cell,
296 const unsigned int face_index,
297 const dealii::NonMatching::LocationToLevelSet &inactive_location =
298 dealii::NonMatching::LocationToLevelSet::inside)
299 {
300 if (cell->at_boundary(face_index))
301 return false;
302
303 const dealii::NonMatching::LocationToLevelSet cell_location =
304 mesh_classifier.location_to_level_set(cell);
305 const dealii::NonMatching::LocationToLevelSet neighbor_location =
306 mesh_classifier.location_to_level_set(cell->neighbor(face_index));
307
308 if (cell_location == dealii::NonMatching::LocationToLevelSet::intersected and
309 neighbor_location != inactive_location)
310 return true;
311
312 if (neighbor_location == dealii::NonMatching::LocationToLevelSet::intersected and
313 cell_location != inactive_location)
314 return true;
315
316 return false;
317 }
