299 template <int dim>
300 bool
301 face_has_ghost_penalty(const dealii::NonMatching::MeshClassifier<dim> &mesh_classifier,
302 const typename dealii::Triangulation<dim>::active_cell_iterator &cell,
303 const unsigned int face_index,
304 const dealii::NonMatching::LocationToLevelSet &inactive_location =
305 dealii::NonMatching::LocationToLevelSet::inside)
306 {
307 if (cell->at_boundary(face_index))
308 return false;
309
310 const dealii::NonMatching::LocationToLevelSet cell_location =
311 mesh_classifier.location_to_level_set(cell);
312 const dealii::NonMatching::LocationToLevelSet neighbor_location =
313 mesh_classifier.location_to_level_set(cell->neighbor(face_index));
314
315 if (cell_location == dealii::NonMatching::LocationToLevelSet::intersected and
316 neighbor_location != inactive_location)
317 return true;
318
319 if (neighbor_location == dealii::NonMatching::LocationToLevelSet::intersected and
320 cell_location != inactive_location)
321 return true;
322
323 return false;
324 }
