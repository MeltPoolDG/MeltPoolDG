160 void
161 add(const InterfaceState<dim, number> &state,
162 const InterfaceJump<number> &jump,
163 InterfaceValueFlux<dim, number> &flux,
164 dealii::VectorizedArray<number> &interface_velocity) const
165 {
166 const auto [flux_liquid, flux_gas, velocity] =
167 calculate_convective_interface_flux_HLLP0(state.liquid,
168 state.gas,
169 state.normal,
170 convective_kernel_liquid,
171 convective_kernel_gas,
172 jump.m_dot_evap);
173
174 // contraction is done with the outward normal of the corresponding phase.
175 // state.normal is defined as pointing out of the liquid phase, so we need to flip the sign
176 // for the gas phase.
177 flux.liquid -= contract_tensor_with_vector(flux_liquid, state.normal);
178 flux.gas -= contract_tensor_with_vector(flux_gas, -state.normal);
179
180 // set interface velocity from the Riemann solver (normal velocity of the interface with
181 // respect to the normal pointing out of the liquid phase)
182 interface_velocity = velocity;
183 }
184
185 private:
187 const CompressibleFlow::ConvectiveFlux<dim, number> convective_kernel_liquid;
188 const CompressibleFlow::ConvectiveFlux<dim, number> convective_kernel_gas;
189 };

255 class SIPGViscousCoupling
256 {
257 public:
267 SIPGViscousCoupling(
268 const CompressibleFlow::MaterialPhaseData<number> &material_liquid,
269 const CompressibleFlow::MaterialPhaseData<number> &material_gas,
270 const typename CompressibleFlowPhaseCouplingData<number>::HLLP0_SIPG &parameters,
271 const number latent_heat_of_vaporization,
272 const int fe_degree)
273 : material_liquid(material_liquid)
274 , material_gas(material_gas)
275 , diffusive_kernel_liquid(material_liquid)
276 , diffusive_kernel_gas(material_gas)
277 , latent_heat_of_vaporization(latent_heat_of_vaporization)
278 , interior_penalty_parameter(parameters.interior_penalty_parameter_interface)
279 , penalty_factor(
280 std::min(material_liquid.dynamic_viscosity / material_liquid.reference_density,
281 material_gas.dynamic_viscosity / material_gas.reference_density) *
282 (fe_degree + 1.) * (fe_degree + 1.))
283 {
284 AssertThrow(dim == 1,
285 dealii::ExcMessage("The SIPG viscous coupling is only supported in 1D."));
286
287 std::tie(weight_liquid, weight_gas) =
288 calculate_viscous_average_weights(material_liquid, material_gas);
289 }

299 void
300 add(const InterfaceState<dim, number> &state,
301 const InterfaceJump<number> &jump,
302 InterfaceValueFlux<dim, number> &value_flux,
303 InterfaceGradientFlux<dim, number> &gradient_flux) const
304 {
305 // calculate the Dirichlet jump in conservative variables for the penalty term in the SIPG
306 // method
307 const auto J_Dir = calculate_Dirichlet_jump_SIPG<dim>(state.liquid,
308 state.gas,
309 material_liquid,
310 material_gas,
311 jump.m_dot_evap,
312 jump.temperature_jump);
313
314 // 1) terms tested with the test functions
315 const number penalty_parameter =
316 penalty_factor / state.cell_size * interior_penalty_parameter;
317
318 const auto [viscous_flux_liquid, viscous_flux_gas] =
319 calculate_viscous_interface_flux_SIPG(state.liquid,
320 state.gas,
321 state.normal,
322 J_Dir,
323 weight_liquid,
324 weight_gas,
325 penalty_parameter,
326 diffusive_kernel_liquid,
327 diffusive_kernel_gas,
328 latent_heat_of_vaporization,
329 jump.m_dot_evap,
330 jump.laser_heat_source);
331
332 value_flux.liquid += viscous_flux_liquid;
333 value_flux.gas -= viscous_flux_gas;
334
335 // 2) terms tested with the gradients of the test functions
336 const auto [gradient_flux_liquid, gradient_flux_gas] =
337 calculate_viscous_interface_flux_gradient_SIPG(state.liquid,
338 state.gas,
339 state.normal,
340 J_Dir,
341 weight_liquid,
342 weight_gas,
343 diffusive_kernel_liquid,
344 diffusive_kernel_gas);
345
346 gradient_flux.liquid -= gradient_flux_liquid;
347 gradient_flux.gas -= gradient_flux_gas;
348 }
349
350 private:
352 const CompressibleFlow::MaterialPhaseData<number> &material_liquid;
353 const CompressibleFlow::MaterialPhaseData<number> &material_gas;
354
356 const CompressibleFlow::DiffusiveFlux<dim, number> diffusive_kernel_liquid;
357 const CompressibleFlow::DiffusiveFlux<dim, number> diffusive_kernel_gas;
358
360 const number latent_heat_of_vaporization;
361
363 const number interior_penalty_parameter;
364
366 const number penalty_factor;
367
369 number weight_liquid;
370 number weight_gas;

483 template <int dim, typename number>
484 ViscousCoupling<dim, number>
485 create_viscous_interface_coupling(
486 const CompressibleFlow::MaterialPhaseData<number> &material_liquid,
487 const CompressibleFlow::MaterialPhaseData<number> &material_gas,
488 const CompressibleFlowPhaseCouplingData<number> &phase_coupling,
489 const number latent_heat_of_vaporization,
490 const int fe_degree,
491 const bool is_viscous)
492 {
493 if (not is_viscous)
494 return InviscidCoupling<dim, number>();
495
496 switch (phase_coupling.type)
497 {
498 case InterfaceNumericalMethod::HLLP0_and_SIPG:
499 return SIPGViscousCoupling<dim, number>(material_liquid,
500 material_gas,
501 phase_coupling.hllp0_and_sipg,
502 latent_heat_of_vaporization,
503 fe_degree);
504 case InterfaceNumericalMethod::HLLP0_and_penalty:
505 return PenaltyViscousCoupling<dim, number>(material_liquid,
506 material_gas,
507 phase_coupling.hllp0_and_penalty,
508 latent_heat_of_vaporization);
509 default:
510 AssertThrow(
511 false,
512 dealii::ExcMessage(
513 "The requested interface numerical method for viscous coupling is not implemented."));
514 }

524 template <int dim, typename number>
525 class InterfaceJumpModel
526 {
527 public:
533 explicit InterfaceJumpModel(
534 const CompressibleFlow::MultiphaseOperationScratchData<dim, number> &data)
535 : laser(data.phase_coupling.laser_heat_source)
536 , m_dot_evap_constant(data.phase_coupling.m_dot_evap)
537 , temperature_jump_constant(data.phase_coupling.temperature_jump)
538 {
539 if (data.phase_coupling.evaporation_model == EvaporationModelType::Knight)
540 evaporation_model_knight.emplace(data.phase_change.liquid_gas.reference_pressure,
541 data.phase_change.liquid_gas.boiling_temperature,
542 data.phase_change.liquid_gas.latent_heat_of_vaporization,
543 data.material_gas.specific_gas_constant,
544 data.material_gas.gamma);
545 else
546 AssertThrow(data.phase_coupling.evaporation_model == EvaporationModelType::constant,
547 dealii::ExcMessage("The given evaporation model is not supported."));
548 }
549
555 void
556 update_laser_heat_source(const number time) const
557 {
558 if (laser.do_ramp and time < laser.ramp_time)
559 {
560 const number factor = 0.5 * (1. - std::cos(std::numbers::pi * time / laser.ramp_time));
561
562 current_laser_heat_source = factor * laser.laser_power_density;
563 }
564 else
565 current_laser_heat_source = laser.laser_power_density;
566 }
567
575 InterfaceJump<number>
576 evaluate(const InterfaceState<dim, number> &state) const
577 {
578 InterfaceJump<number> jump;
579 jump.laser_heat_source = current_laser_heat_source;
580
581 if (evaporation_model_knight)
582 {
583 const dealii::VectorizedArray<number> T_liquid = state.liquid.temperature();
584 const dealii::VectorizedArray<number> vel_n_gas = state.gas.velocity() * state.normal;
585 const dealii::VectorizedArray<number> speed_of_sound_g = state.gas.speed_of_sound();
586 const dealii::VectorizedArray<number> Ma_g = vel_n_gas / speed_of_sound_g;
587
588 evaporation_model_knight->reinit(T_liquid[0], Ma_g[0]);
589 jump.m_dot_evap = evaporation_model_knight->get_evaporative_mass_flux();
590 jump.temperature_jump = evaporation_model_knight->get_temperature_jump();
591 }
592 else
593 {
594 jump.m_dot_evap = m_dot_evap_constant;
595 jump.temperature_jump = temperature_jump_constant;
596 }
597
598 return jump;
599 }
600
601 private:
603 const typename CompressibleFlowPhaseCouplingData<number>::LaserHeatSource laser;
604
606 mutable number current_laser_heat_source = 0.;
607
609 const number m_dot_evap_constant;
610
612 const number temperature_jump_constant;
613
615 mutable std::optional<Evaporation::EvaporationModelKnight<number>> evaporation_model_knight;
616 };
