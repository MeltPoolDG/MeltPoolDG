87 number
88 value(const dealii::Point<dim, number> &loc, const unsigned int component) const override
89 {
90 const auto inflow_ramp_up_scaling = [&]() -> number {
91 if (this->get_time() > inflow_ramp_up_duration or inflow_ramp_up_duration == 0. or
92 ramp_up_type == RampUpType::none)
93 return 1.;
94
95 switch (ramp_up_type)
96 {
97 case RampUpType::linear:
98 return this->get_time() / inflow_ramp_up_duration;
99 break;
100 case RampUpType::exponential:
101 return (std::exp(this->get_time() / inflow_ramp_up_duration) - 1) /
102 (std::numbers::e - 1);
103 break;
104 case RampUpType::cosine:
105 return 0.5 *
106 (1 - std::cos(std::numbers::pi * this->get_time() / inflow_ramp_up_duration));
107 break;
108 default:
109 AssertThrow(false, dealii::ExcMessage("The chosen ramp up type is not supported."));
110 }
111 };
112
113 const auto compute_conserved_energy = [&](const dealii::Point<dim, number> &loc) -> number {
114 dealii::Vector<number> velocity_vec(dim);
115 velocity->vector_value(loc, velocity_vec);
116 velocity_vec *= inflow_ramp_up_scaling() * density->value(loc, 0);
117 return density->value(loc, 0) * (energy->value(loc, 0) + 0.5 * velocity_vec.norm_sqr());
118 };
119
120 switch (component)
121 {
122 case 0:
123 return density->value(loc, 0);
124 break;
125 case 1:
126 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 0);
127 break;
128 case 2:
129 if constexpr (dim >= 2)
130 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 1);
131 else
132 return compute_conserved_energy(loc);
133 break;
134 case 3:
135 if constexpr (dim == 3)
136 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 2);
137 else if constexpr (dim == 2)
138 return compute_conserved_energy(loc);
139 else
140 AssertThrow(false,
141 dealii::ExcMessage(
142 "Component 3 is not valid for dim = " + std::to_string(dim) + "."));
143 break;
144 case 4:
145 if constexpr (dim == 3)
146 return compute_conserved_energy(loc);
147 else
148 AssertThrow(false,
149 dealii::ExcMessage(
150 "Component 4 is not valid for dim = " + std::to_string(dim) + "."));
151 break;
152 default:
153 AssertThrow(false, dealii::ExcMessage("The given component is not valid."));
154 }
155 }

216 std::shared_ptr<dealii::Function<dim>>
217 create_boundary_function(const number start_time)
218 {
219 switch (type)
220 {
221 case (::MeltPoolDG::CompressibleFlow::BoundaryConditionType::inflow): {
222 auto density_boundary_function =
223 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
224 density_boundary_function->initialize(
225 dealii::FunctionParser<dim>::default_variable_names(),
226 density,
227 typename dealii::FunctionParser<dim>::ConstMap());
228
229 auto velocity_boundary_function = std::make_shared<dealii::FunctionParser<dim>>(dim);
230 velocity_boundary_function->initialize(
231 dealii::FunctionParser<dim>::default_variable_names(),
232 velocity,
233 typename dealii::FunctionParser<dim>::ConstMap());
234
235 auto energy_boundary_function =
236 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
237 energy_boundary_function->initialize(
238 dealii::FunctionParser<dim>::default_variable_names(),
239 energy,
240 typename dealii::FunctionParser<dim>::ConstMap());
241
242 return std::make_shared<ConservativeFunction<dim, number>>(start_time,
243 density_boundary_function,
244 velocity_boundary_function,
245 energy_boundary_function,
246 inflow_ramp_up_duration,
247 inflow_ramp_up_type);
248 }
249 case (::MeltPoolDG::CompressibleFlow::BoundaryConditionType::
250 subsonic_outflow_fixed_energy): {
251 auto energy_boundary_function =
252 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
253 energy_boundary_function->initialize(
254 dealii::FunctionParser<dim>::default_variable_names(),
255 energy,
256 typename dealii::FunctionParser<dim>::ConstMap());
257 return energy_boundary_function;
258 }
259 case (::MeltPoolDG::CompressibleFlow::BoundaryConditionType::
260 subsonic_outflow_fixed_pressure): {
261 auto pressure_boundary_function =
262 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
263 pressure_boundary_function->initialize(
264 dealii::FunctionParser<dim>::default_variable_names(),
265 pressure,
266 typename dealii::FunctionParser<dim>::ConstMap());
267 return pressure_boundary_function;
268 }
269 default:
270 return std::make_shared<dealii::Functions::ConstantFunction<dim>>(0.);
271 }
