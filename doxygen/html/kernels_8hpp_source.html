20 template <int dim, typename number, IsConservedStateCompatible<dim> value_type>
21 inline DEAL_II_ALWAYS_INLINE //
22 FluxType<dim, number>
23 convective_flux(const DofStateView<dim, number, const value_type> &conserved_variables)
24 {
25 using FluxType = FluxType<dim, number>;
26
27 FluxType flux;
28 FluxView<dim, FluxType> flux_view(flux);
29 const dealii::VectorizedArray<number> pressure = conserved_variables.pressure();
30
31 for (unsigned int d = 0; d < dim; ++d)
32 {
33 flux_view.density_flux()[d] = conserved_variables.momentum(d);
34 for (unsigned int e = 0; e < dim; ++e)
35 flux_view.momentum_flux(e)[d] =
36 conserved_variables.momentum(e) * conserved_variables.velocity(d);
37 flux_view.momentum_flux(d)[d] += pressure;
38 flux_view.energy_flux()[d] =
39 conserved_variables.velocity(d) * (conserved_variables.total_energy() + pressure);
40 }
41 return flux;
42 }

52 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>>
53 viscous_stress_tensor(

55 &grad_velocity,

57 {

82 template <int dim,
83 typename number,
84 IsConservedStateCompatible<dim> value_type,
85 IsConservedGradientCompatible<dim> gradient_type>
86 inline DEAL_II_ALWAYS_INLINE //
87 FluxType<dim, number>
88 diffusive_flux(
89 const DofValueAndGradientStateView<dim, number, const value_type, const gradient_type>
90 &conserved_variables)
91 {
92 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
93 conserved_variables.velocity();
94
95 const dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>>
96 viscous_stress = viscous_stress_tensor(conserved_variables.grad_velocity(),
97 dealii::VectorizedArray<number>(
98 conserved_variables.dynamic_viscosity()));
99
100 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> neg_heat_flux =
101 conserved_variables.thermal_conductivity() * conserved_variables.grad_temperature();
102
103 FluxType<dim, number> flux;
104 FluxView<dim, FluxType<dim, number>> flux_view(flux);
105 for (unsigned int d = 0; d < dim; ++d)
106 {
107 // density
108 flux_view.density_flux()[d] = 0.0;
109
110 // momentum
111 for (unsigned int e = 0; e < dim; ++e)
112 flux_view.momentum_flux(e)[d] = viscous_stress[e][d];
113
114 // energy
115 flux_view.energy_flux()[d] = neg_heat_flux[d];
116 for (unsigned int e = 0; e < dim; ++e)
117 flux_view.energy_flux()[d] += velocity[e] * viscous_stress[d][e];
118 }
119
120 return flux;
121 }
