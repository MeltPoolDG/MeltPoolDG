143 template <typename number>
144 inline number
145 fresnel_reflectivity(const number n1, const number n2, const number cos_theta_i)
146 {
147 if (n1 == n2)
148 return 0.0;
149
150 const number cos_i = std::clamp(cos_theta_i, number(0), number(1));
151 const number sin2_i = std::max(number(0), number(1) - cos_i * cos_i);
152
153 const number eta = n1 / n2;
154 const number sin2_t = eta * eta * sin2_i;
155
156 // total internal reflection
157 if (sin2_t >= 1.0)
158 return 1.0;
159
160 const number cos_t = std::sqrt(number(1) - sin2_t);
161
162 const number r_s = (n1 * cos_i - n2 * cos_t) / (n1 * cos_i + n2 * cos_t);
163 const number r_p = (n2 * cos_i - n1 * cos_t) / (n2 * cos_i + n1 * cos_t);
164
165 return number(0.5) * (r_s * r_s + r_p * r_p);
166 }

199 template <int dim, typename number>
200 inline Reflection<dim, number>
201 reflect(const dealii::Tensor<1, dim, number> &direction,
202 dealii::Tensor<1, dim, number> normal,
203 const number energy,
204 const number refractive_index_incident,
205 const number refractive_index_material)
206 {
207 const number normal_length = normal.norm();
208 AssertThrow(normal_length > 0, dealii::ExcMessage("The interface normal must be non-zero."));
209 normal /= normal_length;
210 if (direction * normal > 0.0)
211 normal *= -1.0;
212
213 const number cos_theta_i = std::clamp(-(direction * normal), number(0), number(1));
214
215 const number reflectivity =
216 fresnel_reflectivity(refractive_index_incident, refractive_index_material, cos_theta_i);
217
218 Reflection<dim, number> result;
219 result.direction = direction - number(2) * (direction * normal) * normal;
220 result.direction /= result.direction.norm();
221 result.reflected_energy = reflectivity * energy;
222 result.absorbed_energy = (number(1) - reflectivity) * energy;
223 return result;
224 }

249 template <int dim, typename number>
250 std::vector<Ray<dim, number>>
251 make_laser_rays(const Point<dim> &center,
252 const Tensor<1, dim, number> &beam_direction,
253 const Function<dim, number> &intensity_profile,
254 const number radius,
255 const number spacing,
256 const unsigned int rank,
257 const unsigned int n_ranks)
258 {
259 AssertThrow(radius > 0, ExcMessage("The radius must be positive."));
260 AssertThrow(spacing > 0, ExcMessage("The ray spacing must be positive."));
261 AssertThrow(n_ranks > 0 && rank < n_ranks, ExcMessage("Invalid rank."));
262 AssertThrow(beam_direction.norm() > 0, ExcMessage("The beam direction must be non-zero."));
263
264 const Tensor<1, dim, number> direction = beam_direction / beam_direction.norm();
265
266 // Orthonormal basis
267 std::array<Tensor<1, dim, number>, dim - 1> basis;
268 if constexpr (dim == 2)
269 {
270 basis[0][0] = -direction[1];
271 basis[0][1] = direction[0];
272 }
273 else
274 {
275 // Helper axis: the coordinate axis that is most perpendicular to the beam.
276 unsigned int axis = 0;
277 for (unsigned int d = 1; d < dim; ++d)
278 if (std::abs(direction[d]) < std::abs(direction[axis]))
279 axis = d;
280
281 Tensor<1, dim, number> helper;
282 helper[axis] = 1;
283
284 basis[0] = cross_product_3d(direction, helper);
285 basis[0] /= basis[0].norm();
286 basis[1] = cross_product_3d(direction, basis[0]);
287 }
288 // Regular grid over [-R, R] per lateral direction
289 const std::size_t n_per_direction =
290 std::max<std::size_t>(1, static_cast<std::size_t>(std::ceil(2 * radius / spacing)));
291 const number delta = 2 * radius / n_per_direction;
292
293 std::size_t n_rays = 1;
294 number measure = 1; // area (3D) or length (2D) represented by one ray
295 for (unsigned int k = 0; k < dim - 1; ++k)
296 {
297 n_rays *= n_per_direction;
298 measure *= delta;
299 }
300
301 std::vector<Ray<dim, number>> rays;
302 rays.reserve(n_rays / n_ranks + 1);
303
304 // Round-robin over the ranks: ray i is created on rank i % n_ranks
305 for (std::size_t id = rank; id < n_rays; id += n_ranks)
306 {
307 Point<dim> origin = center;
308 std::size_t rest = id;
309 for (unsigned int k = 0; k < dim - 1; ++k)
310 {
311 const number offset = (rest % n_per_direction + number(0.5)) * delta - radius;
312 origin += offset * basis[k];
313 rest /= n_per_direction;
314 }
315
316 rays.push_back({origin, direction, intensity_profile.value(origin) * measure, id});
317 }
318
319 return rays;
320 }
