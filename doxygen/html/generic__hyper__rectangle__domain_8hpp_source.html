88 number
89 value(const dealii::Point<dim, number> &loc, const unsigned int component) const override
90 {
91 const auto inflow_ramp_up_scaling = [&]() -> number {
92 if (this->get_time() > inflow_ramp_up_duration or inflow_ramp_up_duration == 0.)
93 return 1.;
94
95 switch (ramp_up_type)
96 {
97 case RampUpType::linear:
98 return this->get_time() / inflow_ramp_up_duration;
99 break;
100 case RampUpType::exponential:
101 return (std::exp(this->get_time() / inflow_ramp_up_duration) - 1) /
102 (std::numbers::e - 1);
103 break;
104 case RampUpType::cosine:
105 return 0.5 *
106 (1 - std::cos(std::numbers::pi * this->get_time() / inflow_ramp_up_duration));
107 break;
108 default:
109 AssertThrow(false, dealii::ExcInternalError());
110 }
111 };
112
113 const auto compute_conserved_energy = [&](const dealii::Point<dim, number> &loc) -> number {
114 dealii::Vector<number> velocity_vec(dim);
115 velocity->vector_value(loc, velocity_vec);
116 velocity_vec *= inflow_ramp_up_scaling() * density->value(loc, 0);
117 return density->value(loc, 0) * (energy->value(loc, 0) + 0.5 * velocity_vec.norm_sqr());
118 };
119
120 switch (component)
121 {
122 case 0:
123 return density->value(loc, 0);
124 break;
125 case 1:
126 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 0);
127 break;
128 case 2:
129 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 1);
130 break;
131 case 3:
132 if constexpr (dim == 3)
133 return density->value(loc, 0) * inflow_ramp_up_scaling() * velocity->value(loc, 2);
134 else
135 return compute_conserved_energy(loc);
136 break;
137 case 4:
138 if constexpr (dim == 3)
139 return compute_conserved_energy(loc);
140 else
141 AssertThrow(false, dealii::ExcInternalError());
142 break;
143 default:
144 AssertThrow(false, dealii::ExcInternalError());
145 }
146 }

207 std::shared_ptr<dealii::Function<dim>>
208 create_boundary_function(const number start_time)
209 {
210 switch (type)
211 {
212 case (Flow::CompressibleBoundaryConditionType::inflow): {
213 auto density_boundary_function =
214 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
215 density_boundary_function->initialize(
216 dealii::FunctionParser<dim>::default_variable_names(),
217 density,
218 typename dealii::FunctionParser<dim>::ConstMap());
219
220 auto velocity_boundary_function = std::make_shared<dealii::FunctionParser<dim>>(dim);
221 velocity_boundary_function->initialize(
222 dealii::FunctionParser<dim>::default_variable_names(),
223 velocity,
224 typename dealii::FunctionParser<dim>::ConstMap());
225
226 auto energy_boundary_function =
227 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
228 energy_boundary_function->initialize(
229 dealii::FunctionParser<dim>::default_variable_names(),
230 energy,
231 typename dealii::FunctionParser<dim>::ConstMap());
232
233 return std::make_shared<ConservativeFunction<dim, number>>(start_time,
234 density_boundary_function,
235 velocity_boundary_function,
236 energy_boundary_function,
237 inflow_ramp_up_duration,
238 inflow_ramp_up_type);
239 }
240 case (Flow::CompressibleBoundaryConditionType::subsonic_outflow_fixed_energy): {
241 auto energy_boundary_function =
242 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
243 energy_boundary_function->initialize(
244 dealii::FunctionParser<dim>::default_variable_names(),
245 energy,
246 typename dealii::FunctionParser<dim>::ConstMap());
247 return energy_boundary_function;
248 }
249 case (Flow::CompressibleBoundaryConditionType::subsonic_outflow_fixed_pressure): {
250 auto pressure_boundary_function =
251 std::make_shared<dealii::FunctionParser<dim>>(1, start_time);
252 pressure_boundary_function->initialize(
253 dealii::FunctionParser<dim>::default_variable_names(),
254 pressure,
255 typename dealii::FunctionParser<dim>::ConstMap());
256 return pressure_boundary_function;
257 }
258 default:
259 return std::make_shared<dealii::Functions::ConstantFunction<dim>>(0.);
260 }

280 void
281 create_spatial_discretization() override
282 {
283 auto create_point_from_container =
284 []<typename T>(const T &container) -> dealii::Point<dim, number> {
285 dealii::Point<dim, number> point;
286 if constexpr (dim == 2)
287 point = dealii::Point<dim, number>(container[0], container[1]);
288 else if constexpr (dim == 3)
289 point = dealii::Point<dim, number>(container[0], container[1], container[2]);
290 return point;
291 };
292
293 this->triangulation =
294 std::make_shared<dealii::parallel::distributed::Triangulation<dim>>(this->mpi_communicator);
295
296 dealii::Point<dim, number> dimensions =
297 create_point_from_container(scenario_data.domain_dimensions);
298
299 dealii::GridGenerator::subdivided_hyper_rectangle(*this->triangulation,
300 scenario_data.domain_base_discretization,
301 dealii::Point<dim, number>(),
302 dimensions,
303 true);
304
305 this->triangulation->refine_global(this->parameters.base.global_refinements);
306
307 // Perform additional mesh refinement in the user-defined region of interest
308 if (scenario_data.region_of_interest_refinement_times > 0)
309 {
310 AssertThrow(scenario_data.region_of_interest_corner[0].size() == dim &&
311 scenario_data.region_of_interest_corner[1].size() == dim,
312 dealii::ExcMessage(
313 "Invalid size for region-of-interest corner points. "
314 "Both points must have size " +
315 std::to_string(dim) + ", but got sizes " +
316 std::to_string(scenario_data.region_of_interest_corner[0].size()) +
317 " and " +
318 std::to_string(scenario_data.region_of_interest_corner[1].size()) + "."));
319
320
321 std::vector<AMR::AMRRegion<dim, number>> regions;
322 regions.emplace_back(dealii::BoundingBox<dim, number>(
323 {create_point_from_container(scenario_data.region_of_interest_corner[0]),
324 create_point_from_container(scenario_data.region_of_interest_corner[1])}));
325 for (unsigned i = 0; i < scenario_data.region_of_interest_refinement_times; ++i)
326 {
327 AMR::set_refinement_flags_in_regions<dim, number>(*this->triangulation, regions);
328 this->triangulation->execute_coarsening_and_refinement();
329 }
330 }
