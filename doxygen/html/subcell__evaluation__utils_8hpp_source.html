50 // The lanes may lie on boundaries with different ids. The boundary value function is called
51 // once per id, and its result is only used in the lanes on a boundary with this id.
52 std::bitset<n_lanes> remaining_lanes = lanes;
53 while (remaining_lanes.any())
54 {
55 unsigned int first_lane = 0;
56 while (not remaining_lanes[first_lane])
57 ++first_lane;
58 const dealii::types::boundary_id boundary_id = boundary_ids[first_lane];
59
60 const value_type boundary_id_value =
61 boundary_value_function(boundary_point, normal_vector, boundary_id, value);
62 for (unsigned int lane = 0; lane < n_lanes; ++lane)
63 if (remaining_lanes[lane] and boundary_ids[lane] == boundary_id)
64 {
65 copy_simd_lane(boundary_id_value, lane, boundary_value_dst);
66 remaining_lanes.reset(lane);
67 }
68 }
