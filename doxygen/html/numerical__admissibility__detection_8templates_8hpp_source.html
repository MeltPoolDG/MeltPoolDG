22 template <int n_components, typename number>
23 std::bitset<dealii::VectorizedArray<number>::size()>
24 admissibility_mask(const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &lower,
25 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &upper,
26 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &m,
27 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &M)
28 {
29 constexpr number delta_0 = 1e-10;
30 constexpr number epsilon = 1e-5;
31
32 std::bitset<dealii::VectorizedArray<number>::size()> mask;
33 mask.set();
34 for (unsigned int c = 0; c < n_components; ++c)
35 for (unsigned int lane = 0; lane < dealii::VectorizedArray<number>::size(); ++lane)
36 {
37 const number delta = std::max(delta_0, epsilon * (M[c][lane] - m[c][lane]));
38 if (lower[c][lane] < m[c][lane] - delta or upper[c][lane] > M[c][lane] + delta)
39 mask.reset(lane);
40 }
41 return mask;
42 }

84 template <int dim, int n_components, typename number>
85 void
86 CellwiseNumericalAdmissibility<dim, n_components, number>::evaluate(
87 std::vector<MaskType> &troubled_cells_mask)
88 {
89 Assert(
90 n_cell_batches_submitted == n_cell_batches,
91 dealii::ExcMessage(
92 "Not all locally owned cells have been submitted. Call submit_cell() for each cell batch."));
93 Assert(
94 n_ghost_cell_batches == n_ghost_cell_batches_submitted,
95 dealii::ExcMessage(
96 "Not all ghost cells have been submitted. Call submit_ghost_cell() for each ghost cell batch."));
97
98 Assert(
99 troubled_cells_mask.size() >= n_cell_batches,
100 dealii::ExcMessage(
101 "The size of the masks vector must be greater or equal to the number of locally owned cell batches."));
102
103 for (unsigned int cell = 0; cell < n_cell_batches; ++cell)
104 {
105 ValueType m = old_minimum[cell];
106 ValueType M = old_maximum[cell];
107 for (unsigned int lane = 0; lane < n_lanes; ++lane)
108 for (const auto &neighbor : neighbors.neighbors_of_cell(cell * n_lanes + lane))
109 {
110 if (neighbor.cell_index == dealii::numbers::invalid_unsigned_int)
111 continue;
112 for (unsigned int c = 0; c < n_components; ++c)
113 {
114 m[c][lane] =
115 std::min(m[c][lane], old_minimum[neighbor.cell_batch][c][neighbor.lane]);
116 M[c][lane] =
117 std::max(M[c][lane], old_maximum[neighbor.cell_batch][c][neighbor.lane]);
118 }
119 }
120
121 numerically_admissible[cell] =
122 admissibility_mask(candidate_minimum[cell], candidate_maximum[cell], m, M);
123 troubled_cells_mask[cell] |= ~numerically_admissible[cell];
124 }
125 n_cell_batches_submitted = 0;
126 n_ghost_cell_batches_submitted = 0;
127 }

149 void
150 SubcellNumericalAdmissibility<dim, n_components, number>::reinit(
151 const MatrixFreeContext<dim, number> &matrix_free_context)
152 {
153 neighbors.reinit(matrix_free_context.mf);
154
155 n_cell_batches = matrix_free_context.mf.n_cell_batches();
156 n_ghost_cell_batches = matrix_free_context.mf.n_ghost_cell_batches();
157 n_cell_batches_submitted = 0;
158 n_ghost_cell_batches_submitted = 0;
159 subcell_admissibility.resize(n_cell_batches);
160 face_layer_minimum.reinit({n_cell_batches + n_ghost_cell_batches, faces_per_cell});
161 face_layer_maximum.reinit({n_cell_batches + n_ghost_cell_batches, faces_per_cell});
162 deferred_subcells.clear();
163
164 // To prevent multiple memory allocations during the marking of troubled cells, we reserve
165 // memory for the deferred subcells vector. As the exact number of deferred subcells is not
166 // known a priori, we make a conservative estimate.
167 deferred_subcells.reserve(n_cell_batches * n_lanes * faces_per_cell *
168 matrix_free_context.mf.get_dof_handler().get_fe().degree);

171 template <int dim, int n_components, typename number>
172 void
173 SubcellNumericalAdmissibility<dim, n_components, number>::submit_cell(
174 const unsigned int cell_batch,
175 const FESubcellEvaluation<dim, n_components, number> &phi_old,
176 const FESubcellEvaluation<dim, n_components, number> &phi_candidate)
177 {
178 AssertIndexRange(cell_batch, n_cell_batches);
179 compute_face_layer_extrema(cell_batch, phi_old);
180
181 std::bitset<n_lanes> admissible;
182 admissible.set();
183 for (const unsigned int subcell : phi_candidate.subcell_indices())
184 {
185 // bounds from the old solution on the subcell and its neighbors within the cell; a missing
186 // neighbor in the direction of face f means that the subcell is adjacent to face f
187 const std::array<unsigned int, faces_per_cell> subcell_neighbors =
188 phi_old.subcell_neighbor_indices(subcell);
189 ValueType m = phi_old.get_value(subcell);
190 ValueType M = m;
191 std::bitset<faces_per_cell> cell_faces;
192 for (unsigned int f = 0; f < faces_per_cell; ++f)
193 if (subcell_neighbors[f] == dealii::numbers::invalid_unsigned_int)
194 cell_faces.set(f);
195 else
196 {
197 const ValueType value_neighbor = phi_old.get_value(subcell_neighbors[f]);
198 for (unsigned int c = 0; c < n_components; ++c)
199 {
200 m[c] = std::min(m[c], value_neighbor[c]);
201 M[c] = std::max(M[c], value_neighbor[c]);
202 }
203 }
204
205 const ValueType value_candidate = phi_candidate.get_value(subcell);
206 const std::bitset<n_lanes> admissible_subcell =
207 admissibility_mask(value_candidate, value_candidate, m, M);
208
209 // The neighboring cells can only widen the bounds. Hence, a subcell which passes is
210 // admissible, and a subcell in the interior of the cell which fails is not. A failing
211 // subcell at the boundary of the cell is decided in evaluate().
212 if (cell_faces.none())
213 {
214 admissible &= admissible_subcell;
215 continue;
216 }
217 for (unsigned int lane = 0; lane < n_lanes; ++lane)
218 if (not admissible_subcell[lane])
219 {
220 DeferredSubcell deferred{cell_batch, lane, cell_faces, {}, {}, {}};
221 for (unsigned int c = 0; c < n_components; ++c)
222 {
223 deferred.candidate[c] = value_candidate[c][lane];
224 deferred.minimum[c] = m[c][lane];
225 deferred.maximum[c] = M[c][lane];
226 }
227 deferred_subcells.push_back(deferred);
228 }
229 }
230 subcell_admissibility[cell_batch] = admissible;
231 ++n_cell_batches_submitted;
232 }

245 template <int dim, int n_components, typename number>
246 void
247 SubcellNumericalAdmissibility<dim, n_components, number>::compute_face_layer_extrema(
248 const unsigned int cell_batch,
249 const FESubcellEvaluation<dim, n_components, number> &phi_old)
250 {
251 for (unsigned int f = 0; f < faces_per_cell; ++f)
252 for (unsigned int c = 0; c < n_components; ++c)
253 {
254 face_layer_minimum(cell_batch, f)[c] = std::numeric_limits<number>::max();
255 face_layer_maximum(cell_batch, f)[c] = std::numeric_limits<number>::lowest();
256 }
257
258 // a subcell belongs to the layer at face f if it has no neighbor within the cell in the
259 // direction of face f, i.e. if subcell_neighbor_indices returns
260 // dealii::numbers::invalid_unsigned_int for face f
261 for (const unsigned int subcell : phi_old.subcell_indices())
262 {
263 const std::array<unsigned int, faces_per_cell> subcell_neighbors =
264 phi_old.subcell_neighbor_indices(subcell);
265 const ValueType value = phi_old.get_value(subcell);
266 for (unsigned int f = 0; f < faces_per_cell; ++f)
267 if (subcell_neighbors[f] == dealii::numbers::invalid_unsigned_int)
268 for (unsigned int c = 0; c < n_components; ++c)
269 {
270 face_layer_minimum(cell_batch, f)[c] =
271 std::min(face_layer_minimum(cell_batch, f)[c], value[c]);
272 face_layer_maximum(cell_batch, f)[c] =
273 std::max(face_layer_maximum(cell_batch, f)[c], value[c]);
274 }
275 }

278 template <int dim, int n_components, typename number>
279 void
280 SubcellNumericalAdmissibility<dim, n_components, number>::evaluate(
281 std::vector<std::bitset<VectorizedArrayType::size()>> &troubled_cells_mask)
282 {
283 Assert(
284 n_cell_batches_submitted == n_cell_batches,
285 dealii::ExcMessage(
286 "Not all locally owned cells have been submitted. Call submit_cell() for each cell batch."));
287 Assert(
288 n_ghost_cell_batches_submitted == n_ghost_cell_batches,
289 dealii::ExcMessage(
290 "Not all ghost cells have been submitted. Call submit_ghost_cell() for each ghost cell batch."));
291
292 Assert(
293 troubled_cells_mask.size() >= n_cell_batches,
294 dealii::ExcMessage(
295 "The size of the masks vector must be greater or equal to the number of locally owned cell batches."));
296
297 for (const DeferredSubcell &deferred : deferred_subcells)
298 {
299 // the lane may already be inadmissible due to another subcell
300 if (not subcell_admissibility[deferred.cell_batch][deferred.lane])
301 continue;
302
303 // widen the bounds by the face layer extrema of the neighboring cells across the faces
304 // the subcell is adjacent to
305 dealii::Tensor<1, n_components, number> m = deferred.minimum;
306 dealii::Tensor<1, n_components, number> M = deferred.maximum;
307 for (unsigned int f = 0; f < faces_per_cell; ++f)
308 {
309 if (not deferred.faces[f])
310 continue;
311
312 const auto &neighbor = neighbors(deferred.cell_batch * n_lanes + deferred.lane, f);
313 if (neighbor.cell_index == dealii::numbers::invalid_unsigned_int)
314 continue;
315
316 for (unsigned int c = 0; c < n_components; ++c)
317 {
318 m[c] = std::min(m[c],
319 face_layer_minimum(neighbor.cell_batch,
320 neighbor.face_no)[c][neighbor.lane]);
321 M[c] = std::max(M[c],
322 face_layer_maximum(neighbor.cell_batch,
323 neighbor.face_no)[c][neighbor.lane]);
324 }
325 }
326
327 // reuse the vectorized check by broadcasting the values to all lanes
328 ValueType candidate_vectorized, m_vectorized, M_vectorized;
329 for (unsigned int c = 0; c < n_components; ++c)
330 {
331 candidate_vectorized[c] = deferred.candidate[c];
332 m_vectorized[c] = m[c];
333 M_vectorized[c] = M[c];
334 }
335 if (not admissibility_mask(
336 candidate_vectorized, candidate_vectorized, m_vectorized, M_vectorized)[0])
337 subcell_admissibility[deferred.cell_batch].reset(deferred.lane);
338 }
339
340 for (unsigned int cell = 0; cell < n_cell_batches; ++cell)
341 troubled_cells_mask[cell] |= ~subcell_admissibility[cell];
342
343 deferred_subcells.clear();
344 n_cell_batches_submitted = 0;
345 n_ghost_cell_batches_submitted = 0;
346 }
