20 template <int n_components, typename number>
21 std::bitset<dealii::VectorizedArray<number>::size()>
22 admissibility_mask(const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &lower,
23 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &upper,
24 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &m,
25 const dealii::Tensor<1, n_components, dealii::VectorizedArray<number>> &M)
26 {
27 constexpr number delta_0 = 1e-10;
28 constexpr number epsilon = 1e-5;
29
30 std::bitset<dealii::VectorizedArray<number>::size()> mask;
31 mask.set();
32 for (unsigned int c = 0; c < n_components; ++c)
33 for (unsigned int lane = 0; lane < dealii::VectorizedArray<number>::size(); ++lane)
34 {
35 const number delta = std::max(delta_0, epsilon * (M[c][lane] - m[c][lane]));
36 if (lower[c][lane] < m[c][lane] - delta or upper[c][lane] > M[c][lane] + delta)
37 mask.reset(lane);
38 }
39 return mask;
40 }

82 template <int dim, int n_components, typename number>
83 void
84 CellwiseNumericalAdmissibility<dim, n_components, number>::evaluate(
85 std::vector<std::bitset<VectorizedArrayType::size()>> &masks)
86 {
87 Assert(
88 n_cell_batches_submitted == n_cell_batches,
89 dealii::ExcMessage(
90 "Not all locally owned cells have been submitted. Call submit_cell() for each cell batch."));
91 Assert(
92 n_ghost_cell_batches == n_ghost_cell_batches_submitted,
93 dealii::ExcMessage(
94 "Not all ghost cells have been submitted. Call submit_ghost_cell() for each ghost cell batch."));
95
96 Assert(
97 masks.size() >= n_cell_batches,
98 dealii::ExcMessage(
99 "The size of the masks vector must be greater or equal to the number of locally owned cell batches."));
100
101 for (unsigned int cell = 0; cell < n_cell_batches; ++cell)
102 {
103 ValueType m = old_minimum[cell];
104 ValueType M = old_maximum[cell];
105 for (unsigned int lane = 0; lane < n_lanes; ++lane)
106 for (const auto &neighbor : neighbors.neighbors_of_cell(cell * n_lanes + lane))
107 {
108 if (neighbor.cell_index == dealii::numbers::invalid_unsigned_int)
109 continue;
110 for (unsigned int c = 0; c < n_components; ++c)
111 {
112 m[c][lane] =
113 std::min(m[c][lane], old_minimum[neighbor.cell_batch][c][neighbor.lane]);
114 M[c][lane] =
115 std::max(M[c][lane], old_maximum[neighbor.cell_batch][c][neighbor.lane]);
116 }
117 }
118
119 masks[cell] = admissibility_mask(candidate_minimum[cell], candidate_maximum[cell], m, M);
120 }
121 n_cell_batches_submitted = 0;
122 n_ghost_cell_batches_submitted = 0;
123 }
