636 template <EOSIsConservativeValueView ValueView,
637 EOSIsGradientView GradientView,
638 NobleAbelStiffenedGasIsMaterialView MaterialView>
639 static inline DEAL_II_ALWAYS_INLINE //
640 auto
641 grad_pressure(const ValueView &value_view,
642 const GradientView &gradient_view,
643 const MaterialView &material_view)
644 {
645 const auto inv_rho = 1. / value_view.density();
646 const auto heat_capacity_ratio_minus_one = material_view.heat_capacity_ratio() - 1.;
647 const auto inv_specific_volume_minus_covolume = 1. / (inv_rho - material_view.covolume());
648 const auto specific_int_energy = specific_internal_energy(value_view, material_view);
649
650 const auto grad_specific_internal_energy =
651 inv_rho *
652 (gradient_view.grad_total_energy() - specific_int_energy * gradient_view.grad_density() -
653 value_view.density() *
654 matrix_vector_product(transpose(gradient_view.grad_velocity()), value_view.velocity()) -
655 0.5 * scalar_product(value_view.velocity(), value_view.velocity()) *
656 gradient_view.grad_density());
657
658 return heat_capacity_ratio_minus_one * inv_specific_volume_minus_covolume *
659 grad_specific_internal_energy +
660 (specific_int_energy - material_view.heat_bound()) * heat_capacity_ratio_minus_one *
661 inv_specific_volume_minus_covolume * inv_specific_volume_minus_covolume * inv_rho *
662 inv_rho * gradient_view.grad_density();
663 }

1034 dispatch_eos(CompressibleFlow::EquationOfState eos_type, F &&f)
1035 {
1036 auto check_support = [](CompressibleFlow::EquationOfState type) {
1037 if constexpr (HasSupportedEOS<Derived>)
1038 {
1039 if (type == CompressibleFlow::EquationOfState::ideal_gas)
1040 return supports_eos<Derived, CompressibleFlow::EquationOfState::ideal_gas>();
1041 if (type == CompressibleFlow::EquationOfState::stiffened_gas)
1042 return supports_eos<Derived, CompressibleFlow::EquationOfState::stiffened_gas>();
1043 if (type == CompressibleFlow::EquationOfState::noble_abel_stiffened_gas)
1044 return supports_eos<Derived,
1045 CompressibleFlow::EquationOfState::noble_abel_stiffened_gas>();
1046 if (type == CompressibleFlow::EquationOfState::weakly_compressible)
1047 return supports_eos<Derived, CompressibleFlow::EquationOfState::weakly_compressible>();
1048 return false;
1049 }
1050 return true;
1051 };
1052
1053 switch (eos_type)
1054 {
1055 case CompressibleFlow::EquationOfState::ideal_gas:
1056 if constexpr (check_support(CompressibleFlow::EquationOfState::ideal_gas))
1057 return std::forward<F>(f)(IdealGasEOS{});
1058 break;
1059
1060 case CompressibleFlow::EquationOfState::stiffened_gas:
1061 if constexpr (check_support(CompressibleFlow::EquationOfState::stiffened_gas))
1062 return std::forward<F>(f)(StiffenedGasEOS{});
1063 break;
1064
1065 case CompressibleFlow::EquationOfState::noble_abel_stiffened_gas:
1066 if constexpr (check_support(CompressibleFlow::EquationOfState::noble_abel_stiffened_gas))
1067 return std::forward<F>(f)(NobleAbelStiffenedGasEOS{});
1068 break;
1069
1070 case CompressibleFlow::EquationOfState::weakly_compressible:
1071 if constexpr (check_support(CompressibleFlow::EquationOfState::weakly_compressible))
1072 return std::forward<F>(f)(WeaklyCompressibleEOS{});
1073 break;
1074
1075 default:
1076 break;
1077 }
1078
1079 AssertThrow(false, dealii::ExcMessage("The provided EOS is not supported by this View."));
1080 }
