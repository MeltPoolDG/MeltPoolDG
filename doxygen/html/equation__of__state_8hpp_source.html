77 static inline DEAL_II_ALWAYS_INLINE //

79 thermodynamic_pressure(const ValueView &value_view, const MaterialView &material_view)
80 {
81 return (material_view.heat_capacity_ratio() - 1.) *
82 (value_view.total_energy() -
83 value_view.density() * 0.5 *

102 grad_temperature(const ValueView &value_view,
103 const GradientView &gradient_view,
104 const MaterialView &material_view)
105 {
106 const auto grad_E =
107 1. / value_view.density() *
108 (gradient_view.grad_total_energy() -
109 1. / value_view.density() * value_view.total_energy() * gradient_view.grad_density());
110
111 return (material_view.heat_capacity_ratio() - 1.0) / material_view.specific_gas_constant() *
112 (grad_E - matrix_vector_product(gradient_view.grad_velocity(), value_view.velocity()));
113 }
