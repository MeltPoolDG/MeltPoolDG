917 template <typename Derived, typename F>
918 inline DEAL_II_ALWAYS_INLINE decltype(auto)
919 dispatch_eos(CompressibleFlow::EquationOfState eos_type, F &&f)
920 {
921 auto check_support = [](CompressibleFlow::EquationOfState type) {
922 if constexpr (HasSupportedEOS<Derived>)
923 {
924 if (type == CompressibleFlow::EquationOfState::ideal_gas)
925 return supports_eos<Derived, CompressibleFlow::EquationOfState::ideal_gas>();
926 if (type == CompressibleFlow::EquationOfState::stiffened_gas)
927 return supports_eos<Derived, CompressibleFlow::EquationOfState::stiffened_gas>();
928 if (type == CompressibleFlow::EquationOfState::noble_abel_stiffened_gas)
929 return supports_eos<Derived,
930 CompressibleFlow::EquationOfState::noble_abel_stiffened_gas>();
931 if (type == CompressibleFlow::EquationOfState::weakly_compressible)
932 return supports_eos<Derived, CompressibleFlow::EquationOfState::weakly_compressible>();
933 return false;
934 }
935 return true;
936 };
937
938 switch (eos_type)
939 {
940 case CompressibleFlow::EquationOfState::ideal_gas:
941 if constexpr (check_support(CompressibleFlow::EquationOfState::ideal_gas))
942 return std::forward<F>(f)(IdealGasEOS{});
943 break;
944
945 case CompressibleFlow::EquationOfState::stiffened_gas:
946 if constexpr (check_support(CompressibleFlow::EquationOfState::stiffened_gas))
947 return std::forward<F>(f)(StiffenedGasEOS{});
948 break;
949
950 case CompressibleFlow::EquationOfState::noble_abel_stiffened_gas:
951 if constexpr (check_support(CompressibleFlow::EquationOfState::noble_abel_stiffened_gas))
952 return std::forward<F>(f)(NobleAbelStiffenedGasEOS{});
953 break;
954
955 case CompressibleFlow::EquationOfState::weakly_compressible:
956 if constexpr (check_support(CompressibleFlow::EquationOfState::weakly_compressible))
957 return std::forward<F>(f)(WeaklyCompressibleEOS{});
958 break;
959
960 default:
961 break;
962 }
963
964 AssertThrow(false, dealii::ExcMessage("The provided EOS is not supported by this View."));
965 }
