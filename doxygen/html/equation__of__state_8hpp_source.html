641 template <EOSIsConservativeValueView ValueView,
642 EOSIsGradientView GradientView,
643 NobleAbelStiffenedGasIsMaterialView MaterialView>
644 static inline DEAL_II_ALWAYS_INLINE //
645 auto
646 grad_pressure(const ValueView &value_view,
647 const GradientView &gradient_view,
648 const MaterialView &material_view)
649 {
650 const auto inv_rho = 1. / value_view.density();
651 const auto heat_capacity_ratio_minus_one = material_view.heat_capacity_ratio() - 1.;
652 const auto inv_specific_volume_minus_covolume = 1. / (inv_rho - material_view.covolume());
653 const auto specific_int_energy = specific_internal_energy(value_view, material_view);
654
655 const auto grad_specific_internal_energy =
656 inv_rho *
657 (gradient_view.grad_total_energy() - specific_int_energy * gradient_view.grad_density() -
658 value_view.density() *
659 matrix_vector_product(transpose(gradient_view.grad_velocity()), value_view.velocity()) -
660 0.5 * scalar_product(value_view.velocity(), value_view.velocity()) *
661 gradient_view.grad_density());
662
663 return heat_capacity_ratio_minus_one * inv_specific_volume_minus_covolume *
664 grad_specific_internal_energy +
665 (specific_int_energy - material_view.heat_bound()) * heat_capacity_ratio_minus_one *
666 inv_specific_volume_minus_covolume * inv_specific_volume_minus_covolume * inv_rho *
667 inv_rho * gradient_view.grad_density();
668 }

1040 dispatch_eos(CompressibleFlow::EquationOfState eos_type, F &&f)
1041 {
1042 auto check_support = [](CompressibleFlow::EquationOfState type) {
1043 if constexpr (HasSupportedEOS<Derived>)
1044 {
1045 if (type == CompressibleFlow::EquationOfState::ideal_gas)
1046 return supports_eos<Derived, CompressibleFlow::EquationOfState::ideal_gas>();
1047 if (type == CompressibleFlow::EquationOfState::stiffened_gas)
1048 return supports_eos<Derived, CompressibleFlow::EquationOfState::stiffened_gas>();
1049 if (type == CompressibleFlow::EquationOfState::noble_abel_stiffened_gas)
1050 return supports_eos<Derived,
1051 CompressibleFlow::EquationOfState::noble_abel_stiffened_gas>();
1052 if (type == CompressibleFlow::EquationOfState::weakly_compressible)
1053 return supports_eos<Derived, CompressibleFlow::EquationOfState::weakly_compressible>();
1054 return false;
1055 }
1056 return true;
1057 };
1058
1059 switch (eos_type)
1060 {
1061 case CompressibleFlow::EquationOfState::ideal_gas:
1062 if constexpr (check_support(CompressibleFlow::EquationOfState::ideal_gas))
1063 return std::forward<F>(f)(IdealGasEOS{});
1064 break;
1065
1066 case CompressibleFlow::EquationOfState::stiffened_gas:
1067 if constexpr (check_support(CompressibleFlow::EquationOfState::stiffened_gas))
1068 return std::forward<F>(f)(StiffenedGasEOS{});
1069 break;
1070
1071 case CompressibleFlow::EquationOfState::noble_abel_stiffened_gas:
1072 if constexpr (check_support(CompressibleFlow::EquationOfState::noble_abel_stiffened_gas))
1073 return std::forward<F>(f)(NobleAbelStiffenedGasEOS{});
1074 break;
1075
1076 case CompressibleFlow::EquationOfState::weakly_compressible:
1077 if constexpr (check_support(CompressibleFlow::EquationOfState::weakly_compressible))
1078 return std::forward<F>(f)(WeaklyCompressibleEOS{});
1079 break;
1080
1081 default:
1082 break;
1083 }
1084
1085 AssertThrow(false, dealii::ExcMessage("The provided EOS is not supported by this View."));
1086 }
