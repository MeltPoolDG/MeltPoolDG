728 template <typename Derived, typename F>
729 inline DEAL_II_ALWAYS_INLINE decltype(auto)
730 dispatch_eos(CompressibleFlow::EquationOfState eos_type, F &&f)
731 {
732 auto check_support = [](CompressibleFlow::EquationOfState type) {
733 if constexpr (HasSupportedEOS<Derived>)
734 {
735 if (type == CompressibleFlow::EquationOfState::ideal_gas)
736 return supports_eos<Derived, CompressibleFlow::EquationOfState::ideal_gas>();
737 if (type == CompressibleFlow::EquationOfState::stiffened_gas)
738 return supports_eos<Derived, CompressibleFlow::EquationOfState::stiffened_gas>();
739 if (type == CompressibleFlow::EquationOfState::noble_abel_stiffened_gas)
740 return supports_eos<Derived,
741 CompressibleFlow::EquationOfState::noble_abel_stiffened_gas>();
742 return false;
743 }
744 return true;
745 };
746
747 switch (eos_type)
748 {
749 case CompressibleFlow::EquationOfState::ideal_gas:
750 if constexpr (check_support(CompressibleFlow::EquationOfState::ideal_gas))
751 return std::forward<F>(f)(IdealGasEOS{});
752 break;
753
754 case CompressibleFlow::EquationOfState::stiffened_gas:
755 if constexpr (check_support(CompressibleFlow::EquationOfState::stiffened_gas))
756 return std::forward<F>(f)(StiffenedGasEOS{});
757 break;
758
759 case CompressibleFlow::EquationOfState::noble_abel_stiffened_gas:
760 if constexpr (check_support(CompressibleFlow::EquationOfState::noble_abel_stiffened_gas))
761 return std::forward<F>(f)(NobleAbelStiffenedGasEOS{});
762 break;
763
764 default:
765 break;
766 }
767
768 AssertThrow(false, dealii::ExcMessage("The provided EOS is not supported by this View."));
769 }
