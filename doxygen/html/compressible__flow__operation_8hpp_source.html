188 void
189 attach_output_vectors(GenericDataOut<dim, number> &data_out) const
190 {
191 // check, if single-phase or two-phase case is considered
192 const auto &dof_handler = operation_pimpl->get_dof_handler();
193 const bool two_phase = dof_handler.get_fe_collection().n_components() / (dim + 2) == 2;
194
195 std::vector<dealii::DataComponentInterpretation::DataComponentInterpretation> interpretation;
196 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
197 for (unsigned int d = 0; d < dim; ++d)
198 interpretation.push_back(dealii::DataComponentInterpretation::component_is_part_of_vector);
199 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
200
201 // add entries for two-phase case
202 if (two_phase)
203 {
204 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
205 for (unsigned int d = 0; d < dim; ++d)
206 interpretation.push_back(
207 dealii::DataComponentInterpretation::component_is_part_of_vector);
208 interpretation.push_back(dealii::DataComponentInterpretation::component_is_scalar);
209 }
210
211 std::vector<std::string> names_conservative;
212 std::vector<std::string> names_primitive;
213
214 if (not two_phase)
215 {
216 names_conservative.emplace_back("density");
217 for (unsigned int d = 0; d < dim; ++d)
218 names_conservative.emplace_back("momentum");
219 names_conservative.emplace_back("energy");
220
221 names_primitive.emplace_back("pressure");
222 for (unsigned int d = 0; d < dim; ++d)
223 names_primitive.emplace_back("velocity");
224 names_primitive.emplace_back("temperature");
225 }
226 else
227 {
228 names_conservative.emplace_back("density_liquid");
229 for (unsigned int d = 0; d < dim; ++d)
230 names_conservative.emplace_back("momentum_liquid");
231 names_conservative.emplace_back("energy_liquid");
232
233 names_conservative.emplace_back("density_gas");
234 for (unsigned int d = 0; d < dim; ++d)
235 names_conservative.emplace_back("momentum_gas");
236 names_conservative.emplace_back("energy_gas");
237
238 names_primitive.emplace_back("pressure_liquid");
239 for (unsigned int d = 0; d < dim; ++d)
240 names_primitive.emplace_back("velocity_liquid");
241 names_primitive.emplace_back("temperature_liquid");
242
243 names_primitive.emplace_back("pressure_gas");
244 for (unsigned int d = 0; d < dim; ++d)
245 names_primitive.emplace_back("velocity_gas");
246 names_primitive.emplace_back("temperature_gas");
247 }
248
249 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
250 operation_pimpl->get_solution(),
251 names_conservative,
252 interpretation);
253
254 data_out.add_data_vector(operation_pimpl->get_dof_handler(),
255 operation_pimpl->get_solution_in_primitive_variables(),
256 names_primitive,
257 interpretation);
258 }
