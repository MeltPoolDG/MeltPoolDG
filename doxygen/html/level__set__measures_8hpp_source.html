118 const auto &level_set_in = generic_data_out->get_vector(request_variable);
119 const auto &dof_handler = generic_data_out->get_dof_handler(request_variable);
120 const auto &mapping = generic_data_out->get_mapping();
121 const auto &fe = dof_handler.get_fe();
122
123 AssertThrow(not dof_handler.has_hp_capabilities(),
124 dealii::ExcMessage("The level-set DoFHandler must not have hp-capabilities."));
125 AssertThrow(fe.n_components() == 1,
126 dealii::ExcMessage("The level-set field must be a scalar field."));
127 AssertThrow(fe.reference_cell().is_hyper_cube(),
128 dealii::ExcMessage("Only hypercube cells are supported."));
129
130 // The MeshClassifier is only implemented for elements derived from FE_Q_Base.
131 const bool use_mesh_classifier = dynamic_cast<const dealii::FE_Q_Base<dim> *>(&fe) != nullptr;
132
133 // The MeshClassifier reads the level-set values on all faces of the ghost cells, which
134 // requires the locally relevant DoFs as ghost entries. However, the ghost set of a vector
135 // set up by MatrixFree might only contain the DoFs of locally owned cells. In this case,
136 // the level-set field is copied to a vector with a sufficient ghost set.
137 std::decay_t<decltype(level_set_in)> level_set_relevant;
138 bool use_copy = false;
139 if (use_mesh_classifier)
140 {
141 const auto locally_relevant_dofs =
142 dealii::DoFTools::extract_locally_relevant_dofs(dof_handler);
143
144 auto required_ghost_dofs = locally_relevant_dofs;
145 required_ghost_dofs.subtract_set(level_set_in.locally_owned_elements());
146
147 use_copy =
148 not required_ghost_dofs.is_subset_of(level_set_in.get_partitioner()->ghost_indices());
149
150 if (use_copy)
151 {
152 level_set_relevant.reinit(dof_handler.locally_owned_dofs(),
153 locally_relevant_dofs,
154 level_set_in.get_mpi_communicator());
155 level_set_relevant.copy_locally_owned_data_from(level_set_in);
156 level_set_relevant.update_ghost_values();
157 }
158 }
159
160 const auto &level_set = use_copy ? level_set_relevant : level_set_in;
161
162 const bool update_ghosts = not level_set.has_ghost_elements();
163 if (update_ghosts)
164 level_set.update_ghost_values();
165
166 const unsigned int n_q_1d = n_q_points_1d > 0 ? n_q_points_1d : fe.tensor_degree() + 1;
167
168 // the quadrature generator only stores a pointer to the 1D quadrature collection
169 const dealii::hp::QCollection<1> q_collection_1d{dealii::QGauss<1>(n_q_1d)};
170
171 dealii::NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(q_collection_1d,
172 dof_handler,
173 level_set);
174
175 std::optional<dealii::NonMatching::MeshClassifier<dim>> mesh_classifier;
176 if (use_mesh_classifier)
177 {
178 mesh_classifier.emplace(dof_handler, level_set);
179 mesh_classifier->reclassify();
180 }
181
182 // FEValues for cells not intersected by the interface
183 dealii::FEValues<dim> fe_values(mapping,
184 fe,
185 dealii::QGauss<dim>(n_q_1d),
186 dealii::update_JxW_values);
187
188 number volume_positive_local = 0;
189 number volume_negative_local = 0;
190 number surface_area_local = 0;
191
192 const auto integrate_volume = [&](const dealii::Quadrature<dim> &quadrature,
193 const auto &cell) -> number {
194 if (quadrature.empty())
195 return 0;
196
197 dealii::FEValues<dim> fe_values_cut(mapping, fe, quadrature, dealii::update_JxW_values);
198 fe_values_cut.reinit(cell);
199
200 number volume = 0;
201 for (const unsigned int q : fe_values_cut.quadrature_point_indices())
202 volume += fe_values_cut.JxW(q);
203 return volume;
204 };
205
206 for (const auto &cell : dof_handler.active_cell_iterators())
207 {
208 if (not cell->is_locally_owned())
209 continue;
210
211 // dealii::NonMatching classifies negative level-set values as "inside"
212 dealii::NonMatching::LocationToLevelSet location;
213 if (mesh_classifier)
214 location = mesh_classifier->location_to_level_set(cell);
215 else
216 {
217 quadrature_generator.generate(cell);
218
219 const bool has_inside = not quadrature_generator.get_inside_quadrature().empty();
220 const bool has_outside = not quadrature_generator.get_outside_quadrature().empty();
221 const bool has_surface = not quadrature_generator.get_surface_quadrature().empty();
222
223 // Consistent with the MeshClassifier, a cell, of which the interface only
224 // touches a face, is only considered intersected if it lies on the side of
225 // negative level-set values. Thus, an interface coinciding with a cell face is
226 // only counted once.
227 if (has_inside and (has_outside or has_surface))
228 location = dealii::NonMatching::LocationToLevelSet::intersected;
229 else if (has_inside)
230 location = dealii::NonMatching::LocationToLevelSet::inside;
231 else
232 location = dealii::NonMatching::LocationToLevelSet::outside;
233 }
234
235 if (location == dealii::NonMatching::LocationToLevelSet::intersected)
236 {
237 if (mesh_classifier)
238 quadrature_generator.generate(cell);
239
240 const auto &inside_quadrature = quadrature_generator.get_inside_quadrature();
241 const auto &outside_quadrature = quadrature_generator.get_outside_quadrature();
242 const auto &surface_quadrature = quadrature_generator.get_surface_quadrature();
243
244 volume_negative_local += integrate_volume(inside_quadrature, cell);
245 volume_positive_local += integrate_volume(outside_quadrature, cell);
246
247 if (not surface_quadrature.empty())
248 {
249 dealii::NonMatching::FEImmersedSurfaceValues<dim> fe_surface_values(
250 mapping, fe, surface_quadrature, dealii::update_JxW_values);
251 fe_surface_values.reinit(cell);
252
253 for (const unsigned int q : fe_surface_values.quadrature_point_indices())
254 surface_area_local += fe_surface_values.JxW(q);
255 }
256 }
257 else
258 {
259 fe_values.reinit(cell);
260
261 number cell_volume = 0;
262 for (const unsigned int q : fe_values.quadrature_point_indices())
263 cell_volume += fe_values.JxW(q);
264
265 if (location == dealii::NonMatching::LocationToLevelSet::inside)
266 volume_negative_local += cell_volume;
267 else
268 volume_positive_local += cell_volume;
269 }
270 }
271
272 if (update_ghosts)
273 level_set.zero_out_ghost_values();
274
275 const auto mpi_comm = level_set.get_mpi_communicator();
276 volume_positive = dealii::Utilities::MPI::sum(volume_positive_local, mpi_comm);
277 volume_negative = dealii::Utilities::MPI::sum(volume_negative_local, mpi_comm);
278 surface_area = dealii::Utilities::MPI::sum(surface_area_local, mpi_comm);
279 }
