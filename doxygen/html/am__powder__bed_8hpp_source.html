21 template <int dim, typename number>
22 class Inflow : public dealii::Function<dim, number>
23 {
24 public:
25 explicit Inflow(const number time,
26 const number ambient_density,
27 const number ambient_temperature,
28 const number peak_inflow_velocity,
29 const FreeJetProfile free_jet_profile,
30 const dealii::Point<dim, number> free_jet_center,
31 const number free_jet_diameter,
32 const number specific_gas_constant,
33 const number heat_capacity_ratio)
34 : dealii::Function<dim, number>(dim + 2, time)
35 , free_jet_profile(free_jet_profile)
36 , free_jet_center(free_jet_center)
37 , free_jet_radius(free_jet_diameter * 0.5)
38 , ambient_density(ambient_density)
39 , ambient_temperature(ambient_temperature)
40 , peak_inflow_velocity(peak_inflow_velocity)
41 , const_vol_specific_heat(specific_gas_constant / (heat_capacity_ratio - 1.))
42 {}
43
44 number
45 value(const dealii::Point<dim, number> &location, const unsigned int component) const final
46 {
47 // density component
48 if (component == 0)
49 return ambient_density;
50
51 // not in free jet area
52 if (free_jet_center.distance_square(location) > free_jet_radius * free_jet_radius)
53 {
54 if (component == dim + 1)
55 return ambient_density * const_vol_specific_heat * ambient_temperature;
56 else
57 return 0.;
58 }
59
60 // in free jet area
61 number velocity;
62 switch (free_jet_profile)
63 {
64 case FreeJetProfile::cosine: {
65 velocity = peak_inflow_velocity * std::cos(free_jet_center.distance(location) /
66 free_jet_radius * 0.5 * M_PI);
67 break;
68 }
69 case FreeJetProfile::constant: {
70 velocity = peak_inflow_velocity;
71 break;
72 }
73 default: {
74 AssertThrow(false, dealii::ExcMessage("Unknown free jet profile!"));
75 }
76 }
77
78 if (component == dim + 1)
79 return ambient_density *
80 (const_vol_specific_heat * ambient_temperature + velocity * velocity * 0.5);
81 else if (component == dim)
82 return ambient_density * velocity;
83 else
84 return 0.;
85 }
86
87 private:
88 const FreeJetProfile free_jet_profile;
89 const dealii::Point<dim, number> free_jet_center;
90 const number free_jet_radius;
91 const number ambient_density;
92 const number ambient_temperature;
93 const number peak_inflow_velocity;
94 const number const_vol_specific_heat;
95 };
