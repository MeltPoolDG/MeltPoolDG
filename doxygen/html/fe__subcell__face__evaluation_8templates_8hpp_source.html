131 template <int dim, int n_components, typename number>
132 auto
133 FESubcellFaceEvaluation<dim, n_components, number>::get_face_gradient(
134 const unsigned int subcell_index,
135 const FESubcellFaceEvaluation &opposite) const -> gradient_type
136 {
137 AssertIndexRange(subcell_index, cell_index_info.n_subcells_face);
138 Assert(not is_boundary_face,
139 dealii::ExcMessage("Cannot compute face gradient on boundary face with this function."));
140
141 Assert(subcell_values_initialized and opposite.subcell_values_initialized,
142 dealii::ExcNotInitialized());
143 Assert(tangential_gradients_initialized and opposite.tangential_gradients_initialized,
144 dealii::ExcNotInitialized());
145
146 Assert(
147 (opposite.face_batch_index == face_batch_index) and
148 (opposite.is_interior_face != is_interior_face),
149 dealii::ExcMessage(
150 "The opposite FESubcellFaceEvaluation object must represent the opposite side of the face."));
151
152 // The distance of the two subcell centers is computed from the widths of the subcells normal to
153 // the face, i.e. their sizes divided by the subcell face size, and not from their locations,
154 // since the two cells lie at opposite ends of the domain on periodic faces. Both sides compute
155 // the same derivative, swapping the sides flips the sign of both the difference and
156 // inverse_distance.
157 const VectorizedArrayType absolute_distance =
158 0.5 * (subcell_sizes[subcell_index] / subcell_face_sizes[subcell_index] +
159 opposite.subcell_sizes[subcell_index] / opposite.subcell_face_sizes[subcell_index]);
160 const VectorizedArrayType normal_component =
161 subcell_face_normals[subcell_index][face_normal_coordinate_direction];
162 const VectorizedArrayType inverse_distance =
163 (is_interior_face ? normal_component : -normal_component) / absolute_distance;
164
165 const value_type normal_derivative =
166 (opposite.get_value(subcell_index) - get_value(subcell_index)) * inverse_distance;
167
168 const gradient_type &this_tangential_gradient = tangential_subcell_gradients[subcell_index];
169 const gradient_type &opposite_tangential_gradient =
170 opposite.tangential_subcell_gradients[subcell_index];
171 gradient_type face_gradient;
172
173 for (unsigned int d = 0; d < dim; ++d)
174 {
175 if constexpr (n_components == 1)
176 {
177 face_gradient[d] =
178 (d == face_normal_coordinate_direction) ?
179 normal_derivative :
180 0.5 * (this_tangential_gradient[d] + opposite_tangential_gradient[d]);
181 }
182 else
183 {
184 for (unsigned int c = 0; c < n_components; ++c)
185 {
186 face_gradient[c][d] =
187 (d == face_normal_coordinate_direction) ?
188 normal_derivative[c] :
189 0.5 * (this_tangential_gradient[c][d] + opposite_tangential_gradient[c][d]);
190 }
191 }
192 }
193
194 return face_gradient;
195 }

197 template <int dim, int n_components, typename number>
198 auto
199 FESubcellFaceEvaluation<dim, n_components, number>::get_face_gradient(
200 const unsigned int subcell_index) const -> gradient_type
201 {
202 AssertIndexRange(subcell_index, cell_index_info.n_subcells_face);
203 Assert(
204 is_boundary_face,
205 dealii::ExcMessage(
206 "This function can only be used on boundary faces. Use the other get_face_gradient() function for interior faces."));
207
208 Assert(subcell_values_initialized and tangential_gradients_initialized,
209 dealii::ExcNotInitialized());
210
211 // As in FESubcellEvaluation, the boundary value is located on the boundary face, i.e. at half
212 // the width of the subcell normal to the face from the subcell center. The normal vector points
213 // out of the domain, so its component in the face normal direction gives the orientation.
214 const value_type subcell_value = get_value(subcell_index);
215 const value_type boundary_value = boundary_value_function(subcell_face_locations[subcell_index],
216 subcell_face_normals[subcell_index],
217 boundary_id(),
218 subcell_value);
219
220 const VectorizedArrayType distance =
221 0.5 * subcell_sizes[subcell_index] / subcell_face_sizes[subcell_index];
222 const VectorizedArrayType inverse_distance =
223 subcell_face_normals[subcell_index][face_normal_coordinate_direction] / distance;
224
225 const value_type normal_derivative = (boundary_value - subcell_value) * inverse_distance;
226
227 gradient_type face_gradient = tangential_subcell_gradients[subcell_index];
228 if constexpr (n_components == 1)
229 face_gradient[face_normal_coordinate_direction] = normal_derivative;
230 else
231 for (unsigned int c = 0; c < n_components; ++c)
232 face_gradient[c][face_normal_coordinate_direction] = normal_derivative[c];
233
234 return face_gradient;
235 }

379 template <int dim, int n_components, typename number>
380 void
381 FESubcellFaceEvaluation<dim, n_components, number>::integrate()
382 {
383 // To apply the subcell fluxes, we consider the values of all q_points in the interior cell as
384 // zero, and we set the values of the q_points on the face to the submitted fluxes multiplied
385 // by the corresponding subcell face size and divided by the subcell size. We then transform
386 // those values to cells DoF values and finally add the values to the global vector.
387 // This way, we only apply the fluxes on the face and do not affect the values of the other
388 // q_points in the interior cell.
389 dealii::MatrixFreeOperators::CellwiseInverseMassMatrix<dim, -1, n_components, number> inverse(
390 fe_cell_evaluator);
391
392 std::fill(integrated_subcell_values_buffer.begin(),
393 integrated_subcell_values_buffer.end(),
394 VectorizedArrayType(0.0));
395
396 for (unsigned int subcell : subcell_face_indices())
397 {
398 const unsigned int cell_q_index =
399 cell_index_info.face_to_cell_subcell_indices[face_no][subcell];
400 const VectorizedArrayType scaling_factor =
401 subcell_face_sizes[subcell] / subcell_sizes[subcell];
402 if constexpr (n_components == 1)
403 integrated_subcell_values_buffer[cell_q_index] =
404 submitted_fluxes[subcell] * scaling_factor;
405 else
406 for (unsigned int c = 0; c < n_components; ++c)
407 integrated_subcell_values_buffer[c * fe_cell_evaluator.n_q_points + cell_q_index] =
408 submitted_fluxes[subcell][c] * scaling_factor;
409 }
410 inverse.transform_from_q_points_to_basis(n_components,
411 integrated_subcell_values_buffer.data(),
412 fe_cell_evaluator.begin_dof_values());
413 }

465 template <int dim, int n_components, typename number>
466 void
467 FESubcellFaceEvaluation<dim, n_components, number>::fill_halo_subcell_values(
468 const dealii::LinearAlgebra::distributed::Vector<number> &src_vector,
469 const std::bitset<n_lanes> &mask)
470 {
471 // no halo subcells in 1D
472 if constexpr (dim == 1)
473 return;
474
475 dealii::FEEvaluation<dim, -1, 0, n_components, number, VectorizedArrayType>
476 fe_lateral_evaluator(subcell_matrix_free.get_matrix_free(),
477 subcell_matrix_free.get_dof_no(),
478 subcell_matrix_free.get_quad_no());
479
480 for (unsigned int edge_no = 0; edge_no < 2 * (dim - 1); ++edge_no)
481 {
482 const unsigned int face_coordinates_direction = edge_no / 2;
483 const unsigned int cell_coordinates_direction =
484 (face_normal_coordinate_direction + face_coordinates_direction + 1) % dim;
485 const unsigned int lateral_edge_face_no = 2 * cell_coordinates_direction + edge_no % 2;
486
487 const typename SubcellMatrixFree<dim, number>::FaceNeighbors lateral_neighbor_cells =
488 subcell_matrix_free.get_face_neighbors(
489 is_interior_face ? matrix_free.get_face_info(face_batch_index).cells_interior :
490 matrix_free.get_face_info(face_batch_index).cells_exterior,
491 lateral_edge_face_no);
492
493 std::bitset<n_lanes> held_mask;
494 std::bitset<n_lanes> extra_mask;
495 std::bitset<n_lanes> at_boundary_mask;
496 for (unsigned int lane = 0; lane < n_lanes; ++lane)
497 {
498 if (mask[lane])
499 {
500 if (lateral_neighbor_cells.cell_ids[lane] != dealii::numbers::invalid_unsigned_int)
501 held_mask.set(lane);
502 else if (lateral_neighbor_cells.extra_cell_ids[lane] !=
503 dealii::numbers::invalid_unsigned_int)
504 extra_mask.set(lane);
505 else if (lateral_neighbor_cells.boundary_ids[lane] !=
506 dealii::numbers::invalid_boundary_id)
507 at_boundary_mask.set(lane);
508 }
509 }
510
511 // First, we evaluate the lateral neighbors which are held or extra.
512 if (held_mask.any() or extra_mask.any())
513 {
514 fe_lateral_evaluator.reinit(lateral_neighbor_cells.cell_ids);
515 fe_lateral_evaluator.read_dof_values(src_vector, 0, held_mask);
516
517 // The extra neighbors are not held by the dealii::MatrixFree object, so their DoF
518 // values are written into their lanes directly. This must happen after
519 // read_dof_values(), which sets all lanes outside of held_mask to zero.
520 if (extra_mask.any())
521 {
522 Assert(
523 subcell_matrix_free.has_ghost_values_of(src_vector),
524 dealii::ExcMessage(
525 "The DoF values of the extra cells have not been exchanged from the source "
526 "vector. Call SubcellMatrixFree::update_ghost_values() before."));
527 for (unsigned int lane = 0; lane < n_lanes; ++lane)
528 if (extra_mask[lane])
529 subcell_matrix_free.read_extra_cell_dof_values(
530 lateral_neighbor_cells.extra_cell_ids[lane],
531 lane,
532 fe_lateral_evaluator.begin_dof_values());
533 }
534
535 fe_lateral_evaluator.evaluate(dealii::EvaluationFlags::values);
536 }
537
538 const SubcellIndexInfo<dim - 1> &face_index_info = subcell_shape_info.face_index_info;
539 for (unsigned int s = 0; s < face_index_info.n_subcells_face; ++s)
540 {
541 const unsigned int subcell_index =
542 face_index_info.face_to_cell_subcell_indices[edge_no][s];
543
544 const unsigned int haloed_neighbor_index =
545 face_index_info.subcell_index_to_haloed_neighbor_vector_index[subcell_index][edge_no];
546
547 if (held_mask.any() or extra_mask.any())
548 {
549 const unsigned int neighbor_subcell_index =
550 face_index_info.face_to_cell_subcell_indices[edge_no ^ 1][s];
551
552 layer_values[haloed_neighbor_index] = fe_lateral_evaluator.get_value(
553 cell_index_info.face_to_cell_subcell_indices[face_no][neighbor_subcell_index]);
554 }
555
556 if (at_boundary_mask.any())
557 {
558 dealii::Point<dim, VectorizedArrayType> boundary_face_location =
559 subcell_locations[subcell_index];
560 boundary_face_location[cell_coordinates_direction] +=
561 0.5 *
562 layer_extents[face_index_info.subcell_index_to_haloed_vector_index[subcell_index]]
563 [face_coordinates_direction] *
564 (edge_no % 2 == 0 ? -1. : 1.);
565
566 SubcellEvaluation::Internal::set_boundary_values(
567 at_boundary_mask,
568 lateral_neighbor_cells.boundary_ids,
569 boundary_face_location,
570 SubcellEvaluation::Internal::cell_face_normal_vector<dim, VectorizedArrayType>(
571 lateral_edge_face_no),
572 layer_values[face_index_info.subcell_index_to_haloed_vector_index[subcell_index]],
573 layer_values[haloed_neighbor_index],
574 boundary_value_function);

580 template <int dim, int n_components, typename number>
581 void
582 FESubcellFaceEvaluation<dim, n_components, number>::compute_layer_extents()
583 {
584 Assert(fe_cell_evaluator.get_cell_type() == dealii::internal::MatrixFreeFunctions::cartesian,
585 dealii::ExcNotImplemented());
586
587 const SubcellIndexInfo<dim - 1> &face_index_info = subcell_shape_info.face_index_info;
588 AssertDimension(layer_extents.size(), face_index_info.n_haloed_subcells);
589
590 const unsigned int n_subcells_1d = cell_index_info.n_subcells_1d;
591
592 // Face subcells: compute their extents in the face directions from the subcell widths on the
593 // unit cell and the size of the cell.
594 const dealii::Tensor<2, dim, VectorizedArrayType> inverse_jacobian =
595 fe_cell_evaluator.inverse_jacobian(0);
596 for (const unsigned int subcell : subcell_face_indices())
597 {
598 const unsigned int cell_subcell =
599 cell_index_info.face_to_cell_subcell_indices[face_no][subcell];
600 std::array<VectorizedArrayType, dim - 1> &extents =
601 layer_extents[face_index_info.subcell_index_to_haloed_vector_index[subcell]];
602
603 for (unsigned int k = 0; k < dim - 1; ++k)
604 {
605 const unsigned int direction = (face_normal_coordinate_direction + 1 + k) % dim;
606 const unsigned int index_1d =
607 cell_index_info.subcell_cartesian_indices[cell_subcell][direction];
608 extents[k] = subcell_shape_info.subcell_widths_1d[index_1d] /
609 inverse_jacobian[direction][direction];
610 }
611 }
612
613 // The face subcells at the edge edge_no of the face (in 2D: at an end point of the face) need
614 // a halo subcell beyond this edge for the central difference in the face direction k. This
615 // halo subcell lies in the lateral neighbor across the cell face lateral_face_no and is the
616 // subcell of the lateral neighbor touching the face subcell, i.e. its last subcell in the
617 // cell direction `direction` for a lower edge and its first one for an upper edge:
618 // - Its extent in the face direction k is the extent of the lateral neighbor times the width of
619 // this outermost subcell on the unit interval. At the domain boundary, the extent of the
620 // lateral neighbor is zero, such that the halo subcell collapses onto the boundary, where
621 // the boundary value is located.
622 // - In 3D, its extent in the other face direction is that of the face subcell, since the
623 // neighbors are conforming.
624 const dealii::internal::MatrixFreeFunctions::FaceToCellTopology<n_lanes> &face_info =
625 matrix_free.get_face_info(face_batch_index);
626 const std::array<unsigned int, n_lanes> &cell_ids =
627 is_interior_face ? face_info.cells_interior : face_info.cells_exterior;
628 for (unsigned int edge_no = 0; edge_no < 2 * (dim - 1); ++edge_no)
629 {
630 const unsigned int k = edge_no / 2;
631 const unsigned int direction = (face_normal_coordinate_direction + 1 + k) % dim;
632 const unsigned int lateral_face_no = 2 * direction + edge_no % 2;
633 const VectorizedArrayType neighbor_extent =
634 subcell_matrix_free.get_face_neighbors(cell_ids, lateral_face_no).extents;
635 const number outermost_subcell_width =
636 subcell_shape_info.subcell_widths_1d[(edge_no % 2 == 0) ? n_subcells_1d - 1 : 0];
637
638 for (const unsigned int subcell : face_index_info.face_to_cell_subcell_indices[edge_no])
639 {
640 std::array<VectorizedArrayType, dim - 1> &halo_extents =
641 layer_extents[face_index_info
642 .subcell_index_to_haloed_neighbor_vector_index[subcell][edge_no]];
643 halo_extents =
644 layer_extents[face_index_info.subcell_index_to_haloed_vector_index[subcell]];
645 halo_extents[k] = neighbor_extent * outermost_subcell_width;
646 }
647 }
648 }

650 template <int dim, int n_components, typename number>
651 void
652 FESubcellFaceEvaluation<dim, n_components, number>::compute_subcell_tangential_gradients()
653 {
654 const SubcellIndexInfo<dim - 1> &face_index_info = subcell_shape_info.face_index_info;
655
656 AssertDimension(tangential_subcell_gradients.size(), cell_index_info.n_subcells_face);
657 AssertDimension(layer_values.size(), face_index_info.n_haloed_subcells);
658 AssertDimension(layer_extents.size(), face_index_info.n_haloed_subcells);
659
660 Assert(subcell_values_initialized, dealii::ExcNotInitialized());
661
662 for (const unsigned int subcell : subcell_face_indices())
663 {
664 const unsigned int haloed_index =
665 face_index_info.subcell_index_to_haloed_vector_index[subcell];
666
667 tangential_subcell_gradients[subcell] = gradient_type();
668
669 for (unsigned int face_coordinates_direction = 0; face_coordinates_direction < dim - 1;
670 ++face_coordinates_direction)
671 {
672 const unsigned int neighbor_l_index =
673 face_index_info
674 .subcell_index_to_haloed_neighbor_vector_index[subcell]
675 [2 * face_coordinates_direction];
676 const unsigned int neighbor_r_index =
677 face_index_info
678 .subcell_index_to_haloed_neighbor_vector_index[subcell]
679 [2 * face_coordinates_direction + 1];
680
681 const VectorizedArrayType inverse_distance =
682 1. / (layer_extents[haloed_index][face_coordinates_direction] +
683 0.5 * layer_extents[neighbor_l_index][face_coordinates_direction] +
684 0.5 * layer_extents[neighbor_r_index][face_coordinates_direction]);
685
686 const value_type derivative =
687 (layer_values[neighbor_r_index] - layer_values[neighbor_l_index]) * inverse_distance;
688
689 // The derivative is taken along the face coordinate direction (dim-1 dimensions), but
690 // the gradient is stored in cell coordinates. Therefore, we need to map the face
691 // coordinate direction to the cell coordinate direction.
692 const unsigned int cell_coordinates_direction =
693 (face_normal_coordinate_direction + 1 + face_coordinates_direction) % dim;
694 if constexpr (n_components == 1)
695 {
696 tangential_subcell_gradients[subcell][cell_coordinates_direction] = derivative;
697 }
698 else
699 {
700 for (unsigned int c = 0; c < n_components; ++c)
701 {
702 tangential_subcell_gradients[subcell][c][cell_coordinates_direction] =
703 derivative[c];
704 }
705 }
706 }
707 }
708 tangential_gradients_initialized = true;
709 }

713 FESubcellFaceEvaluation<dim, n_components, number>::determine_subcell_locations()
714 {
715 const dealii::internal::MatrixFreeFunctions::FaceToCellTopology<n_lanes> &face_info =
716 matrix_free.get_face_info(face_batch_index);
717 const dealii::Point<dim, VectorizedArrayType> cell_origin = subcell_matrix_free.get_cell_origin(
718 is_interior_face ? face_info.cells_interior : face_info.cells_exterior);
719 const dealii::Tensor<2, dim, VectorizedArrayType> inverse_jacobian =
720 fe_cell_evaluator.inverse_jacobian(0);
721
722 for (const unsigned int subcell_face_index : subcell_face_indices())
723 {
724 const unsigned int subcell_index =
725 cell_index_info.face_to_cell_subcell_indices[face_no][subcell_face_index];
726 for (unsigned int d = 0; d < dim; ++d)
727 subcell_locations[subcell_face_index][d] =
728 cell_origin[d] +
729 subcell_shape_info
730 .subcell_centers_1d[cell_index_info.subcell_cartesian_indices[subcell_index][d]] /
731 inverse_jacobian[d][d];
732 }
