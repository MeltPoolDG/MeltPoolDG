69 number
70 value(const dealii::Point<dim, number> &p, const unsigned int component) const final
71 {
72 std::array<unsigned int, dim + 2> gas_components;
73 std::array<unsigned int, dim + 2> liquid_components;
74
75 for (unsigned int i = 0; i < dim + 2; ++i)
76 {
77 if (gas_phase_is_first)
78 {
79 gas_components[i] = i;
80 liquid_components[i] = i + dim + 2;
81 }
82 else
83 {
84 gas_components[i] = i + dim + 2;
85 liquid_components[i] = i;
86 }
87 }
88
89 // initial phase interface positions
90 constexpr number R_1 = 0.10001;
91 constexpr number R_2 = 0.80001;
92
93 // an initial pressure gradient in the liquid phase triggers the movement of the water column
94 constexpr number p_r = 0.5e5;
95 constexpr number p_l = 1.e5;
96 constexpr number T = 293.15;
97
98 // gas phases
99
100 if (component == gas_components[0])
101 {
102 if (p[0] <= 0.5)
103 return p_l / (gas_material_data.specific_gas_constant * T);
104 else
105 return p_r / (gas_material_data.specific_gas_constant * T);
106 }
107 else if (component == gas_components[1])
108 return 0.;
109 else if (component == gas_components[2])
110 {
111 if (p[0] <= 0.5)
112 return p_l / (gas_material_data.gamma - 1.0);
113 else
114 return p_r / (gas_material_data.gamma - 1.0);
115 }
116
117 // liquid phase
118
119 const number pressure =
120 (p_r - p_l) / (R_2 - R_1) * p[0] + (p_l * R_2 - p_r * R_1) / (R_2 - R_1);
121
122 if (component == liquid_components[0])
123 {
124 return (pressure + liquid_material_data.eos_data.p_inf) /
125 ((liquid_material_data.gamma - 1.) * liquid_material_data.specific_isobaric_heat /
126 liquid_material_data.gamma * T);
127 }
128 else if (component == liquid_components[1])
129 return 0.;
130 else if (component == liquid_components[2])
131 {
132 return (pressure + liquid_material_data.gamma * liquid_material_data.eos_data.p_inf) /
133 (liquid_material_data.gamma - 1.);
134 }
135 else
136 return 0.;
137 }
