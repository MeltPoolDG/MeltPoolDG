159 void
160 post(const bool is_gas)
161 {
162 // Set thermal conductivity, if not explicitly set by the user.
163 // For physical consistency, set thermal conductivity based on the user-defined dynamic
164 // viscosity, gamma and specific gas constant. The Prandtl number is currently set
165 // constant to Pr=0.71 for the gas phase (air) and to Pr=0.01 for the liquid phase (metal).
166 const number Pr = is_gas ? 0.71 : 0.01;
167 if (thermal_conductivity == std::numeric_limits<number>::max())
168 thermal_conductivity =
169 dynamic_viscosity * gamma * specific_gas_constant / (gamma - 1.) * 1. / Pr;
170
171 // Ensure that parameters are set for advanced equations of state
172 if (eos_data.type == EquationOfState::stiffened_gas)
173 AssertThrow(eos_data.p_inf != std::numeric_limits<number>::max(),
174 dealii::ExcMessage(
175 "The parameter p_inf is required for the stiffened gas EOS."));
176 else if (eos_data.type == EquationOfState::noble_abel_stiffened_gas)
177 AssertThrow(eos_data.p_inf != std::numeric_limits<number>::max() and
178 eos_data.b != std::numeric_limits<number>::max() and
179 eos_data.q != std::numeric_limits<number>::min(),
180 dealii::ExcMessage(
181 "The parameters p_inf, b and q are required for the Noble-Abel stiffened"
