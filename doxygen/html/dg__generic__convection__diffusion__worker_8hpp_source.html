121 [[nodiscard]] static FaceFlux
122 face(const ValueType &u_m,
123 const ValueType &u_p,
124 const dealii::Tensor<1, dim, VectorizedArrayType> &normal,
125 const Kernel &kernel,
126 const RiemannSolverType riemann_solver_type = RiemannSolverType::lax_friedrichs)
127 {
128 const FluxType flux_m = kernel.flux(u_m);
129 const FluxType flux_p = kernel.flux(u_p);
130
131 ValueType numerical_flux;
132 switch (riemann_solver_type)
133 {
134 case RiemannSolverType::lax_friedrichs: {
135 const auto local_maximum_wave_speed = kernel.local_maximum_wave_speed(u_m, u_p);
136 numerical_flux =
137 lax_friedrichs_flux(u_m, u_p, flux_m, flux_p, normal, local_maximum_wave_speed);
138 break;
139 }
140 case RiemannSolverType::harten_lax_vanleer: {
141 const auto [smallest_wave_speed, largest_wave_speed] =
142 kernel.smallest_and_largest_wave_speed(u_m, u_p, normal);
143 numerical_flux = harten_lax_van_leer_flux(
144 u_m, u_p, flux_m, flux_p, normal, smallest_wave_speed, largest_wave_speed);
145 break;
146 }
147 default:
148 Assert(false, dealii::ExcNotImplemented());
149 return FaceFlux{};
150 }
151
152 return FaceFlux{.inner_face_value = -numerical_flux, .outer_face_value = numerical_flux};
153 }
154 };

222 [[nodiscard]] static FaceFlux
223 face(const ValueType &u_m,
224 const ValueType &u_p,
225 const GradientType &grad_u_m,
226 const GradientType &grad_u_p,
227 const dealii::Tensor<1, dim, VectorizedArrayType> &normal,
228 const VectorizedArrayType penalty_parameter,
229 const Kernel &kernel)
230 {
231 GradientType flux_m = kernel.flux(u_m, grad_u_m);
232 GradientType flux_p = kernel.flux(u_p, grad_u_p);
233
234 const auto value = contract_average_tensor_with_vector(flux_m, flux_p, normal) -
235 penalty_parameter * (u_m - u_p);
236
237
238 GradientType jump_u = jump(u_m, u_p, normal);
239
240 // use jumps instead of gradients for evaluating the diffusive flux
241 const GradientType penalty_flux_m = 0.5 * kernel.flux(u_m, jump_u);
242 const GradientType penalty_flux_p = 0.5 * kernel.flux(u_p, jump_u);
243
244 return FaceFlux{.inner_face_value = value,
245 .outer_face_value = -value,
246 .inner_face_gradient = penalty_flux_m,
