283 template <typename number, int n_species, typename PrimitiveVariablesView>
284 inline std::bitset<dealii::VectorizedArray<number>::size()>
285 physical_admissibility_mask(const PrimitiveVariablesView &w_view,
286 const unsigned int n_active_lanes)
287 {
288 dealii::VectorizedArray<number> is_admissible = 1.;
289
290 const auto require_greater_than = [&](const dealii::VectorizedArray<number> &value,
291 const dealii::VectorizedArray<number> &threshold) {
292 is_admissible = dealii::compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
293 value, threshold, is_admissible, dealii::VectorizedArray<number>(0.));
294 };
295
296 const auto require_less_than = [&](const dealii::VectorizedArray<number> &value,
297 const dealii::VectorizedArray<number> &threshold) {
298 is_admissible = dealii::compare_and_apply_mask<dealii::SIMDComparison::less_than>(
299 value, threshold, is_admissible, dealii::VectorizedArray<number>(0.));
300 };
301
302 require_greater_than(w_view.density(), dealii::VectorizedArray<number>(0.));
303 require_greater_than(w_view.pressure(), dealii::VectorizedArray<number>(0.));
304
305 if constexpr (n_species > 1)
306 {
307 dealii::VectorizedArray<number> tolerance = 1e-12;
308 for (unsigned int species = 0; species < n_species; ++species)
309 {
310 // mass fractions must be between 0 and 1, with a small tolerance to account for
311 // numerical errors
312 require_greater_than(w_view.mass_fraction(species), -tolerance);
313 require_less_than(w_view.mass_fraction(species),
314 dealii::VectorizedArray<number>(1.) + tolerance);
315 }
316
317 // sum of mass fractions must be equal to 1, with a small tolerance to account for numerical
318 // errors
319 dealii::VectorizedArray<number> mass_fraction_sum = 0.0;
320 for (unsigned int species = 0; species < n_species; ++species)
321 mass_fraction_sum += w_view.mass_fraction(species);
322
323 require_greater_than(mass_fraction_sum, dealii::VectorizedArray<number>(1.) - tolerance);
324 require_less_than(mass_fraction_sum, dealii::VectorizedArray<number>(1.) + tolerance);
325 }
326
327 std::bitset<dealii::VectorizedArray<number>::size()> admissibility_mask;
328 for (unsigned int lane = 0; lane < n_active_lanes; ++lane)
329 admissibility_mask.set(lane, is_admissible[lane] > 0.);
330 return admissibility_mask;
331 }

333 template <int dim, typename Number>
334 void
335 calculate_penalty_parameter(
336 dealii::AlignedVector<dealii::VectorizedArray<Number>> &array_penalty_parameter,
337 const dealii::MatrixFree<dim, Number> &matrix_free,
338 const std::string &domain_representation_type,
339 const unsigned int dof_index,
340 const Number scaling_factor)
341 {
342 const unsigned int n_cells = matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
343 array_penalty_parameter.resize(n_cells);
344
345 dealii::Mapping<dim> const &mapping = *matrix_free.get_mapping_info().mapping;
346 dealii::FiniteElement<dim> const &fe = matrix_free.get_dof_handler(dof_index).get_fe();
347 unsigned int const degree = fe.degree;
348
349 // use penalty factor for hypercube elements according to K. Hillewaert, Development of the
350 // discontinuous Galerkin method for high-resolution, large scale CFD and acoustics in
351 // industrial geometries, PhD thesis, Univ. de Louvain, 2013.
352 const Number fac = scaling_factor * (degree + 1.0) * (degree + 1.0);
353
354 const std::vector<dealii::ReferenceCell<dim>> reference_cells =
355 matrix_free.get_dof_handler(dof_index).get_triangulation().get_reference_cells();
356 AssertThrow(reference_cells.size() == 1, dealii::ExcMessage("No mixed meshes allowed."));
357
358
359 const dealii::Quadrature<dim> quadrature =
360 reference_cells[0].get_gauss_type_quadrature(degree + 1);
361 dealii::FEValues<dim> fe_values(mapping, fe, quadrature, dealii::update_JxW_values);
362
363 const dealii::Quadrature<dim - 1> face_quadrature =
364 reference_cells[0].face_reference_cell(0).get_gauss_type_quadrature(degree + 1);
365 dealii::FEFaceValues<dim> fe_face_values(mapping,
366 fe,
367 face_quadrature,
368 dealii::update_JxW_values);
369
370 if (domain_representation_type == "fitted")
371 {
372 for (unsigned int i = 0; i < n_cells; ++i)
373 {
374 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
375 {
376 typename dealii::DoFHandler<dim>::cell_iterator cell =
377 matrix_free.get_cell_iterator(i, v, dof_index);
378 fe_values.reinit(cell);
379
380 // calculate cell volume
381 Number volume = 0;
382 for (unsigned int q = 0; q < quadrature.size(); ++q)
383 {
384 volume += fe_values.JxW(q);
385 }
386
387 // calculate surface area
388 Number surface_area = 0;
389 for (unsigned int const f : cell->face_indices())
390 {
391 fe_face_values.reinit(cell, f);
392 Number const factor =
393 (cell->at_boundary(f) and not(cell->has_periodic_neighbor(f))) ? 1. : 0.5;
394 for (unsigned int q = 0; q < face_quadrature.size(); ++q)
395 {
396 surface_area += fe_face_values.JxW(q) * factor;
397 }
398 }
399
400 array_penalty_parameter[i][v] = surface_area / volume * fac;
401 }
402 }
403 }
404 else if (domain_representation_type == "cut")
405 {
406 for (unsigned int i = 0; i < n_cells; ++i)
407 {
408 for (unsigned int v = 0; v < matrix_free.n_active_entries_per_cell_batch(i); ++v)
409 {
410 typename dealii::DoFHandler<dim>::cell_iterator cell =
411 matrix_free.get_cell_iterator(i, v, dof_index);
412
413 // simplified computation for hypercube elements
414 array_penalty_parameter[i][v] = fac / cell->minimum_vertex_distance();
415 }
416 }
417 }
418 else
419 AssertThrow(false,
420 dealii::ExcMessage("The domain representation type '" +
421 domain_representation_type + "' is not supported."));
422 }

424 template <int dim, typename number, typename ConservedVariablesType>
425 void
426 update_primitive_variables_solution(
427 dealii::LinearAlgebra::distributed::Vector<number> &solution_primitive_variables,
428 const dealii::LinearAlgebra::distributed::Vector<number> &solution,
429 const ScratchData<dim, dim, number> &scratch_data,
430 const unsigned int dof_idx,
431 const unsigned int quad_idx,
432 const MaterialPhaseData<number> *material_liquid,
433 const MaterialPhaseData<number> *material_gas)
434 {
435 using StateView = DofStateView<dim, number, const ConservedVariablesType>;
436 using PrimitiveValueView = DofPrimitiveValueView<dim, ConservedVariablesType>;
437
438 const dealii::MatrixFree<dim, number, dealii::VectorizedArray<number>> &matrix_free =
439 scratch_data.get_matrix_free();
440 unsigned int n_support_points_per_cell = scratch_data.get_n_dofs_per_cell(dof_idx) / (dim + 2);
441
442 auto process_cell = [&](CutUtil::CellCategory category,
443 unsigned int first_component,
444 const MaterialPhaseData<number> *material,
445 const unsigned int cell_batch) {
446 if (!material)
447 return;
448 auto eval = FECellIntegrator<dim, dim + 2, number>(
449 matrix_free, dof_idx, quad_idx, first_component, category);
450 eval.reinit(cell_batch);
451 eval.read_dof_values(solution);
452
453 for (unsigned int i = 0; i < n_support_points_per_cell; ++i)
454 {
455 const auto &u_cons = eval.get_dof_value(i);
456 StateView u_cons_view(u_cons, *material);
457 ConservedVariablesType u_prim_data;
458 PrimitiveValueView u_prim_view(u_prim_data);
459 conservative_to_primitive<dim, StateView, PrimitiveValueView>(u_cons_view, u_prim_view);
460 eval.submit_dof_value(u_prim_view.value(), i);
461 }
462
463 eval.set_dof_values(solution_primitive_variables);
464 };
465
466 if (!matrix_free.get_dof_info(dof_idx).cell_active_fe_index.empty())
467 {
468 // using hp::FECollection (cutFEM/DG)
469 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
470 {
471 const auto cell_category = matrix_free.get_cell_category(cell_batch);
472
473 if (cell_category == CutUtil::CellCategory::liquid)
474 {
475 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
476 }
477 else if (cell_category == CutUtil::CellCategory::gas)
478 {
479 process_cell(CutUtil::CellCategory::gas, dim + 2, material_gas, cell_batch);
480 }
481 else if (cell_category == CutUtil::CellCategory::intersected)
482 {
483 process_cell(CutUtil::CellCategory::intersected, 0, material_liquid, cell_batch);
484 process_cell(CutUtil::CellCategory::intersected, dim + 2, material_gas, cell_batch);
485 }
486 }
487 }
488 else
489 {
490 // not using hp::FECollection (fitted mesh)
491 for (unsigned int cell_batch = 0; cell_batch < matrix_free.n_cell_batches(); ++cell_batch)
492 {
493 process_cell(CutUtil::CellCategory::liquid, 0, material_liquid, cell_batch);
494 }
495 }
496 }

527 template <typename DofViewType, typename VectorizedArrayType>
528 inline DEAL_II_ALWAYS_INLINE //
529 VectorizedArrayType
530 maximum_local_wave_speed(const DofViewType &u_m, const DofViewType &u_p)
531 {
532 const auto velocity_m = u_m.velocity();
533 const auto velocity_p = u_p.velocity();
534
535 const auto sound_speed_p = u_p.speed_of_sound();
536 const auto sound_speed_m = u_m.speed_of_sound();
537
538 const auto sound_speed_p2 = sound_speed_p * sound_speed_p;
539 const auto sound_speed_m2 = sound_speed_m * sound_speed_m;
540
541 const auto lambda = 0.5 * std::sqrt(std::max(velocity_p.norm_square() + sound_speed_p2,
542 velocity_m.norm_square() + sound_speed_m2));
543 return lambda;
544 }

546 template <int dim, typename DofViewType, typename VectorizedArrayType>
547 inline DEAL_II_ALWAYS_INLINE //
548 std::pair<VectorizedArrayType, VectorizedArrayType>
549 acoustic_wave_speeds(const DofViewType &u_m,
550 const DofViewType &u_p,
551 const dealii::Tensor<1, dim, VectorizedArrayType> &unit_normal)
552 {
553 const VectorizedArrayType average_normal_velocity =
554 0.5 * ((u_m.velocity() + u_p.velocity()) * unit_normal);
555 const VectorizedArrayType average_speed_of_sound =
556 0.5 * (u_m.speed_of_sound() + u_p.speed_of_sound());
557
558 const VectorizedArrayType smallest_wave_speed =
559 std::min(VectorizedArrayType(0.), average_normal_velocity - average_speed_of_sound);
560 const VectorizedArrayType largest_wave_speed =
561 std::max(VectorizedArrayType(0.), average_normal_velocity + average_speed_of_sound);
562 return {smallest_wave_speed, largest_wave_speed};
563 }

565 template <int dim, typename number>
566 inline DEAL_II_ALWAYS_INLINE //
567 dealii::Tensor<2, dim, dealii::VectorizedArray<number>>
568 calculate_cauchy_stress_tensor(
569 const dealii::VectorizedArray<number> &pressure,
570 const dealii::Tensor<2, dim, dealii::VectorizedArray<number>> &viscous_stress_tensor)
571 {
572 auto stress_tensor = viscous_stress_tensor;
573
