301 template <int dim,
302 typename number,
303 typename ConservedVariablesType,
304 typename ConservedVariablesGradType>
305 inline DEAL_II_ALWAYS_INLINE //
306 std::
307 tuple<ConservedVariablesGradType, ConservedVariablesGradType, dealii::VectorizedArray<number>>
308 calculate_convective_interface_flux_HLLP0(
309 const ConservedVariablesType &u_liquid,
310 const ConservedVariablesType &u_gas,
311 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
312 const auto &convective_terms_liquid,
313 const auto &convective_terms_gas,
314 const Flow::CompressibleMultiphaseScratchData<dim, number> &multiphase_scratch_data,
315 const number &m_dot_evap)
316 {
317 // Note: Variables, that are relevant for both the liquid and the gas phase, are considered as
318 // arrays of length 2 in the following. The first element refers to the liquid phase and the
319 // second element to the gas phase.
320 constexpr int liquid = 0;
321 constexpr int gas = 1;
322
323 // enumeration for conserved variables component indices
324 using Idx = std::conditional_t<
325 dim == 1,
326 Flow::Idx1D,
327 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
328
329 // 0) preliminaries
330
331 std::array<ConservedVariablesType, 2> u = {{u_liquid, u_gas}};
332 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel;
333 std::array<dealii::VectorizedArray<number>, 2> pressure;
334 std::array<dealii::VectorizedArray<number>, 2> rho;
335 std::array<dealii::VectorizedArray<number>, 2> rho_E;
336 std::array<dealii::VectorizedArray<number>, 2> speed_of_sound;
337 for (unsigned int i : {0, 1})
338 {
339 vel[i] = Flow::calculate_velocity<dim>(u[i]);
340 rho[i] = u[i][Idx::density];
341 rho_E[i] = u[i][Idx::energy];
342 }
343 pressure[liquid] =
344 multiphase_scratch_data.material_liquid.eos_utils->calculate_thermodynamic_pressure(
345 u[liquid]);
346 pressure[gas] =
347 multiphase_scratch_data.material_gas.eos_utils->calculate_thermodynamic_pressure(u[gas]);
348 speed_of_sound[liquid] =
349 multiphase_scratch_data.material_liquid.eos_utils->calculate_speed_of_sound(u[liquid]);
350 speed_of_sound[gas] =
351 multiphase_scratch_data.material_gas.eos_utils->calculate_speed_of_sound(u[gas]);
352
353 // 1) project velocity and kinetic energy into normal direction of the interface
354
355 std::array<dealii::VectorizedArray<number>, 2> vel_n;
356 std::array<dealii::VectorizedArray<number>, 2> E_n;
357 for (unsigned int i : {0, 1})
358 {
359 vel_n[i] = vel[i] * normal;
360 E_n[i] = rho_E[i] / rho[i] - 0.5 * (vel[i] * vel[i] - vel_n[i] * vel_n[i]);
361 }
362
363 // 2) shock speed estimation
364
365 std::array<dealii::VectorizedArray<number>, 2> shock_speed;
366 shock_speed[liquid] = vel_n[liquid] - speed_of_sound[liquid];
367 shock_speed[gas] = vel_n[gas] + speed_of_sound[gas];
368
369 // 3) calculate helpers using Rankine-Hugoniot conditions
370
371 std::array<dealii::VectorizedArray<number>, 2> m_hat;
372 std::array<dealii::VectorizedArray<number>, 2> I_hat;
373 std::array<dealii::VectorizedArray<number>, 2> E_hat;
374 for (unsigned int i : {0, 1})
375 {
376 m_hat[i] = rho[i] * (vel_n[i] - shock_speed[i]);
377 I_hat[i] = m_hat[i] * vel_n[i] + pressure[i];
378 E_hat[i] = m_hat[i] * E_n[i] + pressure[i] * vel_n[i];
379 }
380
381 // 4) calculate intermediate velocity states
382
383 // TODO: consider surface tension for dim>1 here
384 const dealii::VectorizedArray<number> delta_p = 0.;
385
386 // TODO: consider Hertz-Knudsen theory for evaporation mass flux here
387 std::array<dealii::VectorizedArray<number>, 2> tmp_1;
388 std::array<dealii::VectorizedArray<number>, 2> tmp_2;
389 for (unsigned int i : {0, 1})
390 {
391 tmp_1[i] = m_dot_evap / m_hat[i];
392 tmp_2[i] = m_dot_evap - m_hat[i];
393 }
394
395 std::array<dealii::VectorizedArray<number>, 2> numerator;
396 std::array<dealii::VectorizedArray<number>, 2> denominator;
397
398 numerator[liquid] = tmp_2[gas] *
399 (tmp_1[liquid] * shock_speed[liquid] - tmp_1[gas] * shock_speed[gas]) /
400 (1. - tmp_1[gas]) -
401 I_hat[liquid] - delta_p + I_hat[gas];
402 numerator[gas] = tmp_2[liquid] *
403 (tmp_1[gas] * shock_speed[gas] - tmp_1[liquid] * shock_speed[liquid]) /
404 (1. - tmp_1[liquid]) -
405 I_hat[gas] + delta_p + I_hat[liquid];
406 denominator[liquid] = tmp_2[liquid] - (1. - tmp_1[liquid]) / (1. - tmp_1[gas]) * tmp_2[gas];
407 denominator[gas] = tmp_2[gas] - (1. - tmp_1[gas]) / (1. - tmp_1[liquid]) * tmp_2[liquid];
408
409 std::array<dealii::VectorizedArray<number>, 2> vel_n_star;
410 for (unsigned int i : {0, 1})
411 vel_n_star[i] = numerator[i] / denominator[i];
412
413 // 5) calculate intermediate pressure
414
415 std::array<dealii::VectorizedArray<number>, 2> pressure_star;
416 for (unsigned int i : {0, 1})
417 pressure_star[i] = I_hat[i] - m_hat[i] * vel_n_star[i];
418
419 // 6) re-project normal velocity to Cartesian coordinates
420
421 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel_star_cartesian;
422
423 std::vector<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> tangent;
424 tangent.resize(dim - 1);
425
426 // compute tangential vector for dim=2 and dim=3
427 if constexpr (dim == 2)
428 {
429 tangent[0][0] = normal[1];
430 tangent[0][1] = -normal[0];
431 }
432 else if constexpr (dim == 3)
433 {
434 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec;
435 temp_vec[0] = 1.;
436 // if normal vector is identical with unit vector choose different unit vector to
437 // compute the tangent
438 dealii::VectorizedArray<number> tolerance = 1.e-10;
439 dealii::VectorizedArray<number> norm_diff = (temp_vec - normal).norm();
440 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec_y;
441 temp_vec_y[1] = 1.;
442 for (int i = 0; i < 3; ++i)
443 {
444 temp_vec[i] = compare_and_apply_mask<dealii::SIMDComparison::less_than>(norm_diff,
445 tolerance,
446 temp_vec_y[i],
447 temp_vec[i]);
448 }
449 tangent[0] = temp_vec - (temp_vec * normal) * normal;
450 tangent[1] = dealii::cross_product_3d(normal, tangent[0]);
451 }
452
453 for (unsigned int i : {0, 1})
454 {
455 vel_star_cartesian[i] = vel_n_star[i] * normal;
456 for (unsigned int j = 0; j < dim - 1; ++j)
457 vel_star_cartesian[i] += (vel[i] * tangent[j]) * tangent[j];
458 }
459
460 // 7) calculate conservative variable state vectors of inner states
461
462 std::array<ConservedVariablesType, 2> u_star;
463 for (unsigned int i : {0, 1})
464 {
465 u_star[i][Idx::density] = m_hat[i] / (vel_n_star[i] - shock_speed[i]);
466 for (unsigned int j = 1; j < dim + 1; j++)
467 u_star[i][j] = u_star[i][Idx::density] * vel_star_cartesian[i][0][j - 1];
468 u_star[i][Idx::energy] =
469 (E_hat[i] - pressure_star[i] * vel_n_star[i]) / (vel_n_star[i] - shock_speed[i]) -
470 0.5 * u_star[i][Idx::density] * vel_n_star[i] * vel_n_star[i] +
471 0.5 * u_star[i][Idx::density] * vel_star_cartesian[i] * vel_star_cartesian[i];
472 }
473
474 // 8) calculate phase interface velocity
475
476 dealii::VectorizedArray<number> numerator_normal_vel =
477 vel_n_star[liquid] * u_star[liquid][Idx::density] -
478 vel_n_star[gas] * u_star[gas][Idx::density];
479 dealii::VectorizedArray<number> denominator_normal_vel =
480 u_star[liquid][Idx::density] - u_star[gas][Idx::density];
481 // avoid division by zero
482 dealii::VectorizedArray<number> normal_velocity_interface =
483 compare_and_apply_mask<dealii::SIMDComparison::greater_than>(std::abs(denominator_normal_vel),
484 1.e-12,
485 numerator_normal_vel /
486 denominator_normal_vel,
487 vel_n_star[liquid]);
488
489 // 9) calculate fluxes for the two phases
490
491 std::array<ConservedVariablesGradType, 2> flux;
492 std::array<ConservedVariablesGradType, 2> conv_flux;
493 std::array<ConservedVariablesGradType, 2> shock_flux;
494
495 conv_flux[liquid] = convective_terms_liquid.calculate_convective_flux(u[liquid]);
496 conv_flux[gas] = convective_terms_gas.calculate_convective_flux(u[gas]);
497
498 for (unsigned int i : {0, 1})
499 {
500 shock_flux[i] = dyadic_product(shock_speed[i] * (u_star[i] - u[i]), normal);
501 flux[i] = conv_flux[i];
502 }
503
504 const auto zero_vec = dealii::make_vectorized_array(0.);
505 const auto one_vec = dealii::make_vectorized_array(1.);
506
507 flux[liquid] +=
508 shock_flux[liquid] * compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
509 shock_speed[liquid], zero_vec, zero_vec, one_vec);
510 flux[gas] +=
511 shock_flux[gas] * compare_and_apply_mask<dealii::SIMDComparison::less_than_or_equal>(
512 shock_speed[gas], zero_vec, zero_vec, one_vec);
513
514 return {flux[liquid], flux[gas], normal_velocity_interface};
515 }

543 inline DEAL_II_ALWAYS_INLINE //
544 ConservedVariablesType
545 calculate_Dirichlet_jump_in_conservative_variables(
546 const ConservedVariablesType &u_liquid_cons,
547 const ConservedVariablesType &u_gas_cons,
548 const Flow::CompressibleMultiphaseScratchData<dim, number> &multiphase_scratch_data,
549 const number &m_dot_evap,
550 const number &delta_T)
551 {
552 // enumeration for conserved variables component indices
553 using Idx = std::conditional_t<
554 dim == 1,
555 Flow::Idx1D,
556 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
557
558 auto u_liquid_prim = multiphase_scratch_data.material_liquid.eos_utils
559 ->convert_conservative_into_primitive_variables(u_liquid_cons);
560
561 auto u_gas_prim =
562 multiphase_scratch_data.material_gas.eos_utils->convert_conservative_into_primitive_variables(
563 u_gas_cons);
564
565 // TODO: consider surface tension here
566 const dealii::VectorizedArray<number> delta_p = 0.;
567
568 // TODO: extend to general case dim>1
569 ConservedVariablesType J_Dir;
570
571 J_Dir[Idx::density] = delta_p - m_dot_evap * (u_liquid_prim[1] - u_gas_prim[1]);
572 J_Dir[Idx::momentum_x] = m_dot_evap * (1. / u_liquid_cons[0] - 1. / u_gas_cons[0]);
573 J_Dir[Idx::energy] = delta_T;
574
575 const auto u_liquid_prim_tmp = u_liquid_prim;
576 u_liquid_prim = u_gas_prim + J_Dir;
577 u_gas_prim = u_liquid_prim_tmp - J_Dir;
578
579 const auto u_liquid_cons_star =
580 multiphase_scratch_data.material_liquid.eos_utils
581 ->convert_primitive_into_conservative_variables(u_liquid_prim);
582
583 const auto u_gas_cons_star =
584 multiphase_scratch_data.material_gas.eos_utils->convert_primitive_into_conservative_variables(
585 u_gas_prim);
586
587 J_Dir = u_liquid_cons_star - u_gas_cons_star;
588
589 return J_Dir;
590 }

648 const number &laser_heat_source)
649 {
650 // enumeration for conserved variables component indices
651 using Idx = std::conditional_t<
652 dim == 1,
653 Flow::Idx1D,
654 std::conditional_t<dim == 2, Flow::Idx2D, std::conditional_t<dim == 3, Flow::Idx3D, void>>>;
655
656 // TODO: add contributions for surface tension, interface heat source (laser energy) and
657 // Marangoni forces
658
659 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_liquid =
660 Flow::calculate_velocity<dim>(u_liquid);
661 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_gas =
662 Flow::calculate_velocity<dim>(u_gas);
663
664 const dealii::VectorizedArray<number> vel_n_liquid = vel_liquid * normal;
665 const dealii::VectorizedArray<number> vel_n_gas = vel_gas * normal;
666
667 const auto pressure_liquid =
668 multiphase_scratch_data.material_liquid.eos_utils->calculate_thermodynamic_pressure(u_liquid);
669 const auto pressure_gas =
670 multiphase_scratch_data.material_gas.eos_utils->calculate_thermodynamic_pressure(u_gas);
671
672 // compute Robin-type viscous interface jump conditions
673
674 ConservedVariablesType J_Rob;
675
676 // TODO: Add entries for case dim>1
677 J_Rob[Idx::energy] =
678 (pressure_liquid * vel_n_liquid - pressure_gas * vel_n_gas) +
679 m_dot_evap * (u_liquid[Idx::energy] / u_liquid[Idx::density] -
680 u_gas[Idx::energy] / u_gas[Idx::density]) +
681 laser_heat_source -
682 m_dot_evap * multiphase_scratch_data.phase_change.liquid_gas.latent_heat_of_vaporization;
683
684 const ConservedVariablesGradType viscous_flux_liquid =
685 viscous_terms_liquid.calculate_viscous_flux(u_liquid, grad_u_liquid);
686
687 const ConservedVariablesGradType viscous_flux_gas =
688 viscous_terms_gas.calculate_viscous_flux(u_gas, grad_u_gas);
689
690 ConservedVariablesGradType total_flux_liquid = dyadic_product(J_Rob, normal);
691 total_flux_liquid += viscous_flux_gas;
692 total_flux_liquid *= visc_ave_weight_phase_liquid;
693 total_flux_liquid += visc_ave_weight_phase_gas * viscous_flux_liquid;
694
695 ConservedVariablesGradType total_flux_gas =
696 dyadic_product(J_Rob, -normal); // opposite normal direction for phase 2
697 total_flux_gas += viscous_flux_liquid;
698 total_flux_gas *= visc_ave_weight_phase_gas;
699 total_flux_gas += visc_ave_weight_phase_liquid * viscous_flux_gas;
700
701 // penalty term
702
703 const auto J_Dir_cons =
704 calculate_Dirichlet_jump_in_conservative_variables<dim, number, ConservedVariablesType>(
705 u_liquid, u_gas, multiphase_scratch_data, m_dot_evap, delta_T);
706
707 const number penalty_parameter =
708 std::min(multiphase_scratch_data.material_liquid.data.dynamic_viscosity /
709 multiphase_scratch_data.material_liquid.data.reference_density,
710 multiphase_scratch_data.material_gas.data.dynamic_viscosity /
711 multiphase_scratch_data.material_gas.data.reference_density) *
712 (multiphase_scratch_data.flow_data.fe.degree + 1.) *
713 (multiphase_scratch_data.flow_data.fe.degree + 1.) / cell_size * tau;
714
715 ConservedVariablesGradType penalty_flux_liquid;
716 const auto tmp_m = u_liquid - (u_gas + J_Dir_cons);
717 penalty_flux_liquid = dyadic_product(tmp_m, normal);
718 penalty_flux_liquid *= penalty_parameter;
719
720 ConservedVariablesGradType penalty_flux_gas;
721 const auto tmp_p = u_gas - (u_liquid - J_Dir_cons);
722 penalty_flux_gas = dyadic_product(tmp_p, -normal);
723 penalty_flux_gas *= penalty_parameter;
724
725 total_flux_liquid -= penalty_flux_liquid;
726 total_flux_gas -= penalty_flux_gas;
727
728 return {contract_tensor_with_vector<dim + 2, dim, number>(total_flux_liquid, normal),
729 contract_tensor_with_vector<dim + 2, dim, number>(total_flux_gas, normal)};
730 }

762 typename number,
763 typename ConservedVariablesType,
764 typename ConservedVariablesGradType>
765 inline DEAL_II_ALWAYS_INLINE //
766 std::pair<ConservedVariablesGradType, ConservedVariablesGradType>
767 calculate_viscous_interface_flux_gradient(
768 const ConservedVariablesType &u_liquid,
769 const ConservedVariablesType &u_gas,
770 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
771 const number &visc_ave_weight_phase_liquid,
772 const number &visc_ave_weight_phase_gas,
773 const auto &viscous_terms_liquid,
774 const auto &viscous_terms_gas,
775 const Flow::CompressibleMultiphaseScratchData<dim, number> &multiphase_scratch_data,
776 const number &m_dot_evap,
777 const number &delta_T)
778 {
779 const auto J_Dir_cons =
780 calculate_Dirichlet_jump_in_conservative_variables<dim, number, ConservedVariablesType>(
781 u_liquid, u_gas, multiphase_scratch_data, m_dot_evap, delta_T);
782
783 const auto u_liquid_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
784 visc_ave_weight_phase_gas, u_liquid, visc_ave_weight_phase_liquid, u_gas + J_Dir_cons);
785 const auto u_gas_star = UtilityFunctions::calculate_arithmetic_phase_weighted_average(
786 visc_ave_weight_phase_liquid, u_gas, visc_ave_weight_phase_gas, u_liquid - J_Dir_cons);
787
788 auto tmp_liquid = u_liquid_star - u_liquid;
789 auto tmp_gas = u_gas_star - u_gas;
790
791 ConservedVariablesGradType arg_liquid = dyadic_product(tmp_liquid, normal);
792 ConservedVariablesGradType arg_gas = dyadic_product(tmp_gas, -normal);
793
794 const ConservedVariablesGradType flux_grad_liquid =
795 viscous_terms_liquid.calculate_viscous_flux(u_liquid, arg_liquid);
796 const ConservedVariablesGradType flux_grad_gas =
797 viscous_terms_gas.calculate_viscous_flux(u_gas, arg_gas);
798
799 return {flux_grad_liquid, flux_grad_gas};
800 }

887 u_gas[Idx::energy] / u_gas[Idx::density]) +
888 (pressure_liquid * vel_n_liquid - pressure_gas * vel_n_gas) + laser_heat_source -
889 m_dot_evap * multiphase_scratch_data.phase_change.liquid_gas.latent_heat_of_vaporization;

912 contract_tensor_with_vector<dim + 2, dim, number>(viscous_flux_gas, normal));
