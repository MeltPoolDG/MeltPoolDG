129 // Lambda function to read and parse a single line of the input file. The function reads the
130 // line, computes all particle properties based on the parsed values, and returns the particle
131 // properties and location as a pair. If the line is empty or contains only whitespace, it
132 // returns an empty optional to indicate that no particle should be initialized from this line.
133 // The function throws an exception if the line does not contain the expected number of columns
134 // or if any of the values cannot be converted to a number. Note that no deal.II assertions are
135 // used in this function but only standard exceptions allowing the caller to easily catch and
136 // handle errors in the input file without aborting the entire program.
137 const auto read_values_from_line = [](const std::string &line)
138 -> std::optional<std::pair<std::vector<number>, dealii::Point<dim, number>>> {
139 constexpr unsigned int n_expected_columns = dim + 2; // position (dim) + density + radius
140
141 std::vector<std::string> parsed_strings;
142 parsed_strings.reserve(n_expected_columns);
143
144 std::istringstream data_string(line);
145
146 std::string token;
147 while (std::getline(data_string, token, ','))
148 {
149 parsed_strings.emplace_back(std::move(token));
150 }
151
152 // If the line does not contain any data, return an empty optional to indicate that no
153 // particle should be initialized from this line. This allows for empty lines in the input
154 // file without causing errors.
155 if (parsed_strings.size() == 0 or
156 (parsed_strings.size() == 1 and (parsed_strings[0] == "\r" or parsed_strings[0] == "\n" or
157 parsed_strings[0] == "\t" or parsed_strings[0].empty())))
158 {
159 return std::nullopt;
160 }
161
162 // Check if the expected number of columns is present
163 AssertThrow(parsed_strings.size() == n_expected_columns,
164 ExcInvalidCSVInputColumns(n_expected_columns, parsed_strings.size()));
165
166 // Convert parsed strings to numbers
167 std::vector<number> parsed_values;
168 parsed_values.reserve(n_expected_columns);
169 for (std::string &str : parsed_strings)
170 {
171 // remove trailing whitespace
172 if (!str.empty())
173 {
174 char last_char = str.back();
175
176 if (last_char == '\n' or last_char == '\r' or last_char == '\t')
177 {
178 str.pop_back();
179 }
180 }
181
182 std::size_t characters_processed = 0;
183 parsed_values.emplace_back(std::stod(str, &characters_processed));
184 AssertThrow(characters_processed == str.size(), ExcFailedToConvertStringToNumber(str));
185 }
186
187 // particle location
188 dealii::Point<dim, number> particle_location;
189 for (unsigned int i = 0; i < dim; ++i)
190 particle_location[i] = parsed_values[PropertyCSVColumns::position + i];
191
192 // particle properties
193 std::vector<number> particle_properties(SphericalParticle<dim, number>::n_obstacle_properties,
194 0.0);
195 DEMParticleAccessor<dim, number> accessor(particle_location, particle_properties);
196
197 compute_and_set_spherical_particle_properties(accessor,
198 parsed_values[PropertyCSVColumns::radius],
199 parsed_values[PropertyCSVColumns::density]);
200
201 return std::optional<std::pair<std::vector<number>, dealii::Point<dim, number>>>(
202 {particle_properties, particle_location});
203 };

234 particle_locations.push_back(obstacle_properties.value().second);
235 }

237 }
238 catch (const ExcInvalidCSVInputColumns &e)
239 {

241 }
242 catch (const ExcFailedToConvertStringToNumber &e)
243 {
244 throw_error_with_line_info(line_number, e.what());
245 }
246 }
247 }
