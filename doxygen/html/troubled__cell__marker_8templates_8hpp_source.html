67 template <int dim, int n_components, typename number>
68 void
69 TroubledCellMarker<dim, n_components, number>::mark(
70 const MatrixFreeContext<dim, number> &matrix_free_context,
71 const VectorType &candidate,
72 const VectorType &old_solution)
73 {
74 const dealii::MatrixFree<dim, number> &matrix_free = matrix_free_context.mf;
75
76 const bool old_solution_has_ghost_values = old_solution.has_ghost_elements();
77 if (not old_solution_has_ghost_values)
78 old_solution.update_ghost_values();
79
80 FESubcellEvaluation<dim, n_components, number> phi_candidate(matrix_free,
81 matrix_free_context.dof_idx,
82 matrix_free_context.quad_idx);
83 FESubcellEvaluation<dim, n_components, number> phi_old(matrix_free,
84 matrix_free_context.dof_idx,
85 matrix_free_context.quad_idx);
86
87 for (unsigned int cell = 0; cell < matrix_free.n_cell_batches(); ++cell)
88 {
89 phi_candidate.reinit(cell);
90 phi_candidate.gather_evaluate(candidate, dealii::EvaluationFlags::values);
91 phi_old.reinit(cell);
92 phi_old.gather_evaluate(old_solution, dealii::EvaluationFlags::values);
93
94 if (apply_physical_admissibility_check)
95 physical_admissibility.submit_cell(cell, phi_candidate);
96 if (apply_cellwise_nad)
97 cellwise_numerical_admissibility.submit_cell(cell, phi_old, phi_candidate);
98 if (apply_subcellwise_nad)
99 subcellwise_numerical_admissibility.submit_cell(cell, phi_old, phi_candidate);
100 }
101
102 // the extrema of the old solution on the ghost cells are required for the numerical
103 // admissibility check of the locally owned cells at MPI process boundaries
104 if (apply_cellwise_nad or apply_subcellwise_nad)
105 for (unsigned int cell = matrix_free.n_cell_batches();
106 cell < matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
107 ++cell)
108 {
109 phi_old.reinit(cell);
110 phi_old.gather_evaluate(old_solution, dealii::EvaluationFlags::values);
111 if (apply_cellwise_nad)
112 cellwise_numerical_admissibility.submit_ghost_cell(cell, phi_old);
113 if (apply_subcellwise_nad)
114 subcellwise_numerical_admissibility.submit_ghost_cell(cell, phi_old);
115 }
116
117 if (not old_solution_has_ghost_values)
118 old_solution.zero_out_ghost_values();
119
120 // a lane is troubled if it is marked by at least one of the applied strategies
121 std::fill(troubled_cells_mask.begin(), troubled_cells_mask.end(), MaskType());
122 if (apply_physical_admissibility_check)
123 physical_admissibility.evaluate(troubled_cells_mask);
124 if (apply_cellwise_nad)
125 cellwise_numerical_admissibility.evaluate(troubled_cells_mask);
126 if (apply_subcellwise_nad)
127 subcellwise_numerical_admissibility.evaluate(troubled_cells_mask);
128
129 // only the active lanes are marked, since the unused lanes of the last cell batch may hold
130 // states which are not admissible
131 for (unsigned int cell = 0; cell < matrix_free.n_cell_batches(); ++cell)
132 {
133 MaskType active_lanes;
134 for (unsigned int lane = 0; lane < matrix_free.n_active_entries_per_cell_batch(cell);
135 ++lane)
136 active_lanes.set(lane);
137 marked_cells.cell_data_vector()[cell] =
138 bitset_to_vectorized_mask<number>(troubled_cells_mask[cell] & active_lanes);
139 }
140
141 // the markers of the ghost cells are required at faces between MPI processes
142 marked_cells.update_ghost_values();
143 }
