325 template <InterfaceBandSolverOperatorType<VectorType> OperatorType>
326 unsigned int
327 solve(const OperatorType &op,
328 VectorType &x,
329 const VectorType &b,
330 const LinearSolverData<number> &linear_solver_data,
331 const std::string &identifier = "")
332 {
333 Assert(band.is_initialized(),
334 dealii::ExcMessage("The band solver has not been updated after its last "
335 "reinitialization. Call update() before solve()."));
336
337 const std::vector<unsigned int> &band_dof_indices = band.get_dof_indices();
338
339 if (not vectors_valid)
340 setup_vectors(b);
341
342 AssertDimension(src_full.locally_owned_size(), b.locally_owned_size());
343
344 // Set up reduced vectors.
345 for (unsigned int i = 0; i < band_dof_indices.size(); ++i)
346 {
347 b_band.local_element(i) = b.local_element(band_dof_indices[i]);
348 // Initial guess: previous solution on the band DoFs.
349 x_band.local_element(i) = x.local_element(band_dof_indices[i]);
350 }
351
352 const BandOperator<OperatorType> band_operator{op, band, src_full, dst_full};
353
354 // Solve the band system with a linear solver.
355 // Currently, no preconditioner is used for the band solve, since the band system is small and
356 // well-conditioned.
357 const int n_iterations = LinearSolver::solve<VectorType>(band_operator,
358 x_band,
359 b_band,
360 linear_solver_data,
361 dealii::PreconditionIdentity(),
362 identifier);
363
364 if (x.has_ghost_elements())
365 x.zero_out_ghost_values();
366
367 // Apply the exact cell-wise inverse mass matrix for the cells away from the interface.
368 op.apply_inverse_mass_matrix_outside_band(band, x, b);
369
370 // Copy the band entries of the reduced solution vector into the full vector.
371 for (unsigned int i = 0; i < band_dof_indices.size(); ++i)
372 x.local_element(band_dof_indices[i]) = x_band.local_element(i);
373
374 if (x.has_ghost_elements())
375 x.update_ghost_values();
376
377 return n_iterations;
378 }

395 void
396 vmult(VectorType &dst, const VectorType &src) const
397 {
398 const std::vector<unsigned int> &band_dof_indices = band.get_dof_indices();
399
400 // expand: band entries of the full vectors (the entries of src_full outside the band are
401 // never written and stay zero; they do not influence the band entries of the result of
402 // vmult_band())
403 for (unsigned int i = 0; i < band_dof_indices.size(); ++i)
404 {
405 src_full.local_element(band_dof_indices[i]) = src.local_element(i);
406 dst_full.local_element(band_dof_indices[i]) = number(0.);
407 }
408
409 // solve band system
410 op.vmult_band(band, dst_full, src_full);
411
412 // compress: band entries of the result
413 for (unsigned int i = 0; i < band_dof_indices.size(); ++i)
414 dst.local_element(i) = dst_full.local_element(band_dof_indices[i]);
415 }

424 void
425 setup_vectors(const VectorType &full_vector)
426 {
427 const MPI_Comm mpi_comm = full_vector.get_mpi_communicator();
428
429 const dealii::types::global_dof_index n_locally_owned_band_dofs =
430 band.get_dof_indices().size();
431 // contiguous global numbering of the band DoFs (ordered by MPI rank)
432 const std::vector<dealii::types::global_dof_index> n_band_dofs_per_rank =
433 dealii::Utilities::MPI::all_gather(mpi_comm, n_locally_owned_band_dofs);
434 const unsigned int rank = dealii::Utilities::MPI::this_mpi_process(mpi_comm);
435
436 dealii::types::global_dof_index offset = 0, n_band_dofs = 0;
437 for (unsigned int r = 0; r < n_band_dofs_per_rank.size(); ++r)
438 {
439 if (r < rank)
440 offset += n_band_dofs_per_rank[r];
441 n_band_dofs += n_band_dofs_per_rank[r];
442 }
443
444 dealii::IndexSet locally_owned_band_dofs(n_band_dofs);
445 locally_owned_band_dofs.add_range(offset, offset + n_locally_owned_band_dofs);
446
447 x_band.reinit(locally_owned_band_dofs, mpi_comm);
448 b_band.reinit(locally_owned_band_dofs, mpi_comm);
449
450 // work vectors with the same parallel layout as the full system (zero-initialized)
451 src_full.reinit(full_vector);
452 dst_full.reinit(full_vector);
453
454 vectors_valid = true;
455 }
