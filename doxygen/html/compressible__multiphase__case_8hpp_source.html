161 void
162 print_relative_norm(const GenericDataOut<dim, number> &generic_data_out,
163 dealii::Function<dim> &reference_function,
164 const std::string &norm_name = "norm") const
165 {
166 const dealii::ConditionalOStream pcout(std::cout,
167 dealii::Utilities::MPI::this_mpi_process(
168 this->mpi_communicator) == 0 and
169 parameters.base.verbosity_level >= 1);
170
171 const auto &dof_vector = generic_data_out.get_vector("density_liquid");
172 const auto &dof_handler = generic_data_out.get_dof_handler("density_liquid");
173
174 const unsigned int n_dofs_per_element_per_phase =
175 0.5 * dof_handler.get_fe_collection().max_dofs_per_cell();
176 reference_function.set_time(generic_data_out.get_time());
177 number errors_squared[3] = {};
178
179 // reading DoF-vector and DoFHandler of level-set
180 // (an assert is thrown in the getter functions, if 'level_set' can not be found in
181 // the entries of 'generic_data_out')
182 const auto &level_set = generic_data_out.get_vector("level_set");
183 const auto &level_set_dof_handler = generic_data_out.get_dof_handler("level_set");
184
185 // generate FESystem
186 const unsigned int fe_degree = dof_handler.get_fe_collection().max_degree();
187 dealii::FE_DGQ<dim> fe_q(fe_degree);
188 dealii::FESystem<dim> fe_point_temp(fe_q, dim + 2);
189
190 // classify cells
191 dealii::NonMatching::MeshClassifier<dim> mesh_classifier(level_set_dof_handler, level_set);
192 mesh_classifier.reclassify();
193
194 // compute quadrature
195 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>>
196 mapping_info_cell_liquid(generic_data_out.get_mapping(),
197 dealii::update_values | dealii::update_JxW_values);
198 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>>
199 mapping_info_cell_gas(generic_data_out.get_mapping(),
200 dealii::update_values | dealii::update_JxW_values);
201
202 dealii::hp::QCollection<1> q_collection((dealii::QGauss<1>(fe_degree + 1)));
203 dealii::NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(
204 q_collection, level_set_dof_handler, level_set);
205
206 auto physical_liquid = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
207 return (mesh_classifier.location_to_level_set(i) ==
208 dealii::NonMatching::LocationToLevelSet::intersected) or
209 (mesh_classifier.location_to_level_set(i) ==
210 dealii::NonMatching::LocationToLevelSet::outside);
211 };
212 auto physical_gas = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
213 return (mesh_classifier.location_to_level_set(i) ==
214 dealii::NonMatching::LocationToLevelSet::intersected) or
215 (mesh_classifier.location_to_level_set(i) ==
216 dealii::NonMatching::LocationToLevelSet::inside);
217 };
218
219 const auto physical_active_cell_iterators_liquid =
220 dof_handler.active_cell_iterators() | physical_liquid;
221 const auto physical_active_cell_iterators_gas =
222 dof_handler.active_cell_iterators() | physical_gas;
223
224 std::vector<dealii::Quadrature<dim>> quad_vec_cell_liquid;
225 std::vector<dealii::Quadrature<dim>> quad_vec_cell_gas;
226
227 const unsigned int n_active_cells = std::distance(dof_handler.active_cell_iterators().begin(),
228 dof_handler.active_cell_iterators().end());
229
230 for (const auto &cell : physical_active_cell_iterators_liquid)
231 {
232 quadrature_generator.generate(cell);
233 // liquid phase has positive level-set
234 quad_vec_cell_liquid.push_back(quadrature_generator.get_outside_quadrature());
235 }
236
237 for (const auto &cell : physical_active_cell_iterators_gas)
238 {
239 quadrature_generator.generate(cell);
240 // gas phase has negative level-set
241 quad_vec_cell_gas.push_back(quadrature_generator.get_inside_quadrature());
242 }
243
244 mapping_info_cell_liquid.reinit_cells(physical_active_cell_iterators_liquid,
245 quad_vec_cell_liquid,
246 n_active_cells);
247 mapping_info_cell_gas.reinit_cells(physical_active_cell_iterators_gas,
248 quad_vec_cell_gas,
249 n_active_cells);
250
251 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>>
252 fe_point_eval_liquid(mapping_info_cell_liquid, fe_point_temp);
253 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>>
254 fe_point_eval_gas(mapping_info_cell_gas, fe_point_temp);
255
256 auto compute_error_norm = [&](auto &fe_point_eval,
257 auto &physical_domain,
258 const bool is_gas_phase) {
259 const unsigned int offset = is_gas_phase ? 0 : dim + 2;
260
261 for (const auto &cell : dof_handler.active_cell_iterators())
262 {
263 if (cell->is_artificial() or cell->is_ghost() or (not physical_domain(cell)))
264 continue;
265
266 const unsigned int n_dofs_per_cell = cell->get_fe().dofs_per_cell;
267 std::vector<number> solution_values_in(n_dofs_per_element_per_phase);
268
269 if (n_dofs_per_cell != n_dofs_per_element_per_phase)
270 {
271 // For an intersected element, we have to extract the Dofs, which are relevant for
272 // the current phase
273 std::vector<number> solution_values_in_tmp(n_dofs_per_cell);
274 cell->get_dof_values(dof_vector,
275 solution_values_in_tmp.begin(),
276 solution_values_in_tmp.end());
277
278 if (is_gas_phase)
279 solution_values_in.assign(solution_values_in_tmp.begin() +
280 n_dofs_per_element_per_phase,
281 solution_values_in_tmp.end());
282 else
283 solution_values_in.assign(solution_values_in_tmp.begin(),
284 solution_values_in_tmp.begin() +
285 n_dofs_per_element_per_phase);
286 }
287 else
288 cell->get_dof_values(dof_vector,
289 solution_values_in.begin(),
290 solution_values_in.end());
291
292 fe_point_eval.reinit(cell->active_cell_index());
293 fe_point_eval.evaluate(solution_values_in, dealii::EvaluationFlags::values);
294
295 dealii::VectorizedArray<number> local_errors_squared[3] = {};
296
297 for (const unsigned int q : fe_point_eval.quadrature_point_indices())
298 {
299 const auto solution = fe_point_eval.get_value(q);
300 dealii::VectorizedArray<number> error[dim + 2] = {};
301 for (unsigned int i = 0; i < dim + 2; ++i)
302 {
303 error[i] =
304 VectorTools::evaluate_function_at_vectorized_points<dim, number>(
305 reference_function, fe_point_eval.quadrature_point(q), i + offset) -
306 solution[i];
307 }
308
309 const auto JxW = fe_point_eval.JxW(q);
310 local_errors_squared[0] = error[0] * error[0] * JxW;
311
312 for (unsigned int d = 0; d < dim; ++d)
313 local_errors_squared[1] = (error[d + 1] * error[d + 1]) * JxW;
314
315 local_errors_squared[2] = (error[dim + 1] * error[dim + 1]) * JxW;
316
317 for (unsigned int v = 0; v < fe_point_eval.n_active_entries_per_quadrature_batch(q);
318 ++v)
319 for (unsigned int d = 0; d < 3; ++d)
320 errors_squared[d] += local_errors_squared[d][v];
321 }
322 }
323 };
324
325 compute_error_norm(fe_point_eval_liquid, physical_liquid, false);
326 compute_error_norm(fe_point_eval_gas, physical_gas, true);
327
328 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
329 std::array<number, 3> errors{};
330 for (unsigned int d = 0; d < 3; ++d)
331 errors[d] = std::sqrt(errors_squared[d]);
332
333 std::ostringstream output;
334 output << std::scientific << std::setprecision(4) << norm_name << " rho: " << errors[0]
335 << ", rho * u: " << errors[1] << ", rho * energy: " << errors[2];
336 Journal::print_line(pcout, output.str(), "compressible_multiphase");
337 }
338 };
