198 void
199 print_relative_norm_fitted(const GenericDataOut<dim, number> &generic_data_out,
200 std::vector<DataPostprocessorData> data,
201 const std::string &norm_name = "norm") const
202 {
203 std::map<std::string, number> component_errors;
204
205 std::vector<std::string> labels;
206
207 // In the fitted mesh case, we can use dealii vector tools to compute the error norms
208 // based on the DoFHandler and DoFVector of the solution. We loop over the components of
209 // the solution and compute the error norms separately for each component. We also take
210 // into account the case that some components might be part of a vector (e.g. the velocity
211 // components) and compute the error norm for the entire vector instead of the individual
212 // components in that case.
213 for (const auto &postprocessor_data : data)
214 {
215 const dealii::DataPostprocessor<dim> &post_processor =
216 generic_data_out.get_data_postprocessor(postprocessor_data.name);
217
218 const dealii::DoFHandler<dim> &dof_handler =
219 generic_data_out.get_dof_handler(postprocessor_data.name);
220
221 const auto &solution = generic_data_out.get_vector(postprocessor_data.name);
222
223 // Identify the relevant ranges for the specific name
224 const auto &names = post_processor.get_names();
225 const auto &interpretation = post_processor.get_data_component_interpretation();
226
227 std::pair<unsigned int, unsigned int> component_range = {0, 0};
228 bool name_found = false;
229 unsigned i = 0;
230 while (i < names.size() and not name_found)
231 {
232 if (names[i] == postprocessor_data.name)
233 {
234 name_found = true;
235 component_range.first = i;
236 component_range.second = i + 1;
237 if (interpretation[i] ==
238 dealii::DataComponentInterpretation::component_is_part_of_vector)
239 {
240 while (component_range.second < interpretation.size() and
241 names[component_range.second] == postprocessor_data.name and
242 interpretation[component_range.second] ==
243 dealii::DataComponentInterpretation::component_is_part_of_vector)
244 {
245 ++component_range.second;
246 }
247 }
248 }
249 ++i;
250 }
251
252 AssertThrow(
253 name_found,
254 dealii::ExcMessage(
255 "The name given in the data vector does not have a corresponding registered data "
256 "postprocessor."));
257
258 // Create embedded reference function to match fe element of used dof handler
259 MeltPoolDG::Functions::EmbeddedComponentsFunction<dim, number> reference_function(
260 postprocessor_data.reference_function,
261 dof_handler.get_fe().n_components(),
262 component_range.first);
263
264 dealii::ComponentSelectFunction<dim> component_mask({component_range.first,
265 component_range.second},
266 dof_handler.get_fe().n_components());
267
268 // Compute global error
269 component_errors[postprocessor_data.name] =
270 VectorTools::compute_global_error_norm(solution,
271 dof_handler.get_triangulation(),
272 generic_data_out.get_mapping(),
273 dof_handler,
274 dealii::QGauss<dim>(dof_handler.get_fe().degree),
275 dealii::VectorTools::NormType::L2_norm,
276 reference_function,
277 &component_mask);
278
279 labels.push_back(postprocessor_data.name);
280 }
281
282 pretty_print_norm("L2 " + norm_name, labels, component_errors, 20);
283 }
300 void
301 print_relative_norm_cut(const GenericDataOut<dim, number> &generic_data_out,
302 dealii::Function<dim> &reference_function,
303 const std::string &norm_name = "norm") const
304 {
305 const auto &dof_vector = generic_data_out.get_vector("density");
306 const auto &dof_handler = generic_data_out.get_dof_handler("density");
307
308 reference_function.set_time(generic_data_out.get_time());
309
310 std::map<std::string, number> component_errors;
311
312 std::vector<std::string> labels;
313
314 // reading DoF-vector and DoFHandler of level-set
315 // (an assert is thrown in the getter functions, if 'level_set' can not be found in
316 // the entries of 'generic_data_out')
317 const auto &level_set = generic_data_out.get_vector("level_set");
318 const auto &level_set_dof_handler = generic_data_out.get_dof_handler("level_set");
319
320 number errors_squared[3] = {};
321
322 // generate FESystem
323 const unsigned int fe_degree = dof_handler.get_fe_collection().max_degree();
324 dealii::FE_DGQ<dim> fe_q(fe_degree);
325 dealii::FESystem<dim> fe_point_temp(fe_q, dim + 2);
326
327 // classify cells
328 dealii::NonMatching::MeshClassifier<dim> mesh_classifier(level_set_dof_handler, level_set);
329 mesh_classifier.reclassify();
330
331 // compute quadrature
332 dealii::NonMatching::MappingInfo<dim, dim, dealii::VectorizedArray<number>> mapping_info_cell(
333 generic_data_out.get_mapping(), dealii::update_values | dealii::update_JxW_values);
334
335 dealii::hp::QCollection<1> q_collection((dealii::QGauss<1>(fe_degree + 1)));
336 dealii::NonMatching::DiscreteQuadratureGenerator<dim> quadrature_generator(
337 q_collection, level_set_dof_handler, level_set);
338
339 auto physical = [&](const typename dealii::DoFHandler<dim>::active_cell_iterator &i) {
340 return (mesh_classifier.location_to_level_set(i) ==
341 dealii::NonMatching::LocationToLevelSet::intersected) or
342 (mesh_classifier.location_to_level_set(i) ==
343 dealii::NonMatching::LocationToLevelSet::
344 outside /*active phase has positive level-set*/);
345 };
346
347 const auto physical_active_cell_iterators = dof_handler.active_cell_iterators() | physical;
348
349 std::vector<dealii::Quadrature<dim>> quad_vec_cell;
350
351 unsigned int n_active_cells = std::distance(dof_handler.active_cell_iterators().begin(),
352 dof_handler.active_cell_iterators().end());
353
354 for (const auto &cell : physical_active_cell_iterators)
355 {
356 quadrature_generator.generate(cell);
357 // active phase has positive level-set
358 quad_vec_cell.push_back(quadrature_generator.get_outside_quadrature());
359 }
360
361 mapping_info_cell.reinit_cells(physical_active_cell_iterators, quad_vec_cell, n_active_cells);
362
363 // cell-loop for error-norm computation
364 dealii::FEPointEvaluation<dim + 2, dim, dim, dealii::VectorizedArray<number>> fe_point_eval(
365 mapping_info_cell, fe_point_temp);
366
367 for (const auto &cell : dof_handler.active_cell_iterators())
368 {
369 if (cell->is_artificial() or cell->is_ghost() or (not physical(cell)))
370 continue;
371
372 std::vector<number> solution_values_in(cell->get_fe().dofs_per_cell);
373
374 cell->get_dof_values(dof_vector, solution_values_in.begin(), solution_values_in.end());
375
376 fe_point_eval.reinit(cell->active_cell_index());
377 fe_point_eval.evaluate(solution_values_in, dealii::EvaluationFlags::values);
378
379 std::array<dealii::VectorizedArray<number>, 3> local_errors_squared = {};
380
381 for (const unsigned int q : fe_point_eval.quadrature_point_indices())
382 {
383 const auto error =
384 VectorTools::evaluate_function_at_vectorized_points<dim, number, dim + 2>(
385 reference_function, fe_point_eval.quadrature_point(q)) -
386 fe_point_eval.get_value(q);
387
388 const auto JxW = fe_point_eval.JxW(q);
389 local_errors_squared[0] = error[0] * error[0] * JxW;
390
391 local_errors_squared[1] = 0.;
392 for (unsigned int d = 0; d < dim; ++d)
393 local_errors_squared[1] += (error[d + 1] * error[d + 1]) * JxW;
394 local_errors_squared[2] = (error[dim + 1] * error[dim + 1]) * JxW;
395
396 for (unsigned int v = 0; v < fe_point_eval.n_active_entries_per_quadrature_batch(q);
397 ++v)
398 for (unsigned int d = 0; d < 3; ++d)
399 errors_squared[d] += local_errors_squared[d][v];
400 }
401 }
402
403 dealii::Utilities::MPI::sum(errors_squared, MPI_COMM_WORLD, errors_squared);
404 std::array<number, 3> errors;
405 for (unsigned int d = 0; d < 3; ++d)
406 errors[d] = std::sqrt(errors_squared[d]);
407 labels = {"density", "momentum", "total energy"};
408 component_errors["density"] = errors[0];
409 component_errors["momentum"] = errors[1];
410 component_errors["total energy"] = errors[2];
411
412 pretty_print_norm("L2 " + norm_name, labels, component_errors);
413 }
