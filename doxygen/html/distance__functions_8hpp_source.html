41 Plane(const dealii::Point<dim, number> &center_of_wall,
42 const dealii::Tensor<1, dim, number> &normal,
43 const std::array<number, dim - 1> &half_extents,
44 const dealii::Tensor<1, dim, number> &first_tangent_direction =

47 , normal(normal)

49 {
50 AssertThrow(std::abs(normal.norm() - 1.) < tolerance,
51 dealii::ExcMessage("The normal of the wall plane must be a unit vector."));
52
53 // Building tangent basis
54 if constexpr (dim == 2)
55 {
56 tangents[0][0] = -normal[1];
57 tangents[0][1] = normal[0];
58 }
59 else if constexpr (dim == 3)
60 {
61 const number first_tangent_norm = first_tangent_direction.norm();
62 AssertThrow(first_tangent_norm > tolerance,
63 dealii::ExcMessage("The first tangent direction must be nonzero in 3D."));
64
65 tangents[0] = first_tangent_direction / first_tangent_norm;
66 tangents[1] = dealii::cross_product_3d(normal, tangents[0]);
67 AssertThrow(std::abs(normal * tangents[0]) < tolerance,
68 dealii::ExcMessage(
69 "The first tangent direction must be perpendicular to the normal."));
70 }

86 // At the wall surface, r vanishes and r/r_norm is undefined. Fall back to the wall's unit

169 TruncatedCone(const dealii::Point<dim, number> &axis_center,
170 const dealii::Tensor<1, dim, number> &axis_direction,
171 const number radius_bottom,
172 const number radius_top,
173 const number half_length)
174 : axis_center(axis_center)
175 , radius_bottom(radius_bottom)
176 , radius_top(radius_top)
177 , half_length(half_length)
178 {
179 AssertThrow(axis_direction.norm() > tolerance,
180 dealii::ExcMessage("The axis direction of the truncated cone must not be zero."));
181 AssertThrow(half_length > 0.,
182 dealii::ExcMessage("The half length of the truncated cone must be positive."));
183 AssertThrow(radius_bottom >= 0. && radius_top >= 0.,
184 dealii::ExcMessage("The radii of the truncated cone must not be negative."));
185 AssertThrow(radius_bottom > 0. || radius_top > 0.,
186 dealii::ExcMessage(
187 "At least one radius of the truncated cone must be greater than zero."));
188
189 axis = axis_direction / axis_direction.norm(); // unit vector
190 }

199 dealii::Tensor<1, dim, number>
200 gradient(const dealii::Point<dim, number> &x, const unsigned int component = 0) const override
201 {
202 Assert(component == 0,
203 dealii::ExcMessage("The distance function is scalar; component must be 0."));
204
205 const dealii::Tensor<1, dim, number> d =
206 x - axis_center; // Vector from the axis center to the point
207 const dealii::Tensor<1, dim, number> d_radial =
208 d - (d * axis) * axis; // Vector from the axis (perpendicular) to the point.
209 const number radial_coord = d_radial.norm();
210
211 dealii::Tensor<1, dim, number> radial_direction;
212 if (radial_coord >= tolerance)
213 radial_direction = d_radial / radial_coord;
214 else
215 {
216 // The Cartesian unit vector least aligned with the axis, made perpendicular to it is
217 // used, so the gradient stays well-defined.
218 unsigned int least_aligned = 0;
219 for (unsigned int i = 1; i < dim; ++i)
220 if (std::abs(axis[i]) < std::abs(axis[least_aligned]))
221 least_aligned = i;
222
223 dealii::Tensor<1, dim, number> unit_vector;
224 unit_vector[least_aligned] = 1.;
225 radial_direction = unit_vector - (unit_vector * axis) * axis;
226 radial_direction /= radial_direction.norm();
227 }
228
229 const auto [axial_offset, radial_offset] = meridian_offset_from_wall(x);
230 const number distance = std::hypot(axial_offset, radial_offset);
231
232 // On the surface, fall back to the inward surface normal.
233 if (distance < tolerance)
234 {
235 const number axial_span = 2. * half_length;
236 const number radial_span = radius_top - radius_bottom;
237 const number slant_length = std::hypot(axial_span, radial_span);
238
239 return (radial_span * axis - axial_span * radial_direction) / slant_length;
240 }
241
242 return (radial_offset * radial_direction + axial_offset * axis) / distance;
243 }

258 }

272 std::array<number, 2>
273 meridian_offset_from_wall(const dealii::Point<dim, number> &x) const
274 {
275 const dealii::Tensor<1, dim, number> d =
276 x - axis_center; // Vector from the axis center to the point
277 const number axial_coord = d * axis; // Coordinate of the point along the axis.
278 const number radial_coord =
279 (d - axial_coord * axis).norm(); // Coordinate of the point from the axis.
280
281 const number axial_span = 2. * half_length;
282 const number radial_span = radius_top - radius_bottom;
283
284 // Project (axial_coord, radial_coord) onto the meridian segment, clamped to its end points.
285 const number segment_fraction =
286 std::clamp(((axial_coord + half_length) * axial_span +
287 (radial_coord - radius_bottom) * radial_span) /
288 (axial_span * axial_span + radial_span * radial_span),
289 number(0.),
290 number(1.));
291
292 const number axial_coord_closest = -half_length + segment_fraction * axial_span;
293 const number radial_coord_closest = radius_bottom + segment_fraction * radial_span;
294
295 return {{axial_coord - axial_coord_closest, radial_coord - radial_coord_closest}};
296 }
