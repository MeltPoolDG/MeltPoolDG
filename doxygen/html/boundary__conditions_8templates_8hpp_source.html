17 if (const BoundaryType boundary_type = get_boundary_type(boundary_id);
18 boundary_type == BoundaryType::slip_wall)
19 {
20 // homogeneous Neumann
21 w_p.density() = w_m.density();
22 w_p.grad_density() = -(w_m.grad_density());
23
24 // homogeneous Neumann
25 VectorizedArrayType rho_u_dot_n = 0;
26 for (unsigned int d = 0; d < dim; ++d)
27 rho_u_dot_n += w_m.momentum(d) * normal[d];
28 // symmetry
29 for (unsigned int d = 0; d < dim; ++d)
30 {
31 w_p.momentum(d) = w_m.momentum(d) - 2. * rho_u_dot_n * normal[d];
32 w_p.grad_momentum(d) =
33 w_m.grad_momentum(d) - 2. * scalar_product(w_m.grad_momentum(d), normal) * normal;
34 }
35 // homogeneous Neumann
36 w_p.grad_total_energy() = -(w_m.grad_total_energy());
37 w_p.total_energy() = w_m.total_energy();
38 }
39 else if (boundary_type == BoundaryType::no_slip_wall)
40 {
41 // homogeneous Neumann
42 w_p.density() = w_m.density();
43 w_p.grad_density() = -(w_m.grad_density());
44
45 // Dirichlet
46 for (unsigned int d = 0; d < dim; ++d)
47 {
48 w_p.momentum(d) = 0.;
49 w_p.grad_momentum(d) = w_m.grad_momentum(d);
50 }
51 // homogeneous Neumann
52 w_p.grad_total_energy() = -(w_m.grad_total_energy());
53 w_p.total_energy() = w_m.total_energy();
54 }
55 else if (boundary_type == BoundaryType::inflow)
56 {
57 w_p.density() = get_boundary_value(boundary_id, BoundaryType::inflow, q_point, 0);
58 w_p.grad_density() = w_m.grad_density();
59
60 for (unsigned int d = 0; d < dim; ++d)
61 {
62 w_p.momentum(d) = get_boundary_value(boundary_id, BoundaryType::inflow, q_point, d + 1);
63 w_p.grad_momentum(d) = w_m.grad_momentum(d);
64 }
65
66 w_p.total_energy() =
67 get_boundary_value(boundary_id, BoundaryType::inflow, q_point, dim + 1);
68 w_p.grad_total_energy() = w_m.grad_total_energy();
69 }
70 else if (boundary_type == BoundaryType::subsonic_outflow_fixed_pressure)
71 {
72 // homogeneous Neumann
73 w_p.density() = w_m.density();
74 w_p.grad_density() = -(w_m.grad_density());
75
76 // Dirichlet
77 auto p_dyn = dealii::VectorizedArray<number>(0.);
78 for (unsigned int i = 0; i < dim; ++i)
79 p_dyn += w_m.momentum(i) * w_m.momentum(i);
80
81 p_dyn /= (w_m.density() * 2.);
82 const dealii::VectorizedArray<number> pressure = get_boundary_value(
83 boundary_id, BoundaryType::subsonic_outflow_fixed_pressure, q_point, 0);
84
85 // consider equation of state for computation of inner energy from given pressure
86 const dealii::VectorizedArray<number> inner_energy =
87 w_m.inner_energy_from_pressure(pressure);
88
89 w_p.total_energy() = inner_energy + p_dyn;
90 w_p.grad_total_energy() = w_m.grad_total_energy();
91 }
92 else if (boundary_type == BoundaryType::subsonic_outflow_fixed_energy)
93 {
94 // homogeneous Neumann

101 }
102
103 // Dirichlet
104 w_p.total_energy() =

106 w_p.grad_total_energy() = w_m.grad_total_energy();
107 }

129 if (const BoundaryType boundary_type = get_boundary_type(boundary_id);
130 boundary_type == BoundaryType::inflow)
131 {
132 // For inflow boundary conditions we assume that the partial densities are given and we set
133 // them as Dirichlet values.
134
135 // Remember for compressible flow we only need to prescribe n_species - 1 partial densities.
136 std::array<dealii::VectorizedArray<number>, n_species - 1> dirichlet_boundary_values;
137
138 for (unsigned int species = 0; species < n_species - 1; ++species)
139 dirichlet_boundary_values[species] =
140 get_boundary_value(boundary_id,
141 BoundaryType::inflow,
142 q_point,
143 CompressibleFlow::n_conserved_variables<dim, 1> + species);
144
145
146 SpeciesTransport::set_boundary_value_and_gradient(
147 w_p, w_m, SpeciesTransport::BoundaryConditionType::dirichlet, dirichlet_boundary_values);
148 }
149 else
150 {
151 // For all other boundary conditions we assume homogeneous Neumann for the partial
152 // densities.
