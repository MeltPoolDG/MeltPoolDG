40 inline std::pair<number, number>
41 calculate_viscous_average_weights(
42 const CompressibleFlow::MaterialPhaseData<number> &material_liquid,
43 const CompressibleFlow::MaterialPhaseData<number> &material_gas)
44 {
45 const auto &l = material_liquid;
46 const auto &g = material_gas;
47
48 const number q_liquid =
49 2. * l.dynamic_viscosity + l.thermal_conductivity / (l.specific_isobaric_heat / l.gamma);
50 const number q_gas =
51 2. * g.dynamic_viscosity + g.thermal_conductivity / (g.specific_isobaric_heat / g.gamma);
52
53 const number weight_liquid = q_liquid / (q_liquid + q_gas);
54
55 return {weight_liquid, 1. - weight_liquid};

83 typename number,
84 typename DofStateView,
85 typename ConvectiveKernel,
86 typename ConservedVariablesType = typename DofStateView::state_type,
87 typename ConservedVariablesGradType = typename ConvectiveKernel::PhysicalFluxType>

91 inline DEAL_II_ALWAYS_INLINE //
92 std::
93 tuple<ConservedVariablesGradType, ConservedVariablesGradType, dealii::VectorizedArray<number>>
94 calculate_convective_interface_flux_HLLP0(
95 const DofStateView &liquid_state,
96 const DofStateView &gas_state,
97 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
98 const ConvectiveKernel &convective_kernel_liquid,
99 const ConvectiveKernel &convective_kernel_gas,
100 const number &m_dot_evap)
101 {
102 // Note: Variables, that are relevant for both the liquid and the gas phase, are considered as
103 // arrays of length 2 in the following. The first element refers to the liquid phase and the
104 // second element to the gas phase.
105 constexpr int liquid = 0;
106 constexpr int gas = 1;
107
108 using Idx = CompressibleFlow::ConservedVariableIndex<dim>;
109
110 // 0) preliminaries
111
112 std::array<DofStateView, 2> state = {{liquid_state, gas_state}};
113 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel;
114 std::array<dealii::VectorizedArray<number>, 2> pressure;
115 std::array<dealii::VectorizedArray<number>, 2> rho;
116 std::array<dealii::VectorizedArray<number>, 2> rho_E;
117 std::array<dealii::VectorizedArray<number>, 2> speed_of_sound;
118 for (unsigned int i : {0, 1})
119 {
120 vel[i] = state[i].velocity();
121 rho[i] = state[i].density();
122 rho_E[i] = state[i].total_energy();
123 pressure[i] = state[i].pressure();
124 speed_of_sound[i] = state[i].speed_of_sound();
125 }
126
127 // 1) project velocity and kinetic energy into normal direction of the interface
128
129 std::array<dealii::VectorizedArray<number>, 2> vel_n;
130 std::array<dealii::VectorizedArray<number>, 2> E_n;
131 for (unsigned int i : {0, 1})
132 {
133 vel_n[i] = vel[i] * normal;
134 E_n[i] = rho_E[i] / rho[i] - 0.5 * (vel[i] * vel[i] - vel_n[i] * vel_n[i]);
135 }
136
137 // 2) shock speed estimation
138
139 std::array<dealii::VectorizedArray<number>, 2> shock_speed;
140 shock_speed[liquid] = vel_n[liquid] - speed_of_sound[liquid];
141 shock_speed[gas] = vel_n[gas] + speed_of_sound[gas];
142
143 // 3) calculate helpers using Rankine-Hugoniot conditions
144
145 std::array<dealii::VectorizedArray<number>, 2> m_hat;
146 std::array<dealii::VectorizedArray<number>, 2> I_hat;
147 std::array<dealii::VectorizedArray<number>, 2> E_hat;
148 for (unsigned int i : {0, 1})
149 {
150 m_hat[i] = rho[i] * (vel_n[i] - shock_speed[i]);
151 I_hat[i] = m_hat[i] * vel_n[i] + pressure[i];
152 E_hat[i] = m_hat[i] * E_n[i] + pressure[i] * vel_n[i];
153 }
154
155 // 4) calculate intermediate velocity states
156
157 // TODO: consider surface tension for dim>1 here
158 const dealii::VectorizedArray<number> delta_p = 0.;
159
160 // TODO: consider Hertz-Knudsen theory for evaporation mass flux here
161 std::array<dealii::VectorizedArray<number>, 2> tmp_1;
162 std::array<dealii::VectorizedArray<number>, 2> tmp_2;
163 for (unsigned int i : {0, 1})
164 {
165 tmp_1[i] = m_dot_evap / m_hat[i];
166 tmp_2[i] = m_dot_evap - m_hat[i];
167 }
168
169 std::array<dealii::VectorizedArray<number>, 2> numerator;
170 std::array<dealii::VectorizedArray<number>, 2> denominator;
171
172 numerator[liquid] = tmp_2[gas] *
173 (tmp_1[liquid] * shock_speed[liquid] - tmp_1[gas] * shock_speed[gas]) /
174 (1. - tmp_1[gas]) -
175 I_hat[liquid] - delta_p + I_hat[gas];
176 numerator[gas] = tmp_2[liquid] *
177 (tmp_1[gas] * shock_speed[gas] - tmp_1[liquid] * shock_speed[liquid]) /
178 (1. - tmp_1[liquid]) -
179 I_hat[gas] + delta_p + I_hat[liquid];
180 denominator[liquid] = tmp_2[liquid] - (1. - tmp_1[liquid]) / (1. - tmp_1[gas]) * tmp_2[gas];
181 denominator[gas] = tmp_2[gas] - (1. - tmp_1[gas]) / (1. - tmp_1[liquid]) * tmp_2[liquid];
182
183 std::array<dealii::VectorizedArray<number>, 2> vel_n_star;
184 for (unsigned int i : {0, 1})
185 vel_n_star[i] = numerator[i] / denominator[i];
186
187 // 5) calculate intermediate pressure
188
189 std::array<dealii::VectorizedArray<number>, 2> pressure_star;
190 for (unsigned int i : {0, 1})
191 pressure_star[i] = I_hat[i] - m_hat[i] * vel_n_star[i];
192
193 // 6) re-project normal velocity to Cartesian coordinates
194
195 std::array<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>, 2> vel_star_cartesian;
196
197 std::vector<dealii::Tensor<1, dim, dealii::VectorizedArray<number>>> tangent;
198 tangent.resize(dim - 1);
199
200 // compute tangential vector for dim=2 and dim=3
201 if constexpr (dim == 2)
202 {
203 tangent[0][0] = normal[1];
204 tangent[0][1] = -normal[0];
205 }
206 else if constexpr (dim == 3)
207 {
208 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec;
209 temp_vec[0] = 1.;
210 // if normal vector is identical with unit vector choose different unit vector to
211 // compute the tangent
212 dealii::VectorizedArray<number> tolerance = 1.e-10;
213 dealii::VectorizedArray<number> norm_diff = (temp_vec - normal).norm();
214 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> temp_vec_y;
215 temp_vec_y[1] = 1.;
216 for (int i = 0; i < 3; ++i)
217 {
218 temp_vec[i] = compare_and_apply_mask<dealii::SIMDComparison::less_than>(norm_diff,
219 tolerance,
220 temp_vec_y[i],
221 temp_vec[i]);
222 }
223 tangent[0] = temp_vec - (temp_vec * normal) * normal;
224 tangent[1] = dealii::cross_product_3d(normal, tangent[0]);
225 }
226
227 for (unsigned int i : {0, 1})
228 {
229 vel_star_cartesian[i] = vel_n_star[i] * normal;
230 for (unsigned int j = 0; j < dim - 1; ++j)
231 vel_star_cartesian[i] += (vel[i] * tangent[j]) * tangent[j];
232 }
233
234 // 7) calculate conservative variable state vectors of inner states
235
236 std::array<ConservedVariablesType, 2> u_star;
237 for (unsigned int i : {0, 1})
238 {
239 u_star[i][Idx::density] = m_hat[i] / (vel_n_star[i] - shock_speed[i]);
240 for (unsigned int j = 1; j < dim + 1; j++)
241 u_star[i][j] = u_star[i][Idx::density] * vel_star_cartesian[i][j - 1];
242 u_star[i][Idx::energy] =
243 (E_hat[i] - pressure_star[i] * vel_n_star[i]) / (vel_n_star[i] - shock_speed[i]) -
244 0.5 * u_star[i][Idx::density] * vel_n_star[i] * vel_n_star[i] +
245 0.5 * u_star[i][Idx::density] * vel_star_cartesian[i] * vel_star_cartesian[i];
246 }
247
248 // 8) calculate phase interface velocity
249
250 dealii::VectorizedArray<number> numerator_normal_vel =
251 vel_n_star[liquid] * u_star[liquid][Idx::density] -
252 vel_n_star[gas] * u_star[gas][Idx::density];
253 dealii::VectorizedArray<number> denominator_normal_vel =
254 u_star[liquid][Idx::density] - u_star[gas][Idx::density];
255 // avoid division by zero
256 dealii::VectorizedArray<number> normal_velocity_interface =
257 compare_and_apply_mask<dealii::SIMDComparison::greater_than>(std::abs(denominator_normal_vel),
258 1.e-12,
259 numerator_normal_vel /
260 denominator_normal_vel,
261 vel_n_star[liquid]);
262
263 // 9) calculate fluxes for the two phases
264
265 std::array<ConservedVariablesGradType, 2> flux;
266 std::array<ConservedVariablesGradType, 2> conv_flux;
267 std::array<ConservedVariablesGradType, 2> shock_flux;
268
269 conv_flux[liquid] = convective_kernel_liquid.flux(state[liquid].value());
270 conv_flux[gas] = convective_kernel_gas.flux(state[gas].value());
271
272 for (unsigned int i : {0, 1})
273 {
274 shock_flux[i] = dyadic_product(shock_speed[i] * (u_star[i] - state[i].value()), normal);
275 flux[i] = conv_flux[i];
276 }
277
278 const auto zero_vec = dealii::make_vectorized_array(0.);
279 const auto one_vec = dealii::make_vectorized_array(1.);
280
281 flux[liquid] +=
282 shock_flux[liquid] * compare_and_apply_mask<dealii::SIMDComparison::greater_than>(
283 shock_speed[liquid], zero_vec, zero_vec, one_vec);
284 flux[gas] +=
285 shock_flux[gas] * compare_and_apply_mask<dealii::SIMDComparison::less_than_or_equal>(
286 shock_speed[gas], zero_vec, zero_vec, one_vec);
287
288 return {flux[liquid], flux[gas], normal_velocity_interface};
289 }

321 {

323 using DofPrimitiveValueView =
324 CompressibleFlow::DofPrimitiveValueView<dim, ConservedVariablesType>;
325 using DofPrimitiveStateView =
326 CompressibleFlow::DofPrimitiveStateView<dim, number, const ConservedVariablesType>;
327
328 ConservedVariablesType liquid_prim_data;
329 ConservedVariablesType gas_prim_data;

337
338 // TODO: consider surface tension here
339 const dealii::VectorizedArray<number> delta_p = 0.;

342 ConservedVariablesType J_Dir;
343

345
346 J_Dir[Idx::pressure] =
347 delta_p - m_dot_evap * (liquid_prim_view.velocity(0) - gas_prim_view.velocity(0));
348 J_Dir[Idx::velocity] =

350 J_Dir[Idx::temperature] = delta_T;
351
352 const ConservedVariablesType liquid_prim_tmp = liquid_prim_view.value();

363 DofPrimitiveStateView(liquid_prim_view.value(), material_liquid), liquid_cons_view_star);
364 CompressibleFlow::primitive_to_conservative<dim, DofValueView, DofPrimitiveStateView>(
365 DofPrimitiveStateView(gas_prim_view.value(), material_gas), gas_cons_view_star);
366
367 J_Dir = liquid_cons_view_star.value() - gas_cons_view_star.value();
368

398 template <int dim,
399 typename number,
400 typename DofStateView,
401 typename DiffusiveKernel,
402 typename ConservedVariablesType = typename DofStateView::state_type,
403 typename ConservedVariablesGradType = typename DiffusiveKernel::PhysicalFluxType>

405 ConservedVariablesType,
406 ConservedVariablesGradType>
407 inline DEAL_II_ALWAYS_INLINE //
408 std::pair<ConservedVariablesType, ConservedVariablesType>
409 calculate_viscous_interface_flux_SIPG(
410 const DofStateView &liquid_state,
411 const DofStateView &gas_state,
412 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
413 const ConservedVariablesType &J_Dir,
414 const number weight_liquid,
415 const number weight_gas,
416 const number penalty_parameter,
417 const DiffusiveKernel &diffusive_kernel_liquid,
418 const DiffusiveKernel &diffusive_kernel_gas,
419 const number latent_heat_of_vaporization,
420 const number m_dot_evap,
421 const number laser_heat_source)
422 {
423 using Idx = CompressibleFlow::ConservedVariableIndex<dim>;
424
425 // TODO: add contributions for surface tension, interface heat source (laser energy) and
426 // Marangoni forces
427
428 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_liquid =
429 liquid_state.velocity();
430 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_gas = gas_state.velocity();
431
432 const dealii::VectorizedArray<number> vel_n_liquid = vel_liquid * normal;
433 const dealii::VectorizedArray<number> vel_n_gas = vel_gas * normal;
434
435 const auto pressure_liquid = liquid_state.pressure();
436 const auto pressure_gas = gas_state.pressure();
437
438 // compute Robin-type viscous interface jump conditions
439
440 ConservedVariablesType J_Rob;
441
442 // TODO: Add entries for case dim>1
443 J_Rob[Idx::energy] = (pressure_liquid * vel_n_liquid - pressure_gas * vel_n_gas) +
444 m_dot_evap * (liquid_state.total_energy() / liquid_state.density() -
445 gas_state.total_energy() / gas_state.density()) +
446 laser_heat_source - m_dot_evap * latent_heat_of_vaporization;
447
448 const ConservedVariablesGradType viscous_flux_liquid =
449 diffusive_kernel_liquid.flux(liquid_state.value(), liquid_state.gradient_value());
450
451 const ConservedVariablesGradType viscous_flux_gas =
452 diffusive_kernel_gas.flux(gas_state.value(), gas_state.gradient_value());
453
454 ConservedVariablesGradType total_flux_liquid = dyadic_product(J_Rob, normal);
455 total_flux_liquid += viscous_flux_gas;
456 total_flux_liquid *= weight_liquid;
457 total_flux_liquid += weight_gas * viscous_flux_liquid;
458
459 ConservedVariablesGradType total_flux_gas =
460 dyadic_product(J_Rob, -normal); // opposite normal direction for the gas phase
461 total_flux_gas += viscous_flux_liquid;
462 total_flux_gas *= weight_gas;
463 total_flux_gas += weight_liquid * viscous_flux_gas;
464
465 // penalty term
466 ConservedVariablesGradType penalty_flux_liquid =
467 dyadic_product(liquid_state.value() - (gas_state.value() + J_Dir), normal);
468 penalty_flux_liquid *= penalty_parameter;
469
470 ConservedVariablesGradType penalty_flux_gas =
471 dyadic_product(gas_state.value() - (liquid_state.value() - J_Dir), -normal);
472 penalty_flux_gas *= penalty_parameter;
473
474 total_flux_liquid -= penalty_flux_liquid;
475 total_flux_gas -= penalty_flux_gas;
476
477 return {contract_tensor_with_vector(total_flux_liquid, normal),
478 contract_tensor_with_vector(total_flux_gas, normal)};
479 }

578 template <int dim,
579 typename number,
580 typename DofValueAndGradientStateView,
581 typename DiffusiveKernel,
582 typename ConservedVariablesType = typename DofValueAndGradientStateView::state_type,
583 typename ConservedVariablesGradType = typename DiffusiveKernel::PhysicalFluxType>
584 requires CompressibleFlow::IsFluxKernel<DiffusiveKernel,
585 ConservedVariablesType,
586 ConservedVariablesGradType>
587 inline DEAL_II_ALWAYS_INLINE //
588 std::pair<ConservedVariablesType, ConservedVariablesType>
589 calculate_viscous_interface_flux_penalty(
590 const DofValueAndGradientStateView &liquid_state,
591 const DofValueAndGradientStateView &gas_state,
592 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> &normal,
593 const number weight_liquid,
594 const number weight_gas,
595 const number slip_penalty_parameter,
596 const number temperature_penalty_parameter,
597 const DiffusiveKernel &diffusive_kernel_liquid,
598 const DiffusiveKernel &diffusive_kernel_gas,
599 const number latent_heat_of_vaporization,
600 const number m_dot_evap,
601 const number delta_T,
602 const number laser_heat_source)
603 {
604 using Idx = CompressibleFlow::ConservedVariableIndex<dim>;
605
606 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_liquid =
607 liquid_state.velocity();
608 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> vel_gas = gas_state.velocity();
609
610 const dealii::VectorizedArray<number> vel_n_liquid = vel_liquid * normal;
611 const dealii::VectorizedArray<number> vel_n_gas = vel_gas * normal;
612
613 const dealii::VectorizedArray<number> pressure_liquid = liquid_state.pressure();
614 const dealii::VectorizedArray<number> pressure_gas = gas_state.pressure();
615
616 // compute Robin-type viscous interface jump conditions
617 // TODO: add contributions for surface tension and Marangoni forces (for dim>1)
618
619 ConservedVariablesType J_Rob;
620
621 J_Rob[Idx::energy] = m_dot_evap * (liquid_state.total_energy() / liquid_state.density() -
622 gas_state.total_energy() / gas_state.density()) +
623 (pressure_liquid * vel_n_liquid - pressure_gas * vel_n_gas) +
624 laser_heat_source - m_dot_evap * latent_heat_of_vaporization;
625
626 const ConservedVariablesGradType viscous_flux_liquid =
627 diffusive_kernel_liquid.flux(liquid_state.value(), liquid_state.gradient_value());
628
629 const ConservedVariablesGradType viscous_flux_gas =
630 diffusive_kernel_gas.flux(gas_state.value(), gas_state.gradient_value());
631
632 ConservedVariablesType penalty_term;
633
634 // Weakly enforce tangential no-slip at the interface for dim>1 via a penalty approach.
635 if constexpr (dim > 1)
636 {
637 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> jump_vel =
638 vel_liquid - vel_gas;
639 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> jump_vel_t =
640 jump_vel - (jump_vel * normal) * normal;
641
642 for (unsigned int d = 0; d < dim; ++d)
643 {
644 penalty_term[Idx::momentum + d] = slip_penalty_parameter * jump_vel_t[d];
645 J_Rob[Idx::momentum + d] = m_dot_evap * jump_vel_t[d];
646 }
647
648 // energy contribution from the tangential no-slip penalty term
649 // (jump_vel_t is tangential, so only the tangential part of the averaged velocity
650 // contributes to the work term)
651 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> ave_vel =
652 0.5 * (vel_liquid + vel_gas);
653 penalty_term[Idx::energy] = slip_penalty_parameter * scalar_product(jump_vel_t, ave_vel);
654 }
655
656 penalty_term[Idx::energy] += temperature_penalty_parameter *
657 ((liquid_state.temperature() - gas_state.temperature()) - delta_T);
658
659 const ConservedVariablesType weighted_viscous_flux =
660 UtilityFunctions::calculate_arithmetic_phase_weighted_average(
661 weight_gas,
662 contract_tensor_with_vector(viscous_flux_liquid, normal),
663 weight_liquid,
664 contract_tensor_with_vector(viscous_flux_gas, normal));
665
666 const ConservedVariablesType total_viscous_flux_liquid =
667 J_Rob * weight_liquid + weighted_viscous_flux - penalty_term;
668
669 const ConservedVariablesType total_viscous_flux_gas =
670 J_Rob * weight_gas - weighted_viscous_flux + penalty_term;
671
672 return {total_viscous_flux_liquid, total_viscous_flux_gas};
673 }
