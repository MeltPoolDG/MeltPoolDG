48 template <int dim>
49 class InflowVelocity : public Function<dim>
50 {
51 public:
52 InflowVelocity()
53 : Function<dim>(dim)
54 {}
55
56 double
57 value(const Point<dim> &p, const unsigned int component) const override
58 {
59 if (std::abs(p[0] + width * 0.5) < 1e-12 && p[dim - 1] > delta_h)
60 return (component == 0) ?
61 std::min(inflow_velocity, inflow_velocity * p[dim - 1] / (1.05 * delta_h)) :
62 0.0;
63 else
64 return 0;
65 }

68 template <int dim>
69 class InitialConditionTemperature : public Function<dim>
70 {
71 public:
72 InitialConditionTemperature(const double T_initial_bottom,
73 const double T_initial_top,
74 const double y_min,
75 const double y_max)
76 : Function<dim>()
77 , T_initial_bottom(T_initial_bottom)
78 , T_initial_top(T_initial_top)
79 , y_min(y_min)
80 , grad_T((T_initial_top - T_initial_bottom) / (y_max - y_min))
81 {}
82
83 double
84 value(const Point<dim> &p, const unsigned int /*component*/) const override
85 {
86 if (T_initial_top == T_initial_bottom)
87 return T_initial_top;
88 else
89
90 return T_initial_bottom + grad_T * (p[dim - 1] - y_min);
91 }
92
93 const double T_initial_bottom;
94 const double T_initial_top;
95 const double y_min;
96 const double grad_T;
97 };

250 void
251 set_boundary_conditions() override
252 {
253 // refine global before setting IDs as we consider dimension-dependent
254 // splitting of the boundaries
255 this->triangulation->refine_global(this->parameters.base.global_refinements);
256
257 // Note: For 1D we consider the constraints along the z-axis, i.e. bottom_bc and top_bc.
258 const types::boundary_id substrate_bc = 1;
259 const types::boundary_id gas_top_bc = 2;
260 const types::boundary_id gas_inflow_bc = 3;
261 const types::boundary_id gas_outlet_bc = 4;
262
263 const auto double_eq = [](const Number a, const Number b) {
264 return std::abs(a - b) <= 1e-10;
265 };
266
267 // set boundary ids
268 for (const auto &cell : this->triangulation->cell_iterators())
269 //| IteratorFilters::LocallyOwnedCell())
270 for (const auto &face : cell->face_iterators())
271 if ((face->at_boundary()))
272 {
273 // set boundaries at faces where z=const
274 if (double_eq(face->center()[dim - 1], bottom_left[dim - 1]))
275 face->set_boundary_id(substrate_bc);
276 else if (double_eq(face->center()[dim - 1], top_right[dim - 1]))
277 face->set_boundary_id(gas_top_bc);
278 // set wall boundaries
279 else if (dim > 1)
280 {
281 // substrate: walls (all points with z<delta_h)
282 if (face->center()[dim - 1] <= delta_h)
283 face->set_boundary_id(substrate_bc);
284 // gas inflow: all points with x=0 and z>=delta_h
285 else if (double_eq(face->center()[0], bottom_left[0]) &&
286 (face->center()[dim - 1] > delta_h))
287 face->set_boundary_id(gas_inflow_bc);
288 else if (face->center()[dim - 1] > delta_h)
289 face->set_boundary_id(gas_outlet_bc);
290 else
291 AssertThrow(false, ExcNotImplemented());
292 }
293 }
294
295 // set field-specific BCs
296
298 this->attach_boundary_condition(substrate_bc, "no_slip", "navier_stokes_u");
299 this->attach_boundary_condition(
300 {substrate_bc, std::make_shared<Functions::ConstantFunction<dim>>(T_initial_bottom)},
301 "dirichlet",
302 "heat_transfer");
303 // this->attach_dirichlet_boundary_condition(substrate_bc,
304 // std::make_shared<Functions::ZeroFunction<dim>>(),
305 //"reinitialization");
306
308 this->attach_boundary_condition(gas_top_bc, "symmetry", "navier_stokes_u");
309 // note: isolated for temperature field
310
312 this->attach_boundary_condition({gas_inflow_bc, std::make_shared<InflowVelocity<dim>>()},
313 "dirichlet",
314
315 "navier_stokes_u");
316 this->attach_boundary_condition(
317 {gas_inflow_bc, std::make_shared<Functions::ConstantFunction<dim>>(T_initial_bottom)},
318 "dirichlet",
319 "heat_transfer");
320 this->attach_boundary_condition({gas_inflow_bc,
321 std::make_shared<Functions::ConstantFunction<dim>>(-1.0)},
322 "dirichlet",
323 "level_set");
324 // this->attach_dirichlet_boundary_condition(gas_inflow_bc,
325 // std::make_shared<Functions::ZeroFunction<dim>>(),
326 //"reinitialization");
327
329 this->attach_boundary_condition(
330 {gas_outlet_bc, std::make_shared<Functions::ConstantFunction<dim>>(outlet_pressure)},
331 "open",
332 "navier_stokes_u");
333 // this->attach_dirichlet_boundary_condition(gas_outlet_bc,
334 // std::make_shared<Functions::ZeroFunction<dim>>(),
335 //"reinitialization");
336 // note: isolated for temperature field
337
338 /*
339 * locally refined region described by max. 2 bounding boxes
340 *
341 *
342 * +-------------------------+
343 * | |
344 * | +--------+ |
345 * | |refine 1| |
346 * | +--------+ |
347 * | |
348 * | |
349 * | +--------+ |
350 * | |refine 2| |
351 * | +--------+ |
352 * | |
353 * | |
354 * +-------------------------+
355 *
356 */
357 if (n_local_refinement > 0)
358 {
359 if constexpr (dim == 2)
360 {
361 // 1. region
362 const auto refinement_region =
363 BoundingBox<dim>({local_refinement_1_bottom_left, local_refinement_1_top_right});
364
365 // 2. region
366 const auto refinement_region_2 =
367 BoundingBox<dim>({local_refinement_2_bottom_left, local_refinement_2_top_right});
368
369 for (unsigned int j = 0; j < n_local_refinement; ++j)
370 {
371 for (auto &cell : this->triangulation->active_cell_iterators() |
372 IteratorFilters::LocallyOwnedCell())
373 {
374 for (unsigned int i = 0; i < cell->n_vertices(); ++i)
375 if (refinement_region.point_inside(cell->vertex(i)) ||
376 refinement_region_2.point_inside(cell->vertex(i)))
377 {
378 cell->set_refine_flag();
379 break;
380 }
381 }
382 this->triangulation->execute_coarsening_and_refinement();
383 }
384 }
385 else
386 AssertThrow(false, ExcNotImplemented());
387 }

405 void
406 do_postprocessing(
407 [[maybe_unused]] const GenericDataOut<dim, Number> &generic_data_out) const final
408 {
409 if (this->parameters.output.paraview.enable == false)
410 return;
411
412 n_time_step += 1;
413
414 if ((n_time_step % this->parameters.output.write_frequency) &&
415 generic_data_out.get_time() != this->parameters.time_stepping.end_time)
416 return;
417
418 generic_data_out.get_vector("velocity").update_ghost_values();
419 generic_data_out.get_vector("density").update_ghost_values();
420 generic_data_out.get_vector("rho_cp").update_ghost_values();
421 generic_data_out.get_vector("temperature").update_ghost_values();
422
423 QGauss<dim> quad(generic_data_out.get_dof_handler("density").get_fe().tensor_degree() + 2);
424
425 FEValues<dim> dens_values(generic_data_out.get_mapping(),
426 generic_data_out.get_dof_handler("density").get_fe(),
427 quad,
428 update_values | update_JxW_values);
429
430 FEValues<dim> rho_cp_values(generic_data_out.get_mapping(),
431 generic_data_out.get_dof_handler("rho_cp").get_fe(),
432 quad,
433 update_values);
434
435 FEValues<dim> vel_values(generic_data_out.get_mapping(),
436 generic_data_out.get_dof_handler("velocity").get_fe(),
437 quad,
438 update_values);
439
440 FEValues<dim> temp_values(generic_data_out.get_mapping(),
441 generic_data_out.get_dof_handler("temperature").get_fe(),
442 quad,
443 update_values);
444
445
446 // temporary variables
447 std::vector<Number> density(dens_values.get_quadrature().size());
448 std::vector<Number> rho_cp(dens_values.get_quadrature().size());
449 std::vector<Number> temperature(dens_values.get_quadrature().size());
450 std::vector<Tensor<1, dim>> velocity(density.size(), Tensor<1, dim>());
451 const FEValuesExtractors::Vector velocities(0);
452
453 // output variables
454 Number mass = 0;
455 std::vector<Number> momentum(dim, 0);
456 Number thermal_energy = 0;
457 Number kinetic_energy = 0;
458
459 for (const auto &cell :
460 this->triangulation->active_cell_iterators() | IteratorFilters::LocallyOwnedCell())
461 {
462 {
463 TriaIterator<DoFCellAccessor<dim, dim, false>> dof_cell(
464 &generic_data_out.get_dof_handler("density").get_triangulation(),
465 cell->level(),
466 cell->index(),
467 &generic_data_out.get_dof_handler("density"));
468 dens_values.reinit(dof_cell);
469 dens_values.get_function_values(generic_data_out.get_vector("density"), density);
470 }
471
472 {
473 TriaIterator<DoFCellAccessor<dim, dim, false>> dof_cell(
474 &generic_data_out.get_dof_handler("density").get_triangulation(),
475 cell->level(),
476 cell->index(),
477 &generic_data_out.get_dof_handler("rho_cp"));
478 rho_cp_values.reinit(dof_cell);
479 rho_cp_values.get_function_values(generic_data_out.get_vector("rho_cp"), rho_cp);
480 temp_values.reinit(dof_cell);
481 temp_values.get_function_values(generic_data_out.get_vector("temperature"),
482 temperature);
483 }
484
485 {
486 TriaIterator<DoFCellAccessor<dim, dim, false>> dof_cell(
487 &generic_data_out.get_dof_handler("density").get_triangulation(),
488 cell->level(),
489 cell->index(),
490 &generic_data_out.get_dof_handler("velocity"));
491 vel_values.reinit(dof_cell);
492 vel_values[velocities].get_function_values(generic_data_out.get_vector("velocity"),
493 velocity);
494 }
495
496 for (const unsigned int q : dens_values.quadrature_point_indices())
497 {
498 mass += density[q] * dens_values.JxW(q);
499 thermal_energy += rho_cp[q] * temperature[q] * dens_values.JxW(q);
500 for (unsigned int d = 0; d < dim; ++d)
501 {
502 momentum[d] += density[q] * velocity[q][d] * dens_values.JxW(q);
503 kinetic_energy +=
504 density[q] * velocity[q][d] * velocity[q][d] * dens_values.JxW(q);
505 }
506 }
507 }
508
509 generic_data_out.get_vector("velocity").zero_out_ghost_values();
510 generic_data_out.get_vector("density").zero_out_ghost_values();
511 generic_data_out.get_vector("rho_cp").zero_out_ghost_values();
512 generic_data_out.get_vector("temperature").zero_out_ghost_values();
513
514 mass =
515 Utilities::MPI::sum(mass, generic_data_out.get_vector("density").get_mpi_communicator());
516
517 for (unsigned int d = 0; d < dim; ++d)
518 momentum[d] =
519 Utilities::MPI::sum(momentum[d],
520 generic_data_out.get_vector("density").get_mpi_communicator());
521 kinetic_energy =
522 Utilities::MPI::sum(kinetic_energy,
523 generic_data_out.get_vector("density").get_mpi_communicator());
524 thermal_energy =
525 Utilities::MPI::sum(thermal_energy,
526 generic_data_out.get_vector("density").get_mpi_communicator());
527
528 output_table.add_value("time", generic_data_out.get_time());
529 output_table.add_value("mass", mass);
530 output_table.add_value("kinetic_energy", kinetic_energy);
531 output_table.add_value("thermal_energy", thermal_energy);
532
533 for (unsigned int d = 0; d < dim; ++d)
534 output_table.add_value("momentum_" + std::to_string(d), momentum[d]);
535
536 // write values to file
537 if (Utilities::MPI::this_mpi_process(this->mpi_communicator) == 0)
538 {
539 namespace fs = std::filesystem;
540 std::ofstream output(
541 fs::path(this->parameters.output.directory) /
542 fs::path(this->parameters.output.paraview.filename + "_conservation_variables.txt"));
543
544 output_table.write_text(output);
545 }
546 }
