201 template <int dim, int n_components, typename number>
202 void
203 FESubcellEvaluation<dim, n_components, number>::project_dof_values_to_subcell_values(
204 const dealii::EvaluationFlags::EvaluationFlags evaluation_flags)
205 {
206 AssertDimension(subcell_values.size(), fe_cell_evaluator.n_q_points);
207
208 fe_cell_evaluator.evaluate(evaluation_flags);
209
210 for (unsigned int q : fe_cell_evaluator.quadrature_point_indices())
211 {
212 if (evaluation_flags & dealii::EvaluationFlags::values)
213 subcell_values[q] = fe_cell_evaluator.get_value(q);
214 if (evaluation_flags & dealii::EvaluationFlags::gradients)
215 subcell_gradients[q] = fe_cell_evaluator.get_gradient(q);
216 }
217
218 if (evaluation_flags & dealii::EvaluationFlags::values)
219 subcell_values_initialized = true;
220 if (evaluation_flags & dealii::EvaluationFlags::gradients)
221 subcell_gradients_initialized = true;
222 }

224 template <int dim, int n_components, typename number>
225 void
226 FESubcellEvaluation<dim, n_components, number>::project_subcell_values_to_dof_values()
227 {
228 AssertDimension(submitted_subcell_values.size(), fe_cell_evaluator.n_q_points);
229
230 dealii::MatrixFreeOperators::CellwiseInverseMassMatrix<dim, -1, n_components, number> inverse(
231 fe_cell_evaluator);
232
233 std::vector<dealii::VectorizedArray<number>> subcell_values_dof_vector(
234 n_components * fe_cell_evaluator.n_q_points);
235
236 for (unsigned int subcell : subcell_indices())
237 {
238 if constexpr (n_components == 1)
239 subcell_values_dof_vector[subcell] = submitted_subcell_values[subcell];
240 else
241 for (unsigned int c = 0; c < n_components; ++c)
242 subcell_values_dof_vector[c * fe_cell_evaluator.n_q_points + subcell] =
243 submitted_subcell_values[subcell][c];
244 }
245 inverse.transform_from_q_points_to_basis(n_components,
246 subcell_values_dof_vector.data(),
247 fe_cell_evaluator.begin_dof_values());
248 }
