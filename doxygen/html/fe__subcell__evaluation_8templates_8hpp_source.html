149 template <int dim, int n_components, typename number>
150 void
151 FESubcellEvaluation<dim, n_components, number>::integrate()
152 {
153 Assert(
154 subcell_values_submitted or subcell_fluxes_submitted,
155 dealii::ExcMessage(
156 "Subcell values or fluxes have not been submitted. Call submit_value() or submit_flux() before integrate()."));
157
158 if (subcell_values_submitted)
159 {
160 for (unsigned int subcell : subcell_indices())
161 integrated_values_buffer[subcell] = submitted_subcell_values[subcell];
162 }
163 else
164 {
165 std::ranges::fill(integrated_values_buffer, value_type());
166 }
167
168 if (subcell_fluxes_submitted)
169 {
170 for (unsigned int subcell_face : subcell_face_indices())
171 {
172 const unsigned int interior_subcell_index =
173 face_adjacent_subcell_index(subcell_face, true);
174 const unsigned int exterior_subcell_index =
175 face_adjacent_subcell_index(subcell_face, false);
176
177 integrated_values_buffer[interior_subcell_index] +=
178 intra_cell_face_fluxes[subcell_face] * subcell_face_size(subcell_face) /
179 subcell_size(interior_subcell_index);
180 integrated_values_buffer[exterior_subcell_index] -=
181 intra_cell_face_fluxes[subcell_face] * subcell_face_size(subcell_face) /
182 subcell_size(exterior_subcell_index);
183 }
184 }
185
186 project_subcell_values_to_dof_values(integrated_values_buffer);
187 }

312 template <int dim, int n_components, typename number>
313 void
314 FESubcellEvaluation<dim, n_components, number>::setup_internal_data_structures()
315 {
316 n_subcells_total = fe_cell_evaluator.n_q_points;
317 this->n_subcells_1d = fe_cell_evaluator.get_shape_info().data[0].n_q_points_1d;
318
319 n_intra_cell_faces =
320 dim * (n_subcells_1d - 1) * dealii::Utilities::fixed_power<dim - 1>(n_subcells_1d);
321
322 subcell_values.resize(n_subcells_total);
323 subcell_gradients.resize(n_subcells_total);
324 submitted_subcell_values.resize(n_subcells_total);
325 intra_cell_face_fluxes.resize(n_intra_cell_faces);
326 intra_cell_face_sizes.resize(n_intra_cell_faces);
327 subcell_values_to_project_buffer.resize(n_components * n_subcells_total);
328 integrated_values_buffer.resize(n_subcells_total);
329 }

331 template <int dim, int n_components, typename number>
332 void
333 FESubcellEvaluation<dim, n_components, number>::compute_intra_cell_face_sizes()
334 {
335 Assert(fe_cell_evaluator.get_cell_type() == dealii::internal::MatrixFreeFunctions::cartesian,
336 dealii::ExcNotImplemented());
337
338 // On a Cartesian cell, the extent of a subcell in direction d is h_d * w_{i_d}, with the cell
339 // extent h_d, the subcell index i_d in direction d and the 1D quadrature weight w_{i_d} on the
340 // unit interval. The size of a face is thus the volume of its interior subcell divided by the
341 // extent of that subcell in the face normal direction assuming the inverse Jacobian is diagonal
342 // with the entries 1/h_d.
343 const dealii::Quadrature<1> &quadrature_1d =
344 fe_cell_evaluator.get_shape_info().data[0].quadrature;
345
346 for (const unsigned int subcell_face : subcell_face_indices())
347 {
348 const unsigned int face_normal_direction =
349 compute_reference_coordinates_normal_direction(subcell_face);
350 const unsigned int interior_subcell = face_adjacent_subcell_index(subcell_face, true);
351 const unsigned int interior_subcell_index_1d =
352 (interior_subcell / dealii::Utilities::pow(n_subcells_1d, face_normal_direction)) %
353 n_subcells_1d;
354
355 intra_cell_face_sizes[subcell_face] =
356 fe_cell_evaluator.JxW(interior_subcell) *
357 fe_cell_evaluator.inverse_jacobian(
358 interior_subcell)[face_normal_direction][face_normal_direction] /
359 quadrature_1d.weight(interior_subcell_index_1d);
360 }
361 }

363 template <int dim, int n_components, typename number>
364 void
365 FESubcellEvaluation<dim, n_components, number>::project_dof_values_to_subcell_values(
366 const dealii::EvaluationFlags::EvaluationFlags evaluation_flags)
367 {
368 AssertDimension(subcell_values.size(), fe_cell_evaluator.n_q_points);
369
370 fe_cell_evaluator.evaluate(evaluation_flags);
371
372 for (unsigned int q : fe_cell_evaluator.quadrature_point_indices())
373 {
374 if (evaluation_flags & dealii::EvaluationFlags::values)
375 subcell_values[q] = fe_cell_evaluator.get_value(q);
376 if (evaluation_flags & dealii::EvaluationFlags::gradients)
377 subcell_gradients[q] = fe_cell_evaluator.get_gradient(q);
378 }
379
380 if (evaluation_flags & dealii::EvaluationFlags::values)
381 subcell_values_initialized = true;
382 if (evaluation_flags & dealii::EvaluationFlags::gradients)
383 subcell_gradients_initialized = true;
384 }

386 template <int dim, int n_components, typename number>
387 void
388 FESubcellEvaluation<dim, n_components, number>::project_subcell_values_to_dof_values(
389 const std::vector<value_type> &values_to_project)
390 {
391 AssertDimension(values_to_project.size(), fe_cell_evaluator.n_q_points);
392
393 dealii::MatrixFreeOperators::CellwiseInverseMassMatrix<dim, -1, n_components, number> inverse(
394 fe_cell_evaluator);
395
396 for (unsigned int subcell : subcell_indices())
397 {
398 if constexpr (n_components == 1)
399 subcell_values_to_project_buffer[subcell] = values_to_project[subcell];
400 else
401 for (unsigned int c = 0; c < n_components; ++c)
402 subcell_values_to_project_buffer[c * n_subcells_total + subcell] =
403 values_to_project[subcell][c];
404 }
405 inverse.transform_from_q_points_to_basis(n_components,
406 subcell_values_to_project_buffer.data(),
407 fe_cell_evaluator.begin_dof_values());
408 }

421 template <int dim, int n_components, typename number>
422 unsigned int
423 FESubcellEvaluation<dim, n_components, number>::face_adjacent_subcell_index(
424 const unsigned int subcell_face_index,
425 const bool interior_face) const
426 {
427 AssertIndexRange(subcell_face_index, n_intra_cell_faces);
428
429 const unsigned int n_faces_per_direction =
430 (n_subcells_1d - 1) * dealii::Utilities::fixed_power<dim - 1>(n_subcells_1d);
431
432 const unsigned int face_normal_direction =
433 compute_reference_coordinates_normal_direction(subcell_face_index);
434
435 unsigned int frame_local_index =
436 subcell_face_index - face_normal_direction * n_faces_per_direction;
437 if (face_normal_direction == dim - 1)
438 {
439 if (interior_face)
440 return frame_local_index;
441 else
442 return frame_local_index + dealii::Utilities::pow(n_subcells_1d, face_normal_direction);
443 }
444 else if (dim == 3 and face_normal_direction == 1)
445 {
446 const unsigned int layer = frame_local_index / (n_subcells_1d * (n_subcells_1d - 1));
447 if (interior_face)
448 return frame_local_index + layer * n_subcells_1d;
449 else
450 return frame_local_index + (layer + 1) * n_subcells_1d;
451 }
452 else
453 {
454 const unsigned int layer = frame_local_index / (n_subcells_1d - 1);
455 if (interior_face)
456 return frame_local_index + layer;
457 else
458 return frame_local_index + layer + 1;
459 }
