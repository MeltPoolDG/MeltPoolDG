187 void
188 MeltPoolDG::Postprocessor<dim, number>::output_complex_valued_vector_to_csv(
189 const int n_time_step,
190 const number time,
191 const ComplexVectorType &output_vector,
192 const std::string &filename,
193 const bool force_output) const
194 {
195 if (not(is_output_timestep(n_time_step, time) or force_output))
196 return;
197
198 static int file_count = 0;
199
200 // Only rank 0 writes the output file to avoid multiple processes writing to the same file in
201 // parallel
202 if (dealii::Utilities::MPI::this_mpi_process(mpi_communicator) == 0)
203 {
204 std::string full_filename =
205 output_data.directory + '/' + filename + "_" + std::to_string(file_count) + ".csv";
206
207 std::ofstream file(full_filename, std::ios::out | std::ios::trunc);
208 AssertThrow(file, dealii::ExcMessage("Can not open the file: " + full_filename));
209
210 file << "Time: " << time << '\n';
211 file << "N time step: " << n_time_step << '\n';
212 file << '\n';
213 file << "Real part;Imaginary part\n";
214
215 for (const auto &v : output_vector)
216 file << v.real() << ';' << v.imag() << '\n';
217 }
218
219 ++file_count;
220 }
