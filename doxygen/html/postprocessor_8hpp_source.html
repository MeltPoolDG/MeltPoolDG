107 process_points(const int n_time_step,

111 if (not is_output_timestep(n_time_step, time) or not output_data.point.enable)

113
114 const auto [names, interpretations] =
115 PointDataType::get_property_names_and_component_interpretation();
116 const std::size_t n_properties = names.size();
117
118 std::vector<dealii::Point<dim>> points(point_data.size());
119 std::vector<double> properties(point_data.size() * n_properties);
120 for (std::size_t i = 0; i < point_data.size(); ++i)
121 point_data[i].fill_output(points[i], properties.data() + i * n_properties);
122
123 PointDataOut<dim> point_data_out;
124 point_data_out.build_patches(points, properties, names, interpretations);
125 write_zero_dim_paraview_files(

127 time_at_last_output = time;

273
274 // Only rank 0 writes the output file to avoid multiple processes writing to the same file in

276 if (dealii::Utilities::MPI::this_mpi_process(mpi_communicator) == 0)
277 {
278 std::string full_filename =
279 output_data.directory + '/' + filename + "_" + std::to_string(file_count) + ".csv";
280
281 std::ofstream file(full_filename, std::ios::out | std::ios::trunc);
282 AssertThrow(file, dealii::ExcMessage("Can not open the file: " + full_filename));
283
284 file << "Time: " << time << '\n';
285 file << "N time step: " << n_time_step << '\n';
286 file << '\n';
287 file << "Real part;Imaginary part\n";
288
289 for (const auto &v : output_vector)
290 file << v.real() << ';' << v.imag() << '\n';
291 }

293 ++file_count;
