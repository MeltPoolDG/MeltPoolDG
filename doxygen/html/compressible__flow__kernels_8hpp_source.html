53 dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>>
54 viscous_stress_tensor(

56 &grad_velocity,

58 {
59 const dealii::VectorizedArray<number> div_u = 2. / 3. * trace(grad_velocity);

87 inline DEAL_II_ALWAYS_INLINE //
88 CompressibleFlow::FluxType<dim, number>
89 diffusive_flux(
90 const CompressibleFlow::
91 DofValueAndGradientStateView<dim, number, const value_type, const gradient_type>
92 &conserved_variables)
93 {
94 using FluxType = CompressibleFlow::FluxType<dim, number>;
95
96 dealii::Tensor<1, dim, dealii::VectorizedArray<number>> velocity =
97 conserved_variables.velocity();
98
99 const dealii::Tensor<1, dim, dealii::Tensor<1, dim, dealii::VectorizedArray<number>>>
100 viscous_stress = viscous_stress_tensor(
101 conserved_variables.grad_velocity(conserved_variables.density(), velocity),
102 dealii::VectorizedArray<number>(conserved_variables.dynamic_viscosity()));
103
104 const dealii::Tensor<1, dim, dealii::VectorizedArray<number>> neg_heat_flux =
105 conserved_variables.thermal_conductivity() *
106 conserved_variables.grad_temperature(conserved_variables.value());
107
108 FluxType flux;
109 CompressibleFlow::FluxView<dim, FluxType> flux_view(flux);
110 for (unsigned int d = 0; d < dim; ++d)
111 {
112 // density
113 flux_view.density_flux()[d] = 0.0;
114
115 // momentum
116 for (unsigned int e = 0; e < dim; ++e)
117 flux_view.momentum_flux(e)[d] = viscous_stress[e][d];
118
119 // energy
120 flux_view.energy_flux()[d] = neg_heat_flux[d];
121 for (unsigned int e = 0; e < dim; ++e)
122 flux_view.energy_flux()[d] += velocity[e] * viscous_stress[d][e];
123 }
124
125 return flux;
126 }
