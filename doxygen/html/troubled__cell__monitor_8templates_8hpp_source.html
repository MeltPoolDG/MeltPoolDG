90 template <int dim, int n_components, typename number>
91 void
92 TroubledCellMonitor<dim, n_components, number>::add_info(
93 const MatrixFreeContext<dim, number> &matrix_free_context)
94 {
95 if (verbosity == Profiling::ProfilingVerbosity::none or
96 verbosity == Profiling::ProfilingVerbosity::basic)
97 return;
98
99 const dealii::MatrixFree<dim, number> &matrix_free = matrix_free_context.mf;
100
101 // count the locally owned cells marked by each strategy, where only the active lanes of each
102 // cell batch are considered
103 unsigned int n_troubled_local = 0;
104 unsigned int n_pad_local = 0;
105 unsigned int n_nad_local = 0;
106 unsigned int n_pad_and_nad_local = 0;
107 for (unsigned int cell = 0; cell < matrix_free.n_cell_batches(); ++cell)
108 {
109 std::bitset<dealii::VectorizedArray<number>::size()> active_lanes;
110 for (unsigned int lane = 0; lane < matrix_free.n_active_entries_per_cell_batch(cell);
111 ++lane)
112 active_lanes.set(lane);
113
114 std::bitset<dealii::VectorizedArray<number>::size()> pad_lanes;
115 std::bitset<dealii::VectorizedArray<number>::size()> nad_lanes;
116 if (apply_pad)
117 pad_lanes =
118 ~troubled_cell_marker.physically_admissible_cell_vector()[cell] & active_lanes;
119 if (apply_nad)
120 nad_lanes =
121 ~troubled_cell_marker.numerically_admissible_cell_vector()[cell] & active_lanes;
122
123 n_troubled_local += static_cast<unsigned int>((pad_lanes | nad_lanes).count());
124 n_pad_local += static_cast<unsigned int>(pad_lanes.count());
125 n_nad_local += static_cast<unsigned int>(nad_lanes.count());
126 n_pad_and_nad_local += static_cast<unsigned int>((pad_lanes & nad_lanes).count());
127 }
128
129 const MPI_Comm mpi_communicator =
130 matrix_free.get_dof_handler(matrix_free_context.dof_idx).get_mpi_communicator();
131
132 // sum the counts over all MPI ranks in a single reduction
133 const std::array<unsigned int, 4> local_counts = {
134 {n_troubled_local, n_pad_local, n_nad_local, n_pad_and_nad_local}};
135 std::array<unsigned int, 4> global_counts;
136 dealii::Utilities::MPI::sum(
137 dealii::ArrayView<const unsigned int>(local_counts.data(), local_counts.size()),
138 mpi_communicator,
139 dealii::ArrayView<unsigned int>(global_counts.data(), global_counts.size()));
140
141 n_troubled.add_info(global_counts[0]);
142 if (apply_pad)
143 n_pad.add_info(global_counts[1]);
144 if (apply_nad)
145 n_nad.add_info(global_counts[2]);
146 if (apply_pad and apply_nad)
147 n_pad_and_nad.add_info(global_counts[3]);
148
149 if (verbosity == Profiling::ProfilingVerbosity::detailed)
150 {
151 const dealii::Utilities::MPI::MinMaxAvg min_max_avg =
152 dealii::Utilities::MPI::min_max_avg(static_cast<number>(n_troubled_local),
153 mpi_communicator);
154 // without any troubled cell, the load is defined to be perfectly balanced
155 n_troubled.add_imbalance(min_max_avg.avg > 0. ? min_max_avg.max / min_max_avg.avg : 1.);
156 }
157 }
