194 void
195 compute_jacobian_matrix_representation(
196 const OperatorType &implicit_operator,
197 std::variant<dealii::LinearAlgebra::distributed::Vector<number> *,
198 dealii::TrilinosWrappers::SparseMatrix *> dst,
199 const MatrixRepresentationType type,
200 const dealii::LinearAlgebra::distributed::Vector<number> &current_solution,
201 const dealii::MatrixFree<dim, number> &matrix_free,
202 const unsigned int dof_idx,
203 const unsigned int quad_idx)
204 {
205 FECellIntegrator<dim, n_components, number> phi(matrix_free, dof_idx, quad_idx);
206
207 // ## Cell ## //
208 std::function<void(FECellIntegrator<dim, n_components, number> &)> cell_operation =
209 [&](FECellIntegrator<dim, n_components, number> &delta_phi) -> void {
210 phi.reinit(delta_phi.get_cell_or_face_batch_id());
211 phi.gather_evaluate(current_solution,
212 dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
213 delta_phi.evaluate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
214
215 for (const unsigned int q_index : delta_phi.quadrature_point_indices())
216 {
217 implicit_operator.local_cell_jacobian_kernel(delta_phi, phi, q_index);
218 }
219
220 delta_phi.integrate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
221 };
222
223 // ## Face ##//
224 FEFaceIntegrator<dim, n_components, number> phi_m(matrix_free,
225 true /*is_interior_face*/,
226 dof_idx,
227 quad_idx);
228 FEFaceIntegrator<dim, n_components, number> phi_p(matrix_free,
229 false /*is_interior_face*/,
230 dof_idx,
231 quad_idx);
232 std::function<void(FEFaceIntegrator<dim, n_components, number> &,
233 FEFaceIntegrator<dim, n_components, number> &)>
234 face_operation = [&](FEFaceIntegrator<dim, n_components, number> &delta_phi_m,
235 FEFaceIntegrator<dim, n_components, number> &delta_phi_p) -> void {
236 phi_m.reinit(delta_phi_m.get_cell_or_face_batch_id());
237 phi_p.reinit(delta_phi_p.get_cell_or_face_batch_id());
238 phi_m.gather_evaluate(current_solution,
239 dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
240 phi_p.gather_evaluate(current_solution,
241 dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
242
243 delta_phi_p.evaluate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
244 delta_phi_m.evaluate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
245
246
247 for (const unsigned int q_index : delta_phi_m.quadrature_point_indices())
248 {
249 implicit_operator.local_face_jacobian_kernel(
250 delta_phi_m, delta_phi_p, phi_m, phi_p, q_index);
251 }
252 delta_phi_m.integrate(dealii::EvaluationFlags::values);
253 delta_phi_p.integrate(dealii::EvaluationFlags::values);
254 };
255
256 // ## Boundary face ##//
257 FEFaceIntegrator<dim, n_components, number> phi_boundary(matrix_free,
258 true /*is_interior_face*/,
259 dof_idx,
260 quad_idx);
261 std::function<void(FEFaceIntegrator<dim, n_components, number> &)> boundary_operation =
262 [&](FEFaceIntegrator<dim, n_components, number> &delta_phi_boundary) -> void {
263 phi_boundary.reinit(delta_phi_boundary.get_cell_or_face_batch_id());
264 phi_boundary.gather_evaluate(current_solution,
265 dealii::EvaluationFlags::values |
266 dealii::EvaluationFlags::gradients);

268 dealii::EvaluationFlags::gradients);
269
270
271 for (const unsigned int q_index : delta_phi_boundary.quadrature_point_indices())
272 {
273 implicit_operator.local_boundary_face_jacobian_kernel(delta_phi_boundary,
274 phi_boundary,
275 q_index);
276 }
277 delta_phi_boundary.integrate(dealii::EvaluationFlags::values);
278 };

283 dealii::MatrixFreeTools::compute_diagonal(
284 matrix_free,
285 *std::get<dealii::LinearAlgebra::distributed::Vector<number> *>(dst),
286 cell_operation,

306 DEAL_II_ASSERT_UNREACHABLE();

325 template <int dim, typename number, int n_components>
326 std::vector<std::pair<dealii::Tensor<1, n_components, number>,
327 dealii::Tensor<1, n_components, dealii::Tensor<1, dim, number>>>>
328 compute_cell_average_quantities(
329 const MatrixFreeContext<dim, number> &mf_context,
330 const dealii::LinearAlgebra::distributed::Vector<number> &solution)
331 {
332 using VectorizedArrayType = dealii::VectorizedArray<number>;
333 using ActiveCellIterator = typename dealii::Triangulation<dim>::active_cell_iterator;
334
335 using AveragePairType =
336 std::pair<dealii::Tensor<1, n_components, number>,
337 dealii::Tensor<1, n_components, dealii::Tensor<1, dim, number>>>;
338
339 // A lambda used to exchange the cell average values of ghost cells after they have been
340 // computed for the locally owned cells.
341 const auto exchange_cell_data_to_ghosts =
342 [](const dealii::Triangulation<dim> &triangulation,
343 std::vector<AveragePairType> &cell_average_values) {
344 std::function<std::optional<AveragePairType>(const ActiveCellIterator &)> pack =
345 [&](const ActiveCellIterator &cell) -> std::optional<AveragePairType> {
346 return cell_average_values[cell->active_cell_index()];
347 };
348 std::function<void(const ActiveCellIterator &, const AveragePairType &)> unpack =
349 [&](const ActiveCellIterator &cell, const AveragePairType &value) {
350 cell_average_values[cell->active_cell_index()] = value;
351 };
352 dealii::GridTools::exchange_cell_data_to_ghosts(triangulation, pack, unpack);
353 };
354
355 // A lambda that computes the cell average values for a given range of cells used in the cell
356 // loop of the matrix free object.
357 const std::function<void(const dealii::MatrixFree<dim, number, VectorizedArrayType> &,
358 std::vector<AveragePairType> &,
359 const dealii::LinearAlgebra::distributed::Vector<number> &,
360 const std::pair<unsigned int, unsigned int> &)>
361 cell_loop = [&](const dealii::MatrixFree<dim, number, VectorizedArrayType> &matrix_free,
362 std::vector<AveragePairType> &dst_cell_average_values,
363 const dealii::LinearAlgebra::distributed::Vector<number> &src,
364 const std::pair<unsigned int, unsigned int> &cell_range) {
365 FECellIntegrator<dim, n_components, number> fe_cell_integrator(matrix_free,
366 mf_context.dof_idx,
367 mf_context.quad_idx);
368
369 for (unsigned int cell = cell_range.first; cell < cell_range.second; ++cell)
370 {
371 fe_cell_integrator.reinit(cell);
372 fe_cell_integrator.gather_evaluate(src,
373 dealii::EvaluationFlags::values |
374 dealii::EvaluationFlags::gradients);
375
376 typename FECellIntegrator<dim, n_components, number>::value_type cell_average_value;
377 typename FECellIntegrator<dim, n_components, number>::gradient_type
378 cell_average_gradient;
379 VectorizedArrayType cell_volume = 0.;
380 for (const unsigned int q : fe_cell_integrator.quadrature_point_indices())
381 {
382 cell_average_value += fe_cell_integrator.get_value(q) * fe_cell_integrator.JxW(q);
383 cell_average_gradient +=
384 fe_cell_integrator.get_gradient(q) * fe_cell_integrator.JxW(q);
385 cell_volume += fe_cell_integrator.JxW(q);
386 }
387 cell_average_value /= cell_volume;
388 for (unsigned int c = 0; c < n_components; ++c)
389 cell_average_gradient[c] /= cell_volume;
390
391 const auto cells = cells_in_cell_batch(mf_context.mf, cell);
392
393 for (unsigned int lane = 0; lane < mf_context.mf.n_active_entries_per_cell_batch(cell);
394 ++lane)
395 {
396 for (unsigned int c = 0; c < n_components; ++c)
397 {
398 dst_cell_average_values[cells[lane]->active_cell_index()].first[c] =
399 cell_average_value[c][lane];
400 for (unsigned int d = 0; d < dim; ++d)
401 {
402 dst_cell_average_values[cells[lane]->active_cell_index()].second[c][d] =
403 cell_average_gradient[c][d][lane];
404 }
405 }
406 }
407 }
408 };
409
410 std::vector<AveragePairType> cell_average_values(
411 mf_context.mf.get_dof_handler(mf_context.dof_idx).get_triangulation().n_active_cells());
412 mf_context.mf.cell_loop(cell_loop, cell_average_values, solution);
413
414 exchange_cell_data_to_ghosts(
415 mf_context.mf.get_dof_handler(mf_context.dof_idx).get_triangulation(), cell_average_values);
416
417 return cell_average_values;
418 }
