259 const OperatorType &implicit_operator,
260 std::variant<dealii::LinearAlgebra::distributed::Vector<number> *,
261 dealii::TrilinosWrappers::SparseMatrix *> dst,
262 const MatrixRepresentationType type,
263 const dealii::LinearAlgebra::distributed::Vector<number> &current_solution,
264 const dealii::MatrixFree<dim, number> &matrix_free,
265 const unsigned int dof_idx,

267 {

270 // ## Cell ## //
271 std::function<void(FECellIntegrator<dim, n_components, number> &)> cell_operation =
272 [&](FECellIntegrator<dim, n_components, number> &delta_phi) -> void {
273 phi.reinit(delta_phi.get_cell_or_face_batch_id());
274 phi.gather_evaluate(current_solution,
275 dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
276 delta_phi.evaluate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
277
278 for (const unsigned int q_index : delta_phi.quadrature_point_indices())
279 {
280 implicit_operator.local_cell_jacobian_kernel(delta_phi, phi, q_index);
281 }
282
283 delta_phi.integrate(dealii::EvaluationFlags::values | dealii::EvaluationFlags::gradients);
284 };

287 FEFaceIntegrator<dim, n_components, number> phi_m(matrix_free,
288 true /*is_interior_face*/,
289 dof_idx,

314 }

388 template <int dim, typename number, int n_components>
389 std::vector<std::pair<dealii::Tensor<1, n_components, number>,
390 dealii::Tensor<1, n_components, dealii::Tensor<1, dim, number>>>>
391 compute_cell_average_quantities(
392 const MatrixFreeContext<dim, number> &mf_context,
393 const dealii::LinearAlgebra::distributed::Vector<number> &solution)
394 {
395 using VectorizedArrayType = dealii::VectorizedArray<number>;
396 using ActiveCellIterator = typename dealii::Triangulation<dim>::active_cell_iterator;
397
398 using AveragePairType =
399 std::pair<dealii::Tensor<1, n_components, number>,
400 dealii::Tensor<1, n_components, dealii::Tensor<1, dim, number>>>;
401
402 // A lambda used to exchange the cell average values of ghost cells after they have been
403 // computed for the locally owned cells.
404 const auto exchange_cell_data_to_ghosts =
405 [](const dealii::Triangulation<dim> &triangulation,
406 std::vector<AveragePairType> &cell_average_values) {
407 std::function<std::optional<AveragePairType>(const ActiveCellIterator &)> pack =
408 [&](const ActiveCellIterator &cell) -> std::optional<AveragePairType> {
409 return cell_average_values[cell->active_cell_index()];
410 };
411 std::function<void(const ActiveCellIterator &, const AveragePairType &)> unpack =
412 [&](const ActiveCellIterator &cell, const AveragePairType &value) {
413 cell_average_values[cell->active_cell_index()] = value;
414 };
415 dealii::GridTools::exchange_cell_data_to_ghosts(triangulation, pack, unpack);
416 };
417
418 // A lambda that computes the cell average values for a given range of cells used in the cell
419 // loop of the matrix free object.
420 const std::function<void(const dealii::MatrixFree<dim, number, VectorizedArrayType> &,
421 std::vector<AveragePairType> &,
422 const dealii::LinearAlgebra::distributed::Vector<number> &,
423 const std::pair<unsigned int, unsigned int> &)>
424 cell_loop = [&](const dealii::MatrixFree<dim, number, VectorizedArrayType> &matrix_free,
425 std::vector<AveragePairType> &dst_cell_average_values,
426 const dealii::LinearAlgebra::distributed::Vector<number> &src,
427 const std::pair<unsigned int, unsigned int> &cell_range) {
428 FECellIntegrator<dim, n_components, number> fe_cell_integrator(matrix_free,
429 mf_context.dof_idx,
430 mf_context.quad_idx);
431
432 for (unsigned int cell = cell_range.first; cell < cell_range.second; ++cell)
433 {
434 fe_cell_integrator.reinit(cell);
435 fe_cell_integrator.gather_evaluate(src,
436 dealii::EvaluationFlags::values |
437 dealii::EvaluationFlags::gradients);
438
439 typename FECellIntegrator<dim, n_components, number>::value_type cell_average_value;
440 typename FECellIntegrator<dim, n_components, number>::gradient_type
441 cell_average_gradient;
442 VectorizedArrayType cell_volume = 0.;
443 for (const unsigned int q : fe_cell_integrator.quadrature_point_indices())
444 {
445 cell_average_value += fe_cell_integrator.get_value(q) * fe_cell_integrator.JxW(q);
446 cell_average_gradient +=
447 fe_cell_integrator.get_gradient(q) * fe_cell_integrator.JxW(q);
448 cell_volume += fe_cell_integrator.JxW(q);
449 }
450 cell_average_value /= cell_volume;
451 for (unsigned int c = 0; c < n_components; ++c)
452 cell_average_gradient[c] /= cell_volume;
453
454 const auto cells = cells_in_cell_batch(mf_context.mf, cell);
455
456 for (unsigned int lane = 0; lane < mf_context.mf.n_active_entries_per_cell_batch(cell);
457 ++lane)
458 {
459 for (unsigned int c = 0; c < n_components; ++c)
460 {
461 dst_cell_average_values[cells[lane]->active_cell_index()].first[c] =
462 cell_average_value[c][lane];
463 for (unsigned int d = 0; d < dim; ++d)
464 {
465 dst_cell_average_values[cells[lane]->active_cell_index()].second[c][d] =
466 cell_average_gradient[c][d][lane];
467 }
468 }
469 }
470 }
471 };
472
473 std::vector<AveragePairType> cell_average_values(
474 mf_context.mf.get_dof_handler(mf_context.dof_idx).get_triangulation().n_active_cells());
475 mf_context.mf.cell_loop(cell_loop, cell_average_values, solution);
476
477 exchange_cell_data_to_ghosts(
478 mf_context.mf.get_dof_handler(mf_context.dof_idx).get_triangulation(), cell_average_values);
479
480 return cell_average_values;
481 }

509 void
510 reinit(const dealii::MatrixFree<dim, number, VectorizedArrayType> &matrix_free)
511 {
512 matrix_free.initialize_cell_data_vector(cell_data);
513
514 // set up the internal parallel distributed vector, used to exchange the ghost values of the
515 // cell data, with one entry per active cell
516 const std::shared_ptr<const dealii::Utilities::MPI::Partitioner> partitioner =
517 matrix_free.get_dof_handler()
518 .get_triangulation()
519 .global_active_cell_index_partitioner()
520 .lock();
521 exchange_vector.reinit(partitioner);
522
523 // create the mapping between the entries of the parallel distributed vector and the cell
524 // batches and lanes of the cell data vector
525 owned_slots.clear();
526 ghost_slots.clear();
527
528 // We only need to consider those locally owned cells which are actually ghost cells on
529 // another MPI process, since only their values are sent.
530 std::vector<bool> is_exported(partitioner->locally_owned_size(), false);
531 for (const auto &[begin, end] : partitioner->import_indices())
532 {
533 std::fill(is_exported.begin() + begin, is_exported.begin() + end, true);
534 }
535
536 for (unsigned int batch = 0;
537 batch < matrix_free.n_cell_batches() + matrix_free.n_ghost_cell_batches();
538 ++batch)
539 for (unsigned int lane = 0; lane < matrix_free.n_active_entries_per_cell_batch(batch);
540 ++lane)
541 {
542 const unsigned int local_index = partitioner->global_to_local(
543 matrix_free.get_cell_iterator(batch, lane)->global_active_cell_index());
544 if (batch < matrix_free.n_cell_batches())
545 {
546 if (is_exported[local_index])
547 {
548 owned_slots.push_back({batch, lane, local_index});
549 }
550 }
551 else
552 {
553 ghost_slots.push_back({batch, lane, local_index});
554 }
555 }
556 }
557
565 void
566 update_ghost_values()
567 {
568 for (const auto &[batch, lane, local_index] : owned_slots)
569 exchange_vector.local_element(local_index) = cell_data[batch][lane];
570 exchange_vector.update_ghost_values();
571 for (const auto &[batch, lane, local_index] : ghost_slots)
572 cell_data[batch][lane] = exchange_vector.local_element(local_index);
573 exchange_vector.zero_out_ghost_values();
574 }

655 template <typename number>
656 void
657 reinit(const dealii::MatrixFree<dim, number> &matrix_free)
658 {
659 constexpr unsigned int n_lanes = dealii::VectorizedArray<number>::size();
660 neighbors.reinit(matrix_free.n_cell_batches() * n_lanes,
661 dealii::GeometryInfo<dim>::faces_per_cell);
662
663 for (unsigned int batch = 0; batch < matrix_free.n_cell_batches(); ++batch)
664 for (unsigned int lane = 0; lane < matrix_free.n_active_entries_per_cell_batch(batch);
665 ++lane)
666 {
667 const dealii::TriaIterator<dealii::CellAccessor<dim>> cell =
668 matrix_free.get_cell_iterator(batch, lane);
669 for (const unsigned int f : cell->face_indices())
670 {
671 const bool periodic = cell->has_periodic_neighbor(f);
672 if (cell->at_boundary(f) and not periodic)
673 continue;
674
675 const dealii::TriaIterator<dealii::CellAccessor<dim>> neighbor =
676 cell->neighbor_or_periodic_neighbor(f);
677 Assert(neighbor->is_active() and neighbor->level() == cell->level(),
678 dealii::ExcNotImplemented("Hanging faces are not supported yet."));
679
680 const unsigned int neighbor_mf_index =
681 matrix_free.get_matrix_free_cell_index(neighbor);
682 Assert(
683 neighbor_mf_index != dealii::numbers::invalid_unsigned_int,
684 dealii::ExcMessage(
685 "The face neighbor of a locally owned cell is not stored in the MatrixFree object. "
686 "Set MatrixFree::AdditionalData::hold_all_faces_to_owned_cells = true and enable "
687 "face integrals."));
688 neighbors(batch * n_lanes + lane, f) = {neighbor_mf_index,
689 periodic ?
690 cell->periodic_neighbor_face_no(f) :
691 cell->neighbor_face_no(f),
692 neighbor_mf_index / n_lanes,
693 neighbor_mf_index % n_lanes};
694 }
695 }
696 }
697
706 const Neighbor &
707 operator()(const unsigned int cell_index, const unsigned int face_no) const
708 {
709 return neighbors(cell_index, face_no);
710 }

720 neighbors_of_cell(const unsigned int cell_index) const
721 {
722 return {&neighbors(cell_index, 0), dealii::GeometryInfo<dim>::faces_per_cell};
723 }
