158 {
159 auto integrator = new LowStorageExplicitRungeKuttaIntegrator<number>(

161 [&pde_operator](number time,
162 number,
163 dealii::LinearAlgebra::distributed::Vector<number> &dst,
164 const dealii::LinearAlgebra::distributed::Vector<number> &src,
165 std::function<void(unsigned, unsigned)> post) {
166 pde_operator.apply_operator(time, dst, src, post);
167 });
168 return integrator;
169 }
170 else if (Utils::contains(explicit_rkc_supported_schemes, params.integrator_type))
171 {
172 auto integrator = new ExplicitRungeKuttaChebyshevIntegrator<number>(
173 params,
174 [&pde_operator](number time,
175 number,
176 dealii::LinearAlgebra::distributed::Vector<number> &dst,
177 const dealii::LinearAlgebra::distributed::Vector<number> &src,
178 std::function<void(unsigned, unsigned)> post) {
179 pde_operator.apply_operator(time, dst, src, post);
180 });
181 return integrator;
182 }
183 if (Utils::contains(one_step_theta_supported_schemes, params.integrator_type))
184 return new OneStepTheta<number, PDEOperator>(pde_operator, params, linear_solver_data);
