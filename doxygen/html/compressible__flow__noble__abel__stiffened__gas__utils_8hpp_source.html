146 (material_data.gamma - 1.) / material_data.gamma +
147 material_data.eos_data.b * (u_prim[0] + material_data.eos_data.p_inf));
148 // momentum
149 for (unsigned int i = 1; i < dim + 1; i++)

156 material_data.eos_data.p_inf * (1. / u_cons[0] - material_data.eos_data.b) +
157 material_data.eos_data.q + 0.5 * scalar_product(velocity, velocity));
