114 template <int dim>
115 inline double
116 compute_cell_diameter(const double cell_measure, const unsigned int fe_degree)
117 {
118 if constexpr (dim == 2)
119 return std::sqrt(4. * cell_measure / dealii::numbers::PI) / fe_degree;
120 else if constexpr (dim == 3)
121 return std::cbrt(6. * cell_measure / dealii::numbers::PI) / fe_degree;
122 else
123 AssertThrow(false, ExcMessage("Only dim=2 and dim=3 are supported."));
124 return 0.0;
125 }

200 template <int dim>
201 double
202 find_point_triangle_distance(const std::vector<Point<dim>> &triangle, const Point<dim> &point)
203 {
204 double distance = 0.0;
205
206 const Point<dim> &point_0 = triangle[0];
207 const Point<dim> &point_1 = triangle[1];
208
209 if constexpr (dim == 3)
210 {
211 const Point<dim> &point_2 = triangle[2];
212
213 const Tensor<1, dim> e_0 = point_1 - point_0;
214 const Tensor<1, dim> e_1 = point_2 - point_0;
215
216 const double a = e_0.norm_square();
217 const double b = scalar_product(e_0, e_1);
218 const double c = e_1.norm_square();
219 const double det = a * c - b * b;
220
221 const Tensor<1, dim> vector_to_plane = point_0 - point;
222
223 const double d = scalar_product(e_0, vector_to_plane);
224 const double e = scalar_product(e_1, vector_to_plane);
225
226 double s = b * e - c * d;
227 double t = b * d - a * e;
228
229 if (s + t <= det)
230 {
231 if (s < 0)
232 {
233 if (t < 0)
234 {
235 if (d < 0)
236 {
237 t = 0;
238 if (-d >= a)
239 s = 1;
240 else
241 s = -d / a;
242 }
243 else
244 {
245 s = 0;
246 if (e >= 0)
247 t = 0;
248 else if (-e >= c)
249 t = 1;
250 else
251 t = e / c;
252 }
253 }
254 else
255 {
256 s = 0;
257 if (e >= 0)
258 t = 0;
259 else if (-e >= c)
260 t = 1;
261 else
262 t = -e / c;
263 }
264 }
265 else if (t < 0)
266 {
267 t = 0;
268 if (d >= 0)
269 s = 0;
270 else if (-d >= a)
271 s = 1;
272 else
273 s = -d / a;
274 }
275 else
276 {
277 const double inv_det = 1. / det;
278 s *= inv_det;
279 t *= inv_det;
280 }
281 }
282 else
283 {
284 if (s < 0)
285 {
286 const double tmp0 = b + d;
287 const double tmp1 = c + e;
288 if (tmp1 > tmp0)
289 {
290 const double numer = tmp1 - tmp0;
291 const double denom = a - 2 * b + c;
292 if (numer >= denom)
293 s = 1;
294 else
295 s = numer / denom;
296 t = 1 - s;
297 }
298 else
299 {
300 s = 0;
301 if (tmp1 <= 0)
302 t = 1;
303 else if (e >= 0)
304 t = 0;
305 else
306 t = -e / c;
307 }
308 }
309 else if (t < 0)
310 {
311 const double tmp0 = b + e;
312 const double tmp1 = a + d;
313 if (tmp1 > tmp0)
314 {
315 const double numer = tmp1 - tmp0;
316 const double denom = a - 2 * b + c;
317 if (numer >= denom)
318 t = 1;
319 else
320 t = numer / denom;
321 s = 1 - t;
322 }
323 else
324 {
325 t = 0;
326 if (tmp1 <= 0)
327 s = 1;
328 else if (d >= 0)
329 s = 0;
330 else
331 s = -d / a;
332 }
333 }
334 else
335 {
336 const double numer = (c + e) - (b + d);
337 if (numer <= 0)
338 s = 0;
339 else
340 {
341 const double denom = a - 2 * b + c;
342 if (numer >= denom)
343 s = 1;
344 else
345 s = numer / denom;
346 }
347 t = 1 - s;
348 }
349 }
350
351 const Point<dim> pt_in_triangle = point_0 + s * e_0 + t * e_1;
352 distance = pt_in_triangle.distance(point);
353 }
354
355 if constexpr (dim == 2)
356 {
357 const Tensor<1, dim> d = point_1 - point_0;
358 const double t_bar = d * (point - point_0) / (d.norm() * d.norm() + 1e-12);
359
360 if (t_bar <= 0.0)
361 distance = (point - point_0).norm();
362 else if (t_bar >= 1.0)
363 distance = (point - point_1).norm();
364 else
365 distance = (point - (point_0 + t_bar * d)).norm();
366 }
367
368 return distance;
369 }

382 class CellWiseFunction : public Function<dim>

401 void

406
415 double
416 value(const Point<dim> &point, const unsigned int component = 0) const override
417 {
418 double value = 0.;
419 for (unsigned int i = 0; i < n_cell_wise_dofs; ++i)
420 value += cell_dof_values[i] * fe.shape_value_component(i, point, component);
421 return value;

436 {
437 Tensor<1, dim> gradient;
438 for (unsigned int i = 0; i < n_cell_wise_dofs; ++i)
439 gradient += cell_dof_values[i] * fe.shape_grad_component(i, point, component);
440 return gradient;
441 }

453 SymmetricTensor<2, dim>
454 hessian(const Point<dim> &point, const unsigned int component = 0) const override
455 {
456 Tensor<2, dim> hessian;
457 for (unsigned int i = 0; i < n_cell_wise_dofs; ++i)
458 hessian += cell_dof_values[i] * fe.shape_grad_grad_component(i, point, component);
459 return symmetrize(hessian);
460 }

462 private:
