107 void
108 move_level_set(auto &level_set,
109 const number &time_step,
110 const std::string &case_name,
111 const auto &scratch_data,
112 const int level_set_dof_idx)
113 {
114 if (case_name == "oscillating_water_column")
115 {
116 // two phase boundaries
117 level_set_oscillating_water_column.update_interface_positions(time_step,
118 interface_velocity);
119 level_set.zero_out_ghost_values();
120 dealii::VectorTools::interpolate(scratch_data->get_mapping(),
121 scratch_data->get_dof_handler(level_set_dof_idx),
122 level_set_oscillating_water_column,
123 level_set);
124 level_set.update_ghost_values();
125 }
126 else if (case_name == "two_phase")
127 {
128 // one phase boundary
129
130 // TODO: This is a current work-around, since the level-set advection is not yet fully
131 // implemented. The mean value of the interface velocity is used to advect the level-set
132 // field. In the future, a full level-set advection should be implemented, which uses the
133 // full velocity field to advect the level-set field.
134 const number local_sum =
135 std::accumulate(interface_velocity.begin(), interface_velocity.end(), 0.);
136
137 const unsigned int local_size = interface_velocity.size();
138
139 const number global_sum =
140 dealii::Utilities::MPI::sum(local_sum, scratch_data->get_mpi_comm(level_set_dof_idx));
141
142 const unsigned int global_size =
143 dealii::Utilities::MPI::sum(local_size, scratch_data->get_mpi_comm(level_set_dof_idx));
144
145 const number velocity = global_sum / global_size;
146
147 for (unsigned int i = 0; i < level_set.size(); i++)
148 level_set[i] += time_step * velocity;
149 }
150 else
151 AssertThrow(false,
152 dealii::ExcMessage(
153 "Analytical function for level-set advection is "
154 "only supported for the cases 'oscillating_water_column' and 'two_phase'."));
155 }
