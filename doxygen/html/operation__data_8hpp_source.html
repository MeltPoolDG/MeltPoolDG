206 void
207 post(const FiniteElementData &base_fe_data, const unsigned int base_verbosity_level)
208 {
209 fe.post(base_fe_data);
210 AssertThrow(fe.type == FiniteElementType::FE_DGQ,
211 dealii::ExcMessage(
212 "The compressible flow solver only supports elements of type 'FE_DGQ'."));
213
214 // set default time integration scheme for cut
215 if (domain_representation_type == "cut")
216 {
217 if (time_integrator.integrator_type ==
218 TimeIntegration::TimeIntegratorSchemes::not_initialized)
219 time_integrator.integrator_type =
220 TimeIntegration::TimeIntegratorSchemes::explicit_euler;
221 AssertThrow(
222 time_integrator.integrator_type ==
223 TimeIntegration::TimeIntegratorSchemes::explicit_euler,
224 dealii::ExcMessage(
225 "The cut compressible flow solver only supports explicit Euler time integration."));
226 }
227
228 AssertThrow(
229 not cfl_time_step.enable or (cfl_time_step.viscous_courant_number > 0.0 and
230 cfl_time_step.convective_courant_number > 0.0),
231 dealii::ExcMessage(
232 "If CFL time step limit is enabled, the viscous and convective Courant numbers must be positive."));
233
234 // A CFL time step limit update frequency of 0 is not allowed.
235 AssertThrow(cfl_time_step.update_frequency > 0 or not cfl_time_step.enable,
236 dealii::ExcMessage(
237 "A CFL time step limit update frequency of 0 is not allowed."));
238
239 // Synchronize verbosity with base verbosity if not set explicitly.
240 if (verbosity_level < 0)
241 verbosity_level = base_verbosity_level;
242 }
